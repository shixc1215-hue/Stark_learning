🔍 Step 4：本地关键词搜索模块 search.py ~20 分钟
搜索策略：将用户输入拆分成单个关键词，然后在知识库的 keywords 列表中查找匹配。匹配度 = 匹配到的关键词数量 / 用户输入的关键词总数。返回匹配度最高的几条结果。 
# search.py
import json
import re
from config import KNOWLEDGE_BASE_FILE, MIN_KEYWORD_MATCH, MAX_RESULTS

def load_knowledge_base() -> list:
    """从 JSON 文件加载知识库"""
    try:
        with open(KNOWLEDGE_BASE_FILE, "r", encoding="utf-8") as f:
            return json.load(f)
    except FileNotFoundError:
        print(f"⚠️ 知识库文件 {KNOWLEDGE_BASE_FILE} 不存在")
        return []
    except json.JSONDecodeError as e:
        print(f"⚠️ 知识库 JSON 格式错误：{e}")
        return []


def extract_keywords(text: str) -> list:
    """
    从用户输入中提取关键词

    处理步骤：
    1. 去掉标点符号
    2. 按空格和常见分隔符拆分
    3. 过滤掉太短的无意义词（1个字的通常没意义）
    4. 去重
    """
    # 去掉标点符号
    cleaned = re.sub(r'[，。！？、；：""''（）【】《》\s]+', " ", text)

    # 按空格拆分
    words = cleaned.split()

    # 如果拆分后只有一个长字符串，尝试按2字切分
    if len(words) == 1 and len(words[0]) > 2:
        # 对中文：按2字粒度切分（简单方案，后续可以用分词库）
        words = [words[0][i:i+2] for i in range(len(words[0]) - 1)]

    # 过滤太短的词 + 去重
    keywords = list(set(w for w in words if len(w) >= 2))

    return keywords


def search_knowledge_base(query: str, knowledge_base: list = None) -> list:
    """
    在知识库中搜索匹配的问答对

    匹配逻辑：
    1. 提取用户输入的关键词
    2. 遍历知识库每一条
    3. 计算匹配的关键词数量
    4. 按匹配度排序，返回前N条

    Returns:
        匹配到的问答对列表（按匹配度排序）
    """
    if knowledge_base is None:
        knowledge_base = load_knowledge_base()

    user_keywords = extract_keywords(query)

    # 如果提取不到有效关键词，直接返回空
    if not user_keywords:
        return []

    results = []
    for item in knowledge_base:
        # 计算这条知识库条目匹配了多少个关键词
        matched = set(user_keywords) & set(item["keywords"])
        match_count = len(matched)

        if match_count >= MIN_KEYWORD_MATCH:
            results.append({
                "item": item,
                "match_count": match_count,
                "match_ratio": match_count / len(user_keywords),
                "matched_keywords": list(matched)
            })

    # 按匹配数量降序排列，取前 N 条
    results.sort(key=lambda x: x["match_count"], reverse=True)
    return results[:MAX_RESULTS]

# llm_client.py
from openai import OpenAI
from config import ZHIPU_API_KEY, ZHIPU_BASE_URL, ZHIPU_MODEL, SYSTEM_PROMPT

# ====== 初始化客户端（只创建一次）======
client = OpenAI(
    api_key=ZHIPU_API_KEY,
    base_url=ZHIPU_BASE_URL
)


def ask_llm(
    user_prompt: str,
    system_prompt: str = None,
    temperature: float = 0.7,
    history: list = None
) -> str:
    """
    调用 LLM API 获取回答

    Args:
        user_prompt: 用户的问题
        system_prompt: 系统提示词（默认用 config 里的）
        temperature: 创造性程度（0=稳定, 1=创造性）
        history: 对话历史列表（用于多轮对话）

    Returns:
        LLM 的回答文本

    Raises:
        Exception: API 调用失败时抛出
    """
    if system_prompt is None:
        system_prompt = SYSTEM_PROMPT

    # 构造消息列表
    messages = [{"role": "system", "content": system_prompt}]

    # 加入历史对话（如果有）
    if history:
        messages.extend(history)

    # 加入当前问题
    messages.append({"role": "user", "content": user_prompt})

    try:
        response = client.chat.completions.create(
            model=ZHIPU_MODEL,
            messages=messages,
            temperature=temperature
        )
        return response.choices[0].message.content

    except Exception as e:
        print(f"❌ LLM API 调用失败：{e}")
        return None


# ====== 测试 ======
if __name__ == "__main__":
    print("=== LLM 调用测试 ===\n")

    answer = ask_llm("请用一句话解释什么是净回收率")
    if answer:
        print(f"LLM 回答：{answer}")
    else:
        print("调用失败，请检查 API Key")

main.py 的职责：像项目经理一样把各个模块"组装"起来。它不做具体工作（搜索、调 LLM 都交给专门的模块），只负责控制流程：读用户输入 → 先搜索本地 → 没结果就调 LLM → 显示回答 → 循环。 
# main.py — 不良资产知识问答助手（主程序）
from search import search_knowledge_base, load_knowledge_base
from llm_client import ask_llm
from config import SYSTEM_PROMPT

def format_local_answer(results: list) -> str:
    """将本地搜索结果格式化为友好的回答文本"""
    if not results:
        return None

    answer = "📚 在本地知识库中找到以下相关信息：\n\n"
    for i, r in enumerate(results, 1):
        item = r["item"]
        answer += f"【{i}】{item['question']}\n"
        answer += f"    {item['answer']}\n\n"
        if r["matched_keywords"]:
            answer += f"    （匹配关键词：{', '.join(r['matched_keywords'])}）\n\n"

    return answer


def chat_loop():
    """
    主对话循环

    流程：
    1. 加载知识库
    2. 显示欢迎信息
    3. 循环：读输入 → 本地搜索 → LLM 兜底 → 显示回答
    4. 用户输入 'quit' 或 'exit' 时退出
    """
    # 加载知识库（只加载一次）
    kb = load_knowledge_base()
    if not kb:
        print("⚠️ 知识库加载失败，将直接使用 LLM 回答")

    # 对话历史（用于多轮对话上下文）
    history = []

    # ====== 欢迎信息 ======
    print()
    print("╔══════════════════════════════════════════════╗")
    print("║   AMC 不良资产知识问答助手 v1.0            ║")
    print("║   本地知识库 + LLM 智能回答                  ║")
    print("╠══════════════════════════════════════════════╣")
    print("║  输入问题开始对话                            ║")
    print("║  输入 'quit' 或 'exit' 退出                  ║")
    print("║  输入 'help' 查看帮助                        ║")
    print("╚══════════════════════════════════════════════╝")
    print()

    # ====== 主循环 ======
    while True:
        # 读取用户输入
        try:
            user_input = input("👤 你：").strip()
        except (KeyboardInterrupt, EOFError):
            print("\n👋 再见！")
            break

        # 空输入跳过
        if not user_input:
            continue

        # 退出命令
        if user_input.lower() in ("quit", "exit", "退出", "再见"):
            print("👋 再见！")
            break

        # 帮助命令
        if user_input.lower() in ("help", "帮助"):
            print("""
💡 使用提示：
  - 直接输入 AMC 相关问题即可
  - 系统会先在本地知识库（20条）中搜索
  - 如果没有匹配结果，会调用 LLM 回答
  - 支持：不良资产、处置方式、回收率、数据仓库、LLM概念等
  - 输入 'quit' 退出
""")
            continue

        # ====== 第一步：本地知识库搜索 ======
        local_results = search_knowledge_base(user_input, kb)

        if local_results:
            # 本地找到了！直接用本地答案
            answer = format_local_answer(local_results)
            source = "📚 本地知识库"
        else:
            # 本地没找到，调 LLM
            print("🔍 本地未找到，正在调用 LLM...")
            answer = ask_llm(
                user_prompt=user_input,
                history=history
            )
            source = "🤖 LLM 回答"

            if answer is None:
                print("❌ LLM 调用失败，请稍后重试")
                continue

        # 显示回答
        print()
        print(f"{source}：")
        print(f"{'─' * 40}")
        print(answer)
        print(f"{'─' * 40}")
        print()

        # 更新对话历史
        history.append({"role": "user", "content": user_input})
        history.append({"role": "assistant", "content": answer})

        # 限制历史长度（避免 token 浪费）
        if len(history) > 20:  # 最多保留最近 10 轮（20条消息）
            history = history[-20:]


# ====== 程序入口 ======
if __name__ == "__main__":
    chat_loop()

核心理解：编码器 vs 解码器

用一个简单的比喻理解两大类模型：

编码器模型（BERT）

像一个"阅读理解专家"——给它一段文本，它能理解语义并提取信息。但它不擅长"创造"新文本。

代表作：BERT、RoBERTa

解码器模型（GPT）

像一个"故事讲述者"——给它一个开头，它能逐字生成后续内容。现代 LLM 应用开发主要用这类。

代表作：GPT系列、Qwen、Llama、GLM

LeetCode #349 · 两个数组的交集（Easy）

给定两个数组 nums1 和 nums2，返回它们的交集。结果中的每个元素只出现一次。

示例 1：nums1 = [1,2,2,1], nums2 = [2,2] → [2]
示例 2：nums1 = [4,9,5], nums2 = [9,4,9,8,4] → [4,9]（顺序不限） 

def intersection(nums1: list, nums2: list) -> list:
    """返回两个数组的交集（元素唯一）"""
    # 方法 1：用 set 交集（一行搞定）
    return list(set(nums1) & set(nums2))

# 方法 2：手动实现（理解原理）
def intersection2(nums1: list, nums2: list) -> list:
    set1 = set(nums1)  # 先转集合去重
    result = []
    for n in nums2:
        if n in set1:  # O(1) 查找
            result.append(n)
            set1.remove(n)  # 避免重复
    return result

# 测试
print(intersection([1,2,2,1], [2,2]))             # [2]
print(intersection([4,9,5], [9,4,9,8,4]))        # [4, 9]

关联今天的项目：search.py 里 set(user_keywords) & set(item["keywords"]) 就是求交集——找出用户关键词和知识库关键词的重合部分。这道题让你更理解这个操作的原理。
