# === 3. statistics 统计库 — 一行算出核心统计量 ===
import statistics

# 模拟一组请求延迟数据（毫秒）
latencies = [120, 135, 142, 128, 450, 131, 138, 125, 133, 141]

# 均值 — 受异常值影响
avg = statistics.mean(latencies)  # 154.3ms（被 450 拉高）

# 中位数 — 更稳定，不受极端值影响
median = statistics.median(latencies)  # 134.0ms

# 标准差 — 波动幅度
stdev = statistics.stdev(latencies)  # 95.7ms（波动大说明不稳定）

# 众数 — 出现最频繁的值
# mode = statistics.mode(latencies)

# 分位数 — 压测最常用 P50/P95/P99
p50 = sorted(latencies)[int(len(latencies) * 0.50)]
p95 = sorted(latencies)[int(len(latencies) * 0.95)]
p99 = sorted(latencies)[int(len(latencies) * 0.99)]
# 更精确的分位数计算（处理非整数索引）
def percentile(data: list[float], p: float) -> float:
    """计算任意百分位数"""
    sorted_data = sorted(data)
    k = (len(sorted_data) - 1) * p
    f = int(k)
    c = k - f
    if f + 1 < len(sorted_data):
        return sorted_data[f] + c * (sorted_data[f + 1] - sorted_data[f])
    return sorted_data[f]

@contextmanager
def elapsed_timer():
    """返回一个对象，with 块结束后可读取 elapsed"""
    result = {"elapsed": 0.0}
    start = time.perf_counter()
    try:
        yield result
    finally:
        result["elapsed"] = time.perf_counter() - start


# === Prometheus 格式指标导出器 ===

class PrometheusExporter:
    """
    生成 Prometheus /metrics 格式的文本输出
    Prometheus 是业界标准的监控数据格式，Grafana 可直接消费
    """

    def __init__(self, prefix: str = "llm"):
        self.prefix = prefix
        self._counters: dict[str, float] = {}
        self._gauges: dict[str, float] = {}
        self._histograms: dict[str, list[float]] = {}

    def counter_inc(self, name: str, value: float = 1.0, labels: dict = None):
        """递增计数器（只增不减，用于请求总数、错误数）"""
        key = self._make_key(name, labels)
        self._counters[key] = self._counters.get(key, 0) + value

    def gauge_set(self, name: str, value: float, labels: dict = None):
        """设置仪表盘（可升可降，用于当前延迟、队列长度）"""
        key = self._make_key(name, labels)
        self._gauges[key] = value

    def histogram_observe(self, name: str, value: float, labels: dict = None):
        """记录直方图观测值（用于延迟分布）"""
        key = self._make_key(name, labels)
        if key not in self._histograms:
            self._histograms[key] = []
        self._histograms[key].append(value)

    def export(self) -> str:
        """导出 Prometheus 文本格式"""
        lines = []
        for key, val in self._counters.items():
            name, label_str = self._parse_key(key)
            lines.append(f"# TYPE {name} counter")
            lines.append(f"{name}{label_str} {val}")
        for key, val in self._gauges.items():
            name, label_str = self._parse_key(key)
            lines.append(f"# TYPE {name} gauge")
            lines.append(f"{name}{label_str} {val:.4f}")
        for key, values in self._histograms.items():
            name, label_str = self._parse_key(key)
            lines.append(f"# TYPE {name} histogram")
            for v in values:
                lines.append(f"{name}{label_str} {v:.4f}")
        return "\n".join(lines)

    def _make_key(self, name, labels):
        if labels:
            label_str = ",".join(f'{k}="{v}"' for k, v in labels.items())
            return f"{self.prefix}_{name}{{{label_str}}}"
        return f"{self.prefix}_{name}"

    def _parse_key(self, key):
        """从 key 中分离指标名和标签"""
        if "{" in key:
            name, rest = key.split("{", 1)
            return name, "{" + rest
        return key, ""
  
# === LLM 监控探针（向 LLM API 发请求并采集指标）===

class LLMProbe:
    """
    黑盒监控探针：向 LLM API 发请求，采集 TTFT/TPOT/延迟等指标
    支持 OpenAI 兼容接口（vLLM/Ollama/智谱等）
    """

    def __init__(self, base_url: str, api_key: str = "empty",
                 model: str = "default", timeout: float = 30.0):
        self.base_url = base_url.rstrip("/")
        self.api_key = api_key
        self.model = model
        self.timeout = timeout
        self.exporter = PrometheusExporter(prefix="llm_probe")

    def health_check(self) -> bool:
        """简单的健康检查：GET /health 或 /v1/models"""
        try:
            resp = requests.get(
                f"{self.base_url}/v1/models",
                headers={"Authorization": f"Bearer {self.api_key}"},
                timeout=5
            )
            return resp.status_code == 200
        except Exception:
            return False

    def probe(self, prompt: str, max_tokens: int = 50) -> RequestMetric:
        """
        发送请求并采集完整指标
        使用 stream=True 来精确测量 TTFT
        """
        request_id = str(uuid.uuid4())[:8]
        endpoint = "/v1/chat/completions"

        # 请求体 — 标准 OpenAI 格式
        payload = {
            "model": self.model,
            "messages": [{"role": "user", "content": prompt}],
            "max_tokens": max_tokens,
            "stream": True,  # 流式获取，才能测 TTFT
        }

        headers = {
            "Authorization": f"Bearer {self.api_key}",
            "Content-Type": "application/json",
        }

        # 记录开始时间
        start_time = time.perf_counter()
        ttft = None
        output_tokens = 0
        error_msg = None
        status = 0
        prompt_tokens = 0

        try:
            resp = requests.post(
                f"{self.base_url}{endpoint}",
                json=payload,
                headers=headers,
                stream=True,
                timeout=self.timeout,
            )
            status = resp.status_code

            # 非流式错误直接返回
            if status != 200:
                error_msg = resp.text[:200]
                e2e = time.perf_counter() - start_time
                return RequestMetric(
                    request_id=request_id, endpoint=endpoint,
                    status=status, ttft=None, tpot=None,
                    e2e_latency=e2e, prompt_tokens=0,
                    output_tokens=0, error_msg=error_msg,
                )

            # 流式读取 SSE 事件
            for line in resp.iter_lines(decode_unicode=True):
                if not line or not line.startswith("data: "):
                    continue
                data_str = line[6:]  # 去掉 "data: " 前缀
                if data_str == "[DONE]":
                    break

                chunk = json.loads(data_str)
                delta = chunk["choices"][0]["delta"]

                # 第一个非空 content 就是首 Token
                if delta.get("content") and ttft is None:
                    ttft = time.perf_counter() - start_time

                # 统计输出 token 数
                if delta.get("content"):
                    output_tokens += 1

                # 从 usage 获取输入 token 数
                if "usage" in chunk:
                    prompt_tokens = chunk["usage"].get("prompt_tokens", 0)

        except requests.exceptions.Timeout:
            error_msg = "请求超时"
            status = 0
        except Exception as e:
            error_msg = str(e)[:200

        e2e_latency = time.perf_counter() - start_time

        # 计算 TPOT（平均每个输出 Token 的耗时）
        tpot = None
        if ttft and output_tokens > 1:
            generation_time = e2e_latency - ttft
            tpot = generation_time / (output_tokens - 1)

        metric = RequestMetric(
            request_id=request_id,
            endpoint=endpoint,
            status=status,
            ttft=ttft,
            tpot=tpot,
            e2e_latency=e2e_latency,
            prompt_tokens=prompt_tokens,
            output_tokens=output_tokens,
            error_msg=error_msg,
        )

        # 同步更新 Prometheus 指标
        self.exporter.counter_inc(
            "requests_total",
            labels={"status": str(status)}
        )
        if e2e_latency:
            self.exporter.histogram_observe("latency_seconds", e2e_latency)
            self.exporter.gauge_set("last_latency", e2e_latency)

        return metric
🔑 TTFT 测量原理：TTFT（Time To First Token）是用户体验的核心指标——用户看到第一个字花了多久。只有 stream=True 流式请求才能测 TTFT，因为非流式请求要等全部生成完才返回。
📊 实战 2：指标统计分析器 ⏱ 15min

采集到指标后，需要统计分析：P50/P95/P99 延迟、吞吐量、错误率、告警判断。

import statistics
from dataclasses import dataclass, field

@dataclass
class StatsReport:
    """统计分析报告"""
    total_requests: int
    success_count: int
    error_count: int
    error_rate: float

    # 延迟统计（单位：秒）
    avg_latency: float
    median_latency: float
    p95_latency: float
    p99_latency: float
    min_latency: float
    max_latency: float
    std_latency: float

    # TTFT 统计
    avg_ttft: float | None
    p95_ttft: float | None

    # 吞吐量
    throughput_tps: float     # Token/秒
    qps: float                 # 请求/秒
    total_output_tokens: int

    # 告警
    alerts: list[str] = field(default_factory=list)


class MetricsAnalyzer:
    """
    指标统计分析器
    输入：一批 RequestMetric
    输出：StatsReport + 告警列表
    """

    # 告警阈值（可配置）
    DEFAULT_THRESHOLDS = {
        "error_rate": 0.05,       # 错误率 > 5%
        "p99_latency": 30.0,    # P99 延迟 > 30s
        "p95_ttft": 3.0,        # P95 TTFT > 3s
        "throughput_min": 50,    # 吞吐量 < 50 tok/s
    }

    def __init__(self, thresholds: dict | None = None):
        self.thresholds = {**self.DEFAULT_THRESHOLDS, **(thresholds or {})}
        self.alerts: list[str] = []

    def analyze(self, metrics: list[RequestMetric]) -> StatsReport:
        """分析一批请求指标，生成统计报告"""
        if not metrics:
            raise ValueError("指标列表为空")

        # 基本统计
        total = len(metrics)
        successes = [m for m in metrics if m.status == 200]
        errors = [m for m in metrics if m.status != 200]
        error_rate = len(errors) / total

        # 延迟分布（仅成功的请求）
        latencies = [m.e2e_latency for m in successes]
        ttf_ts = [m.ttft for m in successes if m.ttft is not None]
        total_output = sum(m.output_tokens for m in successes)

        # 吞吐量：总 Token 数 / 总耗时
        time_span = max(m.timestamp for m in metrics) - min(m.timestamp for m in metrics)
        time_span = max(time_span, 0.001)  # 避免除零
        throughput = total_output / time_span
        qps = total / time_span

        # 分位数计算
        p50 = self._percentile(latencies, 0.50)
        p95 = self._percentile(latencies, 0.95)
        p99 = self._percentile(latencies, 0.99)

        report = StatsReport(
            total_requests=total,
            success_count=len(successes),
            error_count=len(errors),
            error_rate=error_rate,
            avg_latency=statistics.mean(latencies) if latencies else 0,
            median_latency=p50,
            p95_latency=p95,
            p99_latency=p99,
            min_latency=min(latencies) if latencies else 0,
            max_latency=max(latencies) if latencies else 0,
            std_latency=statistics.stdev(latencies) if len(latencies) > 1 else 0,
            avg_ttft=statistics.mean(ttf_ts) if ttf_ts else None,
            p95_ttft=self._percentile(ttf_ts, 0.95) if ttf_ts else None,
            throughput_tps=throughput,
            qps=qps,
            total_output_tokens=total_output,
        )

        # 自动告警检测
        report.alerts = self._check_alerts(report)
        return report

    def _percentile(self, data: list[float], p: float) -> float:
        """精确分位数（线性插值）"""
        if not data:
            return 0.0
        s = sorted(data)
        k = (len(s) - 1) * p
        f = int(k)
        c = k - f
        if f + 1 < len(s):
            return s[f] + c * (s[f + 1] - s[f])
        return s[f]

    def _check_alerts(self, report: StatsReport) -> list[str]:
        """根据阈值生成告警"""
        alerts = []
        t = self.thresholds

        if report.error_rate > t["error_rate"]:
            alerts.append(
                f"🔴 错误率 {report.error_rate:.1%} 超过阈值 {t['error_rate']:.1%}"
            )
        if report.p99_latency > t["p99_latency"]:
            alerts.append(
                f"🔴 P99延迟 {report.p99_latency:.1f}s 超过阈值 {t['p99_latency']}s"
            )
        if report.p95_ttft and report.p95_ttft > t["p95_ttft"]:
            alerts.append(
                f"🟡 P95 TTFT {report.p95_ttft:.2f}s 超过阈值 {t['p95_ttft']}s"
            )
        if report.throughput_tps < t["throughput_min"]:
            alerts.append(
                f"🟡 吞吐量 {report.throughput_tps:.0f} tok/s 低于阈值 {t['throughput_min']} tok/s"
            )
        return alerts

    def print_report(self, report: StatsReport):
        """格式化打印报告"""
        print(f"""
╔══════════════════════════════════════════════════╗
║           LLM 服务监控报告                       ║
╠══════════════════════════════════════════════════╣
║  总请求数:  {report.total_requests:>6}                              ║
║  成功:  {report.success_count:>6}   失败:  {report.error_count:>6}                 ║
║  错误率:  {report.error_rate:>6.1%}                            ║
╠══════════════════════════════════════════════════╣
║  延迟统计 (秒)                                   ║
║  平均: {report.avg_latency:>8.3f}  P50: {report.median_latency:>8.3f}           ║
║  P95: {report.p95_latency:>8.3f}  P99: {report.p99_latency:>8.3f}           ║
║  最小: {report.min_latency:>8.3f}  最大: {report.max_latency:>8.3f}           ║
║  标准差: {report.std_latency:>6.3f}                              ║
╠══════════════════════════════════════════════════╣
║  TTFT (首Token时间)                              ║

╠══════════════════════════════════════════════════╣
║  吞吐量: {report.throughput_tps:>8.1f} tok/s   QPS: {report.qps:>5.1f}          ║

        if report.alerts:

# === 异步压测引擎 ===

class LoadTestEngine:
    """
    异步压力测试引擎
    用 Semaphore 控制并发数（Day 24 学过）
    """

    def __init__(self, config: LoadTestConfig):
        self.config = config
        self.metrics: list[RequestMetric] = []

    async def _single_request(self, session: aiohttp.ClientSession,
                             semaphore: asyncio.Semaphore) -> RequestMetric:
        """单次异步请求 + 指标采集"""
        async with semaphore:  # Semaphore 控制并发上限
            request_id = str(uuid.uuid4())[:8]
            start_time = time.perf_counter()
            ttft = None
            output_tokens = 0
            prompt_tokens = 0
            status = 0
            error_msg = None

            payload = {
                "model": self.config.model,
                "messages": [{"role": "user", "content": self.config.prompt}],
                "max_tokens": self.config.max_tokens,
                "stream": True,
            }

            try:
                async with session.post(
                    f"{self.config.base_url}/v1/chat/completions",
                    json=payload,
                    headers={"Authorization": f"Bearer {self.config.api_key}"},
                    timeout=aiohttp.ClientTimeout(total=self.config.timeout),
                ) as resp:
                    status = resp.status
                    if status != 200:
                        error_msg = (await resp.text())[:200]
                    else:
                        # 异步流式读取 SSE
                        async for line in resp.content:
                            decoded = line.decode("utf-8").strip()
                            if not decoded or not decoded.startswith("data: "):
                                continue
                            data_str = decoded[6:]
                            if data_str == "[DONE]":
                                break
                            chunk = json.loads(data_str)
                            delta = chunk["choices"][0]["delta"]
                            if delta.get("content"):
                                if ttft is None:
                                    ttft = time.perf_counter() - start_time
                                output_tokens += 1
                            if "usage" in chunk:
                                prompt_tokens = chunk["usage"].get("prompt_tokens", 0)

            except asyncio.TimeoutError:
                error_msg = "请求超时"
            except Exception as e:
                error_msg = str(e)[:200]

            e2e_latency = time.perf_counter() - start_time
            tpot = None
            if ttft and output_tokens > 1:
                tpot = (e2e_latency - ttft) / (output_tokens - 1)

            return RequestMetric(
                request_id=request_id,
                endpoint="/v1/chat/completions",
                status=status,
                ttft=ttft, tpot=tpot,
                e2e_latency=e2e_latency,
                prompt_tokens=prompt_tokens,
                output_tokens=output_tokens,
                error_msg=error_msg,
            )

    async def run(self) -> list[RequestMetric]:
        """执行压测主流程"""
        cfg = self.config
        semaphore = asyncio.Semaphore(cfg.concurrency)

        print(f"""
🚀 开始压力测试
   目标: {cfg.base_url}
   并发: {cfg.concurrency}  总请求: {cfg.total_requests}
   Prompt: {cfg.prompt[:50]}...
""")

        async with aiohttp.ClientSession() as session:
            # 创建所有任务（受 Semaphore 控制并发）
            tasks = [
                self._single_request(session, semaphore)
                for _ in range(cfg.total_requests)
            ]
            # asyncio.gather 并发执行所有任务
            self.metrics = await asyncio.gather(*tasks)

        print(f"✅ 压测完成，共 {len(self.metrics)} 个请求")
        return self.metrics

    async def run_with_ramp_up(self) -> list[RequestMetric]:
        """带预热阶段的压测（逐步增加并发）"""
        cfg = self.config
        all_metrics = []
        requests_per_step = max(1, cfg.total_requests // 5)
        steps = 5

        async with aiohttp.ClientSession() as session:
            for step in range(1, steps + 1):
                current_concurrency = int(cfg.concurrency * step / steps)
                semaphore = asyncio.Semaphore(current_concurrency)
                print(f"  阶段 {step}/{steps}：并发={current_concurrency}，"
                      f"请求={requests_per_step}")

                tasks = [
                    self._single_request(session, semaphore)
                    for _ in range(requests_per_step)
                ]
                step_metrics = await asyncio.gather(*tasks)
                all_metrics.extend(step_metrics)

        self.metrics = all_metrics
        return all_metrics

    def export_results(self, filepath: str = "load_test_results.json"):
        """导出结果为 JSON 文件"""
        data = [asdict(m) for m in self.metrics]
        with open(filepath, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)
        print(f"📁 结果已保存: {filepath}")

#陷阱症状解法
1 压测把生产打挂 并发压测导致 GPU OOM，服务雪崩 用 Semaphore 限制并发；先小并发测再逐步加大；压测环境与生产隔离
2 TTFT 不准 TTFT 包含了网络传输时间 尽量在同一网络环境压测；对比「本地 vs 远程」TTFT 差异；首 Token 应以 delta.content 为准，排除 usage 事件
3 P99 被异常值污染 1 次超时导致 P99 极高，掩盖真实水位 同时看 P50/P95/P99；用「剔除 P100（最大值）」后的 P99 更稳健；关注分布而非单点
4 只压不监控 压测跑完只看 QPS，不看延迟分布和错误 压测 = 监控 + 统计 + 告警；每次压测都要输出 StatsReport；保存原始 JSON 做历史对比
5 遗忘异步陷阱 aiohttp + requests 混用导致阻塞 压测全程用 aiohttp；不要在 async 函数里调 requests.get()；Day 24 学的异步规则仍然适用
