🐍 一、Python 图算法 — 血缘图上的遍历 25 min

数据血缘本质上是一张有向无环图（DAG）：表/字段是节点，依赖关系是边。Day 65 我们已经用 defaultdict(list) 构建了血缘图，今天先学怎么在图上做遍历。

数仓类比：SQL 解析器就像 ETL 中的 Extract — 从 SQL 文本中提取原始依赖。图算法是 Transform — 把原始边加工成可查询的拓扑结构。影响分析是 Load — 把结果交付给业务方。

1.1 血缘图的两种表示

from collections import defaultdict, deque
from dataclasses import dataclass, field
from typing import Optional
import json

# ── 数据模型 ──

@dataclass
class LineageEdge:
    """血缘关系边"""
    source_table: str           # 上游表
    source_column: str         # 上游字段
    target_table: str           # 下游表
    target_column: str         # 下游字段
    transform_type: str = "direct"  # direct / agg / derive / constant
    confidence: float = 1.0     # 解析置信度 (0~1)
    source: str = "sql_parser"  # sql_parser / llm / manual

@dataclass
class LineageGraph:
    """血缘图 — 同时维护正向和反向邻接表"""
    edges: list[LineageEdge] = field(default_factory=list)
    _downstream: dict = field(default_factory=lambda: defaultdict(set))  # 表A → {表B, 表C}
    _upstream: dict = field(default_factory=lambda: defaultdict(set))    # 表B → {表A}
    _col_down: dict = field(default_factory=lambda: defaultdict(set)) # (表,字段) → {(表,字段)}
    _col_up: dict = field(default_factory=lambda: defaultdict(set))   # 反向

    def add_edge(self, edge: LineageEdge):
        """添加一条血缘边，同时更新四种索引"""
        self.edges.append(edge)
        # 表级索引
        self._downstream[edge.source_table].add(edge.target_table)
        self._upstream[edge.target_table].add(edge.source_table)
        # 字段级索引（用元组做 key）
        src_key = (edge.source_table, edge.source_column)
        tgt_key = (edge.target_table, edge.target_column)
        self._col_down[src_key].add(tgt_key)
        self._col_up[tgt_key].add(src_key)

    def get_all_tables(self) -> set[str]:
        """获取所有表（并集）"""
        return self._downstream.keys() | self._upstream.keys()

    # ── BFS 广度优先：找「直接下游」 ──
    def bfs_downstream(self, start_table: str, max_depth: int = 10) -> dict[str, int]:
        """
        BFS 沿血缘方向（上游→下游）遍历
        返回: {表名: 层级深度}
        """
        visited: dict[str, int] = {start_table: 0}
        queue = deque([start_table])

        while queue:
            current = queue.popleft()
            current_depth = visited[current]
            if current_depth >= max_depth:
                continue
            # 遍历所有直接下游
            for child in self._downstream.get(current, set()):
                if child not in visited:
                    visited[child] = current_depth + 1
                    queue.append(child)

        return visited

    # ── DFS 深度优先：找「完整链路」 ──
    def dfs_all_paths(
        self, start: str, end: str,
        max_depth: int = 10, path: Optional[list] = None
    ) -> list[list[str]]:
        """
        DFS 找从 start 到 end 的所有路径
        用于回答「A 表的数据经过哪些中间表到达 B 表」
        """
        if path is None:
            path = [start]

        if len(path) - 1 >= max_depth:  # 深度限制防死循环
            return []

        paths = []
        if start == end:
            return [path]

        for next_table in self._downstream.get(start, set()):
            if next_table in path:  # 跳过已访问（防环）
                continue
            sub_paths = self.dfs_all_paths(
                next_table, end, max_depth, path + [next_table]
            )
            paths.extend(sub_paths)

        return paths

    # ── 拓扑排序：验证 DAG + 执行顺序 ──
    def topological_sort(self) -> tuple[list[str], bool]:
        """
        Kahn 算法拓扑排序
        返回: (排序列表, 是否存在环)
        """
        all_tables = self.get_all_tables()
        in_degree = {t: 0 for t in all_tables}

        # 计算每个表的入度（有多少上游表指向它）
        for src, targets in self._downstream.items():
            for tgt in targets:
                if tgt in in_degree:
                    in_degree[tgt] += 1

        # 入度为 0 的表先入队（没有依赖的源表）
        queue = deque([t for t, d in in_degree.items() if d == 0])
        result = []

        while queue:
            table = queue.popleft()
            result.append(table)
            for child in self._downstream.get(table, set()):
                if child in in_degree:
                    in_degree[child] -= 1
                    if in_degree[child] == 0:
                        queue.append(child)

        has_cycle = len(result) != len(all_tables)
        return result, has_cycle

    # ── 字段级血缘追溯 ──
    def trace_column(self, table: str, column: str,
                         direction: str = "upstream") -> list[tuple]:
        """
        追溯某字段的上游/下游血缘链
        direction: "upstream" 或 "downstream"
        """
        graph = self._col_up if direction == "upstream" else self._col_down
        start = (table, column)
        visited = {start}
        result = []
        queue = deque([start])

        while queue:
            current = queue.popleft()
            for neighbor in graph.get(current, set()):
                if neighbor not in visited:
                    visited.add(neighbor)
                    result.append(neighbor)
                    queue.append(neighbor)

        return result
BFS vs DFS 选择原则：BFS 适合找「最近/最直接」的关系（影响范围），DFS 适合找「所有可能路径」（完整链路）。拓扑排序验证 DAG — 有环说明存在循环依赖，这在数仓中是严重问题。

🤖 二、LLM 辅助血缘补全 — 补 SQL 解析的盲区 15 min

SQL 解析器只能抓到显式引用（SELECT / FROM 中写的表名），但数仓里还有大量隐式血缘：

  • 配置表/字典表：代码里硬编码关联的 dim 表，SQL 中用 JOIN 但字段名改了
  • 存储过程 / 视图嵌套：A 调用 B 视图，B 引用 C 表 — 需要递归展开
  • 业务语义关系：「不良资产余额」和「风险分类」有逻辑关联，但 SQL 里没有显式 JOIN
核心思路：SQL 解析 = 规则引擎（快但浅），LLM 补全 = 语义理解（慢但深）。两者互补：confidence 为 1.0 的是规则解析的确定结果，LLM 生成的关系标记为 0.7~0.9，供人工审核。

2.1 LLM 血缘补全器

import json
import re

def parse_llm_json_safely(text: str) -> list[dict]:
    """
    三层 fallback 提取 LLM 返回的 JSON 数组
    实战中最常见的 LLM 输出问题：```json 包裹 / 多余文字 / 不合法 JSON
    """
    # 第一层：尝试直接解析
    try:
        result = json.loads(text)
        if isinstance(result, list):
            return result
    except (json.JSONDecodeError, TypeError):
        pass

    # 第二层：正则提取 ```json ... ``` 代码块
    pattern = r"```(?:json)?\s*\n?(.*?)\n?```"
    match = re.search(pattern, text, re.DOTALL)
    if match:
        try:
            result = json.loads(match.group(1))
            if isinstance(result, list):
                return result
        except json.JSONDecodeError:
            pass

    # 第三层：正则提取 [...] 数组
    array_pattern = r"\[.*?\]"
    match = re.search(array_pattern, text, re.DOTALL)
    if match:
        try:
            result = json.loads(match.group(0))
            if isinstance(result, list):
                return result
        except json.JSONDecodeError:
            pass

    return []  # 全部失败，返回空


class LLMLineageCompleter:
    """
    LLM 辅助血缘补全器
    职责：发现 SQL 解析遗漏的隐式血缘关系
    """

    def __init__(self, api_key: str, model: str = "glm-4-flash"):
        self.api_key = api_key
        self.model = model
        self.base_url = "https://open.bigmodel.cn/api/paas/v4/chat/completions"

    def _call_llm(self, prompt: str) -> str:
        """调用智谱 API（同步，简化演示）"""
        import requests
        resp = requests.post(
            self.base_url,
            headers={"Authorization": f"Bearer {self.api_key}"},
            json={
                "model": self.model,
                "messages": [{"role": "user", "content": prompt}],
                "temperature": 0.1,  # 低温度：血缘需要精确
            },
            timeout=30
        )
        resp.raise_for_status()
        return resp.json()["choices"][0]["message"]["content"]

    def _build_prompt(
        self,
        table_name: str,
        table_comment: str,
        columns: list[dict],
        known_upstream: list[str],
        all_tables: list[str]
    ) -> str:
        """构建血缘补全 Prompt"""
        cols_desc = "\n".join(
            f"  - {c['name']}: {c.get('comment','')}(类型:{c.get('type','')})"
            for c in columns
        )
        return f"""你是数据血缘分析专家。请分析以下表的隐式血缘关系。

## 当前表
表名: {table_name}
表注释: {table_comment}
字段:
{cols_desc}

## SQL解析已知的上游表
{', '.join(known_upstream) if known_upstream else '（无）'}

## 数仓中所有可用的表
{', '.join(all_tables)}

## 任务
SQL 解析器只能识别显式引用。请基于业务语义，找出当前表可能存在但 SQL 解析遗漏的血缘关系。
例如：
- 字典表/维度表（如 dim_risk_level）虽然被 JOIN 但字段名被重命名
- 配置表（如 config_xxx）在代码中关联但未出现在 SQL 里
- 语义关联（如金额字段隐含币种依赖）

请只返回 JSON 数组，每个元素包含：
{{"upstream_table": "表名", "upstream_column": "字段名", "relationship": "业务关系说明", "confidence": 0.7}}

如果找不到遗漏关系，返回空数组 []。"""

    def complete(
        self,
        table_name: str,
        table_comment: str,
        columns: list[dict],
        known_upstream: list[str],
        all_tables: list[str]
    ) -> list[LineageEdge]:
        """执行 LLM 血缘补全，返回新增的 LineageEdge"""
        prompt = self._build_prompt(
            table_name, table_comment, columns,
            known_upstream, all_tables
        )
        raw = self._call_llm(prompt)
        suggestions = parse_llm_json_safely(raw)

        new_edges = []
        for item in suggestions:
            edge = LineageEdge(
                source_table=item.get("upstream_table", ""),
                source_column=item.get("upstream_column", "*"),
                target_table=table_name,
                target_column="*",
                transform_type="llm_inferred",
                confidence=item.get("confidence", 0.7),
                source="llm"
            )
            new_edges.append(edge)

        return new_edges
为什么 confidence 标记为 0.7~0.9？LLM 推断的关系未经代码验证，可能存在误判。在血缘图中用 confidence 字段区分来源：1.0 = SQL 解析确定性结果，< 1.0 = LLM 推断需人工确认。这是「人机协同」的核心设计模式。
📊 三、影响分析 — 字段变更会波及谁？ 10 min

数仓开发中最常见的场景：「我要改 ODS 层某字段的类型/名称，下游哪些表会受影响？」这就是影响分析。

数仓类比：影响分析就像「改一个 DIM 表的结构，查一下哪些 DWD/DWS/ADS 的 SQL 会报错」。你在 ETL 里改了源表字段，必须知道下游哪些作业会因此挂掉。

3.1 ImpactAnalyzer 影响分析器

from dataclasses import dataclass, field
from enum import Enum

class ImpactLevel(Enum):
    HIGH = "high"      # 直接引用，必受影响
    MEDIUM = "medium"  # 间接依赖（聚合后引用）
    LOW = "low"        # LLM 推断的语义关联

@dataclass
class ImpactResult:
    """单条影响记录"""
    table: str
    column: str
    level: ImpactLevel
    path: list[str]          # 变更传播路径
    transform_type: str      # direct / agg / derive
    confidence: float = 1.0

@dataclass
class ImpactReport:
    """影响分析报告"""
    source_table: str
    source_column: str
    impacts: list[ImpactResult] = field(default_factory=list)

    def summary(self) -> dict:
        """生成影响摘要统计"""
        from collections import Counter
        level_counts = Counter(i.level for i in self.impacts)
        return {
            "source": f"{self.source_table}.{self.source_column}",
            "total_impacts": len(self.impacts),
            "high": level_counts.get(ImpactLevel.HIGH, 0),
            "medium": level_counts.get(ImpactLevel.MEDIUM, 0),
            "low": level_counts.get(ImpactLevel.LOW, 0),
        }


class ImpactAnalyzer:
    """
    血缘影响分析器
    核心能力：给定一个变更源，沿血缘图 BFS 找出所有受影响的下游
    """

    def __init__(self, graph: LineageGraph):
        self.graph = graph

    def analyze(
        self, table: str, column: str,
        max_depth: int = 10
    ) -> ImpactReport:
        """
        分析某字段变更的下游影响
        BFS 遍历字段级血缘图
        """
        report = ImpactReport(source_table=table, source_column=column)
        start = (table, column)
        visited = {start}
        queue = deque([(start, [table], 1)])  # (当前节点, 路径, 深度)

        while queue:
            (cur_table, cur_col), path, depth = queue.popleft()
            if depth > max_depth:
                continue

            # 查找所有直接下游字段
            cur_key = (cur_table, cur_col)
            for (tgt_table, tgt_col) in self.graph._col_down.get(cur_key, set()):
                if (tgt_table, tgt_col) in visited:
                    continue
                visited.add((tgt_table, tgt_col))

                # 查找这条边的信息
                edge_info = self._find_edge(cur_table, cur_col, tgt_table, tgt_col)
                transform = edge_info.transform_type if edge_info else "unknown"
                conf = edge_info.confidence if edge_info else 0.5

                # 判断影响级别
                if conf >= 0.9 and transform in ("direct", "derive"):
                    level = ImpactLevel.HIGH
                elif conf >= 0.7:
                    level = ImpactLevel.MEDIUM
                else:
                    level = ImpactLevel.LOW

                report.impacts.append(ImpactResult(
                    table=tgt_table,
                    column=tgt_col,
                    level=level,
                    path=path + [tgt_table],
                    transform_type=transform,
                    confidence=conf
                ))
                queue.append(((tgt_table, tgt_col), path + [tgt_table], depth + 1))

        return report

    def _find_edge(self, src_t, src_c, tgt_t, tgt_c) -> Optional[LineageEdge]:
        """根据起止字段找到对应的 Edge 对象"""
        for e in self.graph.edges:
            if (e.source_table == src_t and e.source_column == src_c
                and e.target_table == tgt_t and e.target_column == tgt_c):
                return e
        return None

    def table_level_impact(self, table: str) -> list[str]:
        """表级影响分析：某张表的所有下游表（BFS 表级遍历）"""
        result = self.graph.bfs_downstream(table)
        # 移除自身
        result.pop(table, None)
        return list(result.keys())

🔍 四、血缘异常检测 — 断裂、环路、孤儿节点 10 min

数仓跑久了，血缘图会出现各种「病态」：

异常类型表现数仓场景严重度
血缘断裂上游表被删但下游还在引用ODS 源表下线，DWD 作业报错严重
循环依赖A→B→C→A 形成环两个 DWS 表互相引用严重
孤儿节点表没有任何上下游关系废弃的临时表未清理中等
长链依赖依赖链深度 > 阈值（如 >5）ODS→DWD→DWS→ADS 之外又套了一层中等
宽扇出一张表被 >10 张下游引用公共 DIM 表变更影响面过大低

4.1 LineageAnomalyDetector 异常检测器

from enum import Enum

class AnomalyType(Enum):
    BROKEN_LINEAGE = "血缘断裂"     # 引用了不存在的上游表
    CYCLE = "循环依赖"          # DAG 中存在环
    ORPHAN = "孤儿节点"          # 无上下游关系
    DEEP_CHAIN = "长链依赖"        # 依赖深度超阈值
    WIDE_FANOUT = "宽扇出"        # 被过多下游引用

@dataclass
class Anomaly:
    type: AnomalyType
    severity: str           # "critical" / "warning" / "info"
    description: str
    tables: list[str]

class LineageAnomalyDetector:
    """
    血缘异常检测器
    在图上跑 5 种检测规则
    """

    def __init__(
        self,
        graph: LineageGraph,
        known_tables: set[str],  # 数据库中确实存在的表
        max_depth: int = 5,
        max_fanout: int = 10
    ):
        self.graph = graph
        self.known_tables = known_tables
        self.max_depth = max_depth
        self.max_fanout = max_fanout

    def detect_all(self) -> list[Anomaly]:
        """运行全部检测规则"""
        anomalies = []
        anomalies.extend(self._detect_broken())
        anomalies.extend(self._detect_cycles())
        anomalies.extend(self._detect_orphans())
        anomalies.extend(self._detect_deep_chains())
        anomalies.extend(self._detect_wide_fanout())
        return anomalies

    def _detect_broken(self) -> list[Anomaly]:
        """检测血缘断裂：引用了不存在的上游表"""
        anomalies = []
        for edge in self.graph.edges:
            if edge.source_table not in self.known_tables:
                anomalies.append(Anomaly(
                    type=AnomalyType.BROKEN_LINEAGE,
                    severity="critical",
                    description=
                        f"{edge.target_table} 引用了不存在的上游表 "
                        f"{edge.source_table}（字段: {edge.source_column}）",
                    tables=[edge.target_table, edge.source_table]
                ))
        return anomalies

    def _detect_cycles(self) -> list[Anomaly]:
        """检测循环依赖：拓扑排序后仍有节点未处理 = 有环"""
        _, has_cycle = self.graph.topological_sort()
        if has_cycle:
            # 找出哪些节点参与环路（被拓扑排序遗漏的）
            sorted_tables, _ = self.graph.topological_sort()
            cycle_nodes = [
                t for t in self.graph.get_all_tables()
                if t not in set(sorted_tables)
            ]
            return [Anomaly(
                type=AnomalyType.CYCLE,
                severity="critical",
                description=f"检测到循环依赖，涉及表: {', '.join(cycle_nodes)}",
                tables=cycle_nodes
            )]
        return []

    def _detect_orphans(self) -> list[Anomaly]:
        """检测孤儿节点：既没有上游也没有下游的表"""
        anomalies = []
        for table in self.graph.get_all_tables():
            has_up = table in self.graph._upstream or any(
                e.target_table == table for e in self.graph.edges
            )
            has_down = table in self.graph._downstream or any(
                e.source_table == table for e in self.graph.edges
            )
            if not has_up and not has_down:
                anomalies.append(Anomaly(
                    type=AnomalyType.ORPHAN,
                    severity="warning",
                    description=f"表 {table} 是孤儿节点，无任何血缘关系",
                    tables=[table]
                ))
        return anomalies

    def _detect_deep_chains(self) -> list[Anomaly]:
        """检测长链依赖：BFS 深度超过阈值"""
        anomalies = []
        # 只从入度为 0 的表（源头）开始计算
        all_tables = self.graph.get_all_tables()
        in_degree = {t: 0 for t in all_tables}
        for src, targets in self.graph._downstream.items():
            for tgt in targets:
                if tgt in in_degree:
                    in_degree[tgt] += 1

        sources = [t for t, d in in_degree.items() if d == 0]
        for src in sources:
            depths = self.graph.bfs_downstream(src)
            for table, depth in depths.items():
                if depth > self.max_depth:
                    anomalies.append(Anomaly(
                        type=AnomalyType.DEEP_CHAIN,
                        severity="warning",
                        description=
                            f"{src} → {table} 依赖链深度为 {depth}，"
                            f"超过阈值 {self.max_depth}",
                        tables=[src, table]
                    ))
        return anomalies

    def _detect_wide_fanout(self) -> list[Anomaly]:
        """检测宽扇出：一张表被超过 N 张下游表引用"""
        anomalies = []
        for table, downstream in self.graph._downstream.items():
            if len(downstream) > self.max_fanout:
                anomalies.append(Anomaly(
                    type=AnomalyType.WIDE_FANOUT,
                    severity="info",
                    description=
                        f"{table} 被 {len(downstream)} 张表引用，"
                        f"超过阈值 {self.max_fanout}",
                    tables=[table] + list(downstream)
                ))
        return anomalies

⚠️ 常见陷阱 Top 5
  1. DFS 不设 max_depth 导致栈溢出 — 血缘图如果意外出现环（虽然理论上不该有），DFS 递归会无限展开。
    fix: 始终传 max_depth 参数，并在 visited 中检查已访问节点。
  2. LLM 编造不存在的表名 — LLM 可能「幻觉」出一个不存在的表并声称有血缘关系。
    fix: LLM 返回后必须与 known_tables 交叉验证，过滤不存在的表。
  3. 拓扑排序假设全连通 — Kahn 算法只能检测「图内」的环，如果有孤立子图需要分别处理。
    fix: 对每个连通分量分别做拓扑排序。
  4. 字段级血缘的别名问题 — SELECT a.id AS bid_id 中 id 和 bid_id 是同一字段的不同名称。
    fix: 解析器需要维护别名映射表，归一化后再建边。
  5. 影响分析只看字段级忽略表级 — SELECT * 会让所有字段都变成隐式依赖。
    fix: 遇到 SELECT * 时，自动建立该表所有字段的血缘边。

📝 Day 66 总结

  • Python 图算法：BFS（影响范围）、DFS（完整路径）、Kahn 拓扑排序（DAG 验证 + 环检测） — 血缘图上的三大遍历工具

  • 异常检测：5 种规则（断裂/环/孤儿/长链/宽扇出），复用拓扑排序做环检测
