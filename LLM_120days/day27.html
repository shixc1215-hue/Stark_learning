🔧 一、为什么需要工程化？ ⏱ 5 min

你已经能写面向对象、异步、FastAPI 了。但想想你在做 ETL 时的经验：

SQL 写完 → 在 DBeaver 跑一下 → 能出数 → 丢到调度系统，这不叫"工程化"。
真正的工程化是：日志可查、配置可改、出了 Bug 能定位、新人能接手。

Python 项目的工程化四件套：

能力工具类比数仓
日志logging 模块ETL 跑批日志表
配置pydantic-settings / .env环境变量 / 参数表
测试pytest数据校验 / 对账脚本
部署uvicorn + 进程管理调度系统发布上线
💡 数据开发的优势：你已经有"日志意识"和"配置分离"的直觉，只是需要学会 Python 生态的工具。今天就是把你 SQL/ETL 的工程经验翻译成 Python 工程化。
📝 二、logging 日志系统 ⏱ 15 min

2.1 为什么不用 print？

print 的痛点

  • 无法控制级别（全部输出）
  • 无法输出到文件
  • 没有时间戳、来源信息
  • 生产环境无法关闭调试输出

logging 的优势

  • 5 个级别灵活控制
  • 同时输出到控制台 + 文件
  • 自动带时间、文件名、行号
  • 生产环境一键调级别

2.2 日志五级别

import logging

# 日志级别从低到高，设置级别后，只输出 >= 该级别的日志
# DEBUG    → 开发调试（变量值、流程跟踪）
# INFO     → 正常运行记录（启动、完成、关键节点）
# WARNING  → 警告但不报错（用了废弃 API、配置缺失用默认值）
# ERROR    → 出错了但程序还能跑（单条数据处理失败）
# CRITICAL → 严重错误，程序可能崩溃（数据库连不上）

logging.basicConfig(level=logging.DEBUG)  # 设置最低级别

logging.debug("调试信息：变量 x = %s", x)      # 开发时用
logging.info("信息：LLM API 调用成功")           # 正常流程
logging.warning("警告：API 响应耗时 %.2f 秒", t) # 需关注
logging.error("错误：第 %d 行数据解析失败", i)   # 出了错
logging.critical("严重：数据库连接丢失！")        # 要挂了

2.3 工程级日志配置

实际项目不用 basicConfig，而是用 dictConfig 做精细控制：

# utils/logger.py — 日志配置模块（放在项目里复用）
import logging
import logging.config

LOGGING_CONFIG = {
    "version": 1,
    "disable_existing_loggers": False,

    # 格式定义
    "formatters": {
        "standard": {
            "format": "%(asctime)s | %(levelname)-8s | %(name)s | %(message)s",
            "datefmt": "%Y-%m-%d %H:%M:%S"
        },
        "detailed": {
            "format": "%(asctime)s | %(levelname)-8s | %(name)s | %(funcName)s:%(lineno)d | %(message)s",
            "datefmt": "%Y-%m-%d %H:%M:%S"
        }
    },

    # 输出目标（处理器）
    "handlers": {
        "console": {
            "class": "logging.StreamHandler",
            "formatter": "standard",
            "level": "INFO"  # 控制台只看 INFO 及以上
        },
        "file": {
            "class": "logging.handlers.RotatingFileHandler",
            "filename": "app.log",
            "maxBytes": 10 * 1024 * 1024,  # 10MB 自动轮转
            "backupCount": 5,               # 保留 5 个备份
            "formatter": "detailed",
            "encoding": "utf-8",
            "level": "DEBUG"                # 文件记录所有 DEBUG
        }
    },

    # 记录器（按模块划分）
    "loggers": {
        "": {  # 根 logger
            "handlers": ["console", "file"],
            "level": "DEBUG",
            "propagate": True
        },
        "httpx": {  # 第三方库日志太吵，调高级别
            "level": "WARNING"
        },
        "uvicorn.access": {  # FastAPI 访问日志也调高
            "level": "WARNING"
        }
    }
}

def setup_logging():
    """初始化日志系统，在程序入口调用一次"""
    logging.config.dictConfig(LOGGING_CONFIG)

# 使用方式：
# from utils.logger import setup_logging
# setup_logging()
# logger = logging.getLogger(__name__)  # __name__ = 当前模块名
# logger.info("程序启动")
核心原则：
1. 每个模块用 logger = logging.getLogger(__name__) 创建自己的 logger
2. __name__ 自动等于模块路径（如 services.llm_service），方便定位
3. 生产环境改配置文件里的 level 就行，不用改代码

2.4 在 LLM 服务中使用日志

# services/llm_service.py
import logging
import httpx
import time

logger = logging.getLogger(__name__)

class LLMService:
    """带完整日志记录的 LLM 调用服务"""

    def __init__(self, api_url: str, api_key: str):
        self.api_url = api_url
        self.api_key = api_key
        logger.info("LLMService 初始化, api_url=%s", api_url)

    async def chat(self, prompt: str, model: str = "glm-4-flash") -> str:
        """调用 LLM API，全程日志记录"""
        logger.debug("收到请求: model=%s, prompt前50字=%s", model, prompt[:50])

        start = time.time()
        try:
            async with httpx.AsyncClient(timeout=30.0) as client:
                resp = await client.post(
                    self.api_url,
                    headers={"Authorization": f"Bearer {self.api_key}"},
                    json={"model": model, "messages": [{"role": "user", "content": prompt}]}
                )

            cost = time.time() - start

            if resp.status_code == 200:
                result = resp.json()["choices"][0]["message"]["content"]
                logger.info("LLM调用成功 | 耗时%.2fs | 返回%d字", cost, len(result))
                return result
            else:
                logger.error("LLM返回错误 | status=%d | body=%s", resp.status_code, resp.text[:200])
                raise Exception(f"API 错误: {resp.status_code}")

        except httpx.TimeoutException:
            logger.error("LLM调用超时 | 耗时%.2fs", time.time() - start)
            raise
        except Exception as e:
            logger.critical("LLM调用异常: %s", e, exc_info=True)  # exc_info=True 记录完整堆栈
            raise
💡 与 ETL 日志类比：你的 ETL 日志表可能记录了 etl_run_id, step_name, status, record_count, error_msg。Python logging 的 %(name)s 就像 step_name，%(levelname)s 就像 status，%(message)s 就像 error_msg。思路完全一样，只是实现方式不同。

3.1 为什么配置要分离？

你在数仓里一定做过这些：

3.3 Pydantic Settings 配置类

# config.py — 配置管理（Day26 已学过 Pydantic，这里用 Settings 扩展）
# pip install pydantic-settings
from pydantic_settings import BaseSettings, SettingsConfigDict
from pydantic import Field

class Settings(BaseSettings):
    """应用配置——从环境变量或 .env 文件自动读取"""

    # --- LLM 配置 ---
    llm_api_url: str = "https://open.bigmodel.cn/api/paas/v4/chat/completions"
    llm_api_key: str = ""           # 必填，没有会报验证错误
    llm_model: str = "glm-4-flash"
    llm_timeout: int = Field(default=30, ge=5, le=120)  # 超时 5~120 秒
    llm_max_retries: int = Field(default=3, ge=1, le=10)

    # --- 数据库配置 ---
    db_host: str = "localhost"
    db_port: int = 5432
    db_name: str = ""
    db_user: str = ""
    db_password: str = ""

    # --- 应用配置 ---
    app_env: str = "development"    # development / staging / production
    log_level: str = "INFO"
    api_port: int = 8000

    @property
    def database_url(self) -> str:
        """计算属性：拼接数据库连接串"""
        return f"postgresql://{self.db_user}:{self.db_password}@{self.db_host}:{self.db_port}/{self.db_name}"

    @property
    def is_production(self) -> bool:
        """判断是否生产环境"""
        return self.app_env == "production"

    model_config = SettingsConfigDict(
        env_file=".env",        # 从 .env 文件读取
        env_file_encoding="utf-8",
        case_sensitive=False    # 环境变量不区分大小写
    )

# 全局单例——整个项目共享同一个配置
settings = Settings()

# 使用方式：
# from config import settings
# print(settings.llm_api_url)
# print(settings.database_url)
# print(settings.is_production)
加载优先级：环境变量 > .env 文件 > 代码默认值
这就意味着：本地开发用 .env，部署到服务器时通过环境变量覆盖，代码完全不用改。

🧪 四、单元测试（pytest） ⏱ 18 min

4.1 为什么要写测试？

你做数仓时写对账脚本、数据校验——其实那就是"测试"。pytest 把这个过程标准化了：

  • 改了代码不怕 regression：跑一遍测试就知道有没有破坏已有功能
  • 文档作用：测试用例就是"这个函数应该怎么用"的活文档
  • 重构信心：有测试兜底，敢大胆重构

4.2 第一个测试

# test_calculator.py — 文件名必须以 test_ 开头
# pip install pytest

def add(a, b):
    return a + b

def divide(a, b):
    if b == 0:
        raise ValueError("除数不能为0")
    return a / b

# ===== 测试函数：以 test_ 开头，不用 class =====
def test_add():
    assert add(1, 2) == 3           # 正常加法
    assert add(-1, 1) == 0          # 负数
    assert add(0.1, 0.2) == pytest.approx(0.3)  # 浮点数用 approx

def test_divide_normal():
    assert divide(10, 2) == 5.0
    assert divide(1, 3) == pytest.approx(0.333, rel=0.01)

def test_divide_zero():
    """测试异常：用 pytest.raises"""
    import pytest
    with pytest.raises(ValueError, match="除数不能为0"):
        divide(10, 0)

# 运行：pytest test_calculator.py -v
# -v 显示每个测试的通过/失败状态

测试 LLM 服务时，不想真的调 API（费钱、慢、不稳定）。用 unittest.mock 模拟：

💡 Mock 策略选择：
- 测试业务逻辑（数据处理、校验）→ 不需要 mock
- 测试API 调用（LLM、数据库）→ mock 掉外部依赖
- 测试异常处理 → mock 抛异常，验证你的代码能不能兜住

5.1 标准项目结构

当你写的代码从 1 个文件变成 10+ 文件，就需要组织目录了：

│

分层思维（你一定很熟悉）：
api/ = 接口层（像数仓的 ADS 层，对外暴露）
services/ = 业务层（像 DWD/DWS 层，核心逻辑）
models/ = 数据层（像 ODS 层，定义数据结构）
utils/ = 工具层（通用函数）
每层只调下一层，不能跨层调用。

6.1 完整项目代码

# ===== config.py =====
from pydantic_settings import BaseSettings, SettingsConfigDict
from pydantic import Field

class Settings(BaseSettings):

import logging
from typing import Optional

logger = logging.getLogger(__name__)

class ReportService:
    """AMC 不良资产报表分析服务（带完整日志和错误处理）"""

    # 五级分类标准（逾期天数 → 分类）
    CATEGORY_RULES = [
        (0,   90,  "正常"),    # 逾期 0-90 天
        (91,  180, "关注"),    # 逾期 91-180 天
        (181, 360, "次级"),    # 逾期 181-360 天
        (361, 720, "可疑"),    # 逾期 361-720 天
        (721, 9999,"损失"),    # 逾期 > 720 天
    ]

    def __init__(self):
        logger.info("ReportService 初始化完成")

    def validate_record(self, record: dict) -> Optional[str]:
        """校验单条记录，返回错误信息或 None"""
        required_fields = ["debtor_id", "balance", "overdue_days"]
        for field in required_fields:
            if field not in record:
                return f"缺少必填字段: {field}"

        if record["balance"] < 0:
            return f"余额为负: {record['balance']}"
        if record["overdue_days"] < 0:
            return f"逾期天数为负: {record['overdue_days']}"
        return None

    def classify(self, overdue_days: int) -> str:
        """根据逾期天数分类"""
        for low, high, category in self.CATEGORY_RULES:
            if low <= overdue_days <= high:
                return category
        return "未知"

    def analyze_batch(self, records: list[dict]) -> dict:
        """批量分析——返回统计摘要"""
        logger.info("开始批量分析，共 %d 条记录", len(records))

        valid_records = []
        error_records = []
        stats = {"正常": 0, "关注": 0, "次级": 0, "可疑": 0, "损失": 0}
        total_balance = 0.0

        for i, record in enumerate(records):
            # 1. 校验
            error = self.validate_record(record)
            if error:
                logger.warning("第%d条记录校验失败: %s", i, error)
                error_records.append({"index": i, "reason": error})
                continue

            # 2. 分类
            category = self.classify(record["overdue_days"])
            record["category"] = category
            stats[category] += 1
            total_balance += record["balance"]
            valid_records.append(record)

        logger.info("分析完成: 有效%d条, 异常%d条, 总余额%.2f",
                     len(valid_records), len(error_records), total_balance)

        # 3. 生成摘要文本（可喂给 LLM）
        summary_lines = [f"不良资产分析报告（共{len(records)}条记录）"]
        for cat, count in stats.items():
            pct = count / len(records) * 100 if records else 0
            summary_lines.append(f"  {cat}: {count}笔 ({pct:.1f}%)")
        summary_lines.append(f"  有效余额合计: ¥{total_balance:,.2f}")
        if error_records:
            summary_lines.append(f"  异常记录: {len(error_records)}条")

        summary = "\n".join(summary_lines)
        logger.debug("生成摘要:\n%s", summary)

        return {
            "total": len(records),
            "valid_count": len(valid_records),
            "error_count": len(error_records),
            "stats": stats,
            "total_balance": total_balance,
            "errors": error_records,
            "summary": summary
        }

    async def generate_llm_analysis(self, analysis_result: dict, llm_client=None) -> str:
        """用 LLM 对分析结果做深度解读"""
        import httpx

        prompt = f"""你是一位AMC（资产管理公司）的不良资产分析专家。
请根据以下数据摘要，给出专业分析意见，包括：
1. 整体风险态势评估
2. 重点关注的分类
3. 建议的处置策略

数据：
{analysis_result['summary']}

请用简洁专业的语言回答，200字以内。"""

        logger.info("开始 LLM 深度分析")

        if llm_client is None:
            # 没有传入 LLM 客户端，返回模拟结果
            logger.warning("未配置 LLM 客户端，返回模拟分析")
            return f"[模拟分析] 基于上述{analysis_result['valid_count']}条有效记录的分析..."

        # 真实 LLM 调用
        async with httpx.AsyncClient(timeout=30.0) as client:
            resp = await client.post(
                llm_client["url"],
                headers={"Authorization": f"Bearer {llm_client['key']}"},
                json={"model": "glm-4-flash", "messages": [{"role": "user", "content": prompt}]}
            )
            if resp.status_code == 200:
                result = resp.json()["choices"][0]["message"]["content"]
                logger.info("LLM 分析完成，返回 %d 字", len(result))
                return result
            else:
                logger.error("LLM 调用失败: %d", resp.status_code)
                return f"[错误] LLM 分析失败: {resp.status_code}"

题 155. 最小栈

设计一个支持 push, pop, top, getMin 的栈，getMin 时间复杂度 O(1)。

"""
思路：辅助栈 —— 两个栈同步 push/pop
- 主栈：正常存数据
- 辅助栈：每层存"当前最小值"
  push(x) → 辅助栈 push(min(x, 辅助栈栈顶))
  pop()   → 两个栈同时 pop
  getMin  → 返回辅助栈栈顶

类比数仓：就像维护一个"累积最小值"字段 alongside 原始数据
"""
class MinStack:
    def __init__(self):
        self.stack = []       # 主栈
        self.min_stack = []   # 辅助栈（最小值栈）

    def push(self, val: int) -> None:
        self.stack.append(val)
        # 辅助栈存当前最小值
        if self.min_stack:
            self.min_stack.append(min(val, self.min_stack[-1]))
        else:
            self.min_stack.append(val)

    def pop(self) -> None:
        self.stack.pop()
        self.min_stack.pop()

    def top(self) -> int:
        return self.stack[-1]

    def getMin(self) -> int:
        return self.min_stack[-1]

# 测试
ms = MinStack()
ms.push(-2); ms.push(0); ms.push(-3)
print(ms.getMin())  # -3
ms.pop()
print(ms.top())     # 0
print(ms.getMin())  # -2

# 时间复杂度：所有操作 O(1)
# 空间复杂度：O(n)，辅助栈占用的额外空间
与今天内容的关联：如果你写一个带日志的 MinStack，每次 push/pop 都记日志，这就是"日志 + 数据结构"的结合。工程化不只是 Web 项目的事——任何代码都可以受益。
📋 八、今日自测 ⏱ 5 min
1. logging 的五个级别从低到高是什么？生产环境通常设为什么级别？
DEBUG → INFO → WARNING → ERROR → CRITICAL。生产环境通常设 INFO（只记录重要信息），排查问题时临时改 DEBUG。
2. pydantic-settings 从哪些地方读取配置？优先级是什么？
环境变量 > .env 文件 > 代码默认值。这样本地用 .env 开发，部署时通过环境变量覆盖，代码不用改。
3. pytest 中 fixture 的作用是什么？conftest.py 有什么特殊之处？
fixture 是测试的"准备阶段"，用于共享测试数据或 mock 对象。conftest.py 放在 tests/ 目录下，pytest 自动识别其中的 fixture，所有测试文件都能用，不需要 import。
4. 什么场景下需要用 mock？什么场景不需要？
需要 mock：调外部 API（LLM、第三方服务）、数据库操作、文件系统——因为慢、不稳定、费钱。不需要 mock：纯业务逻辑（数据校验、分类计算、格式转换）——直接测就行。
5. .env 文件为什么必须加入 .gitignore？
.env 里存的是 API Key、数据库密码等敏感信息。一旦提交到 Git（尤其公开仓库），这些密钥就泄露了，别人可以用你的 Key 调 LLM API，产生费用。
