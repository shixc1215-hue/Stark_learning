🧠 LLM 归因分析引擎 20 min

昨天（Day 69）我们构建了规则引擎和统计异常检测。但检测出问题只是第一步——关键是要知道为什么出问题。这就是 LLM 归因分析的价值：把结构化的异常数据喂给大模型，让它用自然语言分析根因、给出修复建议。

核心设计思路：归因分析不能让 LLM 「自由发挥」，必须给它结构化的上下文——包括表元信息、历史基线、异常指标、关联规则结果。这样 LLM 的分析才靠谱，而不是瞎猜。 
# llm_analyzer.py — LLM 归因分析引擎
from openai import OpenAI
from models import InspectionReport, CheckResult, AnomalyResult


class LLMRootCauseAnalyzer:
    """
    LLM 归因分析引擎
    
    职责:
    1. 把巡检报告中的异常结果组织成结构化 Prompt
    2. 调用 LLM 生成自然语言根因分析
    3. 输出修复建议
    
    这就是你在数仓中「看到异常 → 查日志 → 排查ETL → 找到根因」
    这个思维过程的自动化版本。
    """

    def __init__(self, api_key: str, base_url: str = None, model: str = "gpt-4o-mini"):
        """初始化 LLM 客户端（兼容 OpenAI 协议的任意模型）"""
        self.client = OpenAI(api_key=api_key, base_url=base_url)
        self.model = model

    def _build_context(self, report: InspectionReport) -> str:
        """
        从巡检报告中提取异常信息，构建 LLM 上下文
        
        关键：只传异常数据，不传全部结果（减少 Token 消耗）
        同时附上表结构信息帮助 LLM 理解业务含义
        """
        context_parts = []

        # 1. 失败的规则
        failed_rules = [r for r in report.rule_results if not r.passed]
        if failed_rules:
            rules_text = "\n".join([
                f"  - [{r.severity.value}] {r.rule_name}"
                f" | 表: {r.table_name}"
                f" | {r.message}"
                for r in failed_rules
            ])
            context_parts.append(
                f"### 规则检查失败（{len(failed_rules)}项）\n{rules_text}"
            )

        # 2. 统计异常
        anomalies = [a for a in report.anomaly_results if a.is_anomaly]
        if anomalies:
            anomaly_text = "\n".join([
                f"  - {a.table_name}.{a.metric}: {a.description}"
                f" | Z-Score={a.z_score:.2f}, 方向={a.direction}"
                for a in anomalies
            ])
            context_parts.append(
                f"### 统计异常（{len(anomalies)}项）\n{anomaly_text}"
            )

        # 3. 汇总统计
        context_parts.append(
            f"### 巡检概览\n"
            f"  检查表数: {report.tables_checked}\n"
            f"  规则通过率: {report.passed_rules}/{report.total_rules}\n"
            f"  严重问题: {report.critical_count} | 警告: {report.warning_count}"
        )

        return "\n\n".join(context_parts)

    def analyze(self, report: InspectionReport) -> str:
        """
        对巡检报告进行 LLM 归因分析
        
        返回: 结构化的根因分析文本（Markdown 格式）
        """
        context = self._build_context(report)

        # 如果没有异常，直接返回
        if not context.strip().startswith("### 规则") \
           and not context.strip().startswith("### 统计"):
            return "## 归因分析结果\n\n"
                   "✅ 本次巡检未发现异常，数据质量良好。"

        system_prompt = """你是一位资深的数据开发工程师，精通数据仓库、ETL和数
据质量管理。你的任务是对数据质量巡检结果进行根因分析。

请按以下结构输出分析报告：

## 根因分析
对每个异常/失败，分析可能的原因（优先考虑ETL链路问题、
源系统变更、数据同步延迟等数据开发常见场景）。

## 影响评估
评估异常对下游表/报表/业务的影响范围和严重程度。

## 修复建议
给出具体可执行的修复步骤（包括SQL修复语句示例）。

## 预防措施
建议新增哪些质量规则或监控指标来防止类似问题。

注意：
- 分析要具体，不要泛泛而谈
- 修复建议要可直接执行，给出具体SQL
- 考虑数据仓库分层（ODS→DWD→DWS→ADS）的传导影响"""

        # 调用 LLM
        response = self.client.chat.completions.create(
            model=self.model,
            messages=[
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": f"请分析以下数据质量巡检结果：\n\n{context}"},
            ],
            temperature=0.3,  # 低温度：分析报告需要稳定可靠
            max_tokens=2000,
        )

        analysis = response.choices[0].message.content

        # 写入报告对象
        report.llm_analysis = analysis
        return analysis
Temperature = 0.3 的含义：生成任务（文案/对话）用高温度（0.7-1.0）增加创意；分析任务（归因/诊断）用低温度（0.1-0.3）保证稳定。这和你在数仓里「ETL跑批不能有随机性」是一个道理。

# reporter.py — 报告生成器
from datetime import datetime
from models import InspectionReport, Severity


class QualityReportGenerator:
    """数据质量报告生成器"""

    def generate_markdown(self, report: InspectionReport) -> str:
        """
        生成 Markdown 格式的巡检报告
        适合：飞书文档、钉钉消息、邮件正文
        """
        lines = []
        lines.append(f"# 数据质量巡检报告")
        lines.append(f"**巡检ID:** {report.inspection_id}")
        lines.append(
            f"**时间:** {report.started_at.strftime('%Y-%m-%d %H:%M')} "
            f"→ {report.finished_at.strftime('%H:%M') if report.finished_at else 'N/A'}"
        )

        # 汇总
        status_emoji = "✅" if report.failed_rules == 0 else "❌"
        lines.append(f"\n## 汇总 {status_emoji}")
        lines.append(f"- 检查表: {report.tables_checked} 张")
        lines.append(
            f"- 规则: {report.passed_rules}/{report.total_rules} 通过"
        )
        lines.append(f"- 严重: {report.critical_count} | 警告: {report.warning_count}")
        lines.append(f"- 统计异常: {len(report.anomaly_results)} 项")

        # 失败规则详情
        failed = [r for r in report.rule_results if not r.passed]
        if failed:
            lines.append(f"\n## 失败规则（{len(failed)}项）")
            for r in failed:
                icon = "🔴" if r.severity == Severity.CRITICAL else "🟡"
                lines.append(f"\n{icon} **{r.rule_name}**")
                lines.append(f"  - 表: `{r.table_name}`")
                lines.append(f"  - 级别: {r.severity.value}")
                lines.append(f"  - 详情: {r.message}")
                if r.detail_sql:
                    lines.append(
                        f"  - 诊断SQL:\n```sql\n{r.detail_sql}\n```"
                    )

        # 统计异常
        anomalies = [a for a in report.anomaly_results if a.is_anomaly]
        if anomalies:
            lines.append(f"\n## 统计异常（{len(anomalies)}项）")
            for a in anomalies:
                dir_icon = "📈" if a.direction == "spike" else "📉"
                lines.append(
                    f"- {dir_icon} `{a.table_name}` {a.metric}: "
                    f"{a.description}"
                )

        # LLM 分析
        if report.llm_analysis:
            lines.append(f"\n## LLM 根因分析\n")
            lines.append(report.llm_analysis)

        return "\n".join(lines)

    def generate_html(self, report: InspectionReport) -> str:
        """
        生成 HTML 格式的可视化报告
        适合：浏览器查看、邮件附件、归档存储
        """
        # 计算通过率
        pass_rate = (
            report.passed_rules / report.total_rules * 100
            if report.total_rules > 0 else 100
        )

        # 颜色映射
        color = "#16a34a" if report.failed_rules == 0 else "#dc2626"

        html = f"""

巡检报告 {report.inspection_id}


数据质量巡检报告


ID: {report.inspection_id} | 
   {report.started_at.strftime('%Y-%m-%d %H:%M')}




  

    
{report.tables_checked}

    
检查表

  

  

    
{pass_rate:.0f}%

    
通过率

  

  

    
{report.critical_count}

    
严重

  

  

    
{len([a for a in report.anomaly_results if a.is_anomaly])}

    
统计异常

  



"""

        # 失败规则列表
        failed = [r for r in report.rule_results if not r.passed]
        if failed:
            html += '
异常规则
'
            for r in failed:
                cls = "critical" if r.severity == Severity.CRITICAL \
                    else "warning"
                html += (
                    f'
'
                    f''
                    f'{r.severity.value}'
                    f''
                    f'{r.rule_name}'
                    f''
                    f'{r.table_name}'
                    f'
'
                )
            html += '
'

        # LLM 分析
        if report.llm_analysis:
            # 简单的 Markdown → HTML 转换
            analysis_html = report.llm_analysis
            analysis_html = analysis_html.replace("## ", "
")
            analysis_html = analysis_html.replace("\n\n", "
")
            html += (
                f'
LLM 根因分析
'
                f'{analysis_html}
'
            )

        html += ''
        return html

    def save_report(self, report: InspectionReport, output_dir: str = "."):
        """保存 Markdown 和 HTML 两种格式"""
        # Markdown
        md_content = self.generate_markdown(report)
        md_path = f"{output_dir}/report_{report.inspection_id}.md"
        with open(md_path, "w", encoding="utf-8") as f:
            f.write(md_content)

        # HTML
        html_content = self.generate_html(report)
        html_path = f"{output_dir}/report_{report.inspection_id}.html"
        with open(html_path, "w", encoding="utf-8") as f:
            f.write(html_content)

        return md_path, html_path
双格式输出思路：Markdown 给机器人/人看（飞书 Webhook、钉钉机器人直接发），HTML 给浏览器/邮件看。这种「一次生成、多格式分发」的模式和你在数仓中「一张 ADS 表供给多个下游报表」是同一思路。
🔧 告警推送（飞书 Webhook 示例） 10 min

巡检系统最终要「主动通知」人，而不是等人来看。下面展示如何把报告推送到飞书群。 

# alert.py — 告警推送（飞书 Webhook）
import requests
import json
from models import InspectionReport, Severity


class FeishuAlerter:
    """
    飞书群机器人告警推送
    
    通过 Webhook 把巡检结果推送到飞书群。
    数据开发最熟悉的「报表异常@相关人」的自动化版。
    """

    def __init__(self, webhook_url: str):
        self.url = webhook_url

    def send_alert(self, report: InspectionReport):
        """发送巡检告警到飞书群"""
        # 只在有异常时才推送（避免信息轰炸）
        if report.failed_rules == 0 and \
           all(not a.is_anomaly for a in report.anomaly_results):
            print("巡检全部通过，不发送告警")
            return

        # 构建飞书卡片消息
        failed = [r for r in report.rule_results if not r.passed]
        anomalies = [a for a in report.anomaly_results if a.is_anomaly]

        # 构建异常列表文本
        content_lines = []
        for r in failed:
            icon = "🔴" if r.severity == Severity.CRITICAL else "🟡"
            content_lines.append(
                f"{icon} {r.rule_name}（{r.table_name}）"
            )
        for a in anomalies:
            icon = "📈" if a.direction == "spike" else "📉"
            content_lines.append(
                f"{icon} {a.table_name}.{a.metric}: {a.description[:50]}"
            )

        card = {
            "msg_type": "interactive",
            "card": {
                "header": {
                    "title": {"tag": "plain_text",
                            "content": f"数据质量巡检告警"},
                    "template": "red" if report.critical_count > 0 \
                               else "orange",
                },
                "elements": [
                    {
                        "tag": "div",
                        "text": {
                            "tag": "lark_md",
                            "content": (
                                f"**严重:** {report.critical_count} | "
                                f"**警告:** {report.warning_count} | "
                                f"**统计异常:** {len(anomalies)}\n\n"
                                + "\n".join(content_lines)
                            ),
                        },
                    },
                    {
                        "tag": "note",
                        "elements": [
                            {"tag": "plain_text",
                             "content": f"巡检ID: {report.inspection_id}"}
                        ],
                    },
                ],
            },
        }

        # 发送请求
        resp = requests.post(
            self.url,
            headers={"Content-Type": "application/json"},
            data=json.dumps(card, ensure_ascii=False).encode("utf-8"),
        )
        result = resp.json()
        if result.get("code", -1) == 0:
            print("飞书告警推送成功")
        else:
            print(f"推送失败: {result}")

# main.py — 完整版主入口（串联全部组件）
import os
import json
from detectors.inspector import DataQualityInspector
# 以下两个导入需要对应的 API Key/配置
from llm_analyzer import LLMRootCauseAnalyzer
from reporter import QualityReportGenerator
from alert import FeishuAlerter


def run_full_inspection():
    """
    完整的数据质量巡检流程
    
    流程：规则检查 → 统计检测 → LLM归因 → 生成报告 → 推送告警
    
    这就是你每天早上到公司做的第一件事的自动化版本：
    查同步日志 → 跑质量SQL → 看结果 → 分析原因 → 写日报 → 通知相关人
    """
    # 1. 初始化数据库连接
    import pymysql
    db = pymysql.connect(
        host="localhost", port=3306,
        user="root", password=os.getenv("DB_PASSWORD", ""),
        database="ybt_dw", charset="utf8mb4",
        cursorclass=pymysql.cursors.DictCursor,
    )

    try:
        # 2. 执行巡检（规则 + 统计）
        print("='\n📋 Step 1: 执行巡检检测...")
        inspector = DataQualityInspector(db, "rules_config.yaml")
        report = inspector.run_inspection(
            tables=["dwd_bid_detail", "ods_bid_info", "dws_npa_summary"]
        )
        print(f"\n{report.summary()}")

        # 3. LLM 归因分析（仅在有异常时）
        if report.failed_rules > 0 or any(
            a.is_anomaly for a in report.anomaly_results
        ):
            print("\n🧠 Step 2: LLM 归因分析...")
            analyzer = LLMRootCauseAnalyzer(
                api_key=os.getenv("LLM_API_KEY"),
                base_url=os.getenv("LLM_BASE_URL"),
                model="gpt-4o-mini",
            )
            analysis = analyzer.analyze(report)
            print(analysis[:200] + "...")

        # 4. 生成报告
        print("\n📄 Step 3: 生成报告...")
        gen = QualityReportGenerator()
        md_path, html_path = gen.save_report(report, "./reports")
        print(f"  Markdown: {md_path}")
        print(f"  HTML: {html_path}")

        # 5. 推送告警
        webhook = os.getenv("FEISHU_WEBHOOK_URL")
        if webhook:
            print("\n🔔 Step 4: 推送飞书告警...")
            alerter = FeishuAlerter(webhook)
            alerter.send_alert(report)

        # 6. 输出 JSON（供后续审计/展示）
        output_json = report.model_dump_json(indent=2)
        with open("./reports/latest_inspection.json", "w",
                  encoding="utf-8") as f:
            f.write(output_json)

        print(f"\n\n{'='*50}")
        print("巡检完成！报告已保存到 ./reports/")

    finally:
        db.close()


if __name__ == "__main__":
    run_full_inspection()
