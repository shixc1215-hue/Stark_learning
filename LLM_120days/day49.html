1. argparse — 命令行参数解析

当你的程序需要不同参数来控制行为（比如选微调 / 部署 / 压测模式），argparse 是标准库中最专业的方案。

# src/cli.py — AMC LLM 系统的命令行入口
import argparse

def build_parser() -> argparse.ArgumentParser:
    """构建 CLI 参数解析器 — 支持 pipeline 的五个子命令"""
    parser = argparse.ArgumentParser(
        prog="amc-llm",
        description="AMC 金融智能 LLM 系统 — 全链路管理工具",
        formatter_class=argparse.RawDescriptionHelpFormatter,  # 保留格式
        epilog="""示例:
  amc-llm prepare   --input data/raw/ --output data/processed/
  amc-llm finetune  --config configs/train_config.yaml
  amc-llm export    --model ./output/lora_amc --format gguf
  amc-llm serve     --host 0.0.0.0 --port 8000
  amc-llm monitor   --mode load-test --concurrency 10
"""
    )

    # ====== 子命令系统 ======
    subparsers = parser.add_subparsers(dest="command", help="可用子命令")

    # --- prepare: 数据准备 ---
    prep = subparsers.add_parser("prepare", help="数据准备与清洗")
    prep.add_argument("--input", required=True, help="原始数据目录")
    prep.add_argument("--output", required=True, help="输出 JSONL 路径")
    prep.add_argument("--split", type=float, default=0.1,
                      help="验证集比例 (默认 0.1)")
    prep.add_argument("--clean", action="store_true", help="启用数据清洗")

    # --- finetune: 模型微调 ---
    ft = subparsers.add_parser("finetune", help="LoRA/QLoRA 微调训练")
    ft.add_argument("--config", default="configs/train_config.yaml",
                    help="训练配置文件路径")
    ft.add_argument("--data", default="data/processed/",
                    help="训练数据目录")
    ft.add_argument("--output", default="./output/lora_amc",
                    help="模型输出目录")
    ft.add_argument("--quantize", choices=["4bit", "8bit", "none"],
                    default="4bit", help="QLoRA 量化级别")
    ft.add_argument("--dry-run", action="store_true",
                    help="只打印训练参数，不执行")  # action="store_true" 是最常用的参数类型

    # --- export: 模型导出 ---
    exp = subparsers.add_parser("export", help="导出与量化模型")
    exp.add_argument("--model", required=True, help="LoRA 模型路径")
    exp.add_argument("--format", choices=["gguf", "merged"],
                      default="merged", help="导出格式")
    exp.add_argument("--quant-level", choices=["q4_k_m", "q5_k_m", "q8_0"],
                      default="q4_k_m")

    # --- serve: 启动服务 ---
    srv = subparsers.add_parser("serve", help="启动 API 服务")
    srv.add_argument("--host", default="0.0.0.0")
    srv.add_argument("--port", type=int, default=8000)
    srv.add_argument("--workers", type=int, default=1,
                      help="uvicorn worker 数量")
    srv.add_argument("--gpu-memory", type=int, default=4096,
                      help="vLLM GPU 显存分配 (MB)")
    srv.add_argument("--reload", action="store_true", help="开发模式热重载")

    # --- monitor: 监控与压测 ---
    mon = subparsers.add_parser("monitor", help="监控与压力测试")
    mon.add_argument("--mode", choices=["probe", "load-test", "report"],
                      default="probe")
    mon.add_argument("--concurrency", type=int, default=5)
    mon.add_argument("--duration", type=int, default=60,
                      help="压测持续秒数")
    mon.add_argument("--threshold", type=float, default=2.0,
                      help="TTFT 告警阈值 (秒)")

    return parser


# 使用示例
if __name__ == "__main__":
    parser = build_parser()
    args = parser.parse_args()

    # args.command 就是子命令名字
    if args.command == "finetune":
        print(f"开始微调: config={args.config}, data={args.data}")
        print(f"量化: {args.quantize}, dry_run={args.dry_run}")
    elif args.command == "serve":
        print(f"启动服务: {args.host}:{args.port}")
    else:
        parser.print_help()  # 没有子命令时打印帮助
💡 argparse 核心要点
• add_subparsers(dest="command") — 创建 git-style 子命令体系
• choices=[...] — 限制可选值，比手动 if-else 更安全
• action="store_true" — 布尔开关，不加就是 False，加了就是 True
• type=int / type=float — 自动类型转换 + 校验
• default=... — 默认值，让命令更简洁

2. 结构化 JSON 日志 — logging 的生产级用法

生产环境的日志要能被 ELK / Grafana Loki 等系统解析。JSON 格式日志是行业标准。

import logging
import json
from datetime import datetime, timezone
from pathlib import Path


class JSONFormatter(logging.Formatter):
    """把 log record 转成 JSON — 方志 ELK/Loki 解析"""

    def format(self, record: logging.LogRecord) -> str:
        # 提取核心字段
        log_entry = {
            "timestamp": datetime.now(timezone.utc).isoformat(),
            "level": record.levelname,
            "module": record.module,
            "message": record.getMessage(),
        }

        # 附加上下文信息（如果有的话）
        if hasattr(record, "extra"):
            log_entry["context"] = record.extra

        # 异常信息
        if record.exc_info and record.exc_info[0] is not None:
            log_entry["exception"] = self.formatException(record.exc_info)

        return json.dumps(log_entry, ensure_ascii=False)


def setup_logging(log_file: str = "logs/amc-llm.jsonl",
                 level: str = "INFO") -> logging.Logger:
    """初始化结构化日志 — 同时输出到文件和控制台"""
    logger = logging.getLogger("amc-llm")
    logger.setLevel(getattr(logging, level.upper()))

    # 日志文件路径自动创建父目录
    log_path = Path(log_file)
    log_path.parent.mkdir(parents=True, exist_ok=True)

    # ① JSON 文件 Handler — 给 ELK/Loki 解析
    file_handler = logging.FileHandler(log_file, encoding="utf-8")
    file_handler.setFormatter(JSONFormatter())
    logger.addHandler(file_handler)

    # ② 控制台 Handler — 给人看，保持可读性
    console_handler = logging.StreamHandler()
    console_handler.setFormatter(
        logging.Formatter("%(asctime)s [%(levelname)s] %(message)s")
    )
    logger.addHandler(console_handler)

    return logger


# ====== 使用示例 ======
logger = setup_logging()

# 基本日志
logger.info("数据准备开始")

# 带上下文的日志（使用 extra 参数）
try:
    result = process_data("amc_reports.csv")
    logger.info("数据处理完成", extra={
        "extra": {"file": "amc_reports.csv", "rows": 1523, "duration_s": 2.3}

except Exception as e:
    logger.error("数据处理失败", exc_info=True)

3. __main__.py — 让包可以 python -m 运行

在 src/ 目录下放一个 __main__.py，就能用 python -m src 直接运行整个包。这是 Python 项目的标准入口模式。

# src/__main__.py — 包入口
"""
AMC 金融智能 LLM 系统
用法:
    python -m src finetune --config configs/train_config.yaml
    python -m src serve --port 8000
    python -m src monitor --mode load-test
"""
from .cli import build_parser
from .logging_setup import setup_logging


def main():
    parser = build_parser()
    args = parser.parse_args()

    # 初始化日志
    logger = setup_logging()
    logger.info(f"AMC-LLM 系统启动, 命令={args.command}")

    if not args.command:
        parser.print_help()
        return

    # 路由到各模块
    commands = {
        "prepare": "src.data_prep.main",
        "finetune": "src.finetune.main",
        "export": "src.export.main",
        "serve": "src.serve.main",
        "monitor": "src.monitor.main",
    }

    if args.command in commands:
        # 动态导入对应模块（避免全部加载）
        module_path = commands[args.command]
        parts = module_path.split(".")
        mod = __import__(module_path, fromlist=["main"])
        mod.main(args)  # 把 args 传给子模块
    else:
        parser.print_help()


if __name__ == "__main__":
    main()

模块 ① 数据准备 — 复盘 D43-44

# src/data_prep/main.py — 数据准备子命令入口
import json
from pathlib import Path
from dataclasses import dataclass, field


@dataclass
class PrepConfig:
    """数据准备配置 — 对应 prepare 子命令参数"""
    input_dir: str
    output_dir: str
    val_split: float = 0.1
    enable_clean: bool = True
    min_length: int = 10
    max_length: int = 2048


def clean_sample(text: str, config: PrepConfig) -> str | None:
    """单条数据清洗 — 过长/过短/空行过滤"""
    text = text.strip()
    if len(text) < config.min_length:
        return None
    if len(text) > config.max_length:
        return text[:config.max_length]  # 截断而非丢弃
    return text


def load_and_clean(input_dir: str, config: PrepConfig) -> list[dict]:
    """加载原始数据 → 清洗 → 返回有效列表"""
    raw_path = Path(input_dir)
    samples = []

    for file in raw_path.glob("*.jsonl"):
        with open(file, "r", encoding="utf-8") as f:
            for line in f:
                try:
                    data = json.loads(line.strip())
                    # 清洗 instruction 和 output
                    if config.enable_clean:
                        cleaned = clean_sample(data.get("instruction", ""), config)
                        if cleaned is None:
                            continue
                        data["instruction"] = cleaned
                    samples.append(data)
                except json.JSONDecodeError:
                    continue  # 跳过格式错误行

    return samples


def split_dataset(samples: list[dict], val_ratio: float) -> tuple:
    """随机拆分训练/验证集 — 保证可复现"""
    import random
    random.seed(42)  # 固定随机种子
    shuffled = samples.copy()
    random.shuffle(shuffled)

    split_idx = int(len(shuffled) * (1 - val_ratio))
    return shuffled[:split_idx], shuffled[split_idx:]


def save_jsonl(data: list[dict], path: str):
    """保存为 JSONL 格式"""
    Path(path).parent.mkdir(parents=True, exist_ok=True)
    with open(path, "w", encoding="utf-8") as f:
        for item in data:
            f.write(json.dumps(item, ensure_ascii=False) + "\n")


def main(args):
    """数据准备主流程"""
    config = PrepConfig(
        input_dir=args.input,
        output_dir=args.output,
        val_split=args.split,
        enable_clean=args.clean,
    )

    # ① 加载 + 清洗
    samples = load_and_clean(args.input, config)
    print(f"[数据准备] 加载 {len(samples)} 条有效数据")

    # ② 拆分
    train, val = split_dataset(samples, config.val_split)
    print(f"[数据准备] 训练集 {len(train)} / 验证集 {len(val)}")

    # ③ 保存
    save_jsonl(train, str(Path(args.output) / "train.jsonl"))
    save_jsonl(val, str(Path(args.output) / "val.jsonl"))
    print(f"[数据准备] 已保存到 {args.output}/")

模块 ②③④⑤ 聚合 — 全链路一键脚本

# scripts/run_pipeline.py — 全链路一键执行器
"""
AMC LLM 全链路 Pipeline
从数据准备 → 微调 → 导出 → 部署 → 压测 的完整自动化
"""
import subprocess
import sys
import time
from pathlib import Path
from dataclasses import dataclass, field
from enum import Enum


class StageStatus(Enum):
    PENDING = "待执行"
    RUNNING = "执行中"
    SUCCESS = "✅ 成功"
    FAILED  = "❌ 失败"
    SKIPPED = "⏭️ 跳过"


@dataclass
class PipelineStage:
    """Pipeline 中的一个阶段"""
    name: str
    command: str
    status: StageStatus = StageStatus.PENDING
    duration: float = 0.0
    depends_on: list[str] = field(default_factory=list)  # 前置依赖


class Pipeline:
    """全链路 Pipeline 编排器"""

    def __init__(self):
        self.stages: dict[str, PipelineStage] = {}
        self.results: list[dict] = []

    def add_stage(self, name: str, command: str,
                 depends_on: list[str] | None = None) -> "Pipeline":
        """添加阶段（链式调用）"""
        self.stages[name] = PipelineStage(
            name=name,
            command=command,
            depends_on=depends_on or [],
        )
        return self  # 支持链式 .add().add().add()

    def run(self, target: str | None = None):
        """执行 pipeline — target 指定执行到哪个阶段（含）"""
        stage_order = list(self.stages.keys())
        if target and target in stage_order:
            idx = stage_order.index(target)
            stage_order = stage_order[:idx + 1]

        for stage_name in stage_order:
            stage = self.stages[stage_name]

            # 检查依赖是否全部成功
            for dep in stage.depends_on:
                dep_stage = self.stages[dep]
                if dep_stage.status != StageStatus.SUCCESS:
                    stage.status = StageStatus.SKIPPED
                    print(f"  ⏭️ [{stage_name}] 跳过（依赖 {dep} 未成功）")
                    continue

            # 执行
            stage.status = StageStatus.RUNNING
            print(f"\n{'='*50}")
            print(f"  🚀 执行: {stage_name}")
            print(f"{'='*50}")

            start = time.perf_counter()
            try:
                result = subprocess.run(
                    stage.command, shell=True,
                    capture_output=True, text=True,
                    timeout=3600,  # 1小时超时
                )
                stage.duration = time.perf_counter() - start

                if result.returncode == 0:
                    stage.status = StageStatus.SUCCESS
                else:
                    stage.status = StageStatus.FAILED
                    print(f"  ❌ 错误: {result.stderr[:500]}")
            except subprocess.TimeoutExpired:
                stage.status = StageStatus.FAILED
                print(f"  ❌ 超时 (>1h)")
            except Exception as e:
                stage.status = StageStatus.FAILED
                print(f"  ❌ 异常: {e}")

            # 记录结果
            self.results.append({
                "stage": stage_name,
                "status": stage.status.value,
                "duration": round(stage.duration, 2),
            })

        self._print_summary()

    def _print_summary(self):
        """打印 Pipeline 执行摘要"""
        print(f"\n{'='*50}")
        print("  📊 Pipeline 执行摘要")
        print(f"{'='*50}")
        total_time = 0
        for r in self.results:
            icon = "✅" if "成功" in r["status"] else "❌"
            print(f"  {icon} {r['stage']:<20} {r['status']:<10} {r['duration']:.1f}s")
            total_time += r["duration"]
        print(f"  ─────────────────────────────────────")
        print(f"  总耗时: {total_time:.1f}s")


# ====== 构建 AMC Pipeline ======
def build_amc_pipeline() -> Pipeline:
    """构建 AMC 全链路 Pipeline"""
    p = Pipeline()
    p.add_stage("数据准备",
        "python -m src prepare --input data/raw/ --output data/processed/ --clean")
    p.add_stage("LoRA 微调",
        "python -m src finetune --config configs/train_config.yaml --quantize 4bit",
        depends_on=["数据准备"])
    p.add_stage("模型导出",
        "python -m src export --model ./output/lora_amc --format gguf --quant-level q4_k_m",
        depends_on=["LoRA 微调"])
    p.add_stage("服务部署",
        "python -m src serve --host 0.0.0.0 --port 8000",
        depends_on=["模型导出"])
    p.add_stage("压力测试",
        "python -m src monitor --mode load-test --concurrency 10 --duration 60",
        depends_on=["服务部署"])
    return p


if __name__ == "__main__":
    target = sys.argv[1] if len(sys.argv) > 1 else None
    pipeline = build_amc_pipeline()
    pipeline.run(target=target)
    # 用法:
    #   python scripts/run_pipeline.py              — 全部执行
    #   python scripts/run_pipeline.py "LoRA 微调" — 只执行到微调
💡 Pipeline 设计要点
• depends_on 实现依赖检查——上游失败自动跳过下游（类似 Airflow 的任务依赖）
• target 参数支持只执行到某个阶段——调试时不用每次跑全量
• subprocess.run(timeout=3600) 防止挂死
• 数仓类比：Pipeline = ETL 调度，Stage = Task，depends_on = 任务依赖 DAG
⚙️ 配置驱动：YAML + pydantic-settings 全局管理 ~10 min

from pydantic import BaseModel, Field
from typing import Literal
import yaml


class ModelConfig(BaseModel):
    base_model: str = Field(description="基座模型 HuggingFace ID")
    output_dir: str = Field(default="./output")


class LoRAConfig(BaseModel):
    r: int = Field(default=16, ge=1, le=256, description="LoRA 秩")
    alpha: int = Field(default=32, ge=1)
    dropout: float = Field(default=0.05, ge=0, le=1)
    target_modules: list[str] = Field(
        default_factory=lambda: ["q_proj", "v_proj"]
    )


class TrainingConfig(BaseModel):
    num_train_epochs: int = Field(default=3, ge=1)
    per_device_train_batch_size: int = Field(default=4, ge=1)
    gradient_accumulation_steps: int = Field(default=4, ge=1)
    learning_rate: float = Field(default=2e-4, gt=0)
    lr_scheduler_type: Literal["cosine", "linear", "constant"] = "cosine"
    warmup_ratio: float = Field(default=0.1, ge=0, le=1)
    max_seq_length: int = Field(default=2048, le=32768)

    # 计算属性：等效 batch size
    @property
    def effective_batch_size(self) -> int:
        return self.per_device_train_batch_size * self.gradient_accumulation_steps


class TrainPipelineConfig(BaseModel):
    """顶层配置 — 聚合所有子配置"""
    model: ModelConfig
    lora: LoRAConfig = Field(default_factory=LoRAConfig)
    quantization: dict = Field(default_factory=dict)
    training: TrainingConfig = Field(default_factory=TrainingConfig)
    evaluation: dict = Field(default_factory=dict)

    @classmethod
    def from_yaml(cls, path: str) -> "TrainPipelineConfig":
        """从 YAML 文件加载配置"""
        with open(path, "r", encoding="utf-8") as f:
            data = yaml.safe_load(f)
        return cls.model_validate(data)

def format_chatml(example: dict) -> dict:
    """
    将 JSONL 样本转为 ChatML 格式（Qwen 等主流模型的对话模板）
    输入: {"instruction": "...", "output": "..."}
    输出: {"text": "<|im_start|>system\n...\n<|im_start|>user\n...\n<|im_start|>assistant\n..."}
    """
    system_msg = "你是AMC不良资产管理领域的专家助手，擅长金融术语解释、风险分析和报表生成。"
    text = (
        f"<|im_start|>system\n{system_msg}<|im_end|>\n"
        f"<|im_start|>user\n{example['instruction']}<|im_end|>\n"
        f"<|im_start|>assistant\n{example['output']}<|im_end|>"
    )
    return {"text": text}


def build_bnb_config(bits: int = 4) -> BitsAndBytesConfig:
    """构建量化配置（QLoRA 核心）"""
    if bits == 4:
        return BitsAndBytesConfig(
            load_in_4bit=True,
            bnb_4bit_compute_dtype=torch.float16,
            bnb_4bit_quant_type="nf4",           # 正态分布量化（Day 44 重点）
            bnb_4bit_use_double_quant=True,      # 双重量化（省显存）
        )
    elif bits == 8:
        return BitsAndBytesConfig(load_in_8bit=True)
    else:
        return None


def finetune(config: "TrainPipelineConfig"):
    """
    QLoRA 微调主流程 — 整合 Day 43-44 全部知识点
    """
    # ① 加载量化配置
    bnb_config = build_bnb_config(bits=4)

    # ② 加载 Tokenizer + 模型
    tokenizer = AutoTokenizer.from_pretrained(config.model.base_model)
    tokenizer.pad_token = tokenizer.eos_token  # 设置 pad token

    model = AutoModelForCausalLM.from_pretrained(
        config.model.base_model,
        quantization_config=bnb_config,
        device_map="auto",  # 自动分配 GPU/CPU
        torch_dtype=torch.float16,
    )

    # ③ 准备 k-bit 训练（添加可训练的 input embedding）
    model = prepare_model_for_kbit_training(model)

    # ④ 配置 LoRA
    lora_config = LoraConfig(
        r=config.lora.r,
        lora_alpha=config.lora.alpha,
        lora_dropout=config.lora.dropout,
        target_modules=config.lora.target_modules,
        task_type=TaskType.CAUSAL_LM,  # 因果语言模型
        bias="none",
    )

    # ⑤ 获取 PEFT 模型（只训练 LoRA 参数）
    model = get_peft_model(model, lora_config)
    model.print_trainable_parameters()
    # 输出示例: trainable params: 4.2M || all params: 7.0B || trainable%: 0.06

    # ⑥ 加载数据集
    dataset = load_dataset("json", data_files={
        "train": "data/processed/train.jsonl",
        "test": "data/processed/val.jsonl",
    })
    dataset = dataset.map(format_chatml)

    # ⑦ 训练参数
    training_args = SFTConfig(
        output_dir=config.model.output_dir,
        num_train_epochs=config.training.num_train_epochs,
        per_device_train_batch_size=config.training.per_device_train_batch_size,
        gradient_accumulation_steps=config.training.gradient_accumulation_steps,
        learning_rate=config.training.learning_rate,
        lr_scheduler_type=config.training.lr_scheduler_type,
        warmup_ratio=config.training.warmup_ratio,
        max_seq_length=config.training.max_seq_length,
        logging_steps=config.training.logging_steps,
        save_strategy="epoch",
        eval_strategy="epoch",
        bf16=True,               # BF16 混合精度
        gradient_checkpointing=True, # 省显存
        optim="paged_adamw_8bit",  # 分页优化器（Day 44 重点）
        report_to="none",           # 不上传 wandb
    )

    # ⑧ 启动训练
    trainer = SFTTrainer(
        model=model,
        args=training_args,
        train_dataset=dataset["train"],
        eval_dataset=dataset["test"],
        processing_class=tokenizer,
    )

    print("🚀 开始 QLoRA 微调训练...")
    trainer.train()

    # ⑨ 保存 LoRA 权重
    trainer.save_model(config.model.output_dir)
    tokenizer.save_pretrained(config.model.output_dir)
    print(f"✅ 模型已保存到 {config.model.output_dir}")

    return trainer

# src/export/merger.py — LoRA 权重合并
from transformers import AutoModelForCausalLM, AutoTokenizer
from peft import PeftModel


def merge_lora(base_model_id: str, lora_path: str, output_path: str):
    """
    合并 LoRA 权重到基座模型 → 生成完整模型
    Day 43 核心操作: merge_and_unload()
    """
    # 加载基座
    base = AutoModelForCausalLM.from_pretrained(base_model_id, torch_dtype="auto")
    # 加载 LoRA 适配器
    model = PeftModel.from_pretrained(base, lora_path)
    # 合并 + 卸载 PEFT 包装器
    merged = model.merge_and_unload()

    # 保存
    merged.save_pretrained(output_path)
    AutoTokenizer.from_pretrained(lora_path).save_pretrained(output_path)
    print(f"✅ 合并完成 → {output_path}")
    print(f"   合并后可直接用 vLLM / Ollama 部署")
📌 merge_and_unload() 为什么重要？
LoRA 微调只保存了增量权重（ΔW = B × A），推理时需要基座 + LoRA 两步计算。merge_and_unload() 把 ΔW 合并进原模型，生成一个独立完整模型，部署时不再需要 PEFT 库，推理更快更简单。

from openai import OpenAI
from pydantic import BaseModel, Field
from typing import Literal
import time


class ChatRequest(BaseModel):
    """AMC 业务请求体"""
    message: str = Field(min_length=1, max_length=2000)
    mode: Literal["chat", "report", "analyze"] = "chat"
    temperature: float = Field(default=0.7, ge=0, le=2)
    max_tokens: int = Field(default=512, le=4096)


class ChatResponse(BaseModel):
    """统一响应格式"""
    reply: str
    mode: str
    ttft: float   # 首字延迟（秒）
    total_time: float


class LLMClient:
    """vLLM OpenAI 兼容客户端 — Day 46 核心"""

    def __init__(self, base_url: str = "http://localhost:8000"):
        self.client = OpenAI(
            base_url=base_url + "/v1",
            api_key="not-needed",  # vLLM 不需要真实 key
        )
        self.model_name = None  # 启动后自动获取

    def chat(self, request: ChatRequest) -> ChatResponse:
        """发送聊天请求 + 测量 TTFT"""
        start = time.perf_counter()
        first_token_time = None

        # 构建系统提示词
        system_prompts = {
            "chat": "你是AMC不良资产管理领域的专家助手。",
            "report": "你负责生成AMC不良资产分析报告，需包含数据摘要、风险评级和建议。",
            "analyze": "你对AMC数据进行深度分析，找出异常模式和风险信号。",
        }

        # 流式请求以精确测量 TTFT
        stream = self.client.chat.completions.create(
            model=self.model_name or "default",
            messages=[
                {"role": "system", "content": system_prompts[request.mode]},
                {"role": "user", "content": request.message},
            ],
            temperature=request.temperature,
            max_tokens=request.max_tokens,
            stream=True,  # 流式获取 TTFT
        )

        reply_parts = []
        for chunk in stream:
            if chunk.choices[0].delta.content:
                if first_token_time is None:
                    first_token_time = time.perf_counter()  # 记录首字时间
                reply_parts.append(chunk.choices[0].delta.content)

        total_time = time.perf_counter() - start
        return ChatResponse(
            reply="".join(reply_parts),
            mode=request.mode,
            ttft=first_token_time - start if first_token_time else total_time,
            total_time=total_time,
        )

# src/monitor/load_test.py — 压测 AMC API（Day 48 知识整合）
import asyncio
import aiohttp
import time
import json
from dataclasses import dataclass, field


@dataclass
class LoadTestResult:
    """单次请求的压测记录"""
    request_id: int
    status_code: int
    ttft: float          # 首字延迟
    total_time: float     # 总耗时
    success: bool
    error: str | None = None


@dataclass
class LoadTestReport:
    """压测报告"""
    concurrency: int
    total_requests: int
    duration: float
    success_rate: float
    avg_ttft: float
    p50_ttft: float
    p99_ttft: float
    avg_total: float
    throughput: float     # req/s
    results: list[LoadTestResult] = field(default_factory=list)


async def send_request(
    session: aiohttp.ClientSession,
    url: str, req_id: int,
    semaphore: asyncio.Semaphore,
    prompts: list[str],
) -> LoadTestResult:
    """发送单次压测请求（带并发限制）"""
    async with semaphore:  # Semaphore 限流 — Day 24 异步知识
        payload = {
            "message": prompts[req_id % len(prompts)],
            "mode": "chat",
            "max_tokens": 128,  # 压测用短 token 加速
        }
        start = time.perf_counter()
        try:
            async with session.post(url, json=payload, timeout=aiohttp.ClientTimeout(total=30)) as resp:
                data = await resp.json()
                total = time.perf_counter() - start
                return LoadTestResult(
                    request_id=req_id,
                    status_code=resp.status,
                    ttft=data.get("ttft", total),
                    total_time=total,
                    success=resp.status == 200,
                )
        except Exception as e:
            return LoadTestResult(
                request_id=req_id, status_code=0,
                ttft=0, total_time=time.perf_counter()-start,
                success=False, error=str(e),
            )


async def run_load_test(
    url: str = "http://localhost:8000/api/chat",
    concurrency: int = 10,
    total_requests: int = 50,
) -> LoadTestReport:
    """执行压力测试"""
    prompts = [
        "解释一下不良资产包的概念",
        "AMC的风险评级分为哪几级？",
        "什么是NPL？如何计算不良贷款率？",
        "催收流程中的合规要求有哪些？",
    ]

    semaphore = asyncio.Semaphore(concurrency)
    results = []

    async with aiohttp.ClientSession() as session:
        start = time.perf_counter()
        tasks = [
            send_request(session, url, i, semaphore, prompts)
            for i in range(total_requests)
        ]
        results = await asyncio.gather(*tasks)
        duration = time.perf_counter() - start

    # 统计分析
    successes = [r for r in results if r.success]
    ttfts = sorted([r.ttft for r in successes])
    n = len(ttfts)

    return LoadTestReport(
        concurrency=concurrency,
        total_requests=total_requests,
        duration=duration,
        success_rate=len(successes) / total_requests * 100,
        avg_ttft=sum(ttfts) / n if n else 0,
        p50_ttft=ttfts[n//2] if n else 0,
        p99_ttft=ttfts[int(n*0.99)] if n > 1 else 0,
        avg_total=sum(r.total_time for r in successes) / n if n else 0,
        throughput=total_requests / duration if duration > 0 else 0,
        results=list(results),
    )

# LeetCode #239 — 滑动窗口最大值
# 题目：给定数组 nums 和窗口大小 k，返回每个窗口的最大值
from collections import deque
from typing import List


def maxSlidingWindow(nums: List[int], k: int) -> List[int]:
    """
    单调递减队列解法 — O(n)
    核心思想：deque 中维护「候选最大值」的下标
    """
    result = []
    dq = deque()  # 存储 nums 的下标，值单调递减

    for i, num in enumerate(nums):
        # ① 移除 deque 中比当前元素小的（它们不可能成为最大值了）
        while dq and nums[dq[-1]] < num:
            dq.pop()

        # ② 加入当前下标
        dq.append(i)

        # ③ 移除超出窗口范围的下标
        if dq[0] <= i - k:
            dq.popleft()

        # ④ 窗口形成后，记录最大值（deque 头部就是最大值）
        if i >= k - 1:
            result.append(nums[dq[0]])

    return result


# 测试
print(maxSlidingWindow([1,3,-1,-3,5,3,6,7], 3))
# 输出: [3, 3, 5, 5, 6, 7]

# 走一遍过程（k=3）：
# i=0, nums[0]=1 → dq=[0]
# i=1, nums[1]=3 → 弹出0, dq=[1]
# i=2, nums[2]=-1 → dq=[1,2] → 窗口形成 → result=[3]
# i=3, nums[3]=-3 → dq=[1,2,3] → 1不在窗口内, popleft → dq=[2,3] → result=[3,3]
# ...
📌 单调队列核心思想：deque 维护一个单调递减的序列。
• 新来一个大数 → 把比它小的全弹出（它们没机会当最大值了）
• 新来一个小数 → 直接追加（它可能是未来窗口的最大值）
• 窗口滑动 → 过期的下标从左边弹出
• deque 头部永远是当前窗口的最大值 → O(1) 获取
