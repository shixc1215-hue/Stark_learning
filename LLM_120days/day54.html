2.1 enum.Enum — 用枚举管理 Agent 状态

Agent 有多种状态（思考/执行/等待审批/出错），用枚举比字符串更安全：

from enum import Enum, auto

class AgentState(Enum):
    """Agent 生命周期状态 — 比字符串 'thinking' 更安全（拼写错误会直接报错）"""
    IDLE = auto()           # 空闲，等待任务
    THINKING = auto()       # LLM 推理中
    EXECUTING = auto()      # 调用工具中
    WAITING_APPROVAL = auto() # 等待人工审批
    ERROR = auto()          # 出错，等待恢复
    FINISHED = auto()       # 任务完成

# 合法状态转移表 — 防止非法跳转（如 ERROR 直接跳 FINISHED）
TRANSITIONS = {
    AgentState.IDLE:           {AgentState.THINKING},
    AgentState.THINKING:       {AgentState.EXECUTING, AgentState.WAITING_APPROVAL, AgentState.FINISHED},
    AgentState.EXECUTING:      {AgentState.THINKING, AgentState.ERROR, AgentState.FINISHED},
    AgentState.WAITING_APPROVAL: {AgentState.EXECUTING, AgentState.ERROR},
    AgentState.ERROR:         {AgentState.THINKING, AgentState.FINISHED},  # 可重试或终止
}

def transition(current: AgentState, target: AgentState) -> bool:
    """检查状态转移是否合法"""
    if target not in TRANSITIONS.get(current, set()):
        raise ValueError(f"非法转移: {current.name} → {target.name}")
    return True

# 使用
state = AgentState.IDLE
transition(state, AgentState.THINKING)   # ✅ OK
# transition(state, AgentState.ERROR)  # ❌ ValueError!
数仓类比：这就像 ETL 任务的状态机 — PENDING → RUNNING → SUCCESS/FAILED → RETRY，用枚举比在数据库里存字符串 'runnning'（拼错了！不会报错）安全得多。

2.3 异常分类策略 — 不同错误不同处理

class AgentError(Exception):
    """所有 Agent 错误的基类"""
    def __init__(self, message: str, recoverable: bool = True):
        super().__init__(message)
        self.recoverable = recoverable  # 是否可自动恢复

class APIError(AgentError):
    """LLM API 调用失败（超时/限流/500）"""
    pass

class ToolError(AgentError):
    """工具执行失败（SQL语法错/连接断开）"""
    def __init__(self, message: str, tool_name: str = ""):
        super().__init__(message)
        self.tool_name = tool_name

class ContextOverflowError(AgentError):
    """上下文超长 — 必须裁剪，无法恢复"""
    def __init__(self, message: str, current_tokens: int, max_tokens: int):
        super().__init__(message, recoverable=False)
        self.current_tokens = current_tokens
        self.max_tokens = max_tokens

# 使用 — 不同异常不同处理路径
try:
    result = call_llm(prompt)
except ContextOverflowError as e:
    # 不可恢复 → 裁剪后终止
    print(f"上下文溢出: {e.current_tokens}/{e.max_tokens}")
except APIError as e:
    if e.recoverable:
        retry_with_backoff()  # 可恢复 → 重试
    else:
        fallback_to_cached() # 不可恢复 → 降级
关键设计：给异常加 recoverable 字段，让上层统一判断"能不能自动恢复"，而不是根据异常类型硬编码 if-else。这和数仓里用 error_code + error_level 分级处理是一个思路。

3.1 指数退避重试 — Agent 的"三次机会"

LLM API 经常超时或限流（429），不能一错就放弃。指数退避 = 每次等待时间翻倍，给服务端"喘息"机会：

import asyncio, random, time
from dataclasses import dataclass, field

@dataclass
class RetryConfig:
    """重试配置 — 和数仓调度任务的重试策略一模一样"""
    max_retries: int = 3
    base_delay: float = 1.0     # 首次等待 1 秒
    max_delay: float = 30.0     # 最大等待 30 秒
    jitter: bool = True         # 加随机抖动，避免多Agent同时重试"雷群效应"
    retryable_errors: tuple = (Exception,)  # 可重试的异常类型

@dataclass
class RetryResult:
    """重试结果记录 — 用于分析和监控"""
    success: bool
    attempts: int
    total_delay: float
    last_error: str | None = None

def retry_with_backoff(func, config: RetryConfig = RetryConfig()) -> RetryResult:
    """
    同步版指数退避重试
    核心公式: delay = min(base * 2^attempt + random_jitter, max_delay)
    """
    last_error = None
    total_delay = 0.0

    for attempt in range(config.max_retries + 1):  # 0,1,2,3 共4次（含首次）
        try:
            result = func()
            return RetryResult(
                success=True,
                attempts=attempt + 1,
                total_delay=total_delay,
            )
        except config.retryable_errors as e:
            last_error = str(e)
            if attempt < config.max_retries:
                # 计算退避时间: 1s → 2s → 4s → 8s ...
                delay = min(
                    config.base_delay * (2 ** attempt),
                    config.max_delay
                )
                # 加抖动: 避免多Agent同时重试（和ETL多任务错峰调度同理）
                if config.jitter:
                    delay *= random.uniform(0.5, 1.5)
                total_delay += delay
                time.sleep(delay)
                print(f"  ⚠️ 第{attempt+1}次失败，{delay:.1f}秒后重试: {last_error}")

    return RetryResult(
        success=False,
        attempts=config.max_retries + 1,
        total_delay=total_delay,
        last_error=last_error,
    )

# ---- 使用示例 ----
def unstable_api_call():
    """模拟一个不稳定的 API（70% 成功率）"""
    if random.random() < 0.3:
        raise ConnectionError("API 超时")
    return "查询结果: 不良资产余额 3.2亿"

result = retry_with_backoff(unstable_api_call, RetryConfig(max_retries=3))
print(f"成功:{result.success}, 尝试:{result.attempts}次, 总耗时:{result.total_delay:.1f}s")

3.2 降级策略 — 主路径不通时走"备胎"

from typing import Protocol

class FallbackChain:
    """
    降级链: 主方案 → 备选1 → 备选2 → 兜底
    类比: 数仓里 T+1 批处理失败 → 自动切到实时流 → 最后用 T-1 快照兜底
    """
    def __init__(self):
        self.strategies: list[tuple[str, callable]] = []  # (名称, 函数)

    def add(self, name: str, func: callable, priority: int = 0):
        """添加降级策略（priority 越小越优先）"""
        self.strategies.append((name, func, priority))
        self.strategies.sort(key=lambda x: x[2])

    def execute(self, *args, **kwargs):
        """按优先级依次尝试，直到成功"""
        errors = []
        for name, func, _ in self.strategies:
            try:
                result = func(*args, **kwargs)
                if len(errors) > 0:
                    print(f"  ✅ 降级成功: {name} (跳过: {[e[0] for e in errors]})")
                return result
            except Exception as e:
                errors.append((name, str(e)))
                print(f"  ❌ {name} 失败: {e}")

        raise AgentError(f"所有降级策略均失败: {errors}")

# ---- AMC 场景：LLM 生成降级链 ----
chain = FallbackChain()
chain.add("智谱GLM-4", lambda q: call_zhipu(q), priority=0)     # 首选
chain.add("DeepSeek", lambda q: call_deepseek(q), priority=1)  # 备选
chain.add("规则模板", lambda q: rule_based_answer(q), priority=2)  # 兜底

answer = chain.execute("本月不良资产Top5机构")  # 自动降级，用户无感

3.3 死循环防护 — max_steps + 检测重复工具调用

from collections import Counter

class LoopDetector:
    """
    检测 Agent 是否陷入循环 — 两种检测方式：
    1. 同一工具连续调用N次 → 可能参数有误，卡住了
    2. 总步数超限 → 强制终止
    """
    def __init__(self, max_steps: int = 15, max_same_tool: int = 3):
        self.max_steps = max_steps
        self.max_same_tool = max_same_tool
        self.steps: list[str] = []         # 所有工具调用记录
        self.consecutive: Counter = Counter()  # 连续调用计数

    def record(self, tool_name: str) -> bool:
        """记录一次工具调用，返回 True 表示检测到循环"""
        self.steps.append(tool_name)

        # 检测总步数
        if len(self.steps) > self.max_steps:
            return True  # 超过最大步数

        # 检测连续重复调用
        if self.steps[-1:] == self.steps[-2:-1] if len(self.steps) >= 2 else False:
            self.consecutive[tool_name] += 1
            if self.consecutive[tool_name] >= self.max_same_tool:
                return True  # 同一工具连续3次 → 疑似循环
        else:
            self.consecutive.clear()  # 工具变了，重置计数

        return False

# ---- 使用 ----
detector = LoopDetector(max_steps=10, max_same_tool=3)

for tool in ["db_query", "think", "db_query", "think", "db_query", "think", "db_query"]:
    is_loop = detector.record(tool)
    print(f"调用 {tool:12} → 循环检测: {'⚠️ 是!' if is_loop else '正常'}  (已执行{len(detector.steps)}步)")
数仓经验：死循环在 ETL 里也有 — 比如自依赖的 SQL 任务互相等待。生产环境必须有超时机制 + 死锁检测，Agent 系统同理。

4.1 审批策略 — 哪些操作需要人确认？

from enum import Enum, auto
from dataclasses import dataclass, field

class ApprovalPolicy(Enum):
    """审批策略 — 不同操作不同安全级别"""
    ALWAYS = auto()     # 每次都要审批（如：发送外部通知、删除数据）
    NEVER = auto()      # 从不审批（如：只读查询）
    AUTO = auto()       # 自动判断（基于金额/影响范围/置信度）

@dataclass
class ApprovalRule:
    """单条审批规则 — 定义何时需要审批"""
    tool_name: str
    policy: ApprovalPolicy
    threshold: float | None = None  # AUTO 模式的阈值
    description: str = ""

# AMC 业务审批规则 — 参考银行分级授权制度
AMC_APPROVAL_RULES = [
    ApprovalRule("db_query",       ApprovalPolicy.NEVER,  description="只读查询，无需审批"),
    ApprovalRule("knowledge_search", ApprovalPolicy.NEVER,  description="知识库检索，无需审批"),
    ApprovalRule("generate_report", ApprovalPolicy.AUTO,
                 threshold=0.8,  description="报告生成：置信度>0.8自动执行，否则需审批"),
    ApprovalRule("send_notification", ApprovalPolicy.ALWAYS, description="发送通知：必须审批（涉及外部系统）"),
    ApprovalRule("risk_alert",      ApprovalPolicy.ALWAYS, description="风险预警：必须审批（高影响操作）"),
]

4.2 HumanApproval 审批引擎

from datetime import datetime

@dataclass
class ApprovalRequest:
    """审批请求 — Agent 暂停，等待人工处理"""
    tool_name: str
    action_summary: str       # "即将发送预警通知给5个机构"
    params: dict                 # 工具参数
    confidence: float = 0.0
    created_at: str = field(default_factory=lambda: datetime.now().isoformat())
    status: str = "pending"  # pending / approved / rejected

class HumanApproval:
    """
    审批引擎 — Agent 遇到高风险操作时暂停等待
    类比: 银行的"大额转账需主管授权"
    """
    def __init__(self, rules: list[ApprovalRule]):
        self.rules = {r.tool_name: r for r in rules}
        self.history: list[ApprovalRequest] = []  # 审批历史

    def check(self, tool_name: str, params: dict, confidence: float = 0.0) -> bool:
        """
        检查工具调用是否需要人工审批
        返回 True = 可以自动执行, False = 需要暂停等待审批
        """
        rule = self.rules.get(tool_name)

        # 没有配置规则 → 默认允许
        if rule is None:
            return True

        if rule.policy == ApprovalPolicy.NEVER:
            return True  # 免审批

        elif rule.policy == ApprovalPolicy.ALWAYS:
            return self._request_approval(tool_name, params, confidence)

        elif rule.policy == ApprovalPolicy.AUTO:
            if confidence >= rule.threshold:
                return True  # 置信度足够高，自动通过
            return self._request_approval(tool_name, params, confidence)

        return True

    def _request_approval(self, tool_name: str, params: dict, confidence: float) -> bool:
        """发起审批请求 — 实际生产中会发到飞书/企微审批流"""
        req = ApprovalRequest(
            tool_name=tool_name,
            action_summary=self._summarize(tool_name, params),
            params=params,
            confidence=confidence,
        )
        self.history.append(req)

        # 这里模拟人工审批（生产中通过飞书/企微 API 发起）
        print(f"\n🔔 审批请求:")
        print(f"   工具: {tool_name}")
        print(f"   操作: {req.action_summary}")
        print(f"   置信度: {confidence:.0%}")

        # 模拟审批通过（生产中会等待回调）
        req.status = "approved"
        print(f"   ✅ 审批通过\n")
        return True

    def _summarize(self, tool_name: str, params: dict) -> str:
        """生成人可读的操作摘要"""
        summaries = {
            "send_notification": lambda p: f"向 {p.get('target','未知')} 发送通知: {p.get('message','')}",
            "risk_alert": lambda p: f"触发风险预警: {p.get('alert_type','')} (影响 {p.get('institutions',0)} 家机构)",
        }
        fn = summaries.get(tool_name, lambda p: f"执行 {tool_name}")
        return fn(params)

# ---- 使用 ----
approval = HumanApproval(AMC_APPROVAL_RULES)

# 只读查询 → 自动通过
approval.check("db_query", {"sql": "SELECT * FROM npl_assets"})

# 发送通知 → 必须审批
approval.check("send_notification", {
    "target": "风控部",
    "message": "3家机构不良资产超阈值"
})

# 报告生成 → 置信度 0.6 < 0.8 → 需要审批
approval.check("generate_report", {"type": "monthly"}, confidence=0.6)

from dataclasses import dataclass, field
from typing import Protocol

# ---- 定义 Agent 协议（接口） ----
class ExpertAgent(Protocol):
    """专家 Agent 协议 — 每个 Agent 必须有 name 和 can_handle"""
    @property
    def name(self) -> str: ...
    def can_handle(self, query: str) -> float: ...  # 返回匹配度 0~1
    def execute(self, query: str, context: dict) -> str: ...

# ---- 四个 AMC 专家 Agent ----
@dataclass
class DataQueryAgent:
    """数据查询专家 — 专注SQL查询和数据检索"""
    def __post_init__(self):
        self.keywords = ["查询", "查找", "统计", "多少", "top", "排名",
                       "sql", "不良", "余额", "机构", "资产"]

    @property
    def name(self) -> str: return "DataQueryAgent"

    def can_handle(self, query: str) -> float:
        score = sum(1 for kw in self.keywords if kw in query.lower())
        return min(score / 3, 1.0)

    def execute(self, query: str, context: dict) -> str:
        return f"[数据查询] 执行SQL: SELECT ... FROM npl_assets WHERE ... → 返回查询结果"

@dataclass
class ReportAgent:
    """报告生成专家 — 专注报表和分析报告"""
    keywords = ["报告", "报表", "导出", "excel", "pdf",
                "分析", "月报", "周报"]

    @property
    def name(self) -> str: return "ReportAgent"

    def can_handle(self, query: str) -> float:
        score = sum(1 for kw in self.keywords if kw in query.lower())
        return min(score / 3, 1.0)

    def execute(self, query: str, context: dict) -> str:

@dataclass
class NotifyAgent:
    """通知发送专家 — 专注消息推送"""

                "飞书", "告警", "预警"]

# ---- 路由器 ----
@dataclass
class AgentRouter:
    """
    多 Agent 路由器 — 根据用户意图分发到最合适的专家
    核心逻辑: 每个Agent对query打分 → 取最高分 → 分发执行
    """
    agents: list = field(default_factory=list)

    def register(self, agent):
        """注册专家 Agent"""
        self.agents.append(agent)

    def route(self, query: str, context: dict = None) -> list[tuple]:
        """
        路由分发 — 返回排序后的 (agent, score) 列表
        注意: 一个query可能匹配多个Agent（如"导出并发送"→ Report + Notify）
        """
        context = context or {}
        scored = [(agent, agent.can_handle(query)) for agent in self.agents]
        # 过滤掉完全无关的 (score = 0)
        relevant = [(a, s) for a, s in scored if s > 0]
        # 按分数降序
        relevant.sort(key=lambda x: x[1], reverse=True)
        return relevant

    def dispatch(self, query: str, context: dict = None) -> list[str]:
        """路由 + 执行 — 返回所有Agent的执行结果"""
        results = []
        for agent, score in self.route(query, context):
            print(f"  🎯 分发到 {agent.name:20} (匹配度: {score:.0%})")
            result = agent.execute(query, context or {})
            results.append(result)
        return results

from collections import deque
from dataclasses import dataclass, field
from datetime import datetime

@dataclass
class MemoryEntry:
    """单条记忆记录"""
    role: str           # "user" / "assistant" / "system"
    content: str        # 记忆内容
    timestamp: str = field(default_factory=lambda: datetime.now().isoformat())
    is_key: bool = False  # 是否为关键记忆（长期保存）

class AgentMemory:
    """
    Agent 双层记忆系统
    类比: 短期记忆 = 临时表(会话级), 长期记忆 = 持久表(跨会话)
    """
    def __init__(self, short_term_size: int = 20):
        self.short_term: deque[MemoryEntry] = deque(maxlen=short_term_size)
        self.long_term: list[MemoryEntry] = []   # 无大小限制，可持久化到文件/数据库

    def add(self, role: str, content: str, is_key: bool = False):
        """添加记忆 — 同时写入短期；is_key=True 同时写入长期"""
        entry = MemoryEntry(role=role, content=content, is_key=is_key)
        self.short_term.append(entry)
        if is_key:
            self.long_term.append(entry)

    def search_long(self, keyword: str, limit: int = 5) -> list[MemoryEntry]:
        """在长期记忆中按关键词搜索"""
        matched = [e for e in self.long_term if keyword in e.content]
        return matched[:limit]

    def get_context(self, max_tokens: int = 2000) -> list[str]:
        """
        组装上下文 — 供 LLM 使用的对话历史
        策略: 长期关键记忆 + 短期最近对话，按token预算裁剪
        """
        context_parts = []

        # 1. 注入长期记忆中的关键信息（简洁摘要）
        if self.long_term:
            key_memos = [f"[记忆] {e.content}" for e in self.long_term[-5:]]
            context_parts.extend(key_memos)

        # 2. 添加短期记忆（最近的对话）
        for entry in self.short_term:
            context_parts.append(f"{entry.role}: {entry.content}")

        # 3. 简单token估算和裁剪（中文约1字≈2token）
        total_chars = sum(len(p) for p in context_parts)
        estimated_tokens = total_chars * 2

        if estimated_tokens > max_tokens:
            # 从最早的开始裁剪，保留最近的
            while len(context_parts) > 2 and sum(len(p)*2 for p in context_parts) > max_tokens:
                context_parts.pop(0)

        return context_parts

    def clear_short(self):
        """清空短期记忆（新会话开始时）"""
        self.short_term.clear()

    def save_long(self, filepath: str):
        """持久化长期记忆到文件"""
        import json
        data = [{"role": e.role, "content": e.content, "timestamp": e.timestamp}
                for e in self.long_term]
        with open(filepath, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)

    def stats(self) -> dict:
        return {
            "short_term_count": len(self.short_term),
            "long_term_count": len(self.long_term),
            "total_context_chars": sum(len(e.content) for e in self.short_term),
        }

# ---- 使用 ----
memory = AgentMemory(short_term_size=20)

# 模拟多轮对话
memory.add("user", "查询本月不良资产余额")
memory.add("assistant", "本月不良资产余额 3.2亿元", is_key=True)  # 关键信息，长期保存
memory.add("user", "按机构分组看")
memory.add("assistant", "A机构1.2亿, B机构0.8亿, C机构0.6亿", is_key=True)

# 搜索长期记忆
print("搜索'余额':", [e.content for e in memory.search_long("余额")])

# 组装上下文
print("\n上下文内容:")
for line in memory.get_context():
    print(f"  {line}")

print("\n记忆统计:", memory.stats())
数仓类比：短期记忆 = Hive 临时表（会话结束自动删除），长期记忆 = Hive 持久表（跨会话保存）。get_context 的裁剪逻辑就像 SQL 的 LIMIT + WHERE，在预算内取最相关的内容。
