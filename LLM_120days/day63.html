2.1 uuid — 唯一会话 ID 生成

每个对话需要一个唯一 ID 来追踪，uuid4() 生成随机 UUID，比自增 ID 更安全（不泄露用户量）。

import uuid

# uuid4 = 完全随机，适合会话ID、请求追踪ID
session_id = uuid.uuid4()
print(session_id)           # 'a8098c1a-f86e-11da-bd1a-00112444be1e'
print(str(session_id))       # 字符串形式
print(session_id.hex[:8])  # 取前8位做短ID: 'a8098c1a'

# uuid4 vs uuid1 vs uuid5 对比
# uuid4: 随机，最常用（会话ID/追踪ID）
# uuid1: 基于时间戳+MAC地址（可逆，不推荐用于安全场景）
# uuid5: 基于命名空间+名称（确定性，适合生成稳定ID）
# 数仓类比: uuid4 就像雪花算法的分布式ID

2.2 copy.deepcopy — 对话状态的安全复制

对话状态是嵌套字典/列表，浅拷贝 .copy() 只复制第一层，修改内层会污染原始数据。deepcopy 递归复制所有层级。

import copy

# 浅拷贝的陷阱
state = {"slots": {"asset_id": None, "date": None}}
shallow = state.copy()
shallow["slots"]["asset_id"] = "A001"
print(state["slots"]["asset_id"])  # 'A001' — 被污染了！

# deepcopy 安全复制
original = {"slots": {"asset_id": None, "date": None}}
snapshot = copy.deepcopy(original)
snapshot["slots"]["asset_id"] = "A001"
print(original["slots"]["asset_id"])  # None — 原始数据安全
# 数仓类比: deepcopy 就像 INSERT INTO ... SELECT * FROM（完全独立副本）

2.3 match-case 进阶 — 意图路由

Python 3.10 的 match-case 比 if-elif 更清晰地表达"根据值走不同分支"。

from enum import Enum

class Intent(Enum):
    QUERY_BALANCE = "query_balance"
    QUERY_RISK    = "query_risk"
    GREETING      = "greeting"
    UNKNOWN       = "unknown"

# match-case 意图路由
def route_intent(intent: Intent, slots: dict) -> str:
    match intent:
        case Intent.QUERY_BALANCE:
            if slots.get("asset_id"):
                return f"查询 {slots['asset_id']} 余额"
            return "请问要查询哪笔资产的余额？"
        case Intent.QUERY_RISK:
            return "正在查询风险等级..."
        case Intent.GREETING:
            return "您好！请问有什么可以帮您？"
        case _:  # 兜底，相当于 default
            return "抱歉，我没有理解您的意思"

print(route_intent(Intent.QUERY_BALANCE, {"asset_id": "A001"}))
# 输出: 查询 A001 余额
match-case vs if-elif 选择指南：当分支条件是"某个值等于什么"时，用 match-case 更清晰；当分支条件是复杂的布尔表达式时，用 if-elif 更灵活。两者可以混用——外层 match 分大类，case 内部 if 处理细节。
🔄 对话状态管理（ConversationState） 15 min

3.1 状态模型设计

对话状态需要记录三件事：① 对话元信息（ID/创建时间/用户）；② 当前意图与阶段；③ 已填充的槽位信息。

from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
from typing import Optional
import uuid

class DialogPhase(Enum):
    """对话阶段：类比数仓 ETL 的不同 stage"""
    SLOT_FILLING  = "slot_filling"   # 收集信息中（ETL 抽取阶段）
    EXECUTING     = "executing"      # 执行任务中（ETL 转换阶段）
    RESULT_SHOWING = "result_showing"  # 展示结果中（ETL 加载阶段）
    CLOSED        = "closed"          # 对话已结束

@dataclass
class ConversationState:
    """对话状态 — 每个用户会话一份"""
    session_id: str = field(default_factory=lambda: str(uuid.uuid4()))
    user_id: str = "anonymous"
    intent: Optional[str] = None        # 当前意图
    phase: DialogPhase = DialogPhase.SLOT_FILLING
    slots: dict = field(default_factory=dict)  # 已填充的槽位
    history: list = field(default_factory=list)  # 对话历史
    created_at: str = field(
        default_factory=lambda: datetime.now().isoformat()
    )
    turn_count: int = 0                  # 对话轮次计数

    def add_turn(self, user_msg: str, bot_msg: str):
        """记录一轮对话到历史"""
        self.history.append({
            "turn": self.turn_count,
            "user": user_msg,
            "bot": bot_msg,
            "timestamp": datetime.now().isoformat()
        })
        self.turn_count += 1

    def is_slot_filled(self, slot_name: str) -> bool:
        """检查某个槽位是否已填"""
        val = self.slots.get(slot_name)
        return val is not None and val != ""

    def get_missing_slots(self, required: list) -> list:
        """返回尚未填充的必需槽位列表"""
        return [s for s in required if not self.is_slot_filled(s)]

    def to_dict(self) -> dict:
        """序列化为字典（用于持久化）"""
        return {
            "session_id": self.session_id,
            "user_id": self.user_id,
            "intent": self.intent,
            "phase": self.phase.value,
            "slots": self.slots,
            "turn_count": self.turn_count,
        }

3.2 SessionManager — 会话存储管理

生产环境用 Redis 存会话，开发阶段用内存字典即可。

import time
from typing import Optional

class SessionManager:
    """会话存储 — 内存版（生产环境替换为 Redis）"""

    def __init__(self, ttl: int = 1800):
        """ttl: 会话超时秒数，默认30分钟"""
        self._sessions: dict[str, ConversationState] = {}
        self._last_access: dict[str, float] = {}
        self.ttl = ttl

    def get_or_create(self, session_id: str) -> ConversationState:
        """获取已有会话，不存在则创建新会话"""
        if session_id in self._sessions:
            self._last_access[session_id] = time.time()
            return self._sessions[session_id]
        # 创建新会话
        state = ConversationState(session_id=session_id)
        self._sessions[session_id] = state
        self._last_access[session_id] = time.time()
        return state

    def cleanup_expired(self) -> int:
        """清理过期会话，返回清理数量"""
        now = time.time()
        expired = [
            sid for sid, last in self._last_access.items()
            if now - last > self.ttl
        ]
        for sid in expired:
            del self._sessions[sid]
            del self._last_access[sid]
        return len(expired)

    def close(self, session_id: str):
        """关闭指定会话"""
        state = self._sessions.get(session_id)
        if state:
            state.phase = DialogPhase.CLOSED
🧩 槽位填充（Slot Filling） 15 min

4.1 核心原理

槽位填充是"缺少必要信息时主动追问"的机制。就像去银行办业务：柜员发现缺身份证时会先问你"请出示身份证"，而不是直接报错。

数仓类比：槽位填充 = 数仓的 NOT NULL 约束检查。执行任务前，先检查所有必填字段（槽位）是否齐全，缺了就抛异常（追问用户），全了才执行 INSERT（处理任务）。
要素说明AMC 示例
意图→槽位映射每个意图需要哪些必填信息查询余额 → 需要 asset_id
槽位提取从用户输入中提取信息填入槽位"A001余额多少" → asset_id="A001"
追问策略缺哪个槽位就追问哪个"请问要查询哪笔资产？"
完成判定所有必填槽位已填充触发实际查询

4.2 SlotF 实现

from dataclasses import dataclass

@dataclass
class SlotDefinition:
    """单个槽位的定义"""
    name: str           # 槽位名称，如 "asset_id"
    display_name: str   # 追问时显示的名称，如 "资产编号"
    required: bool = True # 是否必填
    prompt: str = ""    # 追问话术模板，如 "请问要查询哪笔{display_name}？"

class SlotFillingManager:
    """
    槽位填充管理器
    职责：检查槽位完整性 → 追问缺失项 → 确认是否可执行
    """

    # 意图 → 需要的槽位定义列表
    INTENT_SLOTS: dict[str, list[SlotDefinition]] = {
        "query_balance": [
            SlotDefinition("asset_id", "资产编号", prompt="请问要查询哪笔资产的余额？请提供资产编号。"),
        ],
        "query_risk": [
            SlotDefinition("debtor_name", "债务人名称", prompt="请问要查询哪位债务人的风险等级？"),
        ],
        "query_detail": [
            SlotDefinition("asset_id", "资产编号"),
            SlotDefinition("report_date", "报告日期", required=False),
        ],
        # greeting / farewell 等不需要槽位
    }

    def get_required_slots(self, intent: str) -> list[SlotDefinition]:
        """获取某意图的必填槽位列表"""
        definitions = self.INTENT_SLOTS.get(intent, [])
        return [s for s in definitions if s.required]

    def check_and_prompt(
        self, intent: str, state: ConversationState
    ) -> Optional[str]:
        """
        检查槽位完整性。
        返回 None 表示全部齐全，返回 str 表示追问话术。
        """
        required = self.get_required_slots(intent)
        if not required:
            return None  # 该意图不需要槽位

        # 找出第一个缺失的必填槽位
        for slot_def in required:
            if not state.is_slot_filled(slot_def.name):
                return slot_def.prompt

        return None  # 全部齐全

    def fill_from_entities(
        self, state: ConversationState, entities: dict
    ) -> int:
        """从实体提取结果填充槽位，返回新填充数量"""
        filled = 0
        for key, value in entities.items():
            if value and key not in state.slots:
                state.slots[key] = value
                filled += 1
        return filled
追问策略要点：每次只追问第一个缺失槽位，不要一次性列出所有缺失项（用户体验差）。追问话术要口语化且具体，不要说"请补充信息"，而要说"请问要查询哪笔资产的余额？"
💬 回复生成（模板 + LLM 混合策略） 20 min

5.1 两层回复策略

策略适用场景优势劣势
模板回复固定意图（问候/告别/常见FAQ）速度快、成本低、100%可控不灵活，覆盖范围有限
LLM 回复开放问答、复杂分析、个性化回复灵活、语义理解强延迟高、成本高、可能幻觉
核心原则：能走模板的绝不走 LLM。模板处理 80% 的常见场景，LLM 处理 20% 的复杂场景。这就像数仓里：简单聚合用物化视图（模板），复杂分析才跑实时 SQL（LLM）。

5.2 TemplateResponder — 模板回复器

from string import Template

class TemplateResponder:
    """模板回复器 — 速度快、零成本、100%可控"""

    # 意图 → 模板列表（按优先级排序）
    TEMPLATES: dict[str, list[str]] = {
        "greeting": [
            "您好！我是AMC智能助手，请问有什么可以帮您？",
            "您好！可以帮您查询不良资产余额、风险等级等信息。",
        ],
        "farewell": [
            "再见！如有其他问题，随时找我。",
            "好的，祝您工作顺利！",
        ],
        "thanks": [
            "不客气！还有其他需要帮助的吗？",
        ],
        "query_balance": [
            "资产 $asset_id 的当前余额为 ¥$amount 元（截至 $date）。",
        ],
        "query_risk": [
            "债务人 $debtor_name 的当前风险等级为 $risk_level。",
        ],
    }

    def can_handle(self, intent: str) -> bool:
        """判断该意图是否可以用模板处理"""
        return intent in self.TEMPLATES

    def respond(
        self, intent: str, slots: dict = None, index: int = 0
    ) -> Optional[str]:
        """
        用模板生成回复。
        slots: 槽位数据，用于替换模板变量
        index: 模板列表索引（轮换不同话术）
        """
        templates = self.TEMPLATES.get(intent)
        if not templates:
            return None

        tpl = templates[index % len(templates)]  # 轮换
        if slots:
            tpl = Template(tpl).safe_substitute(**slots)
        return tpl

    def get_slot_filling_prompt(self, intent: str) -> Optional[str]:
        """获取槽位追问话术（也是模板化处理）"""
        prompts = {
            "query_balance": "请问要查询哪笔资产的余额？请提供资产编号（如 A001）。",
            "query_risk": "请问要查询哪位债务人的风险等级？",
            "query_detail": "请问要查询哪笔资产的详情？请提供资产编号。",
        }
        return prompts.get(intent)

5.3 LLMResponder — LLM 回复器

from typing import Optional
import json, re

class LLMResponder:
    """LLM 回复器 — 处理复杂语义问答"""

    # 系统 Prompt：角色 + 约束 + 上下文
    SYSTEM_PROMPT = """你是AMC（资产管理公司）智能客服助手。
用户可能咨询不良资产相关的问题。

## 回复要求
1. 基于提供的信息回答，不要编造数据
2. 如果提供的信息不足以回答，诚实说明
3. 语气专业但友好
4. 回复控制在 200 字以内
5. 使用 markdown 格式提高可读性"""

    def __init__(self, api_key: str):
        from zhipuai import ZhipuAI
        self.client = ZhipuAI(api_key=api_key)
        self.model = "glm-4-flash"  # 轻量模型，速度快

    def respond(
        self,
        query: str,
        context: str = "",
        history: list = None,
    ) -> str:
        """
        调用 LLM 生成回复
        context: 检索到的参考资料
        history: 对话历史（用于多轮理解）
        """
        messages = [{"role": "system", "content": self.SYSTEM_PROMPT}]

        # 组装参考资料
        if context:
            messages.append({
                "role": "system",
                "content": f"参考资料：\n{context}"
            })

        # 加入对话历史（最近5轮）
        if history:
            recent = history[-10:]  # 最近5轮（每轮user+bot两条）
            messages.extend(recent)

        # 当前问题
        messages.append({"role": "user", "content": query})

        # 调用 API
        response = self.client.chat.completions.create(
            model=self.model,
            messages=messages,
            temperature=0.3,  # 客服场景低温度，保证稳定
            max_tokens=500,
        )
        return response.choices[0].message.content

5.4 HybridResponder — 混合回复路由

class HybridResponder:
    """
    混合回复策略路由：
    模板优先（快+便宜） → LLM 兜底（灵活+智能）
    """

    def __init__(self, template: TemplateResponder, llm: LLMResponder):
        self.template = template
        self.llm = llm

    def respond(
        self, intent: str, query: str,
        slots: dict = None, context: str = "",
        history: list = None,
    ) -> str:
        """
        混合回复：模板优先，LLM 兜底
        """
        # 第一步：尝试模板回复
        if self.template.can_handle(intent):
            reply = self.template.respond(intent, slots)
            if reply:
                return reply

        # 第二步：模板无法处理，走 LLM
        return self.llm.respond(
            query=query,
            context=context,
            history=history,
        )
🔗 ConversationManager — 全流程集成 15 min
import copy
from typing import Optional

class ConversationManager:
    """
    对话管理器 — 串联意图→槽位→回复的完整流程
    类比数仓: ODS(用户输入) → DWD(意图+实体) → DWS(槽位填充) → ADS(回复输出)
    """

    def __init__(self, api_key: str = ""):
        self.sessions = SessionManager(ttl=1800)
        self.slot_mgr = SlotFillingManager()
        self.template = TemplateResponder()
        # LLM Responder（开发阶段可传空 key 走 mock）
        self.llm = LLMResponder(api_key) if api_key else None
        self.responder = HybridResponder(self.template, self.llm) if self.llm else None

    def chat(
        self, session_id: str, user_input: str
    ) -> dict:
        """
        处理一轮对话（完整管线）
        返回: {"reply": str, "intent": str, "phase": str, "slots": dict}
        """
        # 1. 获取/创建会话状态
        state = self.sessions.get_or_create(session_id)

        # 2. 意图识别 + 实体提取（Day 62 的模块）
        intent = self._classify_intent(user_input)
        entities = self._extract_entities(user_input)

        # 3. 意图切换时重置槽位
        if intent != state.intent:
            state.slots.clear()  # 新意图，清空旧槽位
            state.intent = intent

        # 4. 填充槽位
        new_filled = self.slot_mgr.fill_from_entities(state, entities)

        # 5. 槽位检查 + 追问
        missing_prompt = self.slot_mgr.check_and_prompt(intent, state)
        if missing_prompt:
            state.phase = DialogPhase.SLOT_FILLING
            state.add_turn(user_input, missing_prompt)
            return {
                "reply": missing_prompt,
                "intent": intent,
                "phase": state.phase.value,
                "slots": copy.deepcopy(state.slots),
                "action": "ask_slot",
            }

        # 6. 槽位齐全，生成回复
        state.phase = DialogPhase.EXECUTING

        # 执行业务逻辑（如查询数据库）
        business_result = self._execute_business(intent, state.slots)

        # 生成回复
        if self.responder:
            reply = self.responder.respond(
                intent=intent,
                query=user_input,
                slots={**state.slots, **business_result},
                history=self._build_history(state),
            )
        else:
            reply = self._mock_reply(intent, state.slots, business_result)

        state.phase = DialogPhase.RESULT_SHOWING
        state.add_turn(user_input, reply)

        return {
            "reply": reply,
            "intent": intent,
            "phase": state.phase.value,
            "slots": copy.deepcopy(state.slots),
            "action": "reply",
        }

    def _classify_intent(self, text: str) -> str:
        """意图分类（简化版，生产环境用 Day 62 的 IntentClassifier）"""
        text_lower = text.lower()
        if any(w in text_lower for w in ["你好", "在吗", "hello"]):
            return "greeting"
        if any(w in text_lower for w in ["余额", "多少", "多少钱"]):
            return "query_balance"
        if any(w in text_lower for w in ["风险", "等级"]):
            return "query_risk"
        if any(w in text_lower for w in ["再见", "拜拜", "bye"]):
            return "farewell"
        return "unknown"

    def _extract_entities(self, text: str) -> dict:
        """实体提取（简化版正则提取）"""
        entities = {}
        # 提取资产编号: A001, B023 等
        asset_match = re.search(r'[A-Z]\d{3}', text)
        if asset_match:
            entities["asset_id"] = asset_match.group()
        return entities

    def _execute_business(self, intent: str, slots: dict) -> dict:
        """执行业务逻辑（mock 版本）"""
        # 生产环境: 这里调用数据库查询
        mock_results = {
            "query_balance": {"amount": "1,250,000", "date": "2026-08-13"},
            "query_risk": {"risk_level": "关注", "score": "72"},
        }
        return mock_results.get(intent, {})

    def _build_history(self, state: ConversationState) -> list:
        """构建 LLM 所需的对话历史格式"""
        history = []
        for turn in state.history:
            history.append({"role": "user", "content": turn["user"]})
            history.append({"role": "assistant", "content": turn["bot"]})
        return history

    def _mock_reply(self, intent: str, slots: dict, result: dict) -> str:
        """Mock 回复（无 API Key 时使用）"""
        match intent:
            case "greeting":
                return "您好！我是AMC智能助手，请问有什么可以帮您？"
            case "query_balance":
                asset = slots.get("asset_id", "未知")
                amount = result.get("amount", "N/A")

⚠️ 常见陷阱 Top 5
  1. 意图切换不清空槽位 — 用户从"查余额"切到"查风险"，旧的 asset_id 会残留在 slots 中。解决：检测到意图变化时 state.slots.clear()。
  2. 追问死循环 — 用户反复给无效输入（如"不知道"），系统一直追问同一问题。解决：设置最大追问次数（如3次），超限后给出退出选项或转人工。
  3. 对话历史膨胀 — 20轮对话后把全部 history 传给 LLM，Token 爆炸。解决：滑动窗口只保留最近 5-10 轮，或用摘要压缩早期对话。
  4. 模板变量未替换 — Template.safe_substitute 遇到缺失变量会保留 $var 原样。解决：先用 .substitute() 试一下，KeyError 时 fallback 到默认值。
  5. 会话内存泄漏 — 内存版 SessionManager 不会自动清理过期会话。解决：定时调用 cleanup_expired()，或使用 Redis 的 TTL 自动过期。
安全红线：回复中绝对不能暴露系统内部信息（数据库表名、SQL 错误、堆栈跟踪）。所有异常都要 catch 后返回通用话术"暂时无法处理，请稍后再试"。
