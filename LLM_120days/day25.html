🏗️ 一、为什么数据工程师要学 FastAPI？ ~10min
核心动机：你昨天学会了 asyncio 并发调用 LLM。但那些能力只存在于你的脚本里。FastAPI 让你把脚本变成 Web 服务，别人（前端、其他系统、定时任务）都可以通过 HTTP 接口调用你的 LLM 能力。

从 SQL/ETL 视角理解 FastAPI

作为数据开发工程师，你熟悉的调用链是：

  • 数仓视角：SQL 查询 → 结果集（在数据库内部完成）
  • ETL 视角：调度工具 → 脚本 → 写表（定时批量运行）
  • FastAPI 视角：HTTP 请求 → API 函数 → JSON 响应（实时在线服务）

FastAPI 本质上就是 "给 Python 函数套一层 HTTP 外壳"，让任何能发 HTTP 请求的系统都能调用你的代码。

FastAPI vs Flask vs Django

特性FlaskDjangoFastAPI

为什么选 FastAPI？ 因为它天然支持 async/await（跟昨天学的 asyncio 无缝衔接），自带类型校验（Pydantic），自动生成交互式 API 文档，性能极高 —— 这些特性完美契合 LLM 应用场景。

🧱 三、Pydantic 请求体 & 类型校验 ~10min
Pydantic = Python 的"表结构定义"。你用 SQL 定义表的列名和类型，Pydantic 让你用同样的思维定义 API 的输入输出格式。FastAPI 内置了 Pydantic，自动完成校验和文档生成。

3.1 定义请求模型

from pydantic import BaseModel, Field
from typing import Optional, List

# 定义"请求体"的数据结构 —— 类比 CREATE TABLE
class ChatRequest(BaseModel):
    # Field 类似于 SQL 的 NOT NULL / DEFAULT / CHECK 约束
    message: str = Field(..., min_length=1, max_length=4000,
                           description="用户输入的问题")
    model: str = Field("glm-4-flash", description="模型名称")
    temperature: float = Field(0.7, ge=0, le=2, description="温度参数")
    max_tokens: Optional[int] = Field(None, ge=1, le=8192)
    history: List[dict] = Field(default_factory=list, description="对话历史")

    # Pydantic 的模型验证器 —— 类似于 SQL 的 BEFORE INSERT 触发器
    from pydantic import model_validator
    @model_validator(mode='after')
    def check_history_format(self):
        for msg in self.history:
            if "role" not in msg or "content" not in msg:
                raise ValueError("每条历史消息必须包含 role 和 content")
        return self

# 定义"响应体"的数据结构
class ChatResponse(BaseModel):
    reply: str
    model: str
    tokens_used: int
    latency_ms: float

3.2 在路由中使用模型

@app.post("/chat", response_model=ChatResponse)
async def chat(req: ChatRequest):
    # req 已经被 Pydantic 自动校验过了
    # 如果 message 为空、temperature 超出范围，FastAPI 自动返回 422 错误

    import time
    start = time.time()

    # TODO: 调用 LLM（下一节实现）
    reply = f"你说了：{req.message}"

    return ChatResponse(
        reply=reply,
        model=req.model,
        tokens_used=100,
        latency_ms=round((time.time() - start) * 1000, 2)
    )
对比 SQL：ChatRequest 就像定义一张表的 CREATE TABLE，Field(..., min_length=1) 就像 NOT NULL 和 CHECK 约束。FastAPI 在请求进来时自动做校验，不合格直接 422 —— 比你在存储过程里手动检查优雅多了。

4.1 完整的 LLM Chat API 服务

# llm_service.py —— AMC 智能问答 API 服务
import httpx
import time
import os
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel, Field
from typing import Optional, List

app = FastAPI(
    title="AMC 智能问答 API",
    description="基于大模型的不良资产管理智能问答服务",
    version="1.0.0"
)

# ─── 配置（类比 ETL 的配置表） ───
CONFIG = {
    "api_url": "https://open.bigmodel.cn/api/paas/v4/chat/completions",
    "api_key": os.getenv("ZHIPU_API_KEY", "your-api-key-here"),
    "default_model": "glm-4-flash",
    "timeout": 30.0,
}

# AMC 领域系统提示词
SYSTEM_PROMPT = """你是宁波AMC（资产管理公司）的不良资产数据分析师。
回答用户关于不良资产的问题时，请遵循以下规则：
1. 涉及金额时，请用"万元"为单位
2. 提及债务人时注意脱敏（隐藏身份证号中间部分）
3. 如果问题涉及具体数据查询，说明需要查询的表和字段
4. 使用专业但易懂的语言"""

# ─── 请求/响应模型 ───
class Message(BaseModel):
    role: str = Field(..., pattern="^(user|assistant|system)$")
    content: str

class ChatRequest(BaseModel):
    message: str = Field(..., min_length=1, max_length=4000,
                           description="用户问题")
    model: str = Field(CONFIG["default_model"])
    temperature: float = Field(0.7, ge=0, le=2)
    history: List[Message] = Field(default_factory=list)

class ChatResponse(BaseModel):
    reply: str
    model: str
    tokens_used: int
    latency_ms: float

class BatchRequest(BaseModel):
    questions: List[str] = Field(..., min_length=1, max_length=20,
                                 description="批量问题列表，最多20个")
    model: str = Field(CONFIG["default_model"])

class BatchResponse(BaseModel):
    results: List[dict]
    total: int
    total_latency_ms: float

# ─── 核心：调用 LLM 的函数（用 async + httpx，复用 Day24 知识） ───
async def call_llm(
    messages: List[dict],
    model: str = CONFIG["default_model"],
    temperature: float = 0.7
) -> dict:
    """调用智谱 LLM API，返回完整响应"""
    async with httpx.AsyncClient(timeout=CONFIG["timeout"]) as client:
        resp = await client.post(
            CONFIG["api_url"],
            headers={
                "Authorization": f"Bearer {CONFIG['api_key']}",
                "Content-Type": "application/json"
            },
            json={
                "model": model,
                "messages": messages,
                "temperature": temperature,
                "max_tokens": 2048,
            }
        )
        resp.raise_for_status()
        return resp.json()

# ─── API 路由 ───

@app.get("/")
async def root():
    return {
        "service": "AMC 智能问答 API",
        "version": "1.0.0",
        "docs": "/docs"
    }

@app.post("/chat", response_model=ChatResponse)
async def chat(req: ChatRequest):
    """单轮/多轮对话接口"""
    start = time.time()

    # 构建消息列表：系统提示 + 历史 + 当前问题
    messages = [{"role": "system", "content": SYSTEM_PROMPT}]
    messages.extend([m.model_dump() for m in req.history])
    messages.append({"role": "user", "content": req.message})

    try:
        result = await call_llm(messages, req.model, req.temperature)
    except httpx.HTTPStatusError as e:
        raise HTTPException(
            status_code=e.response.status_code,
            detail=f"LLM API 调用失败: {e.response.text}"
        )
    except httpx.RequestError as e:
        raise HTTPException(status_code=503, detail=f"LLM 服务不可用: {e}")

    # 解析响应
    reply = result["choices"][0]["message"]["content"]
    tokens = result["usage"]["total_tokens"]

    return ChatResponse(
        reply=reply,
        model=req.model,
        tokens_used=tokens,
        latency_ms=round((time.time() - start) * 1000, 2)
    )

@app.post("/batch", response_model=BatchResponse)
async def batch_chat(req: BatchRequest):
    """批量问答（复用 Day24 的 asyncio.gather 并发）"""
    import asyncio

    start = time.time()
    tasks = []
    for q in req.questions:
        messages = [
            {"role": "system", "content": SYSTEM_PROMPT},
            {"role": "user", "content": q}
        ]
        tasks.append(call_llm(messages, req.model))

    # 并发调用所有问题（Day24 的知识！asyncio.gather）
    results = await asyncio.gather(*tasks, return_exceptions=True)

    # 组装结果
    output = []
    for i, (q, r) in enumerate(zip(req.questions, results)):
        if isinstance(r, Exception):
            output.append({"question": q, "error": str(r)})
        else:
            output.append({
                "question": q,
                "answer": r["choices"][0]["message"]["content"],
                "tokens": r["usage"]["total_tokens"]
            })

    return BatchResponse(
        results=output,
        total=len(output),
        total_latency_ms=round((time.time() - start) * 1000, 2)
    )

5.1 路径参数 vs 查询参数 vs 请求体

参数类型语法适用场景类比SQL
路径参数 /items/{id} 定位一个具体资源 WHERE id = ?
查询参数 /items?limit=10 过滤/排序/分页 LIMIT 10
请求体 POST /items + JSON body 创建/复杂操作 INSERT INTO ... VALUES
Header Authorization: Bearer xxx 认证/元信息 连接参数

5.2 完整示例：RESTful 风格的"不良资产"API

# restful_api.py —— 模拟不良资产 CRUD（不连接真实数据库）
from fastapi import FastAPI, Query, Path, Header
from pydantic import BaseModel, Field
from typing import Optional, List
from datetime import date

app = FastAPI(title="不良资产管理API")

# 模拟数据库（内存存储）
db: List[dict] = [
    {
        "id": 1,
        "debtor_name": "张*明",
        "principal": 500.0,  # 万元
        "status": "处置中",
        "acquisition_date": "2024-03-15",
    },
    {
        "id": 2,
        "debtor_name": "李*华",
        "principal": 1200.0,
        "status": "已核销",
        "acquisition_date": "2023-11-20",
    },
]
next_id = 3

# ─── 请求模型 ───
class AssetCreate(BaseModel):
    debtor_name: str = Field(..., min_length=2, description="债务人姓名（脱敏）")
    principal: float = Field(..., gt=0, description="本金（万元）")
    status: str = Field("待处置", pattern="^(待处置|处置中|已核销|已转让)$")
    acquisition_date: str = Field(..., description="收购日期 YYYY-MM-DD")

# GET /assets —— 查询列表（带分页和过滤）
@app.get("/assets")
async def list_assets(
    status: Optional[str] = Query(None, description="按状态过滤"),
    min_principal: Optional[float] = Query(None, ge=0, description="最小本金"),
    page: int = Query(1, ge=1, description="页码"),
    page_size: int = Query(10, ge=1, le=100),
):
    """查询不良资产列表 —— 类比 SELECT * FROM assets WHERE ... LIMIT ..."""
    result = db[:]

    # 过滤（类比 WHERE 子句）
    if status:
        result = [a for a in result if a["status"] == status]
    if min_principal is not None:
        result = [a for a in result if a["principal"] >= min_principal]

    # 分页（类比 LIMIT OFFSET）
    total = len(result)
    start = (page - 1) * page_size
    end = start + page_size
    page_data = result[start:end]

    return {
        "total": total,
        "page": page,
        "page_size": page_size,
        "data": page_data
    }

# GET /assets/{asset_id} —— 查询单条（路径参数）
@app.get("/assets/{asset_id}")
async def get_asset(
    asset_id: int = Path(..., gt=0, description="资产ID")
):
    """查询单条资产 —— 类比 SELECT * FROM assets WHERE id = ?"""
    for asset in db:
        if asset["id"] == asset_id:
            return asset
    # FastAPI 会自动把 HTTPException 转成对应的 HTTP 状态码
    from fastapi import HTTPException
    raise HTTPException(status_code=404, detail=f"资产 {asset_id} 不存在")

# POST /assets —— 新增资产（请求体）
@app.post("/assets", status_code=201)
async def create_asset(asset: AssetCreate):
    """新增资产 —— 类比 INSERT INTO assets ..."""
    global next_id
    new_asset = {"id": next_id, **asset.model_dump()}
    db.append(new_asset)
    next_id += 1
    return new_asset

# DELETE /assets/{asset_id} —— 删除资产
@app.delete("/assets/{asset_id}")
async def delete_asset(asset_id: int):
    """删除资产 —— 类比 DELETE FROM assets WHERE id = ?"""
    for i, asset in enumerate(db):
        if asset["id"] == asset_id:
            db.pop(i)
            return {"message": f"资产 {asset_id} 已删除"}
    from fastapi import HTTPException
    raise HTTPException(status_code=404, detail=f"资产 {asset_id} 不存在")

# ─── 启动：uvicorn restful_api:app --reload --port 8001 ───
RESTful 设计规范速记： GET = 查（SELECT），POST = 增（INSERT），PUT = 改（UPDATE），DELETE = 删（DELETE）。URL 用名词复数（/assets），不用动词（/getAssets 是反模式）。

from fastapi import Request
from fastapi.responses import JSONResponse

# 自定义异常
class LLMServiceError(Exception):
    def __init__(self, message: str, code: str = "LLM_ERROR"):
        self.message = message
        self.code = code

# 全局异常处理器 —— 类比 ETL 的全局错误捕获和告警
@app.exception_handler(LLMServiceError)
async def llm_error_handler(request: Request, exc: LLMServiceError):

        status_code=500,
        content={
            "error": exc.code,
            "message": exc.message,
            "path": request.url.path
        }

# 通用兜底异常处理
@app.exception_handler(Exception)

6.3 请求日志中间件

import time
from starlette.middleware.base import BaseHTTPMiddleware

class TimingMiddleware(BaseHTTPMiddleware):
    """记录每个请求的响应时间 —— 类比 SQL 的慢查询日志"""
    async def dispatch(self, request: Request, call_next):
        start = time.time()
        response = await call_next(request)
        duration = round((time.time() - start) * 1000, 2)
        print(f"[{request.method}] {request.url.path} → {response.status_code} ({duration}ms)")
        response.headers["X-Response-Time"] = f"{duration}ms"
        return response

app.add_middleware(TimingMiddleware)

#知识关键点
