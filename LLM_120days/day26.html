🔧 一、依赖注入（Dependency Injection）——FastAPI 的灵魂 20 min
什么是依赖注入？ 就像你写 SQL 时，数据库连接由平台提供而不是你自己拼字符串。FastAPI 的 Depends() 让你把"公共逻辑"抽成独立函数，多个接口复用同一份代码。不用继承、不用装饰器，一个 Depends 搞定。

1.1 最简依赖：公共参数提取

假设你的 API 有 10 个接口都需要 page 和 page_size 分页参数——写 10 遍？不，抽成一个依赖函数：

# common_deps.py —— 公共依赖函数
from fastapi import Query

def common_pagination(
    page: int = Query(1, ge=1, description="页码，从1开始"),
    page_size: int = Query(20, ge=1, le=100, description="每页条数")
):
    """分页参数，所有接口复用"""
    offset = (page - 1) * page_size
    return {"page": page, "page_size": page_size, "offset": offset}
# main.py —— 使用依赖
from fastapi import FastAPI, Depends
from common_deps import common_pagination

app = FastAPI()

@app.get("/amc/assets")
async def list_assets(pag: dict = Depends(common_pagination)):
    # pag 就是 common_pagination 的返回值
    sql = f"SELECT * FROM amc_assets LIMIT {pag['page_size']} OFFSET {pag['offset']}"
    return {"sql": sql, "page": pag["page"]}

@app.get("/amc/debtors")
async def list_debtors(pag: dict = Depends(common_pagination)):
    sql = f"SELECT * FROM debtors LIMIT {pag['page_size']} OFFSET {pag['offset']}"
    return {"sql": sql, "page": pag["page"]}
数据工程师视角：这就像 SQL 里的 WITH 公共表表达式（CTE）——把重复逻辑抽出来，各处引用。依赖注入就是"函数版 CTE"。

1.2 类作为依赖（更强大的状态管理）

当依赖需要维护状态（比如数据库连接池），用类更合适：

from fastapi import Depends

class DBConfig:
    """数据库配置依赖 —— 模拟从环境变量/配置文件读取"""
    def __init__(self, env: str = "dev"):

            "prod": {"host": "10.0.1.100", "port": 3306, "db": "amc_prod"},

        self.config = configs.get(env, configs["dev"])

# 用 Depends(DBConfig) 时，FastAPI 自动实例化 DBConfig()

1.3 依赖链（嵌套依赖）

依赖可以依赖其他依赖，形成一条"调用链"。类似 SQL 中 CTE 嵌套 CTE：

def get_api_key(api_key: str = Header("X-API-Key")):
    if not api_key:
        raise HTTPException(status_code=401, detail="缺少 API Key")
    return api_key

def get_db_session(key: str = Depends(get_api_key)):
    # 先验证 API Key，再创建数据库会话
    # 不同 key 对应不同数据库（多租户）
    session = create_session_by_key(key)
    return session

@app.get("/data")
async def get_data(session = Depends(get_db_session)):
    # 自动执行：get_api_key → get_db_session → get_data
    return session.query(...)
依赖注入的三大好处：
① 复用 —— 一处定义，多处使用
② 可测试 —— 测试时替换为 mock 依赖，无需改业务代码
③ 自动文档 —— 依赖的参数自动出现在 Swagger 文档中

# db_demo.py
from sqlalchemy import create_engine, Column, Integer, String, Float, text
from sqlalchemy.orm import sessionmaker, DeclarativeBase

# 1. 创建引擎（SQLite 文件数据库，适合学习；生产换 MySQL/PostgreSQL）
engine = create_engine("sqlite:///amc.db", echo=False)

# 2. 定义 ORM 模型（表结构）
class Base(DeclarativeBase):
    pass

class NpaAsset(Base):
    __tablename__ = "npa_assets"  # 表名

    id = Column(Integer, primary_key=True, autoincrement=True)
    debtor_name = Column(String(100), nullable=False, comment="债务人名称")
    asset_type = Column(String(20), comment="资产类型: 贷款/债券/保证")
    principal = Column(Float, comment="本金余额(万元)")
    status = Column(String(20), default="持有")

# 3. 建表（如果不存在）
Base.metadata.create_all(engine)

# 4. 创建 Session 工厂
SessionLocal = sessionmaker(bind=engine)

# 5. CRUD 操作
def create_asset(name, asset_type, principal):
    """新增一条不良资产记录"""
    with SessionLocal() as session:
        asset = NpaAsset(
            debtor_name=name,
            asset_type=asset_type,
            principal=principal
        )
        session.add(asset)
        session.commit()
        session.refresh(asset)  # 刷新获取自增ID
        return asset

def list_assets(status=None, limit=20):
    """查询资产列表，可按状态过滤"""
    with SessionLocal() as session:
        query = session.query(NpaAsset)
        if status:
            query = query.filter(NpaAsset.status == status)
        return query.limit(limit).all()

def get_asset(asset_id: int):
    """按ID查询单条"""
    with SessionLocal() as session:
        return session.query(NpaAsset).get(asset_id)

def update_asset_status(asset_id: int, new_status: str):
    """更新资产状态"""
    with SessionLocal() as session:
        asset = session.query(NpaAsset).get(asset_id)
        if not asset:
            return None
        old_status = asset.status
        asset.status = new_status
        session.commit()
        return {"id": asset_id, "old": old_status, "new": new_status}

def delete_asset(asset_id: int):
    """删除资产记录"""
    with SessionLocal() as session:
        asset = session.query(NpaAsset).get(asset_id)
        if not asset:
            return False
        session.delete(asset)
        session.commit()
        return True

2.3 把数据库操作做成 FastAPI 依赖

这是企业级写法——数据库会话的生命周期由框架管理：

# dependencies.py
from db_demo import SessionLocal, NpaAsset

def get_db():
    """数据库会话依赖 —— 请求结束时自动关闭"""
    db = SessionLocal()
    try:
        yield db  # yield 是关键！生成器依赖
    finally:
        db.close()

# api.py —— 使用数据库依赖
from fastapi import FastAPI, Depends, HTTPException
from sqlalchemy.orm import Session
from dependencies import get_db

app = FastAPI()

@app.post("/assets")
async def create_asset_api(
    name: str, asset_type: str, principal: float,
    db: Session = Depends(get_db)  # 注入数据库会话
):
    asset = NpaAsset(debtor_name=name, asset_type=asset_type, principal=principal)
    db.add(asset)
    db.commit()
    db.refresh(asset)
    return {"id": asset.id, "name": asset.debtor_name}

@app.get("/assets")
async def list_assets_api(db: Session = Depends(get_db)):
    assets = db.query(NpaAsset).all()
    return [{"id": a.id, "name": a.debtor_name, "type": a.asset_type} for a in assets]

@app.get("/assets/{asset_id}")
async def get_asset_api(asset_id: int, db: Session = Depends(get_db)):
    asset = db.query(NpaAsset).get(asset_id)
    if not asset:
        raise HTTPException(404, f"资产 {asset_id} 不存在")
    return asset

@app.delete("/assets/{asset_id}")
async def delete_asset_api(asset_id: int, db: Session = Depends(get_db)):
    asset = db.query(NpaAsset).get(asset_id)
    if not asset:
        raise HTTPException(404, f"资产 {asset_id} 不存在")
    db.delete(asset)
    db.commit()
    return {"msg": "删除成功"}
yield 依赖的关键：普通函数用 return，而数据库会话必须用 yield。
yield 之前的代码 = 请求开始时执行
yield 之后的代码（finally 块）= 请求结束后执行（关闭连接）
注意：生产环境千万别用 SQLite！SQLite 不支持并发写入。生产环境用 MySQL（pymysql 驱动）或 PostgreSQL（psycopg2 驱动），只需改连接字符串：
create_engine("mysql+pymysql://user:pass@host:3306/amc_db")

# auth.py —— API Key 认证方案
from fastapi import Header, HTTPException, Depends
from typing import Optional

# 模拟的 API Key 存储（生产环境应存数据库/Redis）
VALID_KEYS = {
    "sk-data-team-001": {"name": "数据团队", "role": "admin"},
    "sk-llm-dev-002":   {"name": "LLM开发", "role": "user"},
}

async def verify_api_key(
    x_api_key: Optional[str] = Header(None)
):
    """
    API Key 认证依赖
    客户端请求时需在 Header 中携带：X-API-Key: sk-xxx
    """
    if not x_api_key:
        raise HTTPException(
            status_code=401,
            detail="缺少 X-API-Key 请求头。请在 Header 中添加 X-API-Key"
        )
    if x_api_key not in VALID_KEYS:
        raise HTTPException(
            status_code=403,
            detail="无效的 API Key"
        )
    return VALID_KEYS[x_api_key]  # 返回用户信息

🤖 四、LLM 实战：带认证 + 数据库的 LLM API 网关 20 min

把今天学的三个知识点串起来，做一个"LLM 资产分析 API 网关"：

# llm_gateway.py —— 完整实战
from fastapi import FastAPI, Depends, HTTPException, Header, Query
from pydantic import BaseModel, Field
from sqlalchemy import create_engine, Column, Integer, String, Float, DateTime, Text
from sqlalchemy.orm import sessionmaker, DeclarativeBase
from datetime import datetime
from typing import Optional, List
import requests as http_requests
import json, os

# ======== 配置 ========
DATABASE_URL = os.getenv("DB_URL", "sqlite:///./llm_gateway.db")
ZHIPU_API_KEY = os.getenv("ZHIPU_API_KEY", "your-key-here")
engine = create_engine(DATABASE_URL)
SessionLocal = sessionmaker(bind=engine)

# ======== 数据库模型 ========
class Base(DeclarativeBase):
    pass

class ChatLog(Base):
    """对话日志表 —— 记录每次 LLM 调用"""
    __tablename__ = "chat_logs"

    id = Column(Integer, primary_key=True, autoincrement=True)
    user_name = Column(String(50), index=True)
    question = Column(Text)
    answer = Column(Text)
    model = Column(String(50))
    tokens_used = Column(Integer, default=0)
    created_at = Column(DateTime, default=datetime.now)

class UserQuota(Base):
    """用户配额表 —— 控制每个用户的调用量"""
    __tablename__ = "user_quotas"

    user_name = Column(String(50), primary_key=True)
    daily_limit = Column(Integer, default=100)
    daily_used = Column(Integer, default=0)
    reset_date = Column(String(10), default="")

Base.metadata.create_all(engine)

# ======== 依赖函数 ========
def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

VALID_KEYS = {
    "sk-amc-admin":  "AMC管理员",
    "sk-data-dev":   "数据开发",
    "sk-analyst":    "分析师",
}

async def auth_dep(x_api_key: Optional[str] = Header(None)):
    """认证 + 配额检查 依赖"""
    if not x_api_key or x_api_key not in VALID_KEYS:
        raise HTTPException(401, "认证失败")
    return VALID_KEYS[x_api_key]

async def check_quota(
    user_name: str = Depends(auth_dep),
    db: Session = Depends(get_db)
):
    """配额检查依赖 —— 依赖链：auth_dep → check_quota"""
    today = datetime.now().strftime("%Y-%m-%d")
    quota = db.query(UserQuota).filter_by(user_name=user_name).first()

    if not quota:
        # 首次使用，自动创建配额
        quota = UserQuota(user_name=user_name, daily_limit=100, daily_used=0, reset_date=today)
        db.add(quota)
        db.commit()

    # 跨天自动重置
    if quota.reset_date != today:
        quota.daily_used = 0
        quota.reset_date = today
        db.commit()

    if quota.daily_used >= quota.daily_limit:
        raise HTTPException(429, f"今日配额已用完（{quota.daily_limit}次），明天重置")

    return {"user_name": user_name, "quota": quota}

# ======== Pydantic 模型 ========
class ChatRequest(BaseModel):
    question: str = Field(..., min_length=1, max_length=2000, description="用户问题")
    model: str = Field(default="glm-4-flash", description="模型名称")

class ChatResponse(BaseModel):
    answer: str
    model: str
    tokens_used: int
    remaining_quota: int

# ======== LLM 调用函数 ========
def call_llm(question: str, model: str = "glm-4-flash") -> dict:
    """调用智谱 AI API"""
    url = "https://open.bigmodel.cn/api/paas/v4/chat/completions"
    headers = {
        "Authorization": f"Bearer {ZHIPU_API_KEY}",
        "Content-Type": "application/json"
    }
    payload = {
        "model": model,
        "messages": [
            {"role": "system", "content": "你是AMC（资产管理公司）不良资产分析专家，用简洁专业的方式回答问题。"},
            {"role": "user", "content": question}
        ],
        "temperature": 0.7,
        "max_tokens": 500
    }
    resp = http_requests.post(url, headers=headers, json=payload, timeout=30)
    resp.raise_for_status()
    data = resp.json()
    return {
        "answer": data["choices"][0]["message"]["content"],
        "tokens": data["usage"].get("total_tokens", 0)
    }

# ======== FastAPI 应用 ========
app = FastAPI(title="LLM 资产分析网关", version="1.0")

@app.post("/chat", response_model=ChatResponse)
async def chat(
    req: ChatRequest,
    ctx: dict = Depends(check_quota),  # 同时注入认证+配额+数据库
    db: Session = Depends(get_db)
):
    """
    核心对话接口
    流程：认证 → 配额检查 → 调用 LLM → 记录日志 → 扣减配额 → 返回结果
    """
    try:
        # 1. 调用 LLM
        result = call_llm(req.question, req.model)

        # 2. 记录日志
        log = ChatLog(
            user_name=ctx["user_name"],
            question=req.question,
            answer=result["answer"],
            model=req.model,
            tokens_used=result["tokens"]
        )
        db.add(log)

        # 3. 扣减配额
        ctx["quota"].daily_used += 1

        db.commit()

        return ChatResponse(
            answer=result["answer"],
            model=req.model,
            tokens_used=result["tokens"],
            remaining_quota=ctx["quota"].daily_limit - ctx["quota"].daily_used
        )
    except http_requests.RequestException as e:
        raise HTTPException(502, f"LLM 服务调用失败: {e}")

@app.get("/logs")
async def get_logs(
    user_name: str = Depends(auth_dep),
    db: Session = Depends(get_db),
    limit: int = Query(20, le=100)
):
    """查询对话日志"""
    logs = db.query(ChatLog).order_by(ChatLog.id.desc()).limit(limit).all()
    return [{
        "id": l.id,
        "time": l.created_at.strftime("%H:%M:%S"),
        "question": l.question[:50],
        "model": l.model,
        "tokens": l.tokens_used
    } for l in logs]

@app.get("/quota")
async def get_quota(
    ctx: dict = Depends(check_quota)
):
    """查看当前配额"""
    q = ctx["quota"]
    return {
        "user": ctx["user_name"],
        "used": q.daily_used,
        "limit": q.daily_limit,
        "remaining": q.daily_limit - q.daily_used
    }

5.1 Pydantic 校验进阶

from pydantic import BaseModel, Field, field_validator
from typing import Literal

class AssetCreate(BaseModel):
    debtor_name: str = Field(..., min_length=2, max_length=100, description="债务人名称")
    asset_type: Literal["贷款", "债券", "保证", "抵债资产"]  # 枚举约束
    principal: float = Field(..., gt=0, description="本金(万元)，必须大于0")
    status: str = Field(default="持有")

    @field_validator("debtor_name")
    def name_must_not_contain_sql(cls, v):
        """防 SQL 注入校验"""
        dangerous = ["DROP", "DELETE", "INSERT", "--", ";"]
        if any(kw in v.upper() for kw in dangerous):
            raise ValueError("名称包含非法字符")
        return v.strip()

    code: int = 0
    message: str = "success"
    data: dict | list | None = None

# 全局异常处理器 —— 所有未被捕获的异常都会走这里
@app.exception_handler(Exception)
async def global_exception_handler(request: Request, exc: Exception):
    return JSONResponse(
        status_code=500,
        content={"code": -1, "message": f"服务器内部错误: {exc}", "data": None}
    )

# 使用统一响应
@app.get("/health")
async def health_check():
    return APIResponse(data={"status": "ok", "version": "1.0"})
