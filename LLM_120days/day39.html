2.1 functools.lru_cache — 内置 LRU 缓存

Python 标准库自带了 lru_cache（Least Recently Used 最近最少使用），它可以用装饰器一键给纯函数加缓存。

lru_cache 基础用法
from functools import lru_cache
import time

# maxsize=128: 缓存最近128次调用的结果
# 参数必须是可哈希的（字符串、数字、元组）
@lru_cache(maxsize=128)
def expensive_compute(n: int) -> int:
    """模拟耗时的幂运算"""
    time.sleep(0.5)  # 模拟耗时
    return n ** n

# 第一次调用：实际计算，耗时0.5秒
t0 = time.time()
result1 = expensive_compute(5)  # 3125
print(f"第一次: {time.time()-t0:.3f}s, 结果={result1}")

# 第二次调用：命中缓存，几乎瞬间
t0 = time.time()
result2 = expensive_compute(5)
print(f"第二次: {time.time()-t0:.6f}s, 结果={result2}")

# 查看缓存状态
print(f"缓存信息: {expensive_compute.cache_info()}")
# CacheInfo(hits=1, misses=1, maxsize=128, currsize=1)
数仓类比：lru_cache 就像物化视图 — 第一次查询走全表扫描（慢），后续相同查询直接读物化结果（快）。maxsize 就是物化视图能存多少种查询模式。
2.2 lru_cache 在 LLM 场景的局限

lru_cache 有两个问题：

  1. 不可变缓存：一旦存入，没有 TTL 过期机制（Embedding 结果可能模型更新后就过期了）
  2. 仅支持可哈希参数：列表、字典等不可哈希类型不能用

我们需要一个支持 TTL 过期 + 可哈希化 key的缓存。手写一个：

自定义 TTL 缓存
import time
import hashlib
import json
from collections import OrderedDict

class TTLCache:
    """带 TTL 过期的 LRU 缓存"""

    def __init__(self, maxsize: int = 128, ttl: int = 3600):
        """maxsize: 最大缓存条目数, ttl: 过期秒数"""
        self._cache: OrderedDict = OrderedDict()  # 有序字典，自动维护 LRU
        self._maxsize = maxsize
        self._ttl = ttl
        self._hits = 0
        self._misses = 0

    def _make_key(self, args, kwargs) -> str:
        """把参数转成可哈希的字符串 key"""
        # json.dumps 会处理列表/字典，然后 MD5 生成固定长度 key
        raw = json.dumps({"args": args, "kwargs": kwargs}, sort_keys=True)
        return hashlib.md5(raw.encode()).hexdigest()

    def get(self, args, kwargs) -> object | None:
        """取值，过期返回 None"""
        key = self._make_key(args, kwargs)
        if key in self._cache:
            value, timestamp = self._cache[key]
            if time.time() - timestamp < self._ttl:
                # 命中且未过期：移到末尾（最近使用）
                self._cache.move_to_end(key)
                self._hits += 1
                return value
            else:
                # 已过期：删除
                del self._cache[key]
        self._misses += 1
        return None

    def set(self, args, kwargs, value):
        """存值，自动淘汰最旧的"""
        key = self._make_key(args, kwargs)
        if key in self._cache:
            del self._cache[key]  # 先删再存，确保移到末尾
        elif len(self._cache) >= self._maxsize:
            self._cache.popitem(last=False)  # 删除最旧的（FIFO端）
        self._cache[key] = (value, time.time())

    def clear(self):
        self._cache.clear()
        self._hits = 0
        self._misses = 0

    def stats(self) -> dict:
        total = self._hits + self._misses
        rate = self._hits / total if total > 0 else 0
        return {
            "hits": self._hits, "misses": self._misses,
            "hit_rate": f"{rate:.1%}",
            "size": len(self._cache),
        }
2.3 状态机模式 — 管理复杂流转

Agent 的执行过程本质是状态机：思考→调用工具→检查结果→成功/失败/重试。用 enum + 条件转移来管理：

状态机基础
from enum import Enum, auto

class AgentState(Enum):
    """Agent 执行状态"""
    THINKING = auto()      # LLM 正在推理
    TOOL_CALL = auto()      # 正在调用工具
    TOOL_RESULT = auto()   # 工具返回了结果
    ERROR = auto()         # 出错了，需要恢复
    WAITING_HUMAN = auto() # 等待人工审批
    FINISHED = auto()      # 任务完成

# 定义合法的状态转移规则
TRANSITIONS = {
    AgentState.THINKING:    [AgentState.TOOL_CALL, AgentState.FINISHED],
    AgentState.TOOL_CALL:  [AgentState.TOOL_RESULT, AgentState.ERROR],
    AgentState.TOOL_RESULT:[AgentState.THINKING, AgentState.WAITING_HUMAN],
    AgentState.ERROR:      [AgentState.THINKING, AgentState.WAITING_HUMAN, AgentState.FINISHED],
    AgentState.WAITING_HUMAN:[AgentState.THINKING, AgentState.FINISHED],
    AgentState.FINISHED:   [],  # 终态，不再转移
}

class StateMachine:
    """通用状态机，控制状态转移"""
    def __init__(self, initial: AgentState):
        self._state = initial
        self._history: list[AgentState] = [initial]

    @property
    def state(self) -> AgentState:
        return self._state

    def transition(self, target: AgentState):
        """尝试状态转移，不合法则抛异常"""
        allowed = TRANSITIONS.get(self._state, [])
        if target not in allowed:
            raise ValueError(
                f"非法转移: {self._state.name} → {target.name}，"
                f"允许: {[s.name for s in allowed]}"
            )
        print(f"  [状态转移] {self._state.name} → {target.name}")
        self._state = target
        self._history.append(target)

    def history(self) -> list[str]:
        return [s.name for s in self._history]
为什么用状态机？没有状态机时，Agent 的执行逻辑全靠 if/else 堆叠，容易遗漏分支。状态机把"从哪来→能到哪去"显式定义，运行时自动校验，防止 Agent 进入非法状态（比如在 ERROR 时直接跳到 FINISHED 而不经过恢复）。

# ── 执行记录（每一步的完整快照）──
@dataclass
class AgentStep:
    """Agent 执行的一步记录"""
    step_num: int                  # 步骤编号
    state: AgentState              # 当前状态
    thought: str = ""          # LLM 的推理内容
    tool_name: str = ""        # 调用的工具名
    tool_input: str = ""       # 工具输入
    tool_output: str = ""      # 工具输出
    error: str = ""              # 错误信息（如果有）
    duration_ms: float = 0     # 本步耗时（毫秒）
    tokens_used: int = 0       # 本步消耗的 Token 数

    def to_dict(self) -> dict:
        return {
            "step": self.step_num,
            "state": self.state.name,
            "thought": self.thought[:100],  # 截断防止过大
            "tool": self.tool_name,
            "input": self.tool_input[:100],
            "output": self.tool_output[:200],
            "error": self.error,
            "duration_ms": round(self.duration_ms, 1),
            "tokens": self.tokens_used,
        }


# ── 带追踪的 ReAct Agent ──
@dataclass
class TraceableReActAgent:
    """带完整追踪的 ReAct Agent"""
    name: str = "Agent"
    max_steps: int = 10
    _steps: list = field(default_factory=list)
    _total_tokens: int = field(default=0, init=False)
    _callbacks: dict = field(default_factory=dict, init=False)

    def register_callback(self, event: str, fn: Callable):
        """注册事件回调（用于日志/监控/审计）"""
        self._callbacks.setdefault(event, []).append(fn)

    def _emit(self, event: str, step: AgentStep):
        """触发回调"""
        for fn in self._callbacks.get(event, []):
            fn(step)

    def _add_step(self, **kwargs) -> AgentStep:
        """记录一步"""
        step = AgentStep(step_num=len(self._steps) + 1, **kwargs)
        self._steps.append(step)
        self._total_tokens += step.tokens_used
        self._emit("on_step", step)
        return step

    def show_trace(self):
        """打印完整执行追踪"""
        print(f"\n{'='*60}")
        print(f"📋 {self.name} 执行追踪")
        print(f"{'='*60}")
        for s in self._steps:
            state_icon = {
                AgentState.THINKING: "🧠",
                AgentState.TOOL_CALL: "🔧",
                AgentState.TOOL_RESULT: "📄",
                AgentState.ERROR: "❌",
                AgentState.WAITING_HUMAN: "👤",
                AgentState.FINISHED: "✅",
            }.get(s.state, "❓")
            print(f"{state_icon} Step {s.step_num}: [{s.state.name}] {s.duration_ms}ms")
            if s.thought:
                print(f"   思考: {s.thought[:80]}")
            if s.tool_name:
                print(f"   工具: {s.tool_name}({s.tool_input[:60]})")
            if s.error:
                print(f"   错误: {s.error}")
        print(f"{'─'*60}")
        print(f"总计: {len(self._steps)} 步, {self._total_tokens} tokens")
可观测性的价值：有了执行追踪，你可以：① 统计每个工具的平均耗时和成功率 ② 发现哪些步骤经常出错 ③ 审计 Agent 做了什么决策 ④ 为用户提供"查看思考过程"的选项。
💾 工具结果缓存 — 省钱省时间 ⏱ 约 15 分钟
4.1 为什么要缓存工具结果？
场景 无缓存 有缓存
相同查询重复调用 Embedding 每次调用 API，0.5s + ¥0.001 命中缓存，<1ms，¥0
数据库查询 每次执行 SQL 短时间窗口内复用结果
多次 ReAct 循环 重复推理同一子问题 缓存中间结果，跳过已解决的问题
缓存不是万能的：查询实时数据（如"今天新增多少不良资产"）绝对不能缓存！每个工具需要独立配置缓存策略。
4.2 CachedTool — 带缓存的工具包装器
cached_tool.py
from dataclasses import dataclass, field
from typing import Callable, Optional
import time
import json

@dataclass
class CachedTool:
    """带 TTL 缓存的工具包装器"""
    name: str                          # 工具名
    description: str                   # 工具描述
    func: Callable                    # 实际执行函数
    ttl: int = 300                      # 缓存TTL（秒），默认5分钟
    maxsize: int = 64                   # 最大缓存条目
    enabled: bool = True                 # 是否启用缓存
    _cache: dict = field(default_factory=dict, init=False, repr=False)
    _stats: dict = field(default_factory=lambda: {"hits": 0, "misses": 0}, init=False, repr=False)

    def _cache_key(self, **kwargs) -> str:
        """根据参数生成缓存 key"""
        raw = json.dumps(kwargs, sort_keys=True, default=str)
        return f"{self.name}:{raw}"

    def execute(self, **kwargs) -> str:
        """执行工具（带缓存）"""
        if not self.enabled:
            return self.func(**kwargs)

        key = self._cache_key(**kwargs)

        # 检查缓存
        if key in self._cache:
            value, ts = self._cache[key]
            if time.time() - ts < self.ttl:
                self._stats["hits"] += 1
                return f"[缓存命中] {value}"
            else:
                del self._cache[key]  # 过期清除

        # 缓存未命中，实际执行
        self._stats["misses"] += 1
        result = self.func(**kwargs)

        # 存入缓存（如果达到上限，清空最早的）
        if len(self._cache) >= self.maxsize:
            oldest_key = next(iter(self._cache))
            del self._cache[oldest_key]
        self._cache[key] = (result, time.time())

        return result

    def clear_cache(self):
        self._cache.clear()

    def stats(self) -> dict:
        total = self._stats["hits"] + self._stats["misses"]
        rate = self._stats["hits"] / total if total > 0 else 0
        return {
            "tool": self.name,
            "hits": self._stats["hits"],
            "misses": self._stats["misses"],
            "hit_rate": f"{rate:.1%}",
            "cache_size": len(self._cache),
        }

5.2 ErrorRecovery 策略引擎
error_recovery.py — 错误恢复策略
from dataclasses import dataclass
from typing import Callable
from enum import Enum, auto
import time

class ErrorAction(Enum):
    """错误恢复动作"""
    RETRY = auto()           # 直接重试
    RETRY_WITH_BACKOFF = auto()  # 指数退避重试
    INFORM_LLM = auto()      # 告诉LLM，让它修正
    SKIP = auto()            # 跳过这个工具
    ABORT = auto()           # 终止任务
    ESCALATE = auto()        # 转人工

@dataclass
class ErrorRule:
    """一条错误匹配规则"""
    error_pattern: str    # 错误信息包含的关键词
    action: ErrorAction     # 要执行的动作
    max_retries: int = 3  # 最大重试次数
    description: str = ""

class ErrorRecovery:
    """错误恢复引擎"""

    # 预置规则
    DEFAULT_RULES = [
        ErrorRule("timeout",    ErrorAction.RETRY_WITH_BACKOFF, 3, "网络超时"),
        ErrorRule("connection", ErrorAction.RETRY_WITH_BACKOFF, 3, "连接错误"),
        ErrorRule("rate limit", ErrorAction.RETRY_WITH_BACKOFF, 5, "API限流"),
        ErrorRule("SyntaxError", ErrorAction.INFORM_LLM, 2, "SQL语法错误"),
        ErrorRule("permission",  ErrorAction.ESCALATE, 0, "权限不足"),
        ErrorRule("not found",   ErrorAction.INFORM_LLM, 1, "资源不存在"),
    ]

    def __init__(self, rules: list[ErrorRule] | None = None):
        self._rules = rules or self.DEFAULT_RULES
        self._retry_counts: dict[str, int] = {}

    def _match_rule(self, error_msg: str) -> ErrorRule | None:
        """根据错误信息匹配规则"""
        for rule in self._rules:
            if rule.error_pattern.lower() in error_msg.lower():
                return rule
        return None

    def handle(self, error_msg: str, context: str = "") -> dict:
        """处理错误，返回恢复指令"""
        rule = self._match_rule(error_msg)

        if not rule:
            # 未知错误：默认告诉 LLM
            return {
                "action": ErrorAction.INFORM_LLM,
                "message": f"遇到未知错误: {error_msg[:200]}，请尝试其他方法。",
                "should_retry": False,
            }

        # 检查重试次数
        count = self._retry_counts.get(rule.error_pattern, 0)
        if count >= rule.max_retries:
            return {
                "action": ErrorAction.ESCALATE,
                "message": f"{rule.description} 已重试 {count} 次仍失败，转人工处理。",
                "should_retry": False,
            }

        self._retry_counts[rule.error_pattern] = count + 1

        # 按规则执行
        if rule.action == ErrorAction.RETRY_WITH_BACKOFF:
            wait = 2 ** count  # 1s, 2s, 4s...
            print(f"  ⚠️ {rule.description}，{wait}s 后重试 ({count+1}/{rule.max_retries})")
            time.sleep(wait)  # 实际场景用 asyncio.sleep
            return {"action": rule.action, "message": "重试中...", "should_retry": True}

        elif rule.action == ErrorAction.INFORM_LLM:
            return {
                "action": rule.action,
                "message": f"{rule.description}: {error_msg[:200]}。请修正后重试。",
                "should_retry": False,  # 不自动重试，让LLM修正后再来
            }

        elif rule.action == ErrorAction.ESCALATE:
            return {
                "action": rule.action,
                "message": f"{rule.description}，需要人工干预。",
                "should_retry": False,
            }

        return {"action": rule.action, "message": "", "should_retry": False}
5.3 把恢复机制集成到 Agent 循环
agent_with_recovery.py — 核心循环（伪代码展示集成思路）
def run_with_recovery(agent, query: str, tools: list, recovery: ErrorRecovery):
    """带错误恢复的 Agent 主循环"""
    for step in range(agent.max_steps):
        # 1. LLM 推理
        response = call_llm(query, history)

        # 2. 解析工具调用
        tool_call = parse_tool_call(response)
        if not tool_call:
            return response  # LLM 直接回答，不需要工具

        # 3. 执行工具（带恢复）
        tool = find_tool(tools, tool_call.name)
        try:
            result = tool.execute(**tool_call.params)
        except Exception as e:
            # ── 错误恢复 kicks in ──
            recovery_result = recovery.handle(str(e))

            if recovery_result["should_retry"]:
                continue  # 自动重试

            elif recovery_result["action"] == ErrorAction.INFORM_LLM:
                # 把错误信息加到上下文，让 LLM 自行修正
                history.append({
                    "role": "tool_error",
                    "content": recovery_result["message"]
                })
                continue

            elif recovery_result["action"] == ErrorAction.ESCALATE:
                return f"❌ 需要人工介入: {recovery_result['message']}"

        # 4. 记录结果，进入下一轮
        history.append({"role": "tool", "content": result})

    return "⚠️ 达到最大步数限制"
👤 Human-in-the-Loop — 人机协同审批 ⏱ 约 15 分钟
6.1 为什么需要人工介入？

Agent 再智能，有些操作也不应该自动执行：

  • 数据变更：删除记录、更新金额 — 影响生产数据
  • 外部调用：发邮件、发消息 — 影响真实用户
  • 高成本操作：大批量 Embedding 计算 — 烧钱
  • 不确定决策：LLM 对多个方案拿不定主意
核心思想：Human-in-the-Loop 不是"让 Agent 每步都问人"（那不叫 Agent，叫半自动），而是在特定节点暂停，请求人工确认或补充信息后继续。
6.2 审批策略设计
策略 触发条件 行为 适用场景
ALWAYS 每次调用此工具 必须等人工确认 删除数据、发邮件
NEVER 不触发 直接执行 查询操作
AUTO 不触发 直接执行，失败时才问人 一般工具调用
THRESHOLD 影响行数/金额超阈值 超阈值才问人 批量更新、导出数据
6.3 实现 HumanApproval 系统
human_approval.py
from enum import Enum, auto
from dataclasses import dataclass, field
from typing import Callable

class ApprovalPolicy(Enum):
    ALWAYS = auto()      # 每次都要审批
    NEVER = auto()       # 不需要审批
    AUTO = auto()        # 自动执行，失败才转人工
    THRESHOLD = auto()   # 超过阈值才审批

@dataclass
class ApprovalRequest:
    """一次人工审批请求"""
    id: int
    tool_name: str
    action: str          # 描述要做什么
    params: dict          # 参数详情
    risk_level: str      # 风险等级: low/medium/high
    reason: str = ""    # 为什么需要审批
    approved: bool | None = None  # None=待审批, True=通过, False=拒绝
    reviewer_note: str = ""

class HumanApproval:
    """人工审批系统"""

    def __init__(self, auto_approve: bool = False):
        """
        auto_approve: 测试模式，自动通过（生产环境必须为 False）
        """
        self._pending: list[ApprovalRequest] = []
        self._history: list[ApprovalRequest] = []
        self._auto_approve = auto_approve
        self._next_id = 1

    def check(self, tool_name: str, policy: ApprovalPolicy,
             params: dict, threshold: dict | None = None) -> bool:
        """
        检查是否需要人工审批。
        返回 True 表示可以直接执行，False 表示需要等待审批。
        """
        # NEVER 策略：直接放行
        if policy == ApprovalPolicy.NEVER:
            return True

        # ALWAYS 策略：必须审批
        if policy == ApprovalPolicy.ALWAYS:
            return self._request_approval(
                tool_name, f"执行 {tool_name}", params, "high",
                "此工具配置为必须审批"
            )

        # THRESHOLD 策略：检查是否超过阈值
        if policy == ApprovalPolicy.THRESHOLD and threshold:
            exceeds = self._check_threshold(params, threshold)
            if exceeds:
                return self._request_approval(
                    tool_name,
                    f"执行 {tool_name}（超阈值）", params, "medium",
                    f"参数超过阈值: {exceeds}"
                )
            return True

        # AUTO 策略：直接放行
        return True

    def _check_threshold(self, params: dict, threshold: dict) -> str | None:
        """检查参数是否超过阈值"""
        # threshold 示例: {"affected_rows": 100, "amount": 100000}
        for key, limit in threshold.items():
            if key in params and params[key] > limit:
                return f"{key}={params[key]} 超过阈值 {limit}"
        return None

    def _request_approval(self, tool_name, action, params, risk, reason) -> bool:
        """发起审批请求"""
        req = ApprovalRequest(
            id=self._next_id,
            tool_name=tool_name,
            action=action,
            params=params,
            risk_level=risk,
            reason=reason,
        )
        self._next_id += 1

        if self._auto_approve:
            # 测试模式：自动通过
            req.approved = True
            self._history.append(req)
            return True

        # 生产模式：暂停等待人工
        self._pending.append(req)
        print(f"\n🔴 [审批请求 #{req.id}] 需要人工确认")
        print(f"   工具: {req.tool_name}")
        print(f"   操作: {req.action}")
        print(f"   风险: {req.risk_level}")
        print(f"   原因: {req.reason}")
        return False  # 表示需要等待

    def approve(self, request_id: int, note: str = ""):
        """人工批准"""
        req = self._find_pending(request_id)
        req.approved = True
        req.reviewer_note = note or "已批准"
        self._pending.remove(req)
        self._history.append(req)
        print(f"✅ [审批 #{request_id}] 已批准")

    def reject(self, request_id: int, note: str = ""):
        """人工拒绝"""
        req = self._find_pending(request_id)
        req.approved = False
        req.reviewer_note = note or "已拒绝"
        self._pending.remove(req)
        self._history.append(req)
        print(f"❌ [审批 #{request_id}] 已拒绝")

    def _find_pending(self, request_id: int) -> ApprovalRequest:
        for req in self._pending:
            if req.id == request_id:
                return req
        raise ValueError(f"未找到待审批请求 #{request_id}")

    def pending_count(self) -> int:
        return len(self._pending)

    def show_history(self):
        print(f"\n📋 审批历史 ({len(self._history)} 条)")
        for req in self._history:
            status = "✅" if req.approved else "❌"
            print(f"  {status} #{req.id} {req.tool_name} - {req.reviewer_note}")

from dataclasses import dataclass, field
from typing import Callable
import time

@dataclass
class ProductionAgent:
    """生产级 Agent：追踪 + 缓存 + 恢复 + 审批"""
    name: str = "ProductionAgent"
    max_steps: int = 10

    # 子系统（注入依赖）
    tools: dict[str, CachedTool] = field(default_factory=dict)
    error_recovery: ErrorRecovery = field(default_factory=ErrorRecovery)
    approval: HumanApproval = field(default_factory=lambda: HumanApproval(True))
    approval_config: dict = field(default_factory=dict)

    # 运行时状态
    _steps: list[AgentStep] = field(default_factory=list, init=False)
    _total_tokens: int = field(default=0, init=False)
    _paused: bool = field(default=False, init=False)

    def run(self, query: str) -> str:
        """执行一次完整的推理循环"""
        print(f"\n🚀 {self.name} 开始处理: {query[:50]}")

        for step_i in range(self.max_steps):
            if self._paused:
                print("⏸️ 等待人工审批中...")
                break

            t0 = time.time()

            # ── Step 1: LLM 推理（模拟）──
            print(f"\n🧠 Step {step_i+1}: LLM 推理中...")
            # 实际: response = call_llm(query, history)
            thought, tool_name, tool_input = self._mock_llm(query, step_i)
            tokens = len(thought.split()) * 3  # 模拟 token 数

            self._add_step(
                state=AgentState.THINKING,
                thought=thought, tokens_used=tokens,
                duration_ms=(time.time()-t0)*1000
            )

            if not tool_name:
                print(f"✅ LLM 直接回答: {thought[:100]}")
                return thought

            # ── Step 2: 审批检查 ──
            config = self.approval_config.get(tool_name, {"policy": ApprovalPolicy.AUTO})
            can_run = self.approval.check(
                tool_name, config["policy"], tool_input,
                config.get("threshold")
            )
            if not can_run:
                self._paused = True
                continue

            # ── Step 3: 执行工具（带缓存）──
            t1 = time.time()
            try:
                tool = self.tools[tool_name]
                result = tool.execute(**tool_input)
                self._add_step(
                    state=AgentState.TOOL_RESULT,
                    tool_name=tool_name, tool_input=str(tool_input),
                    tool_output=result,
                    duration_ms=(time.time()-t1)*1000
                )
            except Exception as e:
                # ── Step 4: 错误恢复 ──
                recovery = self.error_recovery.handle(str(e))
                self._add_step(
                    state=AgentState.ERROR,
                    tool_name=tool_name,
                    error=str(e),
                    duration_ms=(time.time()-t1)*1000
                )
                if recovery["should_retry"]:
                    continue
                if recovery["action"] == ErrorAction.ESCALATE:
                    return recovery["message"]
                # INFORM_LLM: 把错误加入上下文，下一轮让LLM修正
                query += f"\n[工具错误: {recovery['message']}]"
                continue

        return "⚠️ 达到最大步数"

    # ── 辅助方法 ──
    def _mock_llm(self, query, step):
        """模拟 LLM 返回（实际替换为 API 调用）"""
        if step == 0:
            return ("用户想查询不良资产数据，我需要先查询数据库",
                    "query_amc", {"table": "npl_asset", "condition": "status='active'"})
        else:
            return ("根据查询结果，当前有150条活跃不良资产记录，总金额约2.3亿元",
                    None, None)

    def _add_step(self, **kwargs):
        step = AgentStep(step_num=len(self._steps)+1, **kwargs)
        self._steps.append(step)
        self._total_tokens += step.tokens_used


# ── 运行示例 ──
agent = ProductionAgent(
    name="AMC智能助手",
    tools={
        "query_amc": CachedTool(
            name="query_amc", description="查询AMC数据",
            func=lambda table, condition: f"查询{table}: {condition} → 150条",
            ttl=60,
        ),
    },
    approval_config={
        "query_amc": {"policy": ApprovalPolicy.NEVER},
    },
)

print(f"\n🎯 最终结果: {result}")
设计亮点：ProductionAgent 采用组合优于继承的设计 — 追踪、缓存、恢复、审批都是独立模块，通过构造函数注入。这意味着你可以：单独测试每个模块、替换任意模块而不影响其他、按需开启/关闭功能。
⚠️ 常见陷阱 Top 5
1. 缓存未考虑幂等性
POST 请求（创建资源）不能缓存！只有 GET/幂等操作可以。误缓存写入操作会导致数据不一致。
修复：给 CachedTool 加 idempotent=False 标记，非幂等工具强制跳过缓存。
2. 错误恢复导致无限循环
LLM 持续生成错误参数 → 恢复 → LLM 再次生成相同错误 → 无限循环。
修复：给同类错误设置 max_retries，超限后必须转人工或 abort。
3. Human-in-the-Loop 变成 Human-in-the-Bottleneck
太多工具设为 ALWAYS 审批，人工成为瓶颈，Agent 效率还不如纯脚本。
修复：谨慎使用 ALWAYS，优先用 THRESHOLD；对高频低风险操作用 AUTO。
4. 审批信息不够人类可读
Agent 传给人工的信息全是 JSON 参数，人类看不懂到底要做什么。
修复：让 LLM 生成人类可读的审批摘要（"即将删除编号A001的不良资产记录，涉及金额12,500元"）。
5. 追踪数据不截断导致内存爆炸
AgentStep 把完整的 tool_output 存下来，运行几百步后内存爆炸。
修复：AgentStep.to_dict() 中截断大字段，只保留最近 N 步的完整数据。
