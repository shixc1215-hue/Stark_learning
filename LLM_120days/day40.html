1. retry_with_backoff — 指数退避重试

import time
import functools
import random

def retry_with_backoff(max_retries=3, base_delay=1.0, exceptions=(Exception,)):
    """带指数退避的通用重试装饰器"""
    def decorator(func):
        @functools.wraps(func)  # 保留原函数的 __name__ 和 __doc__
        def wrapper(*args, **kwargs):
            for attempt in range(1, max_retries + 1):
                try:
                    return func(*args, **kwargs)
                except exceptions as e:
                    delay = base_delay * (2 ** (attempt - 1))  # 指数退避
                    if attempt == max_retries:
                        raise  # 最后一次失败，抛出异常
                    print(f"  [retry] 第{attempt}次失败: {e}, {delay}s后重试...")
                    time.sleep(delay)
        return wrapper
    return decorator

# ---- 使用示例 ----
@retry_with_backoff(max_retries=3, base_delay=0.5, exceptions=(ConnectionError, TimeoutError))
def call_rag_service(query: str) -> str:
    """模拟可能失败的 RAG 服务调用"""
    if random.random() < 0.6:  # 60% 概率失败，测试重试逻辑
        raise ConnectionError("RAG 服务连接超时")
    return f"检索结果: 关于 '{query}' 的相关知识..."

2. rate_limit — 令牌桶限流

import threading

class TokenBucket:
    """令牌桶限流器 —— 控制单位时间内的调用频率"""
    def __init__(self, rate: float, capacity: int):
        self.rate = rate          # 每秒补充的令牌数
        self.capacity = capacity  # 桶的最大容量
        self.tokens = capacity    # 当前令牌数
        self.last_refill = time.monotonic()
        self._lock = threading.Lock()

    def acquire(self) -> bool:
        """尝试获取一个令牌，成功返回 True"""
        with self._lock:
            now = time.monotonic()
            # 补充令牌（根据经过的时间）
            elapsed = now - self.last_refill
            self.tokens = min(self.capacity, self.tokens + elapsed * self.rate)
            self.last_refill = now
            if self.tokens >= 1:
                self.tokens -= 1
                return True
            return False

def rate_limit(bucket: TokenBucket, timeout: float = 5.0):
    """限流装饰器：超过速率则等待"""
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            start = time.monotonic()
            while not bucket.acquire():
                if time.monotonic() - start > timeout:
                    raise TimeoutError(f"等待令牌超时 ({timeout}s)")
                time.sleep(0.1)  # 短暂等待
            return func(*args, **kwargs)
        return wrapper
    return decorator

# ---- 使用示例 ----
rag_bucket = TokenBucket(rate=2.0, capacity=5)  # 每秒2次，最多积攒5个

@rate_limit(rag_bucket)
def rag_search(query: str) -> str:
    print(f"  正在检索: {query}")
    return f"[检索结果] {query}"

# 连续调用 —— 超过速率会被限流等待

    rag_search(q)

3. log_tool_call — 工具调用日志

import json
from datetime import datetime

def log_tool_call(log_file="tool_calls.jsonl"):
    """记录工具调用到 JSONL 文件（每行一个 JSON）"""
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            start = time.monotonic()
            try:
                result = func(*args, **kwargs)
                status = "success"
            except Exception as e:
                result = str(e)
                status = "error"

            # 构造日志记录
            log_entry = {
                "timestamp": datetime.now().isoformat(),
                "tool": func.__name__,
                "args": str(args[1:]) if args else {},  # 跳过 self
                "kwargs": kwargs,
                "status": status,
                "duration_ms": round((time.monotonic() - start) * 1000, 2),
                "result_preview": str(result)[:200],
            }
            # 追加写入 JSONL
            with open(log_file, "a", encoding="utf-8") as f:
                f.write(json.dumps(log_entry, ensure_ascii=False) + "\n")

            if status == "error":
                raise
            return result
        return wrapper
    return decorator
装饰器叠加顺序：@retry @rate_limit @log — 从下往上执行。先记日志，再限流，最后重试。就像包裹三层包装纸：最外面是 retry，中间是 rate_limit，最里面是 log。

2. SimpleRAGTool —— 基于 ChromaDB 的 RAG 工具

class SimpleRAGTool:
    """基于 ChromaDB 的 RAG 工具（生产环境替换为实际检索逻辑）"""

    def __init__(self, collection_name="amc_knowledge", persist_dir="./chroma_db"):
        self.collection_name = collection_name
        self.persist_dir = persist_dir
        self._collection = None  # 懒加载
        print(f"[RAGTool] 初始化完成, 集合: {collection_name}")

    def _ensure_collection(self):
        """懒加载：第一次调用时才连接 ChromaDB"""
        if self._collection is None:
            import chromadb
            client = chromadb.PersistentClient(path=self.persist_dir)
            self._collection = client.get_or_create_collection(
                name=self.collection_name,
                metadata={"description": "AMC 不良资产管理知识库"}
            )

    @retry_with_backoff(max_retries=2, base_delay=0.5)
    @rate_limit(TokenBucket(rate=5.0, capacity=10))
    @log_tool_call("rag_tool.jsonl")
    def search(self, query: str, top_k: int = 5) -> list[RAGResult]:
        """
        检索知识库，返回 Top-K 相关文档片段

        Args:
            query: 用户问题或搜索关键词
            top_k: 返回结果数量

        Returns:
            按相关性排序的 RAGResult 列表
        """
        self._ensure_collection()

        # 调用 ChromaDB 检索
        results = self._collection.query(
            query_texts=[query],
            n_results=top_k,
            include=["documents", "metadatas", "distances"]
        )

        # 解析结果
        rag_results = []
        if results["documents"] and results["documents"][0]:
            for i, (doc, meta, dist) in enumerate(
                zip(results["documents"][0],
                    results["metadatas"][0],
                    results["distances"][0])
            ):
                rag_results.append(RAGResult(
                    text=doc,
                    score=round(1 - dist, 4),  # 距离转相似度
                    source=meta.get("source", "unknown"),
                    metadata=meta
                ))
        return rag_results

    def get_sources(self) -> list[str]:
        """获取知识库中的所有文档来源"""
        self._ensure_collection()
        all_meta = self._collection.get(include=["metadatas"])["metadatas"]
        sources = list(set(m.get("source", "unknown") for m in all_meta))
        return sorted(sources)

    def tool_description(self) -> str:
        """生成 Agent 工具描述（告诉 Agent 这个工具能做什么）"""
        sources = ", ".join(self.get_sources()[:5])
        return (
            f"知识库检索工具 - 从 AMC 不良资产管理知识库中检索相关信息。\n"
            f"知识来源: {sources}\n"
            f"适用于: 政策法规查询、业务流程查询、风险分类标准、处置方式说明等。\n"
            f"输入: search(query: str, top_k: int = 5) -> list[RAGResult]"
        )

3. 工具注册表 —— Agent 的工具管理器

class ToolRegistry:
    """
    Agent 工具注册表 —— 统一管理所有工具（RAG工具 + 其他工具）
    类比数仓中的: 元数据管理平台，统一注册所有数据表和API
    """
    def __init__(self):
        self._tools: dict[str, dict] = {}  # {name: {"handler": func, "description": str}}

    def register(self, name: str, handler: callable, description: str = ""):
        """注册一个工具"""
        self._tools[name] = {
            "handler": handler,
            "description": description or handler.__doc__ or ""
        }
        print(f"  [注册] 工具 '{name}' 已注册")

    def get_tool(self, name: str) -> dict | None:
        return self._tools.get(name)

    def list_tools(self) -> list[str]:
        return list(self._tools.keys())

    def build_tools_prompt(self) -> str:
        """生成工具描述 Prompt（供 Agent 理解有哪些工具可用）"""
        lines = ["## 可用工具\n"]
        for name, info in self._tools.items():
            lines.append(f"### {name}\n{info['description']}\n")
        return "\n".join(lines)

    def call(self, name: str, **kwargs):
        """调用指定工具"""
        tool = self._tools.get(name)
        if not tool:
            return f"错误: 工具 '{name}' 不存在"
        return tool["handler"](**kwargs)

完整实现：RAGAgent

import re
import json
from dataclasses import dataclass, field

@dataclass
class AgentStep:
    """Agent 单步执行记录"""
    step_num: int
    thought: str      # 思考过程
    action: str       # 要执行的动作/工具
    action_input: dict  # 工具输入参数
    observation: str  # 工具返回结果
    token_count: int = 0

class RAGAgent:
    """
    RAG 增强型 Agent —— 结合 ReAct 推理 + RAG 知识检索 + 多工具调用

    架构流程:
      1. 接收用户问题
      2. 构建系统 Prompt（含工具描述 + RAG 知识库信息）
      3. ReAct 循环:
         a. Thought: 分析问题，决定下一步
         b. Action:   选择工具（rag_search / db_query / calculator）
         c. Observe:  执行工具，获取结果
         d. 判断是否需要继续（是否已有足够信息回答）
      4. 生成最终答案（基于所有工具调用结果）
    """

    def __init__(
        self,
        tool_registry: ToolRegistry,
        api_key: str = "your-api-key",
        model: str = "glm-4-flash",
        max_steps: int = 6,
    ):
        self.registry = tool_registry
        self.api_key = api_key
        self.model = model
        self.max_steps = max_steps
        self.history: list[dict] = []   # 对话历史
        self.steps: list[AgentStep] = []  # 执行轨迹

    def _build_system_prompt(self) -> str:
        """构建系统 Prompt —— 包含 RAG 工具说明和决策指南"""
        tools_desc = self.registry.build_tools_prompt()

        return f"""你是一个 AMC（资产管理公司）智能助手，可以查询知识库、执行SQL、进行计算。

{tools_desc}

## 回答规则
1. 需要查询 AMC 政策/法规/流程时，使用 rag_search 工具
2. 需要查询业务数据时，使用 db_query 工具
3. 需要做数值计算时，使用 calculator 工具
4. 可以组合使用多个工具
5. 每次回复严格使用以下 JSON 格式：

思考阶段（需要更多信息时）:
{{"thought": "你的思考过程", "action": "工具名", "action_input": {{"参数名": "参数值"}}}}

最终回答阶段:
{{"thought": "总结回答", "answer": "最终回答内容"}}

重要：只输出一个 JSON 对象，不要有任何多余文字。"""

    def _parse_response(self, response: str) -> dict:
        """解析 LLM 返回的 JSON 响应"""
        # 尝试提取 JSON
        json_match = re.search(r'\{.*\}', response, re.DOTALL)
        if not json_match:
            return {"thought": "解析失败", "answer": "抱歉，处理过程中出现错误"}
        try:
            return json.loads(json_match.group())
        except json.JSONDecodeError:
            return {"thought": "JSON解析失败", "answer": "抱歉，处理过程中出现格式错误"}

    def _call_llm(self, messages: list[dict]) -> str:
        """
        调用 LLM API（使用智谱 API 作为示例）
        生产环境替换为实际的 API 调用
        """
        # 模拟 LLM 响应（实际使用时替换为 requests.post 调用）
        # 这里为了演示可离线运行，用规则匹配模拟
        last_msg = messages[-1]["content"] if messages else ""

        # 简单的关键词路由模拟
        if "不良资产" in last_msg or "政策" in last_msg or "法规" in last_msg:
            return json.dumps({
                "thought": "用户在询问 AMC 相关政策，需要查询知识库",
                "action": "rag_search",
                "action_input": {"query": last_msg[:100], "top_k": 3}
            })
        elif "SELECT" in last_msg.upper() or "数据" in last_msg:
            return json.dumps({
                "thought": "需要查询业务数据",
                "action": "db_query",
                "action_input": {"sql": last_msg[:200]}
            })
        elif any(c in last_msg for c in "+-*/="):
            return json.dumps({
                "thought": "用户需要进行数学计算",
                "action": "calculator",
                "action_input": {"expression": "1000 * 0.05"}
            })
        else:
            return json.dumps({
                "thought": "已经收集到足够信息，可以回答用户",
                "answer": "根据查询结果，这是您的回答内容。"
            })

    def run(self, question: str) -> str:
        """
        执行 Agent 主循环 —— ReAct（Reasoning + Acting）

        Args:
            question: 用户问题

        Returns:
            Agent 的最终回答
        """
        print(f"\n{'='*60}")
        print(f"🤖 Agent 开始处理: {question}")
        print(f"{'='*60}")

        self.steps = []
        messages = [
            {"role": "system", "content": self._build_system_prompt()},
            {"role": "user", "content": question}
        ]

        for step_i in range(1, self.max_steps + 1):
            print(f"\n--- Step {step_i} ---")

            # 调用 LLM
            llm_response = self._call_llm(messages)
            parsed = self._parse_response(llm_response)

            thought = parsed.get("thought", "")

            # 判断是否是最终回答
            if "answer" in parsed:
                answer = parsed["answer"]
                print(f"  💭 思考: {thought}")
                print(f"  ✅ 最终回答: {answer}")
                self.steps.append(AgentStep(
                    step_num=step_i, thought=thought,
                    action="final_answer", action_input={},
                    observation=answer
                ))
                messages.append({"role": "assistant", "content": llm_response})
                self.history = messages
                return answer

            # 解析工具调用
            action = parsed.get("action", "")
            action_input = parsed.get("action_input", {})
            print(f"  💭 思考: {thought}")
            print(f"  🔧 工具: {action}({action_input})")

            # 执行工具
            try:
                observation = self.registry.call(action, **action_input)
                if isinstance(observation, list):
                    # RAG 结果格式化
                    obs_text = "\n".join(
                        f"[{i+1}] (分数:{r.score:.3f}) {r.text[:100]}..."
                        for i, r in enumerate(observation)
                    )
                else:
                    obs_text = str(observation)
            except Exception as e:
                obs_text = f"工具执行错误: {e}"

            print(f"  📋 结果: {obs_text[:150]}...")

            # 记录步骤
            self.steps.append(AgentStep(
                step_num=step_i, thought=thought,
                action=action, action_input=action_input,
                observation=obs_text[:500]
            ))

            # 将工具结果追加到消息（让 LLM 看到结果）
            messages.append({"role": "assistant", "content": llm_response})
            messages.append({
                "role": "user",
                "content": f"工具 {action} 的执行结果:\n{obs_text}\n\n请根据以上结果继续分析，或直接给出最终回答。"
            })

        return "抱歉，达到最大推理步数仍未得出结论。"

    def show_trace(self):
        """显示完整执行轨迹"""
        print(f"\n📋 执行轨迹 ({len(self.steps)} 步):")
        print("-" * 60)
        for step in self.steps:
            print(f"  Step {step.step_num}: [{step.action}]")
            print(f"    思考: {step.thought}")
            if step.action != "final_answer":
                print(f"    结果: {step.observation[:100]}...")
        print("-" * 60)

关键设计要点：
  • ReAct 循环：Agent 不是"想好了再动手"，而是"边想边做边观察"—— 每一步都根据上一步的结果调整策略
  • 工具结果回传：工具执行结果会追加到对话历史，让 LLM 在下一步推理时能看到
  • 自主决策：Agent 自己决定何时查知识、何时查数据、何时直接回答——不需要人工干预

# 智谱 API 支持 JSON Mode，强制 LLM 返回合法 JSON
import requests
import json

def call_llm_json_mode(prompt: str, api_key: str) -> dict:
    """使用 JSON Mode 调用 LLM，确保返回合法 JSON"""
    url = "https://open.bigmodel.cn/api/paas/v4/chat/completions"
    headers = {
        "Authorization": f"Bearer {api_key}",
        "Content-Type": "application/json"
    }
    payload = {
        "model": "glm-4-flash",
        "messages": [{"role": "user", "content": prompt}],
        "response_format": {  # 关键：指定输出格式为 JSON
            "type": "json_object"
        }
    }
    response = requests.post(url, headers=headers, json=payload, timeout=30)
    result = response.json()
    content = result["choices"][0]["message"]["content"]
    return json.loads(content)  # 保证能正常解析为 dict

2. Few-Shot Prompting —— 少样本提示

def build_fewshot_prompt(question: str) -> str:
    """构建 Few-Shot Prompt —— 用示例教会 LLM 期望的输出格式"""

    # 准备 3 个示例（input → output 对）
    examples = [
        {
            "input": "查询所有不良资产数量",
            "thought": "需要查询 AMC 业务数据，使用 db_query 工具",
            "output": '{"thought": "需要查询业务数据", "action": "db_query", "action_input": {"sql": "SELECT COUNT(*) FROM npl_assets"}}'
        },
        {
            "input": "五级分类标准是什么",
            "thought": "这是 AMC 政策知识问题，使用 rag_search 检索知识库",
            "output": '{"thought": "查询知识库中的分类标准", "action": "rag_search", "action_input": {"query": "五级分类标准", "top_k": 3}}'
        },
        {
            "input": "计算不良率，总资产1亿，不良2000万",
            "thought": "简单的百分比计算，使用 calculator",
            "output": '{"thought": "计算不良率", "action": "calculator", "action_input": {"expression": "20000000 / 100000000 * 100"}}'
        },
    ]

    # 组装 Prompt
    parts = ["你是一个 AMC 智能助手。根据用户问题，选择合适的工具并输出 JSON。\n"]
    for i, ex in enumerate(examples, 1):
        parts.append(f"示例 {i}:")
        parts.append(f"  用户问题: {ex['input']}")
        parts.append(f"  回答: {ex['output']}\n")
    parts.append(f"现在请处理:")
    parts.append(f"  用户问题: {question}")
    parts.append(f"  回答:")

    return "\n".join(parts)


# 使用 Few-Shot Prompt 让 Agent 输出更稳定
prompt = build_fewshot_prompt("不良资产的处置方式有哪些")
print(prompt)
# Few-Shot 让 LLM 通过示例"学会"输出格式，比纯指令描述更稳定

3. Chain-of-Thought (CoT) —— 思维链提示

def build_cot_prompt(question: str, context: str = "") -> str:
    """
    Chain-of-Thought Prompt —— 让 LLM 逐步推理
    核心: 在 Prompt 中明确要求"一步步思考"
    """
    system = """你是一个 AMC 数据分析专家。回答问题时请遵循以下步骤：

第一步：理解问题 - 明确用户想知道什么
第二步：分析数据 - 识别关键数据点和计算逻辑
第三步：执行计算 - 逐步进行数值计算
第四步：得出结论 - 给出明确的数据结论和建议

请严格按这四个步骤组织你的回答。"""

    messages = [
        {"role": "system", "content": system},
    ]
    if context:
        messages.append({"role": "system", "content": f"参考资料:\n{context}"})
    messages.append({"role": "user", "content": question})

    return messages


# 对比：普通 Prompt vs CoT Prompt
# 普通 Prompt: "不良率是多少？" → LLM 可能直接猜一个数字
# CoT Prompt:   要求逐步推理 → LLM 会先列出已知数据、再计算、再给出结论
# 结果更准确、过程可追溯、适合 Agent 的 Thought 输出

4. Prompt 模板管理 —— Jinja2 风格

class PromptTemplate:
    """
    Prompt 模板管理器 —— 类比数仓中的 SQL 模板引擎
    支持变量替换、条件区块、Few-Shot 注入
    """

    def __init__(self, template: str, input_variables: list[str]):
        self.template = template
        self.input_variables = input_variables

    def format(self, **kwargs) -> str:
        """简单的 {variable} 占位符替换"""
        result = self.template
        for key, value in kwargs.items():
            result = result.replace(f"{{{key}}}", str(value))
        # 检查是否有未替换的变量
        unreplaced = set(self.input_variables) - set(kwargs.keys())
        if unreplaced:
            raise ValueError(f"缺少变量: {unreplaced}")
        return result

高级 Prompt 陷阱：
  • 不要过度 Few-Shot：示例超过 5 个会挤占上下文窗口，反而降低效果。一般 2-3 个最佳
  • CoT 不是万能的：简单问题（分类、提取）用 CoT 反而增加延迟。只对推理题和计算题用
  • JSON Mode ≠ 结构化理解：LLM 可能返回合法 JSON 但内容不对。关键业务要做二次校验

⚠️ 常见陷阱 Top 5
陷阱 1：RAG 每次都调用
Agent 拿到问题就先查 RAG，不管需不需要。正确做法：让 Agent 自主判断是否需要检索。简单常识问题直接回答，只有不确定时才查知识库。
陷阱 2：检索结果全部塞进 Prompt
检索返回 50 条结果全部塞进上下文 → Token 爆炸。正确做法：Top-K=3~5，超过 2000 字符就截断，优先保留高分结果。
陷阱 3：工具描述写得太简单
工具描述写"搜索知识库"，Agent 不知道什么场景该用。正确做法：描述中包含适用场景、输入格式、输出格式（参考 rag_tool.tool_description()）。
陷阱 4：装饰器顺序搞反
@retry @rate_limit @log 和 @log @rate_limit @retry 效果不同。重试在最外层，确保限流/日志失败也会被重试；日志在最内层，确保每次实际调用都被记录。
陷阱 5：Few-Shot 示例和实际输出格式不一致
示例里 action_input 用了双引号，但实际输出用了单引号 → JSON 解析失败。确保示例和预期输出格式完全一致。
