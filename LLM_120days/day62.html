1 dataclass + field 默认值工厂

客服系统的数据模型频繁使用可变默认值。dataclass 的 field(default_factory=...) 解决了 Python 经典的"可变默认值陷阱"。

from dataclasses import dataclass, field
from datetime import datetime
from typing import Optional
from enum import Enum

class IntentType(Enum):
    """用户意图分类枚举"""
    QUERY_BALANCE = "query_balance"      # 查询余额
    QUERY_RISK = "query_risk"            # 查询风险等级
    REPORT_ANOMALY = "report_anomaly"    # 举报异常
    POLICY_FAQ = "policy_faq"            # 政策问答
    TRANSFER_HUMAN = "transfer_human"    # 转人工
    UNKNOWN = "unknown"                 # 未知意图

class SessionState(Enum):
    """对话会话状态"""
    ACTIVE = "active"
    WAITING_HUMAN = "waiting_human"     # 等待人工
    CLOSED = "closed"
    TIMEOUT = "timeout"

@dataclass
class Message:
    """单条消息"""
    role: str              # "user" / "assistant" / "system"
    content: str
    timestamp: datetime = field(default_factory=datetime.now)
    intent: Optional[IntentType] = None     # 仅 user 消息有意图
    confidence: float = 0.0                  # 意图置信度

@dataclass
class ChatSession:
    """对话会话"""
    session_id: str
    user_id: str
    messages: list[Message] = field(default_factory=list)
    state: SessionState = SessionState.ACTIVE
    created_at: datetime = field(default_factory=datetime.now)
    metadata: dict = field(default_factory=dict)

    def add_message(self, role: str, content: str,
                    intent: IntentType | None = None,
                    confidence: float = 0.0) -> Message:
        """添加消息到会话"""
        msg = Message(role=role, content=content,
                      intent=intent, confidence=confidence)
        self.messages.append(msg)
        return msg

    @property
    def message_count(self) -> int:
        return len(self.messages)

    def get_recent(self, n: int = 10) -> list[Message]:
        """获取最近 n 条消息（滑动窗口）"""
        return self.messages[-n:]

🔑 field(default_factory=...) 是 dataclass 的核心安全机制。 如果直接写 messages: list = []，所有实例会共享同一个 list（Python 经典坑）。 用 field(default_factory=list) 则每次实例化都创建新 list。
2 正则表达式提取实体

意图分类确定了"用户想干什么"，实体提取确定"用户在说谁/什么"。正则是最直接的方式。

import re
from dataclasses import dataclass

@dataclass
class Entity:
    """提取的实体"""
    entity_type: str    # 实体类型： debtor_id, amount, date, asset_type ...
    value: str          # 原始值
    normalized: str | None = None  # 标准化后的值（如日期格式化）
    start: int = 0      # 在原文中的起始位置
    end: int = 0        # 结束位置

class EntityExtractor:
    """基于正则的实体提取器"""

    def __init__(self):
        # 定义 AMC 业务实体的正则规则
        # 按优先级排序：更具体的规则放前面
        self.rules = [
            {
                "name": "debtor_id",
                # AMC 债务人编号：字母+数字组合，如 D2024001
                "pattern": r'\b([A-Z]\d{7,10})\b',
                "description": "债务人编号",
            },
            {
                "name": "amount",
                # 金额：支持小数、逗号分隔，如 1,234,567.89
                "pattern": r'(\d{1,3}(?:,\d{3})*(?:\.\d{1,2})?)\s*(?:万元|元|块钱)',
                "description": "金额",
            },
            {
                "name": "date",
                # 日期：多种格式 YYYY-MM-DD / YYYY/MM/DD / YYYY年MM月DD日
                "pattern": r'(\d{4})[年/\-](\d{1,2})[月/\-](\d{1,2})[日号]?',
                "description": "日期",
            },
            {
                "name": "risk_level",
                # 风险等级：正常/关注/次级/可疑/损失
                "pattern": r'(正常|关注|次级|可疑|损失)',
                "description": "五级分类风险等级",
            },
            {
                "name": "asset_type",
                # 资产类型关键词
                "pattern": r'(贷款|信用卡|票据|垫款|债券|股权)',
                "description": "资产类型",
            },
        ]

    def extract(self, text: str) -> list[Entity]:
        """从文本中提取所有实体"""
        entities = []
        for rule in self.rules:
            for match in re.finditer(rule["pattern"], text):
                entity = Entity(
                    entity_type=rule["name"],
                    value=match.group(0),
                    start=match.start(),
                    end=match.end(),
                )
                # 日期标准化
                if rule["name"] == "date":
                    entity.normalized = (
                        f"{match.group(1)}-{match.group(2).zfill(2)}"
                        f"-{match.group(3).zfill(2)}"
                    )
                # 金额标准化（去掉逗号和单位）
                elif rule["name"] == "amount":
                    raw = match.group(1).replace(",", "")
                    entity.normalized = raw
                entities.append(entity)
        # 按出现位置排序
        return sorted(entities, key=lambda e: e.start)

3 编辑距离 + 相似度匹配

用户输入常与标准表述有偏差（错别字、口语化），编辑距离用于模糊匹配意图关键词。

def levenshtein_distance(s1: str, s2: str) -> int:
    """计算两个字符串的编辑距离（Levenshtein）
    支持三种操作：插入、删除、替换
    """
    m, n = len(s1), len(s2)
    # dp[i][j] = s1[:i] 和 s2[:j] 的最小编辑距离
    dp = [[0] * (n + 1) for _ in range(m + 1)]

    # 初始化：空串到另一个串的距离 = 长度
    for i in range(m + 1):
        dp[i][0] = i
    for j in range(n + 1):
        dp[0][j] = j

    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if s1[i-1] == s2[j-1]:
                # 字符相同，不需要操作
                dp[i][j] = dp[i-1][j-1]
            else:
                # 三种操作取最小值 + 1
                dp[i][j] = 1 + min(
                    dp[i-1][j],      # 删除 s1[i-1]
                    dp[i][j-1],      # 插入 s2[j-1]
                    dp[i-1][j-1],    # 替换
                )
    return dp[m][n]

def similarity_ratio(s1: str, s2: str) -> float:
    """相似度 = 1 - 编辑距离 / max(len1, len2)
    返回 0~1 的浮点数，1 表示完全相同
    """
    if not s1 or not s2:
        return 0.0
    dist = levenshtein_distance(s1, s2)
    max_len = max(len(s1), len(s2))
    return 1.0 - dist / max_len

# 测试：模糊匹配意图关键词
keywords = ["余额查询", "风险等级", "不良率", "转让审批"]
queries = ["余额查询", "余额茶询", "风险等级", "风显等级", "不良利率"]

for q in queries:
    best_kw, best_sim = "", 0.0
    for kw in keywords:
        sim = similarity_ratio(q, kw)
        if sim > best_sim:
            best_sim, best_kw = sim, kw
    status = "✅ 精确" if best_sim >= 0.9 else "⚠️ 模糊" if best_sim >= 0.7 else "❌ 未匹配"
    print(f"  '{q}' → '{best_kw}' (相似度: {best_sim:.2f}) {status}")
⚠️ 注意：编辑距离对长度敏感。短字符串（2-3字）"余额"和"余额茶"的相似度 = 1 - 1/3 = 0.67，看起来低但只差1个字。 实际项目建议配合 前缀匹配 + 关键词命中 组合判断，而非仅靠相似度阈值。

1 三层意图识别策略
策略原理优点缺点适用
规则匹配 关键词/正则/编辑距离 快、确定性强、可解释 覆盖面窄、维护成本高 高频标准意图（转人工/常见FAQ）
LLM 分类 Prompt + JSON输出 覆盖面广、理解语义 有延迟、偶尔误判 复杂语义、长句、模糊表达
混合策略 规则优先 → LLM兜底 兼顾速度与覆盖 复杂度稍高 生产环境推荐 ✅
2 混合意图分类器完整实现
import json
import re
from dataclasses import dataclass, field
from enum import Enum
from typing import Optional
from zhipuai import ZhipuAI  # 或 openai / 其他 LLM SDK

class IntentType(Enum):
    QUERY_BALANCE = "query_balance"
    QUERY_RISK = "query_risk"
    QUERY_NPL_RATIO = "query_npl_ratio"
    REPORT_ANOMALY = "report_anomaly"
    POLICY_FAQ = "policy_faq"
    TRANSFER_HUMAN = "transfer_human"
    GREETING = "greeting"
    UNKNOWN = "unknown"

@dataclass
class IntentResult:
    """意图识别结果"""
    intent: IntentType
    confidence: float
    strategy: str       # "rule" / "llm" / "fallback"
    raw_response: str = ""

class IntentClassifier:
    """混合意图分类器：规则优先 + LLM 兜底"""

    # ── 规则层：关键词映射 ──
    RULE_KEYWORDS: dict[IntentType, list[str]] = {
        IntentType.TRANSFER_HUMAN: [
            "转人工", "人工客服", "真人", "投诉",
            "找人工", "人工服务", "不智能",
        ],
        IntentType.GREETING: [
            "你好", "在吗", "hello", "hi",
            "在不在", "帮我查", "请问",
        ],
        IntentType.QUERY_BALANCE: [
            "余额", "多少钱", "剩余", "贷款余额",
            "欠款", "还剩多少", "账面",
        ],
        IntentType.QUERY_RISK: [
            "风险", "等级", "分类", "五级",
            "正常吗", "次级", "可疑", "损失",
        ],
        IntentType.QUERY_NPL_RATIO: [
            "不良率", "不良贷款率", "NPL",
            "违约率", "逾期率",
        ],
        IntentType.REPORT_ANOMALY: [
            "举报", "异常", "错误", "数据不对",
            "有问题", "金额异常", "余额为负",
        ],
    }

    def __init__(self, api_key: str = "", model: str = "glm-4-flash"):
        self.model = model
        # 实际使用: self.client = ZhipuAI(api_key=api_key)
        self.client = None  # mock 模式
        # 预编译正则：性能优化，避免每次调用都编译
        self._compiled_rules = self._compile_rules()

    def _compile_rules(self) -> list[tuple[IntentType, re.Pattern]]:
        """预编译所有正则表达式"""
        compiled = []
        for intent, keywords in self.RULE_KEYWORDS.items():
            # 用 | 连接同意图的关键词
            pattern = "|".join(re.escape(kw) for kw in keywords)
            compiled.append((intent, re.compile(pattern)))
        return compiled

    def classify_rule(self, text: str) -> Optional[IntentResult]:
        """规则匹配层：遍历预编译正则，命中即返回"""
        text_lower = text.lower().strip()
        for intent, pattern in self._compiled_rules:
            if pattern.search(text_lower):
                # 计算置信度：关键词占文本长度的比例（简单启发式）
                match = pattern.search(text_lower)
                confidence = min(0.95, 0.7 + len(match.group()) / len(text_lower) * 0.3)
                return IntentResult(
                    intent=intent,
                    confidence=round(confidence, 2),
                    strategy="rule",
                )
        return None  # 规则未命中

    def classify_llm(self, text: str) -> IntentResult:
        """LLM 分类层：语义理解兜底"""
        intent_descriptions = "\n".join(
            f"- {item.value}: {item.name}"
            for item in IntentType if item != IntentType.UNKNOWN
        )
        system_prompt = f"""你是一个金融客服系统的意图分类器。
根据用户输入，判断意图类型。

可选意图类型：
{intent_descriptions}

输出严格的JSON格式（不要有其他文字）：
{{"intent": "意图value", "confidence": 0.0-1.0, "reason": "一句话理由"}}"""

        # 实际调用 LLM
        # response = self.client.chat.completions.create(
        #     model=self.model,
        #     messages=[
        #         {"role": "system", "content": system_prompt},
        #         {"role": "user", "content": text},
        #     ],
        #     temperature=0.1,  # 低温度减少随机性
        # )
        # raw = response.choices[0].message.content

        # ── mock LLM 响应 ──
        raw = self._mock_llm(text)

        # 三层容错解析 JSON
        result = self._parse_intent_json(raw, text)
        return result

    def classify(self, text: str) -> IntentResult:
        """主入口：规则优先 → LLM 兜底"""
        # 第一步：规则快速匹配
        rule_result = self.classify_rule(text)
        if rule_result and rule_result.confidence >= 0.8:
            return rule_result

        # 第二步：LLM 语义理解
        try:
            llm_result = self.classify_llm(text)
            # 如果 LLM 和规则都命中，取置信度更高的
            if rule_result and llm_result.confidence < rule_result.confidence:
                return rule_result
            return llm_result
        except Exception as e:
            # 第三步：异常兜底
            if rule_result:
                return rule_result
            return IntentResult(
                intent=IntentType.UNKNOWN,
                confidence=0.3,
                strategy="fallback",
                raw_response=f"LLM异常: {e}",
            )

    def _parse_intent_json(self, raw: str, text: str) -> IntentResult:
        """三层容错解析 LLM 返回的 JSON"""
        # 尝试1：直接解析
        try:
            data = json.loads(raw)
            intent = IntentType(data["intent"])
            return IntentResult(
                intent=intent,
                confidence=float(data.get("confidence", 0.7)),
                strategy="llm",
                raw_response=data.get("reason", ""),
            )
        except (json.JSONDecodeError, KeyError, ValueError):
            pass

        # 尝试2：提取 JSON 块（LLM 可能包裹在 ```json 中）
        json_match = re.search(r'\{[^}]+\}', raw)
        if json_match:
            try:
                data = json.loads(json_match.group())
                intent = IntentType(data["intent"])
                return IntentResult(
                    intent=intent, confidence=float(data.get("confidence", 0.7)),
                    strategy="llm", raw_response=data.get("reason", ""),
                )
            except (json.JSONDecodeError, KeyError, ValueError):
                pass

        # 尝试3：关键词猜测兜底
        return IntentResult(
            intent=IntentType.UNKNOWN, confidence=0.3,
            strategy="fallback", raw_response=raw[:100],
        )

    def _mock_llm(self, text: str) -> str:
        """模拟 LLM 响应（实际项目替换为真 API 调用）"""
        mock_map = {
            "查一下": '{"intent":"query_balance","confidence":0.85,"reason":"用户想查询余额"}',
            "这个编号D2024": '{"intent":"query_balance","confidence":0.9,"reason":"通过编号查询"}',

            "数据好像不对": '{"intent":"report_anomaly","confidence":0.82,"reason":"反馈数据异常"}',
        }
        for key, resp in mock_map.items():
            if key in text:
                return resp
        return '{"intent":"policy_faq","confidence":0.7,"reason":"通用咨询"}'

1 文本预处理
import re
import html

class TextPreprocessor:
    """文本预处理：清洗、标准化、分段"""

    @staticmethod
    def clean(text: str) -> str:
        """清洗文本"""
        # 1. HTML 解码（处理 & < 等转义）
        text = html.unescape(text)
        # 2. 去除多余空白
        text = re.sub(r'\s+', ' ', text).strip()
        # 3. 去除特殊控制字符（保留中文、英文、数字、标点）
        text = re.sub(r'[\x00-\x08\x0b\x0c\x0e-\x1f\x7f]', '', text)
        return text

    @staticmethod
    def normalize(text: str) -> str:
        """标准化：全角→半角、繁简等"""
        # 全角数字/字母 → 半角
        fullwidth = "０１２３４５６７８９ＡＢＣＤＥＦＧＨＩＪＫＬＭＮＯＰＱＲＳＴＵＶＷＸＹＺ"
        halfwidth = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ"
        table = str.maketrans(fullwidth, halfwidth)
        text = text.translate(table)
        # 统一引号
        text = text.replace('"', '"').replace('"', '"')
        return text

    def process(self, text: str) -> str:
        """完整预处理流水线"""
        text = self.clean(text)
        text = self.normalize(text)
        return text

@dataclass
class ProcessedRequest:
    """处理后的用户请求"""
    original_text: str         # 原始文本
    cleaned_text: str          # 预处理后文本
    intent: IntentResult      # 意图识别结果
    entities: list[Entity]    # 提取的实体
    session_id: str = ""

class RequestProcessor:
    """请求处理器：串联 预处理→意图→实体"""

    def __init__(self, api_key: str = ""):
        self.preprocessor = TextPreprocessor()
        self.classifier = IntentClassifier(api_key=api_key)
        self.entity_extractor = EntityExtractor()

    def process(self, user_input: str,
                session_id: str = "") -> ProcessedRequest:
        """处理一条用户输入"""
        # Step 1: 预处理
        cleaned = self.preprocessor.process(user_input)

        # Step 2: 意图识别
        intent = self.classifier.classify(cleaned)

        # Step 3: 实体提取（不是所有意图都需要）
        entities = []
        if intent.intent in (
            IntentType.QUERY_BALANCE,
            IntentType.QUERY_RISK,
            IntentType.REPORT_ANOMALY,
        ):
            entities = self.entity_extractor.extract(cleaned)

        return ProcessedRequest(
            original_text=user_input,
            cleaned_text=cleaned,
            intent=intent,
            entities=entities,
            session_id=session_id,
        )

🚨 常见陷阱 Top 5 ⏱ 3min
陷阱1：意图粒度过细 — 30个意图 → LLM 频繁误判。 建议：先用 5-8 个大意图，后期根据真实数据拆分。
陷阱2：只依赖 LLM，没有规则兜底 — "转人工" 这种简单请求不应该打 LLM。 建议：高频 + 确定性意图走规则，LLM 只处理语义模糊场景。
陷阱3：正则实体提取没处理边界 — "D2024001234" 被 "D202400123" 先匹配走了。 建议：按长度降序排列规则，或用负向前瞻 (?!\d)。
陷阱4：JSON 解析太脆弱 — LLM 返回带 markdown 代码块、多余空格、BOM 头。 建议：三层容错（直接解析 → 正则提取 → 关键词兜底）。
陷阱5：可变默认值 — dataclass 的 list/dict 默认值必须用 field(default_factory=...)。 否则所有实例共享同一个对象（经典 Python 坑）。
✅ 最佳实践速查表
领域最佳实践
意图分类规则优先(高频确定) → LLM兜底(语义模糊) → UNKNOWN兜底
置信度阈值规则 ≥ 0.8 直接采纳；LLM ≥ 0.7 采纳；否则走澄清追问
实体提取正则快但脆 → 生产环境可加 NER 模型或 LLM 辅助

正则优化启动时预编译(compile)，避免每次请求重复编译
dataclass所有可变默认值用 field(default_factory=...)
测试策略构造 20+ 测试用例，覆盖所有意图+边界+异常输入

📅 明日预告：Day 63 — 对话管理 + 回复生成

今天搭好了"识别"和"提取"，明天进入对话的核心：多轮上下文管理和回复生成。

  • SessionManager：会话生命周期管理（创建/活跃/超时/关闭）+ 多轮上下文窗口
  • ContextBuilder：组装对话历史给 LLM（滑动窗口 + Token 预算控制）

  • Handler 策略模式：每种意图对应一个处理器，统一接口
  • 人工接管：低置信度或复杂场景自动触发转人工流程
