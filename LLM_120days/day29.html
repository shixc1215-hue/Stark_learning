📖 一、为什么需要文本分块？ 15 min
核心概念：LLM 有上下文窗口限制（如 128K tokens），但一份 PDF 可能几百页。即使窗口够大，把整份文档塞进去会导致：检索噪音多、回答不聚焦、token 费用暴涨。分块 = 把大文档切成有意义的"知识碎片"，让检索系统精准命中。

1.2 分块的核心矛盾

因素块太大（如整章）块太小（如每句）
检索精度❌ 噪音多，无关信息多❌ 缺少上下文，语义不完整
回答质量LLM 难以聚焦关键信息LLM 缺乏足够的背景知识
Token 成本❌ 每次检索塞大量无用 token✅ 节省 token
适用场景需要全面分析的综述题精确匹配的简单事实查询
黄金法则：好的分块 = 在 "语义完整性" 和 "粒度精细度" 之间找平衡。一"块"应该是一段自包含、有意义的信息单元。

2.3 生成器表达式（内存高效的文本处理）

为什么用生成器？处理大文档时，[f(x) for x in data] 列表推导会一次性把所有结果加载到内存。生成器表达式 (f(x) for x in data) 是"懒加载"的——一次只处理一个元素，内存占用恒定。 
# ========== 列表推导 vs 生成器表达式 ==========
text = "Python 是一门优秀的编程语言，特别适合数据开发和 AI 工程"

# 列表推导 — 一次性生成全部结果，占内存
words_list = [w for w in text.split() if len(w) > 1]
print(type(words_list))  # <class 'list'>

# 生成器表达式 — 惰性求值，逐个产出
words_gen = (w for w in text.split() if len(w) > 1)
print(type(words_gen))   # <class 'generator'>

# 惰性的体现：只有在"用"的时候才计算
for word in words_gen:   # 每次循环才产出下一个值
    print(word)

# ========== 分块场景中的实际应用 ==========
def read_lines(filepath):
    """逐行读取大文件（生成器，不一次性加载到内存）"""
    with open(filepath, "r", encoding="utf-8") as f:
        for line in f:
            yield line.strip()

# 使用生成器逐行处理
line_count = sum(1 for line in read_lines("large_report.txt"))  # 不占内存
print(f"文件共 {line_count} 行")

# 也可以用生成器做分块预处理
chunks = ("段落" + str(i) for i in range(1000))  # 不占内存
next(chunks)  # '段落0'
next(chunks)  # '段落1'

def chunk_by_size(text, chunk_size, overlap=0):
    """按固定大小分块（后面会完整实现）"""

    start = 0
    while start < len(text):
        end = start + chunk_size
        chunks.append(text[start:end])
        start = end - overlap  # 重叠部分
    return chunks

3.1 策略一：固定字符分块（Fixed-size Chunking）

最简单的策略：按固定字符数切割，支持重叠（overlap）。

def fixed_size_chunks(
    text: str,
    chunk_size: int = 500,
    overlap: int = 50,
    separator: str = " "
) -> list[str]:
    """
    固定大小分块（最基础的策略）

    参数:
        text: 原始文本
        chunk_size: 每块的目标字符数
        overlap: 块之间的重叠字符数（防止信息被切断）
        separator: 分割单位，" " 按词切（英文），"" 按字符切（中文）

    返回:
        分块列表
    """
    if chunk_size <= overlap:
        raise ValueError(f"chunk_size({chunk_size}) 必须 > overlap({overlap})")

    if separator:  # 英文：按词分割后拼接
        words = text.split(separator)
        chunks = []
        current_chunk = []
        current_len = 0

        for word in words:
            word_len = len(word) + len(separator)  # 词 + 分隔符
            # 如果加入这个词会超限，且当前块不为空，先存起来
            if current_len + word_len > chunk_size and current_chunk:
                chunks.append(separator.join(current_chunk))
                # 保留 overlap 部分的词（从尾部取）
                overlap_words = []
                overlap_len = 0
                for w in reversed(current_chunk):
                    w_len = len(w) + len(separator)
                    if overlap_len + w_len > overlap:
                        break
                    overlap_words.insert(0, w)
                    overlap_len += w_len
                current_chunk = overlap_words
                current_len = overlap_len
            current_chunk.append(word)
            current_len += word_len

        # 别忘了最后一部分
        if current_chunk:
            chunks.append(separator.join(current_chunk))
        return chunks
    else:  # 中文/通用：直接按字符位置切
        chunks = []
        start = 0
        while start < len(text):
            end = start + chunk_size
            chunks.append(text[start:end])
            start = end - overlap
        return chunks

3.2 策略二：按分隔符分块（Separator-based / Recursive Chunking）

按"自然的分隔符"切分——段落、标题、句子——优先级从大到小递归。

def recursive_split(
    text: str,
    separators: list[str] = ["\n\n", "\n", "。", "，", " "],
    chunk_size: int = 500,
) -> list[str]:
    """
    递归分隔符分块（LangChain 的 RecursiveCharacterTextSplitter 的简化版）

    思路：从最"粗"的分隔符开始切（如段落），
    如果某块仍然太大，就用更"细"的分隔符继续切（如句子），
    直到每块都在 chunk_size 以内。

    参数:
        text: 原始文本
        separators: 分隔符列表，从粗到细排列
        chunk_size: 每块的最大字符数
    """
    if len(text) <= chunk_size:
        return [text]

    # 尝试第一个分隔符
    separator = separators[0]

    # 按分隔符切分
    parts = text.split(separator)
    # 过滤空块
    parts = [p for p in parts if p.strip()]

    # 如果只剩一个块且太大，或者没有分隔符可用
    if len(parts) <= 1:
        if len(separators) > 1:
            # 用下一个更细的分隔符继续递归
            return recursive_split(text, separators[1:], chunk_size)
        else:
            # 实在切不动了，强制按字符数切
            return fixed_size_chunks(text, chunk_size, separator="")

    # 合并小块，直到接近 chunk_size
    chunks = []
    current = ""
    for part in parts:
        candidate = current + separator + part if current else part
        if len(candidate) <= chunk_size:
            current = candidate
        else:
            if current:
                chunks.append(current)
            # 如果单个 part 就超大了，递归切它
            if len(part) > chunk_size and len(separators) > 1:
                sub_chunks = recursive_split(part, separators[1:], chunk_size)
                chunks.extend(sub_chunks)
            else:
                current = part
    if current:
        chunks.append(current)

    return chunks

3.3 策略三：按文档结构分块（Structure-based / Markdown-aware）

利用文档自带的层级结构（Markdown 标题、HTML 标签等）来分块。

import re

def markdown_chunk(doc: str, max_size: int = 500) -> list[dict]:
    """
    按Markdown结构分块

    思路：
    1. 用正则把文档拆成"节"（以 # 标题开头）
    2. 每节带着标题信息，保证语义完整
    3. 如果某节太长，递归用更细粒度切

    返回: 带元信息的块列表，每个块包含 {title, content, level}
    """
    # 按Markdown标题分割：匹配 # 开头的行
    # (?=...) 是前向断言，标题本身保留在匹配结果中
    sections = re.split(r"(?=^#{1,6}\s)", doc, flags=re.MULTILINE)
    sections = [s.strip() for s in sections if s.strip()]

    chunks = []
    for section in sections:
        # 提取标题和级别
        first_line = section.split("\n")[0]
        title_match = re.match(r"^(#{1,6})\s+(.+)$", first_line)

        if title_match:
            level = len(title_match.group(1))
            title = title_match.group(2).strip()
            content = "\n".join(section.split("\n")[1:]).strip()
            header_info = f"[{'#' * level} {title}]"
        else:
            level = 0
            title = ""
            content = section.strip()
            header_info = ""

        # 如果该节太大，递归切分正文
        if len(content) > max_size:
            sub_parts = recursive_split(content, chunk_size=max_size)
            for j, part in enumerate(sub_parts):
                prefix = f"{header_info}" if j == 0 else f"{header_info} (续{ j+1})"
                chunks.append({
                    "title": title or "无标题",
                    "content": part,
                    "header": prefix,
                    "level": level,
                })
        else:
            chunks.append({
                "title": title or "无标题",
                "content": content,
                "header": header_info,
                "level": level,
            })

    return chunks

3.4 策略四：语义分块（Semantic Chunking）

def semantic_chunk(
    text: str,
    embedding_fn,           # Embedding 函数（明天 Day30 详解）
    similarity_threshold: float = 0.3,
    min_chunk_size: int = 100,
) -> list[str]:
    """
    语义分块核心思路（概念版，明天配 Embedding 后可直接运行）

    算法：
    1. 把文本切成句子
    2. 计算相邻句子的 Embedding 相似度
    3. 相似度骤降的地方 = 主题跳变 = 切分点
    4. 在切分点处断开，形成语义连贯的块

    参数:
        text: 原始文本
        embedding_fn: 向量化函数（明天实现）
        similarity_threshold: 相似度低于此阈值时切分
        min_chunk_size: 块的最小字符数
    """
    # 第一步：分句
    sentences = re.split(r'(?<=[。！？\n])\s*', text)
    sentences = [s.strip() for s in sentences if s.strip()]

    if len(sentences) <= 2:
        return [text]

    # 第二步：计算相邻句子的相似度差异
    # （这里用概念伪代码，明天换成真实 Embedding）
    # similarities[i] = cosine_sim(embed(sentence[i]), embed(sentence[i+1]))
    similarities = []
    for i in range(len(sentences) - 1):
        # TODO: 明天替换为真实的 embedding 计算
        sim = compute_similarity_placeholder(sentences[i], sentences[i+1])
        similarities.append(sim)

    # 第三步：找到语义跳变点
    breakpoints = [0]  # 永远从第0句开始
    for i, sim in enumerate(similarities):
        if sim < similarity_threshold:
            breakpoints.append(i + 1)  # 下一句是新块的起点

    # 第四步：根据断点合并句子为块
    chunks = []
    for i in range(len(breakpoints)):
        start_idx = breakpoints[i]
        end_idx = breakpoints[i + 1] if i + 1 < len(breakpoints) else len(sentences)
        chunk = "".join(sentences[start_idx:end_idx])
        if len(chunk.strip()) >= min_chunk_size:
            chunks.append(chunk.strip())
        elif chunks:  # 太小的块合并到上一个
            chunks[-1] = chunks[-1] + chunk.strip()

    return chunks


def compute_similarity_placeholder(s1: str, s2: str) -> float:
    """占位相似度计算（明天替换为真实 Embedding）"""
    # 简单的基于共同字符的粗糙相似度
    set1 = set(s1)
    set2 = set(s2)
    intersection = set1 & set2
    union = set1 | set2
    return len(intersection) / len(union) if union else 0.0

固定字符 实现简单、可预测 可能切断语义 快速原型、结构化文本

import re
import json
from dataclasses import dataclass, field
from typing import Callable


@dataclass
class TextChunk:
    """文本块数据类——每个分块的标准化表示"""
    text: str              # 块内容
    chunk_id: int = 0      # 块编号
    char_count: int = 0    # 字符数
    token_estimate: int = 0  # token 估算（中文约 1字≈1.5 token）
    metadata: dict = field(default_factory=dict)  # 元数据（来源、章节等）

    def __post_init__(self):
        if self.char_count == 0:
            self.char_count = len(self.text)
        if self.token_estimate == 0:
            self.token_estimate = int(self.char_count * 1.5)

    def __str__(self):
        return f"Chunk #{self.chunk_id} ({self.char_count}字, ~{self.token_estimate}tokens)"


class DocumentChunker:
    """
    文档分块器——统一的分块入口

    整合多种分块策略，提供统一的接口和统计信息。
    你可以把它想象成 ETL 中的"数据分片工具"。
    """

    def __init__(self, default_strategy: str = "recursive"):
        self.default_strategy = default_strategy
        self._strategies = {
            "fixed": self._chunk_fixed,
            "recursive": self._chunk_recursive,
            "markdown": self._chunk_markdown,
        }

    def chunk(
        self,
        text: str,
        strategy: str | None = None,
        chunk_size: int = 500,
        overlap: int = 50,
        **kwargs,
    ) -> list[TextChunk]:
        """统一分块接口"""
        strategy = strategy or self.default_strategy
        if strategy not in self._strategies:
            raise ValueError(f"未知策略: {strategy}，可选: {list(self._strategies.keys())}")

        raw_chunks = self._strategies[strategy](
            text, chunk_size=chunk_size, overlap=overlap, **kwargs
        )

        # 包装成 TextChunk 对象
        chunks = []
        for i, raw in enumerate(raw_chunks):
            if isinstance(raw, dict):
                chunks.append(TextChunk(
                    text=raw["content"],
                    chunk_id=i,
                    metadata={"title": raw.get("title", ""), "header": raw.get("header", "")},
                ))
            else:
                chunks.append(TextChunk(text=raw, chunk_id=i))

        return chunks

    def _chunk_fixed(self, text, chunk_size, overlap, **kw):
        """固定字符分块"""
        return fixed_size_chunks(text, chunk_size, overlap, separator="")

    def _chunk_recursive(self, text, chunk_size, overlap, **kw):
        """递归分隔符分块"""
        return recursive_split(text, chunk_size=chunk_size)

    def _chunk_markdown(self, text, chunk_size, overlap, **kw):
        """Markdown结构分块"""
        return markdown_chunk(text, max_size=chunk_size)

    def stats(self, chunks: list[TextChunk]) -> dict:
        """统计分块信息"""
        if not chunks:
            return {"total_chunks": 0}
        counts = [c.char_count for c in chunks]
        return {
            "total_chunks": len(chunks),
            "total_chars": sum(counts),
            "avg_chars": round(sum(counts) / len(counts), 1),
            "min_chars": min(counts),
            "max_chars": max(counts),
            "total_tokens_est": sum(c.token_estimate for c in chunks),
        }

    def visualize(self, chunks: list[TextChunk], bar_width: int = 50):
        """可视化分块效果——用字符柱状图展示每块大小"""
        print(f"\n{'='*60}")
        print(f" 分块可视化 | 共 {len(chunks)} 块")
        print(f"{'='*60}")

        max_size = max(c.char_count for c in chunks) if chunks else 1
        for chunk in chunks:
            bar_len = int(chunk.char_count / max_size * bar_width)
            bar = "█" * bar_len + "░" * (bar_width - bar_len)
            print(f"  #{chunk.chunk_id:02d} | {bar} | {chunk.char_count}字")

        print(f"{'='*60}")

📊 五、分块效果评估指标 5 min

好的分块应该满足以下标准，我们可以量化评估：

def evaluate_chunks(chunks: list[TextChunk], target_size: int = 300) -> dict:
    """
    评估分块质量（不需要 Embedding 的基础指标）

    指标：
    - 块数/平均大小 — 是否合理
    - 大小方差 — 块大小是否均匀（方差越小越好）
    - 过小块占比 — 太小的块说明切分可能有问题
    - 过大块占比 — 太大的块检索噪音大
    """
    import statistics

    if not chunks:
        return {"error": "无分块数据"}

    sizes = [c.char_count for c in chunks]
    avg = statistics.mean(sizes)
    stdev = statistics.stdev(sizes) if len(sizes) > 1 else 0
    cv = stdev / avg if avg > 0 else 0  # 变异系数（越小越均匀）

    too_small = sum(1 for s in sizes if s < target_size * 0.3)  # 不到目标30%
    too_large = sum(1 for s in sizes if s > target_size * 2.0)  # 超过目标2倍

    return {
        "块数量": len(chunks),
        "平均大小": round(avg, 1),
        "大小标准差": round(stdev, 1),
        "变异系数": round(cv, 3),  # < 0.3 算均匀
        "过小块(＜30%目标)": too_small,
        "过大块(＞200%目标)": too_large,
        "评分": _score(cv, too_small, too_large, len(chunks)),
    }


def _score(cv: float, too_small: int, too_large: int, n: int) -> str:
    """简单评分"""
    if cv < 0.3 and too_small == 0 and too_large == 0:
        return "⭐ 优秀 — 块大小均匀，无极端块"
    elif cv < 0.5 and too_small <= 1 and too_large <= 1:
        return "✅ 良好 — 基本合理，有少量极端块"
    elif cv < 0.8:
        return "⚠️ 一般 — 块大小波动较大，建议调整参数"
    else:
        return "❌ 较差 — 分块质量不佳，建议更换策略或参数"


# --- 使用 ---

for key, val in result.items():
    print(f"  {key}: {val}")
