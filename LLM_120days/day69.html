🏗️ 项目背景与架构设计 15 min

作为数据开发工程师，你每天面对的核心挑战之一就是数据质量。传统方式靠写 SQL 手动巡检——SELECT COUNT(*) WHERE amount < 0——费时且覆盖面有限。今天我们用 Python + LLM 构建一个智能巡检系统，能自动执行规则检测、统计异常发现，并让 LLM 进行根因分析。

核心思路：把你的数仓质量检查经验「编码」为规则引擎，再用统计学方法（Z-Score/IQR）自动发现异常模式，最后调用 LLM 对异常进行自然语言归因分析。三层互补：规则兜底已知问题，统计发现未知异常，LLM 提供可读性报告。

📋 数据模型设计（Pydantic） 10 min

用 Pydantic 定义清晰的数据模型，让整个系统的数据流类型安全、可序列化。 

# models.py — 数据质量巡检系统的核心数据模型
from pydantic import BaseModel, Field
from enum import Enum
from datetime import datetime
from typing import Optional, Any


class Severity(str, Enum):
    """问题严重级别"""
    CRITICAL = "critical"   # 严重：数据不可用，阻塞下游
    WARNING = "warning"     # 警告：存在质量问题需关注
    INFO = "info"         # 提示：统计偏差，建议人工确认


class RuleType(str, Enum):
    """规则类型"""
    NULL_CHECK = "null_check"         # 空值检查
    RANGE_CHECK = "range_check"       # 范围检查
    UNIQUE_CHECK = "unique_check"     # 唯一性检查
    FRESHNESS = "freshness"           # 新鲜度检查
    VOLUME_CHANGE = "volume_change"   # 数据量波动
    CUSTOM_SQL = "custom_sql"         # 自定义SQL


class QualityRule(BaseModel):
    """单条质量规则定义"""
    rule_id: str = Field(..., description="规则唯一标识")
    rule_name: str = Field(..., description="规则名称")
    rule_type: RuleType
    table_name: str
    column_name: Optional[str] = None
    # 规则参数（灵活字典）
    params: dict[str, Any] = Field(default_factory=dict)
    severity: Severity = Severity.WARNING
    enabled: bool = True


class CheckResult(BaseModel):
    """单条规则检测结果"""
    rule_id: str
    rule_name: str
    table_name: str
    passed: bool
    severity: Severity
    message: str
    # 检测详情
    actual_value: Optional[Any] = None   # 实际值
    expected_value: Optional[Any] = None # 期望值
    affected_rows: int = 0               # 影响行数
    detail_sql: Optional[str] = None     # 诊断SQL
    checked_at: datetime = Field(default_factory=datetime.now)


class AnomalyResult(BaseModel):
    """统计异常检测结果"""
    table_name: str
    column_name: str
    metric: str           # 检测指标：mean/std/count/null_ratio
    current_value: float
    historical_mean: float
    historical_std: float
    z_score: float        # Z-Score
    is_anomaly: bool
    direction: str        # "spike" 突增 / "drop" 突降
    description: str


class InspectionReport(BaseModel):
    """巡检报告（聚合所有结果）"""
    inspection_id: str
    started_at: datetime
    finished_at: Optional[datetime] = None
    tables_checked: int
    rule_results: list[CheckResult] = []
    anomaly_results: list[AnomalyResult] = []
    llm_analysis: Optional[str] = None
    # 汇总统计
    total_rules: int = 0
    passed_rules: int = 0
    failed_rules: int = 0
    critical_count: int = 0
    warning_count: int = 0

    def summary(self) -> str:
        """生成汇总文本"""
        status = "✅ 全部通过" if self.failed_rules == 0 else f"❌ {self.failed_rules} 项异常"
        return (
            f"巡检 {self.inspection_id} | {status}\n"
            f"检查表: {self.tables_checked} | 规则: {self.passed_rules}/{self.total_rules} 通过\n"
            f"严重: {self.critical_count} | 警告: {self.warning_count} | 异常: {len(self.anomaly_results)}"
        )

3.2 规则引擎核心

# rules/engine.py — 规则引擎：把 YAML 规则翻译成 SQL 并执行
import yaml
from datetime import datetime, timedelta
from models import QualityRule, CheckResult, RuleType, Severity


class RuleEngine:
    """规则引擎：将规则定义翻译为 SQL 并执行"""

    def __init__(self, db_conn, config_path: str = "rules_config.yaml"):
        """
        db_conn: pymysql/aiomysql 连接对象
        config_path: YAML 规则配置文件路径
        """
        self.db = db_conn
        self.rules: list[QualityRule] = []
        self.config = {}
        self._load_config(config_path)

    def _load_config(self, path: str):
        """加载 YAML 配置"""
        with open(path, "r", encoding="utf-8") as f:
            data = yaml.safe_load(f)
        self.config = data.get("global", {})
        # 解析规则列表
        for r in data.get("rules", []):
            if r.get("enabled", True):
                self.rules.append(QualityRule(**r))

    def _execute_sql(self, sql: str) -> any:
        """执行 SQL 并返回结果"""
        with self.db.cursor() as cur:
            cur.execute(sql)
            return cur.fetchone()[0]

    def _build_check_sql(self, rule: QualityRule) -> tuple[str, str]:
        """
        根据规则类型生成检测 SQL 和诊断 SQL
        返回: (check_sql, detail_sql)
        """
        t = rule.table_name
        c = rule.column_name
        p = rule.params

        if rule.rule_type == RuleType.NULL_CHECK:
            # 空值率检查
            max_ratio = p.get("max_null_ratio",
                             self.config["thresholds"]["null_ratio_max"])
            check_sql = f"""
                SELECT CASE 
                    WHEN (SELECT COUNT(*) FROM {t} WHERE {c} IS NULL) 
                         * 1.0 / NULLIF((SELECT COUNT(*) FROM {t}), 0) 
                    <= {max_ratio} THEN 1 ELSE 0 END AS passed
            """
            detail_sql = f"SELECT COUNT(*) AS null_count, 
                COUNT(*) * 1.0 / NULLIF((SELECT COUNT(*) FROM {t}), 0) AS null_ratio 
                FROM {t} WHERE {c} IS NULL"
            return check_sql, detail_sql

        elif rule.rule_type == RuleType.RANGE_CHECK:
            # 范围检查
            min_v = p.get("min_value")
            max_v = p.get("max_value")
            condition = []
            if min_v is not None:
                condition.append(f"{c} >= {min_v}")
            if max_v is not None:
                condition.append(f"{c} <= {max_v}")
            cond_str = " AND ".join(condition)
            check_sql = f"SELECT CASE 
                WHEN (SELECT COUNT(*) FROM {t} WHERE NOT ({cond_str})) = 0 
                THEN 1 ELSE 0 END"
            detail_sql = f"SELECT {c}, COUNT(*) AS cnt 
                FROM {t} WHERE NOT ({cond_str}) GROUP BY {c} LIMIT 10"
            return check_sql, detail_sql

        elif rule.rule_type == RuleType.UNIQUE_CHECK:
            # 唯一性检查：比较 COUNT(DISTINCT) 和 COUNT(*)
            check_sql = f"SELECT CASE 
                WHEN (SELECT COUNT(DISTINCT {c}) FROM {t}) 
                   = (SELECT COUNT(*) FROM {t}) THEN 1 ELSE 0 END"
            detail_sql = f"SELECT {c}, COUNT(*) AS dup_count 
                FROM {t} GROUP BY {c} HAVING COUNT(*) > 1 LIMIT 10"
            return check_sql, detail_sql

        elif rule.rule_type == RuleType.FRESHNESS:
            # 新鲜度检查：最新更新时间是否在阈值内
            max_hours = p.get("max_hours", 48)
            check_sql = f"SELECT CASE 
                WHEN (SELECT MAX({c}) FROM {t}) 
                   >= DATE_SUB(NOW(), INTERVAL {max_hours} HOUR) 
                THEN 1 ELSE 0 END"
            detail_sql = f"SELECT MAX({c}) AS last_update, 
                TIMESTAMPDIFF(HOUR, MAX({c}), NOW()) AS hours_ago 
                FROM {t}"
            return check_sql, detail_sql

        elif rule.rule_type == RuleType.CUSTOM_SQL:
            # 自定义 SQL 检查
            sql = p["check_sql"]
            expected_max = p.get("expected_max", 0)
            check_sql = f"SELECT CASE WHEN ({sql}) <= {expected_max} 
                THEN 1 ELSE 0 END"
            detail_sql = sql
            return check_sql, detail_sql

    def check_rule(self, rule: QualityRule) -> CheckResult:
        """执行单条规则检测"""
        try:
            check_sql, detail_sql = self._build_check_sql(rule)
            passed = bool(self._execute_sql(check_sql))

            result = CheckResult(
                rule_id=rule.rule_id,
                rule_name=rule.rule_name,
                table_name=rule.table_name,
                passed=passed,
                severity=rule.severity,
                message=f"{'通过' if passed else '失败'}: {rule.rule_name}",
                detail_sql=detail_sql,
            )
            return result
        except Exception as e:
            return CheckResult(
                rule_id=rule.rule_id,
                rule_name=rule.rule_name,
                table_name=rule.table_name,
                passed=False,
                severity=Severity.CRITICAL,
                message=f"规则执行异常: {e}",
            )

    def run_all_rules(self) -> list[CheckResult]:
        """执行所有启用的规则"""
        results = []
        for rule in self.rules:
            result = self.check_rule(rule)
            results.append(result)
            # 打印即时反馈
            icon = "✅" if result.passed else "❌"
            print(f"  {icon} [{rule.rule_id}] {rule.rule_name}")
        return results
SQL 注入风险：这里直接拼接了表名和列名到 SQL 中。生产环境应使用白名单校验（表名/列名是否在 information_schema 中存在），防止配置文件被篡改后造成注入。当前作为学习项目简化处理，但要记住这个安全意识。
📊 统计异常检测（Z-Score + IQR） 20 min

规则引擎只能发现你已经知道的问题。但数据异常往往是你没想到的——比如今天的数据量突然比平时多了 3 倍。统计异常检测用历史基线自动发现这类问题。

两种统计方法：
Z-Score：衡量当前值偏离历史均值多少个标准差。适合近似正态分布的数据（如日数据量）。
IQR（四分位距）：用 Q1 和 Q3 定义正常范围，对极端值更鲁棒，不要求正态分布。 
# detectors/stat_anomaly.py — 统计异常检测器
import statistics
from models import AnomalyResult


class StatAnomalyDetector:
    """
    基于历史统计数据的异常检测器
    
    核心思路（用数仓的语言说）：
    - 把每张表的每日「行数、空值率、均值」当成时间序列指标
    - 用历史 N 天的数据算基线（均值+标准差）
    - 当今天的值偏离基线超过阈值时，标记为异常
    """

    def __init__(self, z_threshold: float = 2.5, iqr_factor: float = 1.5):
        self.z_threshold = z_threshold
        self.iqr_factor = iqr_factor

    def z_score_check(
        self,
        table_name: str,
        column_name: str,
        metric: str,
        current_value: float,
        historical_values: list[float],
    ) -> AnomalyResult:
        """
        Z-Score 异常检测
        
        Z = (当前值 - 历史均值) / 历史标准差
        |Z| > 2.5 → 异常
        
        参数:
            historical_values: 历史N天的指标值列表
            current_value: 今天的指标值
        """
        if len(historical_values) < 7:
            # 历史数据太少，跳过检测
            return AnomalyResult(
                table_name=table_name, column_name=column_name,
                metric=metric, current_value=current_value,
                historical_mean=0, historical_std=0,
                z_score=0, is_anomaly=False, direction="N/A",
                description="历史数据不足，跳过检测（需至少7天）",
            )

        mean = statistics.mean(historical_values)
        std = statistics.stdev(historical_values)

        # 标准差为0说明历史值全部相同，无法计算Z-Score
        if std == 0:
            return AnomalyResult(
                table_name=table_name, column_name=column_name,
                metric=metric, current_value=current_value,
                historical_mean=mean, historical_std=std,
                z_score=0, is_anomaly=False, direction="stable",
                description=f"历史值稳定在 {mean:.2f}，无波动",
            )

        z = (current_value - mean) / std
        is_anomaly = abs(z) > self.z_threshold
        direction = "spike" if z > 0 else "drop"

        if is_anomaly:
            desc = (f"{metric} 异常{'突增' if direction=='spike' else '突降'}: "
                   f"当前 {current_value:.2f}, 基线 {mean:.2f}±{std:.2f}, "
                   f"Z-Score={z:.2f}")
        else:
            desc = f"{metric} 正常: 当前 {current_value:.2f}, 基线 {mean:.2f}±{std:.2f}"

        return AnomalyResult(
            table_name=table_name, column_name=column_name,
            metric=metric, current_value=current_value,
            historical_mean=mean, historical_std=std,
            z_score=round(z, 4),
            is_anomaly=is_anomaly, direction=direction,
            description=desc,
        )

    def iqr_check(
        self,
        table_name: str, column_name: str, metric: str,
        current_value: float,
        historical_values: list[float],
    ) -> AnomalyResult:
        """
        IQR 异常检测（对极端值更鲁棒）
        
        Q1 = 25分位数, Q3 = 75分位数
        IQR = Q3 - Q1
        正常范围: [Q1 - 1.5*IQR, Q3 + 1.5*IQR]
        """
        sorted_vals = sorted(historical_values)
        n = len(sorted_vals)
        if n < 7:
            return AnomalyResult(
                table_name=table_name, column_name=column_name,
                metric=metric, current_value=current_value,
                historical_mean=0, historical_std=0,
                z_score=0, is_anomaly=False, direction="N/A",
                description="历史数据不足",
            )

        # 计算四分位数
        q1 = sorted_vals[n // 4]
        q3 = sorted_vals[3 * n // 4]
        iqr = q3 - q1
        lower = q1 - self.iqr_factor * iqr
        upper = q3 + self.iqr_factor * iqr

        is_anomaly = current_value < lower or current_value > upper
        direction = "spike" if current_value > upper else "drop"

        return AnomalyResult(
            table_name=table_name, column_name=column_name,
            metric=metric, current_value=current_value,
            historical_mean=statistics.mean(historical_values),
            historical_std=statistics.stdev(historical_values)
                if len(historical_values) > 1 else 0,
            z_score=0,  # IQR 不使用 Z-Score
            is_anomaly=is_anomaly, direction=direction,
            description=(f"IQR检测: 当前{current_value:.2f}, "
                       f"正常范围[{lower:.2f}, {upper:.2f}]"),
        )

# detectors/inspector.py — 巡检执行器：串联规则引擎和统计检测
import uuid
from datetime import datetime
from rules.engine import RuleEngine
from detectors.stat_anomaly import StatAnomalyDetector
from models import InspectionReport, CheckResult, Severity


class DataQualityInspector:
    """数据质量巡检执行器 — 系统的总调度"""

    def __init__(self, db_conn, config_path: str = "rules_config.yaml"):
        self.engine = RuleEngine(db_conn, config_path)
        self.detector = StatAnomalyDetector()

    def collect_table_metrics(self, table: str) -> dict:
        """
        采集单表的统计指标
        返回: {"row_count": int, "null_ratios": {col: float}, ...}
        """
        db = self.engine.db
        metrics = {}

        # 1. 行数
        with db.cursor() as cur:
            cur.execute(f"SELECT COUNT(*) FROM {table}")
            metrics["row_count"] = cur.fetchone()[0]

        # 2. 各列空值率
        with db.cursor() as cur:
            cur.execute(f"""
                SELECT COLUMN_NAME, 
                       SUM(CASE WHEN IS_NULLABLE='NO' THEN 0 
                                ELSE 1 END) AS nullable_count
                FROM INFORMATION_SCHEMA.COLUMNS 
                WHERE TABLE_SCHEMA=DATABASE() AND TABLE_NAME='{table}'
            """)
            metrics["null_ratios"] = {}
            for row in cur.fetchall():
                col = row[0]
                cur2 = db.cursor()
                cur2.execute(
                    f"SELECT COUNT(*) FROM {table} WHERE {col} IS NULL")
                null_count = cur2.fetchone()[0]
                total = metrics["row_count"]
                if total > 0:
                    metrics["null_ratios"][col] = null_count / total
                cur2.close()

        return metrics

    def run_inspection(self, tables: list[str] = None) -> InspectionReport:
        """
        执行完整巡检
        
        流程:
        1. 规则引擎逐条检查
        2. 统计检测器扫描各表指标
        3. 聚合结果生成报告
        """
        report = InspectionReport(
            inspection_id=str(uuid.uuid4())[:8],
            started_at=datetime.now(),
            tables_checked=len(tables) if tables else 0,
        )

        # 第一步：执行所有规则
        print(f"\n{'='*50}")
        print(f"📋 执行规则检查（{len(self.engine.rules)} 条规则）")
        print(f"{'='*50}")
        report.rule_results = self.engine.run_all_rules()

        # 第二步：统计异常检测
        # （实际生产中从历史记录表读取历史值，这里用模拟数据演示）
        print(f"\n📊 统计异常检测...")
        # 模拟历史数据 + 当前值做演示
        mock_history = {
            ("dwd_bid_detail", "row_count"): {
                "history": [15200, 15800, 16100, 15900, 15500, 16000, 16200],
                "current": 45000,  # 模拟一个突增异常
            },
            ("dwd_bid_detail", "null_ratio_project_name"): {
                "history": [0.002, 0.003, 0.001, 0.002, 0.002, 0.001, 0.003],
                "current": 0.15,  # 模拟空值率飙升
            },
        }

        for (tbl, metric_name), data in mock_history.items():
            col = "*" if metric_name == "row_count" else metric_name.replace("null_ratio_", "")
            result = self.detector.z_score_check(
                table_name=tbl, column_name=col, metric=metric_name,
                current_value=data["current"],
                historical_values=data["history"],
            )
            report.anomaly_results.append(result)
            icon = "🔴" if result.is_anomaly else "🟢"
            print(f"  {icon} {tbl}.{metric_name}: {result.description}")

        # 第三步：汇总统计
        report.total_rules = len(report.rule_results)
        report.passed_rules = sum(1 for r in report.rule_results if r.passed)
        report.failed_rules = report.total_rules - report.passed_rules
        report.critical_count = sum(
            1 for r in report.rule_results
            if not r.passed and r.severity == Severity.CRITICAL
        )
        report.warning_count = sum(
            1 for r in report.rule_results
            if not r.passed and r.severity == Severity.WARNING
        )
        report.finished_at = datetime.now()

        return report
数据开发视角：Z-Score 检测本质上就是在做「日环比/周环比」的高级版。你平时用 SQL 写 WHERE cnt > AVG(cnt)*1.3 就是手动版 Z-Score。这里只是把它工程化、自动化了。

知识点说明数仓关联
Pydantic BaseModel 类型安全的数据模型，自动校验+序列化 像 DDL 定义表结构一样定义 Python 数据结构
YAML 配置驱动 规则从代码解耦到配置文件 类似数仓中把 ETL 规则配置化，避免硬编码
策略模式 不同规则类型对应不同的 SQL 生成策略 每个 DWD 表的清洗规则不同，但流程一样
Z-Score / IQR 统计异常检测的两种方法 比手动写「环比 > 30%」更科学、自适应
statistics 模块 Python 内置统计函数（mean/stdev） 替代 SQL 的 AVG/STDDEV，适合内存计算
