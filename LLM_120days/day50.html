1. tomllib — Python 3.11+ 内置 TOML 解析

Python 3.11 引入了 tomllib（只读），Python 3.12 引入了 tomllib 的写入增强。TOML 是比 JSON/YAML 更适合配置文件的格式：

TOML 优势

  • 类型明确（不像 YAML 隐式转换）
  • 支持注释（JSON 不行）
  • 比 YAML 语法简单，不易出错

# 加载 TOML 配置
import tomllib  # Python 3.11+
import os
from pathlib import Path

def load_config(config_path: str = "config.toml") -> dict:
    """加载 TOML 配置文件，自动替换环境变量占位符"""
    path = Path(config_path)
    if not path.exists():
        raise FileNotFoundError(f"配置文件不存在: {config_path}")

    # tomllib.open 返回 dict-like 对象
    with tomllib.open(path, "rb") as f:
        config = dict(f)  # 转为普通 dict 便于后续修改

    # 递归替换 ${ENV_VAR} 占位符
    config = _resolve_env_vars(config)
    return config


def _resolve_env_vars(obj):
    """递归遍历字典，替换 ${KEY} 格式的环境变量"""
    if isinstance(obj, dict):
        return {k: _resolve_env_vars(v) for k, v in obj.items()}
    elif isinstance(obj, list):
        return [_resolve_env_vars(item) for item in obj]
    elif isinstance(obj, str) and obj.startswith("${") and obj.endswith("}"):
        # 提取环境变量名
        env_key = obj[2:-1]
        env_val = os.environ.get(env_key, "")
        if not env_val:
            raise ValueError(f"环境变量 {env_key} 未设置！")
        return env_val
    return obj

Protocol vs ABC：Protocol 是"隐式接口"——只要类实现了对应方法就自动满足协议，不需要显式继承。ABC 是"显式接口"——必须继承并实现所有抽象方法。

实战选型：项目内部组件用 Protocol（灵活替换实现），对外暴露的 API 契约用 ABC（强制子类实现）。

# src/chunking/loader.py
"""文档加载器 — 支持 .md / .txt / .csv 文件"""

from pathlib import Path
from dataclasses import dataclass, field
import csv
import logging

logger = logging.getLogger(__name__)


@dataclass
class Document:
    """统一文档数据结构"""
    content: str                    # 文档正文
    metadata: dict = field(default_factory=dict)  # 元数据（来源/日期等）

    def __post_init__(self):
        # 自动统计字符数和段落数
        self.metadata["char_count"] = len(self.content)
        self.metadata["paragraph_count"] = len(self.content.split("\n\n"))


class DocumentLoader:
    """文档加载器 — 从目录批量加载文档"""

    # 支持的文件扩展名
    SUPPORTED_EXTENSIONS = {".md", ".txt", ".csv"}

    def __init__(self, data_dir: str):
        self.data_dir = Path(data_dir)
        if not self.data_dir.exists():
            raise FileNotFoundError(f"数据目录不存在: {data_dir}")

    def load_directory(self, recursive: bool = True) -> list[Document]:
        """批量加载目录下所有支持的文件"""
        pattern = "**/*" if recursive else "*"
        documents = []

        for file_path in sorted(self.data_dir.glob(pattern)):
            if file_path.suffix.lower() in self.SUPPORTED_EXTENSIONS:
                try:
                    doc = self.load_file(file_path)
                    documents.append(doc)
                    logger.info("加载文档: %s (%d 字符)",
                                file_path.name, doc.metadata["char_count"])
                except Exception as e:
                    logger.warning("跳过文件 %s: %s", file_path.name, e)

        logger.info("共加载 %d 个文档", len(documents))
        return documents

    def load_file(self, file_path: Path) -> Document:
        """加载单个文件（自动识别格式）"""
        suffix = file_path.suffix.lower()

        if suffix == ".csv":
            content = self._load_csv(file_path)
        else:  # .md / .txt
            content = file_path.read_text(encoding="utf-8")

        # 构造元数据
        metadata = {
            "source": str(file_path.relative_to(self.data_dir)),
            "filename": file_path.name,
            "extension": suffix,
        }

        return Document(content=content, metadata=metadata)

    def _load_csv(self, file_path: Path) -> str:
        """将 CSV 转为可读文本（每行用逗号分隔）"""
        rows = []
        with open(file_path, encoding="utf-8") as f:
            reader = csv.reader(f)
            headers = for i, row in enumerate(reader):
                if i == 0:
                    headers = row
                    continue
                # 把每行数据格式化为 "字段: 值" 的可读形式
                parts = [f"{headers[j]}: {val}"
                         for j, val in enumerate(row) if j < len(headers)]
                rows.append(" | ".join(parts))
        return "\n".join(rows)

# src/chunking/recursive.py
"""递归分隔符分块器 — 最实用的分块策略"""

from src.chunking.base import BaseChunker, Chunk


class RecursiveCharacterChunker(BaseChunker):
    """
    递归分隔符分块器

    核心思路（数仓类比）：
    像 Hive 的 STR_TO_MAP一样，先按大分隔符切，切不动了再按小分隔符切。
    分隔符优先级：段落 → 行 → 句子 → 词 → 字符
    """

    # 默认分隔符优先级（中文场景优化）
    DEFAULT_SEPARATORS = ["\n\n", "\n", 。", "！", "？",
                          ".", " ", "，", ""]

    def __init__(
        self,
        chunk_size: int = 512,
        chunk_overlap: int = 64,
        separators: Optional[list[str]] = None,
    ):
        self.chunk_size = chunk_size
        self.chunk_overlap = chunk_overlap
        self.separators = separators or self.DEFAULT_SEPARATORS

    def split(self, text: str, metadata: Optional[dict] = None) -> list[Chunk]:
        """递归分块主方法"""
        raw_chunks = self._recursive_split(text, self.separators)
        return self._merge_small_chunks(raw_chunks, metadata or {})

    def _recursive_split(self, text: str, separators: list[str]) -> list[str]:
        """
        递归分块核心算法：
        1. 用当前分隔符分割文本
        2. 如果分割后的所有片段都 ≤ chunk_size，直接返回
        3. 如果有片段 > chunk_size，用下一个分隔符继续分割该片段
        4. 直到用完所有分隔符，最后按字符数硬切
        """
        if not text.strip():
            return []

        # 没有更多分隔符了，按字符硬切
        if not separators:
            return self._hard_split(text)

        sep = separators[0]
        remaining_seps = separators[1:]

        # 用当前分隔符分割
        splits = text.split(sep)
        splits = [s for s in splits if s.strip()]  # 过滤空串

        result = []
        for split in splits:
            if len(split) <= self.chunk_size:
                # 片段足够小，直接加入
                result.append(split)
            else:
                # 片段太大，用下一个分隔符继续切
                sub_splits = self._recursive_split(split, remaining_seps)
                result.extend(sub_splits)

        return result

    def _hard_split(self, text: str) -> list[str]:
        """兜底：按固定字符数切分（带 overlap）"""
        chunks = []
        start = 0
        while start < len(text):
            end = start + self.chunk_size
            chunks.append(text[start:end])
            # 下一个块的起始位置 = end - overlap（保证相邻块有重叠）
            start = end - self.chunk_overlap
        return chunks

    def _merge_small_chunks(
        self,
        chunks: list[str],
        metadata: dict,
    ) -> list[Chunk]:
        """
        合并过小的块（避免"一句话一个块"浪费检索配额）

        数仓类比：就像小文件合并（CombineHiveInputFormat），
        把太小的 chunk 拼到前一个 chunk 里。
        """
        result = []
        buffer = ""
        index = 0

        for chunk_text in chunks:
            if not buffer:
                buffer = chunk_text
            elif len(buffer) + len(chunk_text) + 1 <= self.chunk_size:
                # 合并到缓冲区
                buffer = buffer + "\n" + chunk_text
            else:
                # 缓冲区已满，保存当前块
                result.append(Chunk(
                    text=buffer,
                    chunk_index=index,
                    metadata={**metadata, "chunk_index": index}
                ))
                index += 1
                buffer = chunk_text

        # 保存最后一个缓冲区
        if buffer:
            result.append(Chunk(
                text=buffer,
                chunk_index=index,
                metadata={**metadata, "chunk_index": index}
            ))

        return result
chunk_overlap 的作用：就像数仓 ETL 中"上游分区字段冗余存储到下游"——overlap 确保被分隔符切断的上下文在相邻块中也有副本，避免检索时丢失关键信息。典型值：chunk_size 的 10%~20%。

# src/embedding/zhipu_embedder.py
"""智谱 Embedding API 封装 — 支持批量 + 缓存"""

import hashlib
import json
import logging
from functools import lru_cache
from pathlib import Path

logger = logging.getLogger(__name__)


class ZhipuEmbedder:
    """
    智谱 Embedding API 封装

    核心优化：
    1. 批量请求（减少 HTTP 往返）
    2. MD5 缓存（相同文本不重复计算）
    3. 指数退避重试（API 偶尔 429）
    """

    # 智谱 Embedding API 端点
    API_URL = "https://open.bigmodel.cn/api/paas/v4/embeddings"
    # 最大批量大小（智谱限制）
    MAX_BATCH = 50

    def __init__(self, api_key: str, model: str = "embedding-3"):
        import requests
        self.api_key = api_key
        self.model = model
        self._session = requests.Session()
        self._session.headers.update({
            "Authorization": f"Bearer {api_key}"
        })

    def embed_texts(self, texts: list[str]) -> list[list[float]]:
        """批量向量化（自动分批）"""
        all_embeddings = []

        # 按最大批量分批处理
        for i in range(0, len(texts), self.MAX_BATCH):
            batch = texts[i : i + self.MAX_BATCH]
            embeddings = self._embed_batch(batch)
            all_embeddings.extend(embeddings)
            logger.debug("已嵌入 %d/%d 文本", min(i + self.MAX_BATCH, len(texts)), len(texts))

        return all_embeddings

    def embed_query(self, query: str) -> list[float]:
        """单条查询向量化"""
        return self.embed_texts([query])[0]

    def _embed_batch(self, texts: list[str]) -> list[list[float]]:
        """单批次 API 调用（带重试）"""
        import time

        for attempt in range(3):  # 最多重试 3 次
            try:
                resp = self._session.post(
                    self.API_URL,
                    json={
                        "model": self.model,
                        "input": texts,
                    },
                    timeout=30,
                )
                resp.raise_for_status()

                data = resp.json()
                # 按原始顺序排列（API 可能不按顺序返回）
                sorted_results = sorted(data["data"], key=lambda x: x["index"])
                return [item["embedding"] for item in sorted_results]

            except Exception as e:
                logger.warning("Embedding API 第 %d 次调用失败: %s", attempt + 1, e)
                if attempt < 2:
                    # 指数退避：1s, 2s, 4s
                    time.sleep(2 ** attempt)
                else:
                    raise

# src/store/chroma_store.py
"""ChromaDB 向量存储封装 — 持久化 + 元数据过滤"""

import logging
from pathlib import Path

logger = logging.getLogger(__name__)


class ChromaVectorStore:
    """
    ChromaDB 向量存储

    核心设计：
    1. 持久化到磁盘（重启不丢数据）
    2. 元数据过滤（按来源/类型检索）
    3. 幂等写入（upsert 语义）
    """

    def __init__(self, persist_dir: str, collection_name: str = "default"):
        import chromadb

        # 确保持久化目录存在
        Path(persist_dir).mkdir(parents=True, exist_ok=True)

        # 创建持久化客户端
        self.client = chromadb.PersistentClient(path=persist_dir)

        # 获取或创建 Collection
        self.collection = self.client.get_or_create_collection(
            name=collection_name,
            metadata={"description": "AMC 知识库"}
        )
        logger.info("向量库初始化: %s (已有 %d 条)",
                    collection_name, self.collection.count())

    def add_documents(
        self,
        ids: list[str],
        embeddings: list[list[float]],
        documents: list[str],
        metadatas: list[dict],
    ) -> int:
        """批量添加文档（upsert 语义，同 ID 覆盖）"""

        # ChromaDB 单次限制 5000 条，自动分批
        batch_size = 1000
        total_added = 0

        for i in range(0, len(ids), batch_size):
            end = min(i + batch_size, len(ids))
            self.collection.upsert(
                ids=ids[i:end],
                embeddings=embeddings[i:end],
                documents=documents[i:end],
                metadatas=metadatas[i:end],
            )
            total_added += (end - i)
            logger.info("入库进度: %d/%d", total_added, len(ids))

        return total_added

    def search(
        self,
        query_embedding: list[float],
        top_k: int = 5,
        where: Optional[dict] = None,
    ) -> list[dict]:
        """向量相似度检索"""
        results = self.collection.query(
            query_embeddings=[query_embedding],
            n_results=top_k,
            where=where,  # 元数据过滤条件
            include=["documents", "metadatas", "distances"]
        )

        # 组装结果
        retrieved = []
        for i in range(top_k):
            retrieved.append({
                "text": results["documents"][0][i],
                "metadata": results["metadatas"][0][i],
                "distance": results["distances"][0][i],
                "score": 1 - results["distances"][0][i],  # 余弦相似度 = 1 - 距离
            })

        return retrieved

    def count(self) -> int:
        """返回当前文档总数"""
        return self.collection.count()

    def delete_collection(self):
        """删除整个集合（重建索引时用）"""
        self.client.delete_collection(self.collection.name)
        logger.warning("已删除集合: %s", self.collection.name)
ChromaDB 的 upsert 语义：就像数仓的 MERGE INTO（INSERT ON DUPLICATE KEY UPDATE）——相同 ID 的文档会被覆盖更新，不会产生重复。这保证了索引构建的幂等性：不管你跑多少次，最终结果都一样。

# src/pipeline/index_pipeline.py
"""离线索引管线 — 文档 → 分块 → Embedding → 入库

这个 Pipeline 就是 RAG 的"离线 ETL"：
  ODS（原始文档）→ DWD（分块）→ DWS（Embedding）→ ADS（向量库）
"""

import logging
import time
from dataclasses import dataclass, field
from typing import Optional

from src.chunking.loader import DocumentLoader, Document
from src.chunking.recursive import RecursiveCharacterChunker
from src.embedding.zhipu_embedder import ZhipuEmbedder
from src.store.chroma_store import ChromaVectorStore

logger = logging.getLogger(__name__)


@dataclass
class IndexStats:
    """索引统计信息 — 就像 ETL 的运行报告"""
    total_documents: int = 0
    total_chunks: int = 0
    total_chars: int = 0
    elapsed_seconds: float = 0
    chunks_per_doc: float = 0
    avg_chunk_size: float = 0

    def summary(self) -> str:
        return f"""
  ========== 索引构建报告 ==========
  文档数量:    {self.total_documents}
  分块数量:    {self.total_chunks}
  总字符数:    {self.total_chars:,}
  耗时:        {self.elapsed_seconds:.1f}s
  平均每文档:  {self.chunks_per_doc:.1f} 块
  平均块大小:  {self.avg_chunk_size:.0f} 字符
  ================================="""


class IndexPipeline:
    """
    离线索引管线 — RAG 系统的 ETL

    数据流：
    documents/ → DocumentLoader → Chunks → Embedder → ChromaDB

    设计要点：
    1. 组件可替换（通过 Protocol 接口）
    2. 配置驱动（从 config.toml 读取参数）
    3. 幂等执行（rebuild=True 先清空再重建）
    4. 统计报告（运行后输出完整报告）
    """

    def __init__(
        self,
        data_dir: str,
        api_key: str,
        persist_dir: str = "./data/chroma_db",
        collection_name: str = "amc_knowledge",
        chunk_size: int = 512,
        chunk_overlap: int = 64,
        embedding_model: str = "embedding-3",
    ):
        # 初始化各组件
        self.loader = DocumentLoader(data_dir)
        self.chunker = RecursiveCharacterChunker(
            chunk_size=chunk_size,
            chunk_overlap=chunk_overlap,
        )
        self.embedder = ZhipuEmbedder(api_key=api_key, model=embedding_model)
        self.store = ChromaVectorStore(persist_dir, collection_name)

        logger.info("IndexPipeline 初始化完成")

    def run(self, rebuild: bool = False) -> IndexStats:
        """
        执行完整索引流程

        Args:
            rebuild: 是否重建（先删除旧数据再重新索引）
        Returns:
            IndexStats 索引统计报告
        """
        start_time = time.perf_counter()
        stats = IndexStats()

        # 1. 重建模式：清空旧数据
        if rebuild:
            logger.warning("重建模式：清空旧索引...")
            self.store.delete_collection()
            self.store = ChromaVectorStore(
                self.store.client,  # 需要重新创建
            )

        # 2. 加载所有文档
        logger.info("📖 Step 1: 加载文档...")
        documents = self.loader.load_directory()
        stats.total_documents = len(documents)
        if not documents:
            logger.warning("未找到任何文档！")
            return stats

        # 3. 分块
        logger.info("✂️  Step 2: 文档分块...")
        all_chunks = []
        for doc in documents:
            chunks = self.chunker.split(doc.content, doc.metadata)
            all_chunks.extend(chunks)
            logger.info("  %s → %d 个块", doc.metadata["filename"], len(chunks))

        stats.total_chunks = len(all_chunks)
        stats.total_chars = sum(c.char_count for c in all_chunks)

        # 4. Embedding（批量）
        logger.info("🔢 Step 3: 文本向量化（%d 块）...", len(all_chunks))
        texts = [chunk.text for chunk in all_chunks]
        embeddings = self.embedder.embed_texts(texts)

        # 5. 入库
        logger.info("📦 Step 4: 向量入库...")
        ids = [f"doc_{chunk.metadata.get('source', 'unknown')}"
                f"_chunk_{chunk.chunk_index}"
                for chunk in all_chunks]
        documents_text = [chunk.text for chunk in all_chunks]
        metadatas = [chunk.metadata for chunk in all_chunks]

        self.store.add_documents(ids, embeddings, documents_text, metadatas)

        # 6. 统计
        stats.elapsed_seconds = time.perf_counter() - start_time
        stats.chunks_per_doc = stats.total_chunks / max(stats.total_documents, 1)
        stats.avg_chunk_size = stats.total_chars / max(stats.total_chunks, 1)

        logger.info(stats.summary())
        return stats

import random
import numpy as np

# === 模拟 Embedder（随机向量，但维度一致） ===
class MockEmbedder:
    """模拟 Embedder — 用随机向量代替真实 API 调用"""

    def __init__(self, dim: int = 128):
        self.dim = dim
        # 固定随机种子，保证结果可复现
        rng = np.random.RandomState(42)

    def embed_texts(self, texts: list[str]) -> list[list[float]]:
        # 基于文本长度生成伪向量（相同长度 → 相同向量，保证缓存测试有效）
        return [
            self._hash_to_vector(text)
            for text in texts
        ]

    def _hash_to_vector(self, text: str) -> list[float]:
        """用文本哈希生成确定性向量"""
        hash_val = hash(text)
        rng = np.random.RandomState(hash_val % (2**31))
        vec = rng.randn(self.dim).tolist()
        # 归一化（和真实 Embedding 一样）
        norm = sum(v**2 for v in vec) ** 0.5
        return [v / norm for v in vec]

#陷阱症状解法
1 chunk_size 过大 检索出整个文档，无关信息淹没答案 从 256 开始，根据检索效果调到 512-1024
2 overlap 过大 检索结果高度重复，浪费 Token chunk_size 的 10%-20%，不要超过 25%
3 sys.path 导入失败 ModuleNotFoundError: No module named 'src' 入口脚本顶部加 sys.path.insert(0, project_root)
4 ChromaDB upsert ID 冲突 不同文档的 chunk 被覆盖 ID 包含来源文件名 + chunk_index：f"{source}_chunk_{idx}"
5 API Key 硬编码 代码推到 git，Key 泄漏 放 .env 文件，config.toml 用 ${VAR} 占位
