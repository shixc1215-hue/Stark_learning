2.4 自定义过滤器

# 数仓人最爱：金额格式化（万/亿）
def format_amount(value, unit="万"):
    """将数字格式化为带单位的金额"""
    if value >= 10000:
        return f"¥{value / 10000:.2f}亿"
    return f"¥{value:,.2f}{unit}"

# 注册到 Jinja2 环境
env.filters["amount"] = format_amount

# 模板中使用：{{ 384715600 | amount }}
# 输出：¥3.85亿

# --- 变化率颜色标记过滤器 ---
def trend_color(value):
    """根据变化率返回颜色样式"""
    if value > 0:
        return "color:red"    # 涨 = 红色（中国股市惯例）
    elif value < 0:
        return "color:green"  # 跌 = 绿色
    return "color:gray"

env.filters["trend"] = trend_color
注意：Jinja2 模板不是 Python 代码！模板里不能 import、不能调用任意函数、不能写复杂逻辑。复杂运算应在 Python 代码中完成，模板只负责"填空"。

4.1 报表生成的 Prompt 设计

核心思路：把 SQL 结果（列名 + 数据行）喂给 LLM，让它做三件事：① 描述数据概况 → ② 发现异常/趋势 → ③ 给出建议。这就是一个标准的"数据分析报告"结构。 
from dataclasses import dataclass, field
from typing import Optional
from datetime import datetime

# --- 报表上下文（SQL结果 + 元数据） ---
@dataclass
class ReportContext:
    """报表生成所需的上下文信息"""
    query: str                          # 用户原始问题
    sql: str                            # 生成的 SQL
    columns: list[str]                  # 查询结果的列名
    data: list[dict]                    # 查询结果（字典列表）
    report_type: str = "analysis"    # analysis / summary / comparison
    period: str = "2026年上半年"     # 时间范围
    max_rows: int = 20               # 最大展示行数

    def format_data(self) -> str:
        """将数据格式化为 LLM 易读的文本"""
        # 只取前 max_rows 行，避免 Token 爆炸
        rows = self.data[:self.max_rows]
        header = " | ".join(self.columns)
        lines = [header, "-" * len(header)]
        for row in rows:
            line = " | ".join(str(row.get(c, "")) for c in self.columns)
            lines.append(line)
        return "\n".join(lines)


# --- 报表 Prompt 构建器 ---
class ReportPromptBuilder:
    """构建报表生成的 Prompt"""

    SYSTEM_ANALYSIS = """你是一位专业的金融数据分析师，擅长不良资产管理领域。
请根据查询结果生成一份简洁、专业的分析报告。

报告结构：
1. **数据概况**：一句话总结查询范围和总体情况
2. **关键发现**：列出 2-3 个最重要的数据洞察（用具体数字）
3. **趋势/异常**：标注同比/环比变化、异常值
4. **建议**：基于数据给出 1-2 条可执行的建议

格式要求：
- 使用中文
- 数字用 ¥ 和万/亿为单位
- 变化用 ↑↓ 箭头标记
- 语气专业但不冗余，每段不超过 3 句话"""

    SYSTEM_SUMMARY = """你是一位数据分析师。请生成一份简短的数据摘要：
- 总体概况（1句话）
- TOP3 关键指标
- 结论（1句话）
控制在 200 字以内。"""

    @classmethod
    def build(cls, ctx: ReportContext) -> str:
        """构建完整的 Prompt"""
        system = (cls.SYSTEM_ANALYSIS if ctx.report_type == "analysis"
                  else cls.SYSTEM_SUMMARY)

        user_prompt = f"""## 查询背景
- 用户问题：{ctx.query}
- 时间范围：{ctx.period}
- 查询 SQL：```sql
{ctx.sql}
```

## 查询结果（共 {len(ctx.data)} 行，展示前 {ctx.max_rows} 行）
{ctx.format_data()}

## 要求
请根据以上数据，生成{ctx.period}的{'详细分析报告' if ctx.report_type == 'analysis' else '数据摘要'}。
重点关注数据中的异常值、趋势变化和排名变化。"""
        return system, user_prompt

4.2 完整报表生成引擎

import json
import re
from zhipuai import ZhipuAI

class ReportGenerator:
    """基于 LLM 的智能报表生成引擎"""

    def __init__(self, api_key: str, model: str = "glm-4-flash"):
        self.client = ZhipuAI(api_key=api_key)
        self.model = model

    def generate(self, ctx: ReportContext) -> str:
        """生成报表文本"""
        system_prompt, user_prompt = ReportPromptBuilder.build(ctx)

        response = self.client.chat.completions.create(
            model=self.model,
            messages=[
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": user_prompt},
            ],
            temperature=0.3,  # 低温度：报表要准确，不要太发散
        )
        return response.choices[0].message.content

    def generate_with_fallback(self, ctx: ReportContext, retries: int = 2) -> str:
        """带降级重试的报表生成"""
        models = [self.model, "glm-4-flash"]  # 降级模型列表
        for model in models:
            for attempt in range(retries + 1):
                try:
                    # 临时切换模型
                    orig_model = self.model
                    self.model = model
                    result = self.generate(ctx)
                    self.model = orig_model
                    return result
                except Exception as e:
                    self.model = orig_model
                    if attempt == retries:
                        continue  # 下一个模型
        return f"[报表生成失败] 查询：{ctx.query}\n原始数据行数：{len(ctx.data)}"

4.3 LLM 驱动的图表推荐

进阶技巧：让 LLM 根据 SQL 结果的特征自动推荐合适的图表类型。比如有日期列 → 推荐折线图，有分类列 + 数值列 → 推荐柱状图。这比硬编码判断更灵活。 
# --- 图表推荐（基于 LLM 推理） ---
CHART_RECOMMEND_PROMPT = """根据以下 SQL 查询的列信息，推荐最合适的图表类型。

列信息：{columns}
样本数据：{sample}
查询类型：{query_type}

可选图表类型：
- bar: 柱状图（适合分类对比）
- line: 折线图（适合时间趋势）
- pie: 饼图（适合占比分布，不超过6类）
- scatter: 散点图（适合相关性分析）
- table: 纯表格（不适合可视化的数据）

请只返回 JSON 格式：
{"chart_type": "bar", "x_column": "...", "y_column": "...", "title": "...", "reason": "..."}"""

def recommend_chart(columns: list[str], sample: list[dict],
                   query: str, client: ZhipuAI) -> dict:
    """让 LLM 推荐合适的图表类型"""
    prompt = CHART_RECOMMEND_PROMPT.format(
        columns=columns,
        sample=json.dumps(sample[:3], ensure_ascii=False),
        query_type="分析类" if "分析" in query else "汇总类",
    )
    resp = client.chat.completions.create(
        model="glm-4-flash",
        messages=[{"role": "user", "content": prompt}],
        temperature=0.1,
    )
    # 提取 JSON（容错处理）
    content = resp.choices[0].message.content
    json_match = re.search(r'\{[^}]+\}', content)
    if json_match:
        return json.loads(json_match.group())
    return {"chart_type": "table", "reason": "无法解析推荐结果"}

import base64
from pathlib import Path
from io import BytesIO
import matplotlib.pyplot as plt

class ChartRenderer:
    """自动生成图表并转为 base64 内嵌到 HTML"""

    def __init__(self):
        plt.rcParams["font.sans-serif"] = ["SimHei", "Microsoft YaHei"]
        plt.rcParams["axes.unicode_minus"] = False

    def render_to_base64(self, chart_func) -> str:
        """执行绘图函数，返回 base64 编码的 PNG"""
        buf = BytesIO()
        fig = chart_func()  # 调用绘图函数，返回 fig
        fig.savefig(buf, format="png", dpi=150,
                   bbox_inches="tight", facecolor="white")
        plt.close(fig)
        buf.seek(0)
        b64 = base64.b64encode(buf.read()).decode("utf-8")
        return f"data:image/png;base64,{b64}"

    def render_bar(self, names, values, title="",
                 colors=None) -> str:
        """生成柱状图 base64"""
        def draw():
            fig, ax = plt.subplots(figsize=(8, 4))
            c = colors or ["#3b82f6"] * len(names)
            bars = ax.bar(names, values, color=c, width=0.5)
            for bar, v in zip(bars, values):
                ax.text(bar.get_x() + bar.get_width()/2, bar.get_height(),
                        str(v), ha="center", va="bottom", fontsize=10)
            ax.set_title(title, fontsize=13, fontweight=bold)
            ax.spines["top"].set_visible(False)
            ax.spines["right"].set_visible(False)
            return fig
        return self.render_to_base64(draw)

    def render_line(self, x_labels, y_values, title="",
                  label="", color="#dc2626") -> str:
        """生成折线图 base64"""
        def draw():
            fig, ax = plt.subplots(figsize=(8, 4))
            ax.plot(x_labels, y_values, "o-", color=color,
                    linewidth=2, markersize=6, label=label)
            ax.fill_between(range(len(x_labels)), y_values,
                            alpha=0.1, color=color)
            ax.legend()
            ax.set_title(title, fontsize=13, fontweight=bold)
            ax.grid(axis="y", alpha=0.3)
            return fig
        return self.render_to_base64(draw)


# --- 完整报表构建器 ---
class ReportBuilder:
    """整合数据 + LLM 分析 + 图表 + 模板的报表构建器"""

    # HTML 报表模板
    TEMPLATE = """
<html lang="zh-CN">
<head><meta charset="UTF-8">
<title>{{ title }}</title>
<style>
  body { font-family: sans-serif; max-width: 800px;
         margin: 0 auto; padding: 20px; background: #f8fafc; }
  .card { background: white; border-radius: 12px;
          padding: 24px; margin: 16px 0;
          border: 1px solid #e2e8f0; }
  h1 { color: #1e293b; font-size: 22px; }
  h2 { color: #3b82f6; font-size: 17px; margin-top: 20px; }
  .meta { color: #94a3b8; font-size: 13px; }
  .content { color: #475569; line-height: 1.8; font-size: 14px; }
  img { max-width: 100%; border-radius: 8px; margin: 12px 0; }
  table { width: 100%; border-collapse: collapse; margin: 12px 0; }
  th { background: #f1f5f9; padding: 8px 12px;
       text-align: left; font-size: 13px; }
  td { padding: 8px 12px; border-bottom: 1px solid #e2e8f0;
       font-size: 13px; color: #475569; }
</style>
</head>
<body>
  <div class="card">
    <h1>{{ title }}</h1>
    <p class="meta">生成时间：{{ created_at }} | 查询：{{ query }}</p>
  </div>

  <div class="card">
    <h2>📊 数据分析</h2>
    <div class="content">{{ analysis | safe }}</div>
  </div>

  {% if chart_image %}
  <div class="card">
    <h2>📈 数据可视化</h2>
    <img src="{{ chart_image }}" alt="数据图表">
  </div>
  {% endif %}

  {% if data %}
  <div class="card">
    <h2>📋 原始数据</h2>
    <table>
      <tr>{% for col in columns %}<th>{{ col }}</th>{% endfor %}</tr>
      {% for row in data %}
      <tr>{% for col in columns %}<td>{{ row[col] }}</td>{% endfor %}</tr>
      {% endfor %}
    </table>
  </div>
  {% endif %}
</body>
</html>
"""

    def __init__(self, report_gen: ReportGenerator,
                 chart_renderer: ChartRenderer):
        self.report_gen = report_gen
        self.chart = chart_renderer

    def build(self, ctx: ReportContext,
              chart_type: str = "auto") -> str:
        """构建完整的 HTML 报表"""
        # Step 1: LLM 生成分析文本
        analysis = self.report_gen.generate(ctx)

        # Step 2: 生成图表（如果有数值列）
        chart_image = None
        if chart_type != "none" and ctx.data:
            chart_image = self._auto_chart(ctx, chart_type)

        # Step 3: Jinja2 渲染 HTML
        tpl = Template(self.TEMPLATE)
        html = tpl.render(
            title=f"AMC 数据分析报告 — {ctx.period}",
            created_at=datetime.now().strftime("%Y-%m-%d %H:%M"),
            query=ctx.query,
            analysis=analysis,     # | safe 允许 HTML 标签
            chart_image=chart_image,
            data=ctx.data[:50],  # 最多展示50行
            columns=ctx.columns,
        )
        return html

    def _auto_chart(self, ctx: ReportContext,
                    chart_type: str) -> Optional[str]:
        """根据数据自动选择图表类型"""
        try:
            # 简单策略：有日期列用折线，有名称+数值用柱状
            date_cols = [c for c in ctx.columns
                       if "日期" in c or "时间" in c or "月份" in c]
            name_cols = [c for c in ctx.columns
                       if "名称" in c or "行业" in c or "企业" in c]
            num_cols = [c for c in ctx.columns
                       if isinstance(ctx.data[0].get(c), (int, float))]

            if date_cols and num_cols:
                # 折线图
                x = [str(r[date_cols[0]) for r in ctx.data]
                y = [r[num_cols[0]] for r in ctx.data]
                return self.chart.render_line(x, y, title=ctx.query)
            elif name_cols and num_cols:
                # 柱状图
                x = [str(r[name_cols[0]) for r in ctx.data[:10]]
                y = [r[num_cols[0]] for r in ctx.data[:10]]
                return self.chart.render_bar(x, y, title=ctx.query)
        except Exception:
            pass  # 图表生成失败不影响报表
        return None

    def save(self, html: str, output_path: str):
        """保存报表到文件"""
        Path(output_path).parent.mkdir(parents=True, exist_ok=True)
        Path(output_path).write_text(html, encoding="utf-8")
        print(f"报表已保存: {output_path}")

⚡ Python 知识补充：base64 与 BytesIO 5 min

上面用到了 base64 和 BytesIO，这两个在报表/图片处理中非常常用。

import base64
from io import BytesIO
from pathlib import Path

# --- BytesIO：内存中的"文件" ---
# 就像 SQL 临时表：在内存中操作，不落盘
buf = BytesIO()
buf.write(b"hello")
buf.seek(0)          # 指针回到开头（重要！）
content = buf.read()  # b'hello'

# matplotlib 的 fig.savefig() 可以直接写 BytesIO
fig = plt.figure()
plt.plot([1, 2, 3], [4, 5, 6])
buf = BytesIO()
fig.savefig(buf, format="png", dpi=150)
buf.seek(0)
png_bytes = buf.read()  # PNG 二进制数据，没落盘
plt.close(fig)

# --- base64：二进制 ↔ 字符串 ---
# 为什么需要？HTML 的 <img src="data:image/png;base64,..."> 需要 base64
b64_str = base64.b64encode(png_bytes).decode("utf-8")
img_tag = f'<img src="data:image/png;base64,{b64_str}">'

# 反向：base64 → 文件
# Path("output.png").write_bytes(base64.b64decode(b64_str))

# --- 文件 vs 内存 对比 ---
# 写文件：fig.savefig("chart.png")  ← 落盘，需要管理文件
# 写内存：fig.savefig(BytesIO())     ← 纯内存，适合嵌入 HTML/JSON
数仓类比：BytesIO 就像 SQL 的 WITH cte AS (...) —— 在内存中暂存中间结果，不需要创建物理临时表。base64 编码就像把 BLOB 字段转成 VARCHAR 存储 —— 用字符串形式携带二进制数据。
⚠️ 常见陷阱 Top 5
1 Jinja2 的 | safe 过滤器：LLM 返回的文本可能包含 <b>、<h3> 等 HTML 标签。不写 | safe，Jinja2 会自动转义成 &lt;b&gt;，格式全没了。但 | safe 有 XSS 风险 —— 确保数据来源可信。
2 matplotlib 中文字体：Windows 上 SimHei（黑体）和 Microsoft YaHei（微软雅黑）通常可用，但 Linux/Mac 上不一定。如果出现方块，需要指定系统自带的中文字体路径。
3 matplotlib 内存泄漏：每次创建 fig, ax 后一定要 plt.close(fig)。否则 figure 对象会一直堆积在内存里，生成几百张图后 OOM。
4 Token 爆炸：SQL 查出 10000 行直接喂给 LLM？不行。ReportContext.max_rows 控制展示行数，图表也只取 Top-N。大数据量时考虑先做聚合。
5 LLM 编造数据：LLM 生成报告时可能"创造"原文没有的数字。 mitigation：在 Prompt 中强调"仅基于查询结果中的数据，不要编造或推测具体数字"。

练习 3：LeetCode #49 — 字母异位词分组

用今天学的 defaultdict + sorted 完成。思路：对每个字符串排序后作为 key，相同 key 的字符串放到同一个 list。

from collections import defaultdict

def groupAnagrams(strs: list[str]) -> list[list[str]]:
    groups = defaultdict(list)
    for s in strs:
        key = "".join(sorted(s))  # "eat" → "aet"
        groups[key].append(s)
    return list(groups.values())

# 测试
print(groupAnagrams(["eat","tea","tan","ate","nat","bat"]))
# [["eat","tea","ate"], ["tan","nat"], ["bat"]]
