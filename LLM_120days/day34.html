经过前 5 天的学习，RAG 管线我们已经走了 80%：

用户问题 → D32 检索（从向量库召回 Top-20） → D33 Reranker（交叉编码器精排 Top-5） → D34 生成（把上下文+问题拼成 Prompt，调 LLM） ← 今天 → 最终回答
生成 = Prompt Engineering + LLM API 调用
RAG 的"生成"不是凭空编造，而是 将检索到的真实文档作为上下文注入 Prompt，让 LLM "基于材料作答"。类比数仓：离线层已经建好了宽表（知识库+索引），现在到了报表查询层的最后一步——把数据塞进模板生成报表。

核心问题：上下文塞不进去怎么办？

  • Token 限制：每个 LLM 都有最大上下文窗口（如 GLM-4 128K tokens），检索到的文档可能超出
  • 上下文污染：不相关的文档塞进去反而干扰 LLM 判断
  • 答案分散：答案可能跨越多个文档，需要合并
  • 格式要求：用户可能要表格、JSON、摘要等不同输出格式

# 多行拼接——RAG Prompt 模板的核心技巧
def build_prompt(question: str, docs: list[str], max_docs: int = 3) -> str:
    """构建 RAG Prompt：注入检索到的文档作为上下文"""
    context = "\n\n".join(
        f"【文档{i+1}】{doc}"
        for i, doc in enumerate(docs[:max_docs])
    )
    return f"""你是 AMC（资产管理公司）专业顾问。
请根据以下检索到的文档回答用户问题。
如果文档中没有足够信息，请明确说明。

{context}

用户问题：{question}

请给出准确、简洁的回答，并在回答末尾标注引用的文档编号。"""

# 使用
docs = [
    "AMC 2024年Q3不良资产余额为1250亿元，同比增长3.2%...",
    "不良资产处置方式包括：债务重组、债权转让、资产证券化...",
    "AMC 监管报送要求：每月5日前提交EAST格式数据..."
]
prompt = build_prompt("AMC的不良资产余额是多少？", docs)
print(prompt)
技巧：f-string 的大括号内可以写任意 Python 表达式（包括函数调用、列表推导、三元运算），但不要写太复杂的逻辑——可读性优先。复杂逻辑提取成变量再插值。

策略一：Stuff（全部塞进去）

原理：把所有检索到的文档一次性拼进 Prompt 的上下文区域。最简单直接，适用于文档少、总 token 不超限的情况。 
def stuff_generation(question: str, docs: list[str]) -> str:
    """Stuff 策略：全部文档塞进一个 Prompt"""
    context = "\n---\n".join(docs)  # 用分隔线区分文档
    prompt = f"""基于以下参考文档回答问题。

参考文档：
{context}

问题：{question}

回答："""
    return prompt

# 场景：检索到 3 条文档，总长约 2000 tokens
docs = [
    "AMC不良资产余额1250亿元...",
    "处置方式：债务重组、债权转让...",
    "EAST报送要求：每月5日前提交..."
]
prompt = stuff_generation("不良资产怎么处置？", docs)
print(prompt)  # 一次拼接，一次调用，一次返回
Stuff 策略优劣势 ✅ 优势实现简单，一次 LLM 调用就搞定；LLM 能看到全部文档，全局推理能力强 ❌ 劣势受上下文窗口限制（文档多就塞不下）；噪声文档会干扰答案质量 📌 适用文档 ≤ 5 条、总 token ≤ 8K、质量较高的场景

策略二：Map-Reduce（分而治之）

原理：文档太多塞不下？先把文档分组，每组单独问 LLM 得到部分答案（Map），再把所有部分答案合并，让 LLM 综合出最终答案（Reduce）。类比 SQL 的 GROUP BY + 聚合。 
import asyncio

# 模拟 LLM 调用（实际替换为真实 API）
async def mock_llm(prompt: str) -> str:
    """模拟 LLM 返回（实际项目中替换为智谱/OpenAI API）"""
    await asyncio.sleep(0.1)  # 模拟延迟
    return f"[模拟回答] 基于给定上下文的分析结果..."

async def map_reduce_generation(question: str, docs: list[str],
                                 chunk_size: int = 2) -> str:
    """
    Map-Reduce 策略：
    1. Map 阶段：文档分组 → 并行调 LLM → 各自产出部分答案
    2. Reduce 阶段：汇总部分答案 → 调 LLM → 生成最终答案
    """
    # ---- Map：分组处理 ----
    chunks = [docs[i:i+chunk_size] for i in range(0, len(docs), chunk_size)]

    map_tasks = []
    for i, chunk in enumerate(chunks):
        context = "\n".join(chunk)
        map_prompt = f"""基于以下文档片段回答问题。
文档片段：
{context}

问题：{question}
请给出该片段能回答的部分内容，如果没有相关信息请回答"无相关信息"。"""
        map_tasks.append(mock_llm(map_prompt))

    # 并发调用所有 Map 任务
    partial_answers = await asyncio.gather(*map_tasks)
    print(f"Map 阶段：{len(chunks)} 组 → {len(partial_answers)} 个部分答案")

    # ---- Reduce：合并汇总 ----
    summary = "\n".join(f"部分答案{i+1}：{ans}" for i, ans in enumerate(partial_answers))
    reduce_prompt = f"""以下是针对同一个问题的多个部分答案，请综合它们给出完整的最终回答。

{summary}

原始问题：{question}

请给出综合后的完整回答："""
    final_answer = await mock_llm(reduce_prompt)

    return final_answer

# 使用：10 条文档，每组 2 条 → 5 组 Map → 1 次 Reduce
docs = [f"文档{i+1}的内容..." for i in range(10)]
result = await map_reduce_generation("AMC不良资产处置流程是什么？", docs)
print(f"最终答案：{result}")

策略三：Refine（逐步精炼）

原理：先看第 1 个文档生成初始答案，再把答案 + 第 2 个文档一起给 LLM 精炼，如此迭代。每一步都"在上一步答案基础上改进"。类比 SQL 的 窗口函数累进计算。 
async def refine_generation(question: str, docs: list[str]) -> str:
    """
    Refine 策略：逐文档精炼
    初始答案 → +doc1 → 精炼1 → +doc2 → 精炼2 → ... → 最终答案
    """
    # Step 0：基于第一个文档生成初始答案
    initial_prompt = f"""基于以下文档回答问题。
文档：{docs[0]}

问题：{question}
回答："""
    current_answer = await mock_llm(initial_prompt)

    # Step 1..N：逐步精炼
    for i in range(1, len(docs)):
        refine_prompt = f"""你之前对问题的回答如下：
---
{current_answer}
---

现在又获得了一份新文档：
{docs[i]}

原始问题：{question}

请根据新文档中的信息，修正或补充你之前的回答。如果新文档没有相关新信息，保持原回答不变。
修正后的回答："""
        current_answer = await mock_llm(refine_prompt)
        print(f"  精炼第 {i+1} 轮完成")

    return current_answer

# 使用
docs = [f"文档{i+1}的内容..." for i in range(4)]
result = await refine_generation("AMC监管报送有哪些要求？", docs)
策略LLM 调用次数适用场景类比 Stuff 1 次 文档少（≤5），token 够用 SELECT * WHERE ... Map-Reduce N+1 次（N=分组数） 文档多，答案分散 GROUP BY + 聚合 Refine N 次（N=文档数） 需要逐步推理，顺序敏感 窗口函数累进
选型建议：实际项目 80% 用 Stuff 就够了（配合 Reranker 把文档精排到 Top-3~5）。Map-Reduce 适合批量处理大量文档；Refine 适合需要严格推理链条的场景（如法律条文解读、合同审核）。

4.2 关键技巧

def build_rag_prompt(question: str, docs: list[dict],
                       strategy: str = "stuff") -> str:
    """
    构建 RAG Prompt 的通用函数
    docs 格式：[{"id": 1, "text": "...", "score": 0.95}, ...]
    """
    # 技巧1：按相似度分数排序（高分在前）
    docs_sorted = sorted(docs, key=lambda d: d["score"], reverse=True)

    # 技巧2：给每段文档编号 + 来源标记
    context_parts = []
    for doc in docs_sorted:
        context_parts.append(
            f"[来源:{doc['id']}] {doc['text']}"
        )
    context_block = "\n\n".join(context_parts)

    # 技巧3：明确告知 LLM 约束条件
    prompt = f"""## 角色设定
你是 AMC（资产管理公司）的资深数据分析师，擅长从文档中提取关键信息并给出专业解读。

## 参考文档（按相关度排序）
{context_block}

## 回答要求
1. 仅基于以上参考文档中的信息作答，不要编造
2. 如果文档中没有足够信息，请明确说明"参考文档中未找到相关信息"
3. 回答时引用来源编号，如"根据[来源:1]..."
4. 如果涉及数值，请给出具体数字和单位

## 用户问题
{question}

## 回答
"""
    return prompt

4.3 上下文窗口管理：Token 预算分配

import textwrap

# Token 估算函数（中文约 1.5 字/token，英文约 0.75 词/token）
def estimate_tokens(text: str) -> int:
    """粗略估算 token 数（中英混合）"""
    chinese_chars = sum(1 for c in text if '\u4e00' <= c <= '\u9fff')
    other_chars = len(text) - chinese_chars
    return int(chinese_chars * 1.5 + other_chars * 0.25)

def allocate_budget(total_budget: int = 8000) -> dict:
    """
    Token 预算分配
    类比 SQL：total_budget = 上下文窗口 - 输出预留
    """
    return {
        "system_prompt": int(total_budget * 0.05),   # 5% → 400
        "context_docs": int(total_budget * 0.70),     # 70% → 5600
        "user_question": int(total_budget * 0.05),    # 5% → 400
        "output_reserve": int(total_budget * 0.20),   # 20% → 1600
    }

def fit_docs_in_budget(docs: list[str], budget: int) -> list[str]:
    """在 token 预算内尽量塞入最多的文档"""
    fitted = []
    used = 0
    for doc in docs:
        doc_tokens = estimate_tokens(doc)
        if used + doc_tokens <= budget:
            fitted.append(doc)
            used += doc_tokens
        else:
            # 尝试截断最后一份文档
            remaining = budget - used
            if remaining > 100:
                char_limit = int(remaining / 1.5)  # 中文粗估
                truncated = textwrap.shorten(doc, width=char_limit, placeholder="...")
                fitted.append(truncated)
                used += estimate_tokens(truncated)
            break
    print(f"Token 预算：{budget}，实际使用：{used}（{used/budget:.0%}）")
    return fitted

# 使用
budget = allocate_budget(8000)
print(f"Token 预算分配：{budget}")

docs = ["文档A..." * 500, "文档B..." * 300, "文档C..." * 400, "文档D..." * 600]
fitted = fit_docs_in_budget(docs, budget["context_docs"])

"""
完整 RAG 生成器
将 Day 32（检索）+ Day 33（Reranker）+ Day 34（生成）串成完整管线
"""
import asyncio
from dataclasses import dataclass, field
from string import Template
from typing import Callable, Awaitable

# ============================================================
# 数据模型
# ============================================================
@dataclass
class Document:
    """检索到的文档"""
    id: str
    text: str
    score: float        # 检索相似度分数
    rerank_score: float = 0.0  # 重排序分数（Day33 已讲）


@dataclass
class RAGConfig:
    """RAG 配置"""
    model: str = "glm-4"
    temperature: float = 0.3
    max_context_tokens: int = 5600   # 上下文 token 预算
    max_docs: int = 5                # 最大文档数
    score_threshold: float = 0.6    # 重排分数阈值


# ============================================================
# Token 管理
# ============================================================
def estimate_tokens(text: str) -> int:
    """粗略估算 token 数"""
    chinese = sum(1 for c in text if '\u4e00' <= c <= '\u9fff')
    other = len(text) - chinese
    return int(chinese * 1.5 + other * 0.25)

def fit_docs(documents: list[Document], budget: int) -> list[Document]:
    """在 token 预算内选文档（高分优先）"""
    result, used = [], 0
    for doc in documents:
        tokens = estimate_tokens(doc.text)
        if used + tokens <= budget:
            result.append(doc)
            used += tokens
        else:
            break
    return result

# Prompt 模板
# ============================================================

# ============================================================
# RAG 生成器核心类
# ============================================================
class RAGGenerator:
    """RAG 生成器：检索 → 重排 → 生成"""

    def __init__(self, config: RAGConfig | None = None,
                 llm_func: Callable[[str], Awaitable[str]] | None = None):
        self.config = config or RAGConfig()
        self.llm_func = llm_func  # LLM 调用函数，可替换为真实 API
        self._call_count = 0  # 统计 LLM 调用次数

    async def _call_llm(self, prompt: str) -> str:
        """调用 LLM（可替换为真实 API）"""
        self._call_count += 1
        if self.llm_func:
            return await self.llm_func(prompt)
        # 模拟返回
        return f"[模拟回答] 基于检索到的文档内容..."

    def _build_context_block(self, docs: list[Document]) -> str:
        """将文档列表格式化为上下文文本"""
        parts = []
        for doc in docs:
            # 显示重排分数（如果有的话）
            score_str = f"（相关度:{doc.rerank_score:.2f}）" if doc.rerank_score else ""
            parts.append(f"[来源:{doc.id}] {doc.text}")
        return "\n\n".join(parts)

    async def generate(self, question: str,
                       documents: list[Document]) -> dict:
        """
        完整 RAG 生成流程
        返回：{"answer": "...", "used_docs": [...], "tokens_used": N}
        """
        print(f"\n{'='*50}")
        print(f"RAG 生成开始 | 问题：{question}")
        print(f"{'='*50}")

        # Step 1：按重排分数排序 + 过滤低分文档
        docs = sorted(documents,
                      key=lambda d: d.rerank_score or d.score,
                      reverse=True)
        docs = [d for d in docs
                if (d.rerank_score or d.score) >= self.config.score_threshold]
        print(f"  过滤后文档数：{len(docs)}")

        # Step 2：Token 预算裁剪
        docs = fit_docs(docs, self.config.max_context_tokens)[:self.config.max_docs]
        print(f"  最终使用文档数：{len(docs)}")

        for doc in docs:
            score = doc.rerank_score or doc.score
            print(f"    [来源:{doc.id}] 相关度={score:.2f}")

        # Step 3：构建 Prompt
        context = self._build_context_block(docs)
        prompt = STUFF_PROMPT_TEMPLATE.substitute(
            context=context,
            question=question
        )
        prompt_tokens = estimate_tokens(prompt)
        print(f"  Prompt token 估算：{prompt_tokens}")

        # Step 4：调用 LLM
        answer = await self._call_llm(prompt)
        answer_tokens = estimate_tokens(answer)

        # Step 5：返回结果
        result = {
            "answer": answer,
            "used_docs": [{"id": d.id, "score": d.rerank_score or d.score}
                          for d in docs],
            "prompt_tokens": prompt_tokens,
            "answer_tokens": answer_tokens,
            "llm_calls": self._call_count,
        }
        print(f"  LLM 调用次数：{self._call_count}")
        print(f"  输出 token 估算：{answer_tokens}")

        return result

#290. 单词模式（Easy）

与 Prompt 模板中的变量替换逻辑高度相关——本质是 双射映射。

"""
290. Word Pattern
给定 pattern 和字符串 s，判断 s 是否遵循相同的模式。
pattern = "abba", s = "dog cat cat dog" → True
pattern = "abba", s = "dog cat cat fish" → False
pattern = "aaaa", s = "dog dog dog dog" → True

思路：双向映射（类似 SQL 的 UNIQUE 约束）
  a → dog 且 dog → a，不能 a→dog 又 a→cat（一对多）
"""
def word_pattern(pattern: str, s: str) -> bool:
    words = s.split()
    if len(pattern) != len(words):
        return False

    # 两个方向的映射
    char_to_word = {}  # pattern_char → word
    word_to_char = {}  # word → pattern_char

    for ch, word in zip(pattern, words):
        # 检查正向映射
        if ch in char_to_word:
            if char_to_word[ch] != word:
                return False
        else:
            char_to_word[ch] = word

        # 检查反向映射
        if word in word_to_char:
            if word_to_char[word] != ch:
                return False
        else:
            word_to_char[word] = ch

    return True

# 测试
print(word_pattern("abba", "dog cat cat dog"))   # True
print(word_pattern("abba", "dog cat cat fish"))  # False
print(word_pattern("aaaa", "dog dog dog dog"))    # True
print(word_pattern("abba", "dog dog dog dog"))    # False
