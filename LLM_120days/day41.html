2.1 字符串裁剪与 Token 估算

处理长上下文时，最常用的操作就是裁剪字符串。Python 提供了几种方式：

# === 方法1: 按字符数裁剪 ===
text = "这是一段很长的文本..." * 100
truncated = text[:2000]  # 直接截取前2000字符
print(f"原始长度: {len(text)}, 裁剪后: {len(truncated)}")

# === 方法2: textwrap.shorten 按词/字边界裁剪 ===
import textwrap
short = textwrap.shorten(text, width=500, placeholder="...[已截断]")
print(f"缩短后: {len(short)} 字符")

# === 方法3: 按句子边界裁剪（不切断句子） ===
import re

def truncate_by_sentence(text: str, max_chars: int) -> str:
    """按句子边界裁剪，确保不切断完整句子"""
    if len(text) <= max_chars:
        return text
    # 在限制范围内找最后一个句号/感叹号/问号
    boundary = max_chars
    for marker in ('。', '！', '？', '.', '!', '?', '\n'):
        last_pos = text.rfind(marker, 0, max_chars)
        if last_pos > boundary * 0.5:  # 至少保留一半以上
            boundary = max(boundary, last_pos + 1)
    return text[:boundary] + "...[已截断]"

result = truncate_by_sentence("第一句话。第二句话。第三句话。第四句话。", 20)
print(result)  # 输出: 第一句话。第二句话。...[已截断]

from typing import Optional

2.3 JSON Schema 校验 + Pydantic 自动验证

对 LLM 输出做严格校验，比正则更可靠：

from pydantic import BaseModel, Field, field_validator
from typing import Literal
from enum import Enum

class RiskLevel(str, Enum):
    LOW = "低"
    MEDIUM = "中"
    HIGH = "高"
    CRITICAL = "极高"

class RiskAnalysis(BaseModel):
    """LLM 输出的风险分析结果 —— 用 Pydantic 强约束"""
    risk_level: RiskLevel = Field(..., description="风险等级")
    risk_score: float = Field(
        ..., ge=0, le=100,
        description="风险分数 0-100"
    )
    key_factors: list[str] = Field(
        ..., min_length=1, max_length=5,
        description="关键风险因素，最多5个"
    )
    amount: float = Field(
        ..., gt=0,
        description="涉及金额（元），必须大于0"
    )
    recommendation: str = Field(
        ..., min_length=10, max_length=500,
        description="处置建议，10-500字"
    )

    @field_validator('risk_score')
    @classmethod
    def validate_score_consistency(cls, v, info):
        """风险分数必须与风险等级一致"""
        data = info.data
        if 'risk_level' in data:
            level_ranges = {
                RiskLevel.LOW: (0, 30),
                RiskLevel.MEDIUM: (30, 60),
                RiskLevel.HIGH: (60, 80),
                RiskLevel.CRITICAL: (80, 100),
            }
            low, high = level_ranges[data['risk_level']]
            if not (low <= v <= high):
                raise ValueError(
                    f"风险等级={data['risk_level'].value} 对应分数范围 [{low},{high}]，"
                    f"但实际分数={v}"
                )
        return v

# === 测试校验 ===
good_data = {
    "risk_level": "高",
    "risk_score": 72,
    "key_factors": ["不良率上升", "集中度超标"],
    "amount": 5000000,
    "recommendation": "建议立即增加拨备，并对重点客户进行压力测试。"
}
result = RiskAnalysis(**good_data)
print(f"校验通过: {result.risk_level}, 分数={result.risk_score}")

# 错误数据 —— 分数与等级不匹配
try:
    RiskAnalysis(**{**good_data, "risk_score": 20})  # "高"等级却只有20分
except Exception as e:
    print(f"校验失败: {e}")

# === 自动从 JSON 字符串解析 + 校验 ===
import json

def parse_llm_output(raw: str) -> Optional[RiskAnalysis]:
    """从 LLM 原始输出中解析并校验"""
    # 步骤1: 尝试提取 JSON
    for pattern in [r'```json\s*(.*?)\s*```', r'\{[\s\S]*\}']:
        match = re.search(pattern, raw, re.DOTALL)
        if match:
            try:
                data = json.loads(match.group(1))
                return RiskAnalysis(**data)  # 步骤2: Pydantic 校验
            except (json.JSONDecodeError, Exception) as e:
                print(f"解析/校验失败: {e}")
                return None
    return None
Pydantic vs 正则：正则适合从自由文本中提取片段；Pydantic 适合对提取后的数据做结构化校验。 实战中通常是正则提取 → json.loads → Pydantic 校验，三层防御。
🔒 核心主题一：输出约束 — 让 LLM 按规矩来 ⏱ 20 min

3.1 为什么要约束输出？

LLM 本质是"概率性文本生成器"，没有内置的"类型系统"。你不约束它，它就可能：

  • 该返回数字时返回文字："风险等级约为七成"
  • 该返回 JSON 时加上一堆废话："好的，这是您要的JSON："
  • 该返回 3 项时返回 20 项，或返回 0 项
  • 该返回专业术语时用口语化表达
输出约束的三个层次：
① 格式约束 — 输出是什么格式？（JSON / 纯文本 / Markdown 表格 / SQL）
② 内容约束 — 输出必须/不能包含什么？（必填字段 / 禁止词汇 / 值域范围）
③ 行为约束 — 输出该遵循什么行为？（只基于给定信息 / 不编造数据 / 遇到不确定要说明）

3.3 内容约束 + Pydantic 双重校验

最佳实践是 Prompt 里写约束 + 代码里做校验，形成双保险：

import json
from pydantic import BaseModel, Field, field_validator
from typing import Optional

class ConstrainedOutput(BaseModel):
    """带严格约束的 LLM 输出模型"""
    # 格式约束: 枚举值
    category: str = Field(..., pattern=r"^(高|中|低)$")

    # 内容约束: 范围
    score: int = Field(..., ge=0, le=100)

    # 内容约束: 长度
    summary: str = Field(..., min_length=20, max_length=500)

    # 内容约束: 正则
    report_id: str = Field(..., pattern=r"^RPT-\d{8}-\d{4}$")

    # 可选字段
    notes: Optional[str] = None

    @field_validator('summary')
    @classmethod
    def no_forbidden_words(cls, v):
        forbidden = ['幻觉', '猜测', '不确定', '大概']
        for word in forbidden:
            if word in v:
                raise ValueError(f"摘要中不允许包含'{word}'")
        return v


def call_llm_with_constraint(prompt: str, context: str) -> dict:
    """
    带 Prompt 约束 + Pydantic 校验 的 LLM 调用
    如果校验失败，自动重试（最多3次）
    """
    max_retries = 3
    for attempt in range(max_retries):
        # 1. 调用 LLM
        raw_response = call_llm(prompt, context)  # 假设的 LLM 调用

        # 2. 提取 JSON
        try:
            json_str = extract_json(raw_response)
            data = json.loads(json_str)
        except (json.JSONDecodeError, ValueError) as e:
            print(f"第{attempt+1}次: JSON解析失败 - {e}")
            # 自动追加约束提示，引导 LLM 纠正
            prompt += f"\n\n上次输出格式错误，请严格只输出JSON，不要任何其他文字。错误: {e}"
            continue

        # 3. Pydantic 校验
        try:
            validated = ConstrainedOutput(**data)
            return validated.model_dump()
        except Exception as e:
            print(f"第{attempt+1}次: 校验失败 - {e}")
            prompt += f"\n\n上次输出内容不符合要求。错误: {e}。请按格式重新输出。"
            continue

    raise RuntimeError(f"经过{max_retries}次尝试仍无法获得合规输出")


def extract_json(text: str) -> str:
    """从 LLM 输出中提取 JSON 字符串（支持多种格式）"""
    import re
    # 尝试: markdown代码块
    m = re.search(r'```(?:json)?\s*(\{.*?\})\s*```', text, re.DOTALL)
    if m:
        return m.group(1)
    # 尝试: 直接匹配最外层 {}
    m = re.search(r'\{.*\}', text, re.DOTALL)
    if m:
        return m.group(0)
    raise ValueError("未找到有效JSON内容")

📚 核心主题二：长上下文管理 — 信息太多怎么办 ⏱ 20 min

4.1 长上下文的问题

现代 LLM 支持 128K 甚至 1M Token 的上下文窗口，但"支持"≠"效果好"。现实问题：

  • Lost in the Middle — 中间的信息容易被忽略，首尾的信息更容易被关注
  • 注意力稀释 — 上下文越长，每个信息被"关注"的权重越低
  • Token 成本 — 长上下文 = 高 API 费用 + 高延迟
  • 关键信息淹没 — 100 页报告里的一句话关键条款，可能被完全忽略
核心原则：不是把所有信息塞给 LLM，而是只给最相关的信息。
这和数仓的分区裁剪一样 —— 你不会 SELECT * 全表扫描，而是 WHERE 条件精确过滤。

4.2 四种长上下文管理策略

原始文档 → 预处理 → 智能选择 → 结构化组织 → 精简Prompt 
from dataclasses import dataclass, field
from typing import Optional
import re

# ============================================================
# 策略1: 摘要压缩（Summarization）
# 将长文档先压缩成摘要，再给 LLM 做二次分析
# ============================================================

def summarize_document(text: str, max_chars: int = 4000) -> str:
    """
    分段摘要策略：把长文档分成多段，每段生成摘要，再合并
    适用于: 超长报告、批量文档分析
    """
    chunk_size = max_chars // 3  # 留余量给合并摘要
    chunks = [text[i:i+chunk_size] for i in range(0, len(text), chunk_size)]

    summaries = []
    for i, chunk in enumerate(chunks, 1):
        prompt = f"请用不超过200字概括以下内容的要点：\n\n{chunk}"
        summary = call_llm(prompt)  # 模拟调用
        summaries.append(f"【第{i}部分摘要】{summary}")

    # 用摘要再做一次合并摘要
    combined = "\n".join(summaries)
    final_prompt = f"请根据以下各部分摘要，生成一份综合摘要（不超过300字）：\n\n{combined}"
    return call_llm(final_prompt)


# ============================================================
# 策略2: 关键段落提取（Keyword / Section Extraction）
# 按关键词或段落标题定位相关部分，跳过无关内容
# ============================================================

def extract_relevant_sections(text: str, query: str, max_chars: int = 6000) -> str:
    """
    基于段落标题和关键词的相关性提取
    适用于: 结构化文档（有标题/章节的）
    """
    # 1. 提取关键词
    keywords = extract_keywords_from_query(query)

    # 2. 按段落分割
    sections = re.split(r'\n(?=#{1,3}\s)', text)  # 按 Markdown 标题分割

    # 3. 给每个段落打分
    scored_sections = []
    for section in sections:
        if not section.strip():
            continue
        score = sum(1 for kw in keywords if kw in section)
        # 标题匹配加分
        title_match = re.match(r'#{1,3}\s*(.+)', section)
        if title_match and any(kw in title_match.group(1) for kw in keywords):
            score += 3  # 标题匹配权重更高
        scored_sections.append((score, section))

    # 4. 按分数排序，截取
    scored_sections.sort(key=lambda x: x[0], reverse=True)

    result = []
    total_chars = 0
    for score, section in scored_sections:
        if score > 0 and total_chars + len(section) <= max_chars:
            result.append(section)
            total_chars += len(section)

    return "\n".join(result)


# ============================================================
# 策略3: Lost-in-the-Middle 缓解（信息重排）
# 把最重要的信息放在开头和结尾，不重要的放中间
# ============================================================

def reorder_for_attention(
    items: list[dict],
    key_field: str = "relevance_score",
) -> list[dict]:
    """
    将检索结果重排，高相关的放首尾，低相关的放中间
    缓解 "Lost in the Middle" 效应
    """
    if len(items) <= 2:
        return items

    # 按相关性排序
    sorted_items = sorted(items, key=lambda x: x[key_field], reverse=True)

    # 交替放入首尾
    reordered = []
    left, right = 0, len(sorted_items) - 1
    put_left = True  # 标记当前该放左边还是右边

    while left <= right:
        if put_left:
            reordered.append(sorted_items[left])
            left += 1
        else:
            reordered.insert(0, sorted_items[right])
            right -= 1
        put_left = not put_left

    return reordered


# ============================================================
# 策略4: 结构化 Prompt 组织（Section-Based Prompting）
# 用清晰的标记和分隔符组织长上下文
# ============================================================

@dataclass
class PromptBuilder:
    """结构化 Prompt 构建器 —— 管理长上下文"""
    system_prompt: str = ""
    context_sections: list[tuple[str, str]] = field(default_factory=list)  # (标题, 内容)
    question: str = ""
    max_context_chars: int = 8000  # 上下文总预算

    def add_section(self, title: str, content: str):
        """添加一个上下文段落"""
        self.context_sections.append((title, content))

    def build(self) -> str:
        """构建完整 Prompt（自动裁剪超长内容）"""
        parts = [f"# 系统指令\n{self.system_prompt}\n"]

        # 预算分配：每个段落平均分配额度
        n = len(self.context_sections)
        if n == 0:
            budget_per_section = self.max_context_chars
        else:
            budget_per_section = self.max_context_chars // n

        for i, (title, content) in enumerate(self.context_sections, 1):
            # 裁剪过长的段落
            if len(content) > budget_per_section:
                content = content[:budget_per_section] + "\n...[段落过长已截断]"
            parts.append(f"## 参考资料 {i}: {title}\n{content}\n")

        parts.append(f"# 问题\n{self.question}")
        return "\n---\n".join(parts)

    def char_count(self) -> int:
        """统计当前 Prompt 字符数（近似 Token 数）"""
        full_prompt = self.build()
        return len(full_prompt)


# === 使用示例 ===
builder = PromptBuilder(
    system_prompt="你是一个AMC不良资产分析专家，根据参考资料回答问题。",
    max_context_chars=6000,
)
builder.add_section("不良资产概览", overview_text)
builder.add_section("近期风险预警", warning_text)
builder.add_section("行业对比数据", industry_data)
builder.question = "本月不良资产的主要风险因素是什么？请给出处置建议。"

prompt = builder.build()
print(f"Prompt 长度: {len(prompt)} 字符")

4.3 长上下文最佳实践

✓ 推荐做法

• 先摘要再分析（两阶段）

• 关键信息放首尾

• 用分隔符标记不同段落

• 压缩无关的填充词

• 监控 Token 用量

✗ 常见错误

• 原文直接塞进 Prompt

• 所有检索结果平铺罗列

• 不标注信息来源

• 忽略 Token 成本

• 假设"窗口够大就够用"

🔄 核心主题三：多轮对话优化 — 保持记忆与一致性 ⏱ 15 min

5.1 多轮对话的挑战

单轮问答很简单，但多轮对话会引入一系列问题：

问题表现SQL 类比
上下文膨胀对话越长，Prompt 越大，成本越高每次全表扫描而非增量
角色漂移聊着聊着，AI 忘了自己的人设ETL 字段含义变更未同步
指代消解失败用户说"那个"或"它"，AI 不知道指什么JOIN 条件不明确导致笛卡尔积
信息遗忘早期对话中的重要信息被忽略历史数据被覆盖
重复/矛盾AI 在不同轮次给出矛盾的答案数据版本不一致

5.2 对话管理器：Memory + Context Window + Persona Guard

from collections import deque
from dataclasses import dataclass, field
from typing import Optional
import hashlib

@dataclass
class Message:
    """单条对话消息"""
    role: str           # "system" | "user" | "assistant"
    content: str
    timestamp: float = 0.0
    token_count: int = 0

@dataclass
class ConversationManager:
    """
    多轮对话管理器 — 三大核心能力：
    1. 滑动窗口：控制上下文长度（类似增量 ETL 只加载必要数据）
    2. 关键记忆：重要信息永不遗忘
    3. 角色守护：防止 AI 角色漂移
    """
    system_prompt: str = ""
    # 对话历史：deque 滑动窗口
    history: deque = field(default_factory=lambda: deque(maxlen=50))
    # 关键记忆：主动提取并持久化
    key_memories: list[str] = field(default_factory=list)
    # 角色设定
    persona: str = ""
    # Token 预算
    max_context_tokens: int = 4000
    # 约每 1.5 个中文字 ≈ 1 Token 的粗略估算
    chars_per_token: float = 1.5

    def __post_init__(self):
        if self.system_prompt:
            self._add_msg("system", self.system_prompt)

    def _add_msg(self, role: str, content: str):
        msg = Message(
            role=role,
            content=content,
            token_count=int(len(content) / self.chars_per_token),
        )
        self.history.append(msg)

    def add_user_message(self, content: str):
        self._add_msg("user", content)

    def add_assistant_message(self, content: str):
        self._add_msg("assistant", content)

    def build_context(self) -> list[dict]:
        """
        构建给 LLM 的上下文（自动裁剪）
        策略: system + 关键记忆 + 最近对话
        """
        messages = []
        total_tokens = 0

        # 1. System Prompt（始终保留）
        if self.system_prompt:
            sys_tokens = int(len(self.system_prompt) / self.chars_per_token)
            messages.append({"role": "system", "content": self.system_prompt})
            total_tokens += sys_tokens

        # 2. 关键记忆摘要（压缩版，始终保留）
        if self.key_memories:
            memory_text = "## 已知关键信息\n" + "\n".join(
                f"- {m}" for m in self.key_memories
            )
            memory_tokens = int(len(memory_text) / self.chars_per_token)
            messages.append({"role": "system", "content": memory_text})
            total_tokens += memory_tokens

        # 3. 最近对话（从后往前添加，直到预算用完）
        budget_left = self.max_context_tokens - total_tokens
        recent_msgs = list(self.history)

        # 跳过 system 消息（已添加）
        non_system = [m for m in recent_msgs if m.role != "system"]

        selected = []
        used = 0
        for msg in reversed(non_system):
            if used + msg.token_count <= budget_left:
                selected.append(msg)
                used += msg.token_count

        selected.reverse()  # 恢复时间顺序
        for msg in selected:
            messages.append({"role": msg.role, "content": msg.content})

        return messages

    def extract_key_info(self, text: str) -> Optional[str]:
        """
        从对话中提取需要长期记忆的关键信息
        例如: 用户姓名、偏好、重要决定
        """
        prompt = f"""从以下对话中提取需要长期记住的关键信息（用户偏好、重要决定、核心数据）。
如果没有关键信息，返回"无"。

对话: {text}

请用简洁的要点列出（每条不超过50字）:"""

        result = call_llm(prompt)  # 模拟调用
        if result.strip() != "无" and len(result) < 200:
            return result.strip()
        return None

    def update_memory(self, user_msg: str, assistant_msg: str):
        """每轮对话后更新关键记忆"""
        # 每 5 轮检查一次是否有新的关键信息
        if len(self.history) % 5 == 0:
            recent_text = "\n".join(
                m.content for m in list(self.history)[-6:]
            )
            key_info = self.extract_key_info(recent_text)
            if key_info:
                self.key_memories.append(key_info)
                # 限制记忆数量
                if len(self.key_memories) > 10:
                    self.key_memories.pop(0)

    def enforce_persona(self, prompt: str) -> str:
        """
        角色守护：在每轮 Prompt 中注入角色提醒
        防止 AI 在多轮对话后"忘掉"自己的角色
        """
        if not self.persona:
            return prompt
        # 在 Prompt 末尾追加轻量角色提醒
        reminder = f"\n\n[角色提醒: 你是{self.persona}。请保持专业、简洁的风格。]"
        return prompt + reminder

import json
import re
from pydantic import BaseModel, Field, field_validator
from typing import Optional, Literal
from collections import deque

# 1. 输出模型 —— Pydantic 强约束
# ============================================================

class RiskFactor(BaseModel):
    """单个风险因素"""
    name: str = Field(..., min_length=2, max_length=50)
    severity: Literal["低", "中", "高", "极高"]
    description: str = Field(..., min_length=10, max_length=200)

class RiskReport(BaseModel):
    """完整风险报告 —— LLM 输出目标"""
    report_id: str = Field(..., pattern=r"^RPT-\d{8}-\d{4}$")
    region: str = Field(..., min_length=2)
    period: str = Field(..., pattern=r"^\d{4}年\d{1,2}月$")
    overall_risk: Literal["低", "中", "高", "极高"]
    overall_score: float = Field(..., ge=0, le=100)
    total_amount: float = Field(..., gt=0, description="总涉及金额（元）")
    factors: list[RiskFactor] = Field(..., min_length=1, max_length=8)
    trend: Literal["上升", "持平", "下降"]
    recommendations: list[str] = Field(..., min_length=1, max_length=5)
    summary: str = Field(..., min_length=50, max_length=500)

    @field_validator('overall_score')
    @classmethod
    def score_matches_level(cls, v, info):
        mapping = {"低": (0,30), "中": (30,60), "高": (60,80), "极高": (80,100)}
        level = info.data.get('overall_risk', '')
        if level in mapping:
            lo, hi = mapping[level]
            if not (lo <= v <= hi):
                raise ValueError(f"{level}风险对应分数{lo}-{hi}，实际{v}")
        return v

# ============================================================
# 3. 安全输出提取器 —— 正则 + JSON 双保险
# ============================================================

def safe_extract_json(text: str) -> Optional[dict]:
    """多层容错的 JSON 提取"""
    # 第1层: 尝试直接解析
    text = text.strip()
    if text.startswith('{'):
        try:
            return json.loads(text)
        except json.JSONDecodeError:
            pass

    # 第2层: 提取 markdown 代码块
    m = re.search(r'```(?:json)?\s*(\{.*?\})\s*```', text, re.DOTALL | re.IGNORECASE)
    if m:
        try:
            return json.loads(m.group(1))
        except json.JSONDecodeError:
            pass

    # 第3层: 找到第一个 { 到最后一个 }
    first_brace = text.find('{')
    last_brace = text.rfind('}')
    if first_brace != -1 and last_brace > first_brace:
        try:
            return json.loads(text[first_brace:last_brace+1])
        except json.JSONDecodeError:
            pass

    return None

class ConstrainedReportGenerator:
    """
    带严格约束的 LLM 报告生成器
    整合: 输出约束 + 长上下文管理 + 多轮优化
    """
    def __init__(self, max_retries: int = 3):
        self.max_retries = max_retries
        self.prompt_builder = ReportPromptBuilder()
        self.conversation = deque(maxlen=6)  # 保留最近3轮（每轮2条消息）
        self.reports_generated = 0

    def _trim_context(self, context: str, max_chars: int = 4000) -> str:
        """裁剪过长的上下文"""
        if len(context) <= max_chars:
            return context
        # 保留开头和结尾各40%，中间20%
        head = int(max_chars * 0.4)
        tail = int(max_chars * 0.4)
        mid = max_chars - head - tail
        return (
            context[:head]
            + f"\n...[省略{len(context) - head - tail}字符]...\n"
            + context[-tail:]
        )

    def _persona_reminder(self) -> str:
        """角色守护提醒"""
        self.reports_generated += 1
        return f"[提醒: 你是AMC风险分析专家，已生成{self.reports_generated}份报告。保持一致性。]"

    def generate(
        self,
        context: str,
        region: str,
        period: str,
    ) -> Optional[RiskReport]:
        """生成风险报告（自动重试 + 自动纠正）"""
        context = self._trim_context(context)
        base_prompt = self.prompt_builder.build(
            context=context,
            persona_reminder=self._persona_reminder(),
        )

        for attempt in range(self.max_retries):
            raw_output = call_llm(base_prompt)  # 模拟 LLM 调用

            # 记录对话
            self.conversation.append(("user", base_prompt))
            self.conversation.append(("assistant", raw_output))

            # 提取 JSON
            data = safe_extract_json(raw_output)
            if data is None:
                base_prompt += (
                    f"\n\n[自动纠正 第{attempt+1}次] "
                    "输出未包含有效JSON。请严格只输出JSON对象，不要任何其他文字。"
                )
                continue

            # Pydantic 校验
            try:
                report = RiskReport(**data)
                print(f"报告生成成功: {report.report_id}, 风险={report.overall_risk}")
                return report
            except Exception as e:
                base_prompt += (
                    f"\n\n[自动纠正 第{attempt+1}次] "
                    f"校验失败: {e}。请修正后重新输出JSON。"
                )
                continue

        print(f"经过 {self.max_retries} 次尝试仍无法生成合规报告")
        return None

⚠️ 常见陷阱 Top 5
  1. 只靠 Prompt 不做代码校验
    Prompt 约束是"软约束"，LLM 不保证遵守。关键场景必须 Pydantic 校验兜底。
  2. 长上下文不管 Token 成本
    把 10 页文档塞进 Prompt 调用 API，一次就烧掉几万 Token。先摘要/先分块/先检索再调用。
  3. Lost in the Middle 不做重排
    10 条检索结果平铺罗列，中间的关键信息被忽略。高相关的放首尾。
  4. 多轮对话不设滑动窗口
    聊了 50 轮还不裁剪历史，Token 膨胀 + 成本爆炸 + 早期信息失效。
  5. System Prompt 太长太模糊
    System Prompt 里写了 2000 字的角色设定，反而挤占了有效上下文。精简 + 结构化 + 分层。
