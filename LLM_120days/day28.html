分层原则

层职责类比数仓关键约束
models定义数据结构建表 DDL只放类定义，不放业务逻辑
services业务逻辑ETL 脚本不依赖 FastAPI，可独立测试
apiHTTP 接口报表服务接口只做参数校验 + 调 service
core基础设施调度系统/元数据被其他所有层引用
核心设计原则：api 层调 services 层，services 层调 core 层。
想象数仓：ADS 调 DWS，DWS 调 DWD，DWD 调 ODS —— 数据只能自上而下引用，不能反向。

3.1 AMC 业务模型 — models/amc.py

用 Day 19-21 学的 OOP，把 AMC 的核心业务实体建模成类：

# models/amc.py
from dataclasses import dataclass, field
from datetime import date
from typing import Optional
from enum import Enum


class AssetStatus(str, Enum):
    """资产状态枚举 —— 用 Day 22 学的枚举模式"""
    NORMAL = "正常"
    OVERDUE = "逾期"
    NPL = "不良"       # Non-Performing Loan
    WRITE_OFF = "核销"
    DISPOSED = "处置"


@dataclass
class Debtor:
    """债务人类 —— 类比数仓 dim_debtor 表"""
    debtor_id: str
    name: str
    id_type: str            # 证件类型
    id_number: str          # 证件号（脱敏）
    industry: Optional[str] = None
    region: Optional[str] = None

    @property
    def masked_name(self) -> str:
        """姓名脱敏 —— @property 计算属性（Day 21 学的）"""
        if len(self.name) <= 2:
            return self.name[0] + "*"
        return self.name[0] + "*" * (len(self.name) - 2) + self.name[-1]


@dataclass
class NonPerformingAsset:
    """不良资产 —— 类比数仓 dwd_npl_asset 表"""
    asset_id: str
    debtor: Debtor
    principal: float        # 本金余额
    interest: float         # 利息余额
    status: AssetStatus
    category: str           # 资产类别（贷款/债券/垫款）
    acquire_date: date      # 收购日期
    dispose_date: Optional[date] = None

    @property
    def total_balance(self) -> float:
        """总余额 = 本金 + 利息"""
        return self.principal + self.interest

    @property
    def days_held(self) -> int:
        """持有天数"""
        end = self.dispose_date or date.today()
        return (end - self.acquire_date).days

    @property
    def is_over_threshold(self) -> bool:
        """是否超过管理阈值（本金 > 1000万）"""
        return self.principal > 10_000_000

    def __str__(self) -> str:
        """自定义打印 —— Day 19 学的 __str__"""
        return (
            f"[{self.asset_id}] {self.debtor.masked_name} | "
            f"本金: {self.principal:,.0f} | 状态: {self.status.value}"
        )

    def __lt__(self, other: "NonPerformingAsset") -> bool:
        """按本金排序 —— Day 21 学的 __lt__"""
        return self.principal < other.principal

    def to_dict(self) -> dict:
        """转为字典，方便序列化"""
        return {
            "asset_id": self.asset_id,
            "debtor_name": self.debtor.masked_name,
            "principal": self.principal,
            "interest": self.interest,
            "total_balance": self.total_balance,
            "status": self.status.value,
            "category": self.category,
            "days_held": self.days_held,
        }


@dataclass
class AMCReport:
    """AMC 报表 —— 类比数仓 ads 层的汇总表"""
    report_date: date
    total_assets: int = 0          # 资产总数
    total_principal: float = 0     # 本金总额
    total_interest: float = 0      # 利息总额
    npl_count: int = 0             # 不良笔数
    npl_ratio: float = 0.0         # 不良率

    @classmethod
    def from_assets(cls, assets: list[NonPerformingAsset]) -> "AMCReport":
        """工厂方法 —— Day 21 学的 @classmethod"""
        total = len(assets)
        npls = [a for a in assets if a.status == AssetStatus.NPL]
        return cls(
            report_date=date.today(),
            total_assets=total,
            total_principal=sum(a.principal for a in assets),
            total_interest=sum(a.interest for a in assets),
            npl_count=len(npls),
            npl_ratio=len(npls) / total if total > 0 else 0.0,
        )
💡 数据开发视角：每个 dataclass 就是一张表的 Schema 定义。
@property 就像 SQL 的计算列：SELECT principal + interest AS total_balance。
from_assets 工厂方法就像 INSERT INTO ads_report SELECT ... FROM dwd_asset GROUP BY。

这是整个平台的核心 —— LLM 调用服务。把 Day 24 学的异步 + Day 23 学的策略模式结合起来：

# services/llm_service.py
import asyncio
import logging
from abc import ABC, abstractmethod
from typing import Optional
import httpx

from config import settings
from core.exceptions import LLMTimeoutError

logger = logging.getLogger("amc.llm")


# ===== 策略模式：LLM 提供商抽象基类（Day 23 设计模式）=====
class LLMProvider(ABC):
    """LLM 提供商抽象基类 —— 策略模式中的 Strategy"""

    @abstractmethod
    async def chat(self, messages: list[dict],
                   temperature: float = 0.7) -> str:
        """发送对话请求"""
        ...

    @abstractmethod
    async def count_tokens(self, text: str) -> int:
        """估算 token 数"""
        ...


class ZhipuProvider(LLMProvider):
    """智谱 GLM 提供商 —— 具体策略"""

    def __init__(self):
        self.api_key = settings.llm_api_key
        self.base_url = settings.llm_base_url
        self.model = settings.llm_model
        self._client: Optional[httpx.AsyncClient] = None

    async def _get_client(self) -> httpx.AsyncClient:
        """延迟创建 HTTP 客户端（懒加载）"""
        if self._client is None:
            self._client = httpx.AsyncClient(
                base_url=self.base_url,
                headers={"Authorization": f"Bearer {self.api_key}"},
                timeout=settings.llm_timeout
            )
        return self._client

    async def chat(self, messages: list[dict],
                   temperature: float = 0.7) -> str:
        """调用智谱 API —— Day 24 学的异步 HTTP"""
        client = await self._get_client()

        # 指数退避重试（Day 24 学的）
        last_error = None
        for attempt in range(settings.llm_max_retries):
            try:
                logger.debug(f"LLM 请求尝试 {attempt + 1}/{settings.llm_max_retries}")

                response = await client.post(
                    "/chat/completions",
                    json={
                        "model": self.model,
                        "messages": messages,
                        "temperature": temperature,
                    }
                )
                response.raise_for_status()

                data = response.json()
                content = data["choices"][0]["message"]["content"]
                tokens = data.get("usage", {}).get("total_tokens", 0)

                logger.info(f"LLM 响应成功，tokens={tokens}")
                return content

            except httpx.TimeoutException as e:
                last_error = e
                # 指数退避：1s, 2s, 4s
                wait = 2 ** attempt
                logger.warning(f"LLM 超时，{wait}s 后重试...")
                await asyncio.sleep(wait)

            except httpx.HTTPStatusError as e:
                last_error = e
                if e.response.status_code == 429:  # 限流
                    wait = 2 ** attempt
                    logger.warning(f"LLM 限流 429，{wait}s 后重试...")
                    await asyncio.sleep(wait)
                else:
                    raise  # 非 429 错误直接抛出

        raise LLMTimeoutError()

    async def count_tokens(self, text: str) -> int:
        """简单估算：中文约 1.5 字/token，英文约 4 字母/token"""
        chinese_chars = sum(1 for c in text if '\u4e00' <= c <= '\u9fff')
        other_chars = len(text) - chinese_chars
        return int(chinese_chars / 1.5 + other_chars / 4)


# ===== 工厂模式：LLM 服务入口（Day 23 学的）=====
def create_llm_provider() -> LLMProvider:
    """工厂函数：根据配置创建 LLM 提供商"""
    return ZhipuProvider()


# ===== 单例模式：全局 LLM 服务（Day 23 + Day 27 学的）=====
class LLMService:
    """
    LLM 服务门面类（Facade 模式）
    封装了：Provider 选择 + Prompt 构建 + 重试逻辑
    对外只暴露一个简单的 ask() 方法
    """

    _instance = None

    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            cls._instance._provider = create_llm_provider()
            cls._instance._semaphore = asyncio.Semaphore(
                settings.max_concurrent  # 并发限流（Day 24 学的）
            )
        return cls._instance

    async def ask(self, question: str,
                  system_prompt: str = "") -> tuple[str, int]:
        """
        发送问题到 LLM

        Returns:
            (回答文本, token 消耗)
        """
        messages = []
        if system_prompt:
            messages.append({"role": "system", "content": system_prompt})
        messages.append({"role": "user", "content": question})

        # 信号量限流，防止并发过高
        async with self._semaphore:
            answer = await self._provider.chat(messages)
            tokens = self._provider.count_tokens(
                question + answer
            )
            return answer, tokens

    async def batch_ask(self, questions: list[str],
                        system_prompt: str = "") -> list[tuple[str, int]]:
        """
        批量并发提问 —— Day 24 学的 asyncio.gather

        Returns:
            [(回答, tokens), ...]  按原始顺序返回
        """
        tasks = [
            self.ask(q, system_prompt)
            for q in questions
        ]
        results = await asyncio.gather(*tasks, return_exceptions=True)

        # 处理异常
        final = []
        for i, r in enumerate(results):
            if isinstance(r, Exception):
                logger.error(f"批量提问第 {i+1} 个失败: {r}")
                final.append((f"[错误] {r}", 0))
            else:
                final.append(r)
        return final
这里用到的设计模式：
  • 策略模式：LLMProvider 抽象基类 → 未来切换 DeepSeek/OpenAI 只需加一个子类
  • 工厂模式：create_llm_provider() 根据配置自动选择提供商
  • 单例模式：LLMService.__new__ 保证全局只有一个实例 + 一个 HTTP 连接池
  • 门面模式：LLMService.ask() 隐藏了 Provider/重试/限流等复杂度
就像数仓里，你用视图封装了底层多表关联，对外只暴露一个简单字段。

class QueryService:
    """自然语言查询服务"""

    def __init__(self):
        self.llm = LLMService()  # 单例，不会重复创建

    async def ask(self, question: str, user_id: str = "") -> dict:
        """
        处理用户的自然语言查询

        流程：用户提问 → LLM 生成 SQL → 模拟执行 → 返回结果

        Returns:
            {"answer": "中文回答", "sql": "生成的SQL", "data": [...]}

        logger.info(f"用户 {user_id} 提问: {question[:50]}...")

        answer, tokens = await self.llm.ask(question, SYSTEM_PROMPT)

        if sql:

        return {
            "answer": self._clean_answer(answer),
            "sql": sql,
            "data": data,
            "tokens_used": tokens,
        }

    def _extract_sql(self, text: str) -> str | None:
        """从 LLM 回答中提取 SQL"""
        import re
        match = re.search(r"```sql\s*(.*?)```", text, re.DOTALL)
        if match:
            return match.group(1).strip()
        return None

    def _mock_execute(self, sql: str) -> list[dict]:
        """模拟 SQL 执行 —— 真实项目中替换为数据库查询"""
        # 简单过滤模拟
        sql_lower = sql.lower()

        if "count" in sql_lower:
            return [{"count": len(MOCK_DATA)}]
        elif "sum" in sql_lower:
            total = sum(r["principal"] for r in MOCK_DATA)
            return [{"total_principal": total}]
        elif "group by" in sql_lower:
            # 按地区分组
            from collections import defaultdict
            groups = defaultdict(list)
            for r in MOCK_DATA:
                groups[r["region"]].append(r)
            return [
                {"region": k, "count": len(v),
                 "total": sum(x["principal"] for x in v)}
                for k, v in groups.items()
            ]
        else:
            return MOCK_DATA[:5]  # 默认返回前5条

    def _clean_answer(self, answer: str) -> str:
        """清理回答（去掉 SQL 代码块标记）"""
        import re
        return re.sub(r"```sql.*?```", "[SQL 已生成]", answer, flags=re.DOTALL)
💡 数据开发视角：AMC_SCHEMA 就是你在做 NL2SQL 时的"表结构字典"。
这个模式和你日常做的事一样 —— 写文档告诉别人表结构，只不过这里的"别人"是 LLM。

    user_id: str = Depends(verify_api_key)
) -> str:

    配额检查 —— 依赖链：先认证 → 再查配额
    这里用到了 Day 26 学的依赖链概念
    """
    used = _quota_store.get(user_id, 0)

            detail=f"今日已用 {used}/{settings.max_daily_queries} 次"
        )
    _quota_store[user_id] = used + 1
    logger.info(f"用户 {user_id} 配额: {used+1}/{settings.max_daily_queries}")
    return user_id

练习 3（LeetCode #146 LRU 缓存，预热 Phase 2）：

实现一个 LRU 缓存类，支持 get/put 操作，容量有限，超出时淘汰最近最少使用的 key。

from collections import OrderedDict

class LRUCache:
    """LRU 缓存 —— Day 26 学过的 OrderedDict 正好能实现"""

    def __init__(self, capacity: int):
        self.capacity = capacity
        self.cache: OrderedDict = OrderedDict()

    def get(self, key: int) -> int:
        if key not in self.cache:
            return -1
        # 命中后移到末尾（标记为最近使用）
        self.cache.move_to_end(key)
        return self.cache[key]

    def put(self, key: int, value: int) -> None:
        if key in self.cache:
            self.cache.move_to_end(key)
        self.cache[key] = value
        # 超出容量，移除最前面的（最久未使用）
        if len(self.cache) > self.capacity:
            self.cache.popitem(last=False)  # FIFO 方向弹出

OrderedDict 的 move_to_end + popitem(last=False) 天然就是 LRU 的实现。面试常考，也和 Phase 2 的缓存策略直接相关。
