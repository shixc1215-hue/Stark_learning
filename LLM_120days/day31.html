暴力搜索 vs ANN（近似最近邻）

Day 30 中我们手写的是「暴力搜索」——拿 query 向量和 所有 向量逐一算余弦相似度。这在数据量小的时候没问题，但当数据量从 100 → 100万 时：

# 暴力搜索 —— O(N×d)，N=文档数，d=向量维度
def brute_search(query_vec, all_vectors, top_k=5):
    scores = [(i, cosine_sim(query_vec, v)) for i, v in enumerate(all_vectors)]
    scores.sort(key=lambda x: x[1], reverse=True)
    return scores[:top_k]

# 1000 条 → 毫秒级  ✅ 可接受
# 100 万条 → 每次查询 0.5~2 秒  ⚠️ 太慢
# 1 亿条 → 每次查询 50~200 秒  ❌ 不可用
ANN（Approximate Nearest Neighbor）的核心思想：不用和 每条 数据比，通过索引结构快速定位到「大概率最近的候选集」，再精确计算。
HNSW（Hierarchical Navigable Small World）是最主流的 ANN 算法 —— 把向量组织成多层图结构，类似"先坐飞机到大致区域，再坐车到具体地点"。精度损失不到 1%，速度提升 100~1000 倍。

1. @contextmanager 装饰器 —— 更灵活的上下文管理

Day 21 学过 __enter__/__exit__ 魔术方法来实现 with 语句。Python 还提供了更简洁的方式：

from contextlib import contextmanager

# 用装饰器写上下文管理器，比写类简单很多
@contextmanager
def timer(name: str):
    """计时上下文：自动记录代码块执行时间"""
    import time
    start = time.perf_counter()
    print(f"[{name}] 开始...")
    try:
        yield  # ← yield 之前的代码 = __enter__，yield 之后 = __exit__
    finally:
        elapsed = time.perf_counter() - start
        print(f"[{name}] 完成，耗时 {elapsed:.3f}s")

# 使用
with timer("向量化"):
    embeddings = [get_embedding(t) for t in texts]
# 输出：[向量化] 开始...
# 输出：[向量化] 完成，耗时 1.234s


# 结合连接/关闭数据库的场景
@contextmanager
def db_connection(db_path):
    """模拟数据库连接的上下文管理"""
    print(f"  连接数据库: {db_path}")
    conn = {"path": db_path, "active": True}  # 模拟连接对象
    try:
        yield conn  # 把连接对象传给 as 后面的变量
    finally:
        conn["active"] = False
        print(f"  关闭数据库连接")

with db_connection("./data/amc.db") as conn:
    print(f"  当前连接状态: {conn['active']}")
    # ... 执行查询 ...
# 输出：连接数据库: ./data/amc.db
# 输出：当前连接状态: True
# 输出：关闭数据库连接


# 错误处理：yield 处可以 yield 一个值
@contextmanager
def safe_file_write(filepath):
    """安全的文件写入：写入临时文件，成功后原子替换"""
    import os
    tmp_path = filepath + ".tmp"
    f = open(tmp_path, "w", encoding="utf-8")
    try:
        yield f
        f.close()
        # 成功后原子替换（数据仓库常用技巧，避免写到一半断电）
        os.replace(tmp_path, filepath)
    except Exception as e:
        f.close()
        if os.path.exists(tmp_path):
            os.remove(tmp_path)
        raise  # 重新抛出异常

# 使用
with safe_file_write("output.json") as f:
    f.write('{"data": "test"}')  # 成功时才替换原文件

4. 自定义 JSONEncoder —— 处理特殊类型

import json
from datetime import datetime
from pathlib import Path

# 标准 JSON 只支持 dict/list/str/int/float/bool/None
# 遇到 datetime、set、Path 就会报 TypeError

class AdvancedEncoder(json.JSONEncoder):
    """自定义 JSON 编码器，处理特殊类型"""
    def default(self, obj):
        if isinstance(obj, datetime):
            return obj.isoformat()       # datetime → 字符串
        if isinstance(obj, set):
            return list(obj)            # set → list
        if isinstance(obj, Path):
            return str(obj)             # Path → 字符串
        if isinstance(obj, bytes):
            return obj.decode("utf-8")  # bytes → 字符串
        # 其他类型交给父类处理（会抛 TypeError）
        return super().default(obj)

# 使用
data = {
    "report_date": datetime(2026, 6, 30),
    "categories": {"不良", "关注", "次级"},   # set
    "file_path": Path("D:/data/report.csv"),
}
json_str = json.dumps(data, cls=AdvancedEncoder, ensure_ascii=False, indent=2)
print(json_str)
# {
#   "report_date": "2026-06-30T00:00:00",
#   "categories": ["次级", "关注", "不良"],   ← set 变成了 list

5. 元数据过滤 —— where 参数

这是向量库最强大的功能之一：语义匹配 + 结构化过滤 组合查询。类比 SQL：

-- SQL 类比
SELECT * FROM amc_knowledge_base
WHERE chapter = '分类'
ORDER BY similarity(embedding_col, query_vec)
LIMIT 3;
# ---- 基础过滤 ----
results = collection.query(
    query_texts=["贷款分类标准"],
    where={"source": "regulation"},  # 只在 regulation 类文档中搜索
    n_results=3,
)

# ---- $and 组合条件 ----
results = collection.query(
    query_texts=["资产处置"],
    where={
        "$and": [
            {"source": "regulation"},
            {"chapter": {"$ne": "总则"}},  # ne = not equal
        ]
    },
    n_results=3,
)

# ---- $or 或条件 ----
results = collection.query(
    query_texts=["AMC业务"],
    where={
        "$or": [
            {"source": "regulation"},
            {"source": "wiki"},
        ]
    },
    n_results=3,
)

# ---- $in 包含于 ----
results = collection.query(
    query_texts=["不良资产"],
    where={"chapter": {"$in": ["分类", "处置"]}},  # chapter 是 "分类" 或 "处置"
    n_results=3,
)

# ---- 数值比较 ----
collection.add(
    documents=["2024年Q3不良资产报告", "2025年Q1不良资产报告"],
    ids=["doc_100", "doc_101"],
    metadatas=[{"year": 2024, "quarter": 3}, {"year": 2025, "quarter": 1}],
)
results = collection.query(
    query_texts=["最新报告"],
    where={"year": {"$gte": 2025}},  # gte = greater than or equal
    n_results=3,
)

# ---- where_document：在原文上过滤 ----
results = collection.query(
    query_texts=["资产处置"],
    where_document={"$contains": "转让"},  # 原文包含 "转让" 的才匹配
    n_results=3,
)
💡 where 过滤速查表：
等值: {"field": "value"} · 不等于: {"$ne": "value"}
包含: {"$in": ["a","b"]} · 数值: {"$gt"/"$gte"/"$lt"/"$lte": num}
组合: {"$and": [...]} · {"$or": [...]}
原文过滤: {"$contains": "keyword"}（在 where_document 里）

🔗 自定义 Embedding Function —— 接入智谱 API ⏱ 10 min

Chroma 默认用英文模型 all-MiniLM-L6-v2。我们的 AMC 知识库是中文的，需要换成中文 Embedding：

import chromadb
import requests
from typing import List, Optional

class ZhipuEmbeddingFunction:
    """
    自定义 Embedding 函数：对接智谱 embedding-3 API
    Chroma 要求实现 __call__(self, input: List[str]) -> List[List[float]]
    """

    def __init__(self, api_key: str, model: str = "embedding-3"):
        self.api_key = api_key
        self.model = model
        self.url = "https://open.bigmodel.cn/api/paas/v4/embeddings"

    def __call__(self, input: List[str]) -> List[List[float]]:
        """
        Chroma 在 add()/query() 时会自动调用这个方法
        input: 文本列表（可能是一条，也可能是多条）
        返回: 向量列表（每条文本对应一个向量）
        """
        # 批量请求
        resp = requests.post(
            self.url,
            headers={"Authorization": f"Bearer {self.api_key}"},
            json={"model": self.model, "input": input},
            timeout=30,
        )
        resp.raise_for_status()

        # 解析响应，按原顺序排列
        data_list = resp.json()["data"]
        data_list.sort(key=lambda x: x["index"])  # 确保顺序一致
        return [item["embedding"] for item in data_list]


# ---- 使用自定义 Embedding Function ----
embedding_fn = ZhipuEmbeddingFunction(api_key="your_api_key")

collection = client.get_or_create_collection(
    name="amc_zhipu",
    embedding_function=embedding_fn,  # 传入自定义函数
)

# 之后的 add/query 会自动用智谱 API 计算 embedding
collection.add(
    documents=["不良资产是指借款人未能按期偿还的贷款"],
    ids=["zh_001"],
)
results = collection.query(query_texts=["什么是不良资产"], n_results=1)
print(results["documents"][0])
Chroma EmbeddingFunction 接口规范：
只需实现一个 __call__ 方法：input: List[str] → List[List[float]]
Chroma 在 add() 和 query() 时自动调用。你可以接入任意 API（智谱/OpenAI/BGE/本地模型）。 
# ---- 纯自定义 EmbeddingFunction（不调用API）----
import numpy as np

class RandomEmbeddingFunction:
    """Demo 用：随机向量（仅用于测试，不要用于生产！）"""
    def __init__(self, dim: int = 512):
        self.dim = dim
    def __call__(self, input: List[str]) -> List[List[float]]:
        return [np.random.rand(self.dim).tolist() for _ in input]

class CachedEmbeddingFunction:
    """带缓存的 Embedding：避免重复计算"""
    def __init__(self, real_fn):
        self.real_fn = real_fn
        self.cache = {}
    def __call__(self, input: List[str]) -> List[List[float]]:
        results = []
        for text in input:
            if text not in self.cache:
                self.cache[text] = self.real_fn([text])[0]
            results.append(self.cache[text])
        return results

# 用法：把任意 EmbeddingFunction 包上缓存
cached_fn = CachedEmbeddingFunction(ZhipuEmbeddingFunction(api_key="xxx"))

把前面的知识整合起来，封装一个完整的 AMCKnowledgeBase 类：

import chromadb
import json
import time
from pathlib import Path
from contextlib import contextmanager
from typing import List, Dict, Optional, Any


class AMCKnowledgeBase:
    """
    AMC 不良资产知识库
    整合：Chroma 持久化存储 + 自定义 Embedding + 元数据过滤 + 批量导入
    """

    def __init__(self, persist_dir: str = "./chroma_amc",
                 collection_name: str = "amc_docs",
                 api_key: str = ""):
        """初始化知识库"""
        self.persist_dir = Path(persist_dir)
        self.persist_dir.mkdir(parents=True, exist_ok=True)

        # 使用持久化客户端
        self.client = chromadb.PersistentClient(path=str(self.persist_dir))

        # 创建/获取 Collection
        self.collection = self.client.get_or_create_collection(
            name=collection_name,
            metadata={"created": time.strftime("%Y-%m-%d %H:%M:%S")},
        )
        print(f"[AMCKnowledgeBase] 初始化完成，共 {self.collection.count()} 条文档")

    def __enter__(self):
        """支持 with 语法"""
        return self

    def __exit__(self, exc_type, exc_val, exc_tb):
        """with 退出时自动关闭"""
        pass  # Chroma PersistentClient 不需要显式关闭

    @contextmanager
    def _timer(self, name: str):
        """计时上下文（内部工具方法）"""
        start = time.perf_counter()
        yield
        elapsed = time.perf_counter() - start
        print(f"  [{name}] 耗时 {elapsed:.3f}s")

    def batch_import(self, documents: List[str], ids: List[str],
                     metadatas: Optional[List[Dict]] = None,
                     batch_size: int = 100):
        """
        批量导入文档（分批 upsert，避免单次请求过大）

        Args:
            documents: 文本列表
            ids: ID 列表（必须唯一）
            metadatas: 元数据列表（可选）
            batch_size: 每批大小（默认 100）
        """
        total = len(documents)
        print(f"[导入] 开始导入 {total} 条文档，每批 {batch_size} 条")

        for i in range(0, total, batch_size):
            batch_end = min(i + batch_size, total)
            batch_docs = documents[i:batch_end]
            batch_ids = ids[i:batch_end]
            batch_meta = metadatas[i:batch_end] if metadatas else None

            with self._timer(f"批次 {i//batch_size+1}"):
                self.collection.upsert(
                    documents=batch_docs,
                    ids=batch_ids,
                    metadatas=batch_meta,
                )
            print(f"  进度: {batch_end}/{total} ({batch_end*100//total}%)")

        print(f"[导入] 完成，当前共 {self.collection.count()} 条文档")

    def search(self, query: str, top_k: int = 5,
               source: Optional[str] = None,
               year: Optional[int] = None,
               keyword: Optional[str] = None) -> List[Dict]:
        """
        语义检索 + 元数据过滤

        Args:
            query: 自然语言查询
            top_k: 返回条数
            source: 按来源过滤（如 "regulation"/"report"）
            year: 按年份过滤
            keyword: 原文关键词过滤

        Returns:
            检索结果列表，每项包含 doc, metadata, distance
        """
        # 构建 where 过滤条件
        conditions = []
        if source:
            conditions.append({"source": source})
        if year:
            conditions.append({"year": year})

        where = {"$and": conditions} if len(conditions) > 1 else (
            conditions[0] if conditions else None
        )

        # 构建查询参数
        query_params = {
            "query_texts": [query],
            "n_results": top_k,
        }
        if where:
            query_params["where"] = where
        if keyword:
            query_params["where_document"] = {"$contains": keyword}

        results = self.collection.query(**query_params)

        # 格式化输出
        formatted = []
        for doc, meta, dist in zip(
            results["documents"][0],
            results["metadatas"][0],
            results["distances"][0],
        ):
            formatted.append({
                "text": doc,
                "metadata": meta,
                "distance": round(dist, 4),
            })
        return formatted

    def delete_by_source(self, source: str):
        """按来源批量删除"""
        count_before = self.collection.count()
        self.collection.delete(where={"source": source})
        count_after = self.collection.count()
        print(f"[删除] source={source}，删除 {count_before - count_after} 条")

    def export_metadata(self, output_path: str):
        """导出所有文档的元数据为 JSON（方便审计）"""
        all_data = self.collection.get(include=["documents", "metadatas"])
        export = []
        for doc_id, doc, meta in zip(
            all_data["ids"],
            all_data["documents"],
            all_data["metadatas"],
        ):
            export.append({
                "id": doc_id,
                "metadata": meta or {},
                "text_preview": (doc or "")[:100],
            })

        output = Path(output_path)
        output.parent.mkdir(parents=True, exist_ok=True)
        output.write_text(
            json.dumps(export, ensure_ascii=False, indent=2),
            encoding="utf-8",
        )
        print(f"[导出] {len(export)} 条元数据 → {output_path}")

    def stats(self) -> Dict[str, Any]:
        """统计信息"""
        return {
            "collection_name": self.collection.name,
            "total_docs": self.collection.count(),
            "persist_dir": str(self.persist_dir),
            "db_size_mb": sum(
                f.stat().st_size for f in self.persist_dir.rglob("*") if f.is_file()
            ) / (1024 * 1024),
        }

@contextmanager
def safe_file_write(filepath):
    import os

        with open(tmp, "w", encoding="utf-8") as f:
            yield f
        os.replace(tmp, filepath)  # 原子替换
    except:
        if os.path.exists(tmp):
            os.remove(tmp)
