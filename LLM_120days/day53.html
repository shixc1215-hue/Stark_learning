🐍 2. Python 知识 — Protocol + 工具注册表模式 15 min

2.1 typing.Protocol — 结构化子类型协议

Protocol 是 Python 3.8+ 引入的 隐式接口：不要求继承，只要类实现了相同方法就自动满足协议。

# 对比：ABC（显式继承）vs Protocol（隐式满足）

# ABC 方式 — 必须显式继承
from abc import ABC, abstractmethod

class ToolBase(ABC):
    @abstractmethod
    def execute(self, **kwargs) -> dict:
        ...

# Protocol 方式 — 只要有 execute 方法就行，无需继承
from typing import Protocol, runtime_checkable

@runtime_checkable  # 允许 isinstance() 检查
class ToolLike(Protocol):
    def execute(self, **kwargs) -> dict: ...
    def describe(self) -> str: ...

# 任意类只要实现了 execute + describe，自动满足协议
class DBQueryTool:
    def execute(self, **kwargs) -> dict:
        return {"result": "查询完成"}

    def describe(self) -> str:
        return "查询 AMC 数据库"

# isinstance 检查通过，无需继承！
print(isinstance(DBQueryTool(), ToolLike))  # True
Protocol vs ABC 选型指南
用 Protocol：工具/插件/策略 — 你想"鸭子类型"但需要类型检查
用 ABC：基类/框架 — 你要求子类必须显式继承

2.2 工具注册表模式 — 动态注册 + 查找

核心思想：用一个字典存储"工具名 → 工具实例"的映射，支持动态增删查。

from typing import Protocol, runtime_checkable, Any
from dataclasses import dataclass, field

# ========== 第一步：定义工具协议 ==========
@runtime_checkable
class ToolLike(Protocol):
    """所有工具必须实现的协议"""
    name: str           # 工具唯一标识
    description: str    # 工具描述（给 LLM 看）

    def execute(self, **kwargs) -> dict: ...
    def validate_args(self, **kwargs) -> list[str]: ...
    def describe(self) -> str: ...


# ========== 第二步：参数定义（让 LLM 知道该传什么参数）==========
@dataclass
class ToolParam:
    """工具参数的结构化定义"""
    name: str
    type: str           # "string" / "number" / "boolean"
    description: str    # 参数含义
    required: bool = True
    default: Any = None
    enum: list[str] = field(default_factory=list)


# ========== 第三步：工具注册表 ==========
class ToolRegistry:
    """
    工具注册中心 — 单例模式管理所有可用工具
    类比：数仓的元数据管理系统（数据字典）
    """

    def __init__(self):
        self._tools: dict[str, ToolLike] = {}  # name → tool

    def register(self, tool: ToolLike) -> None:
        """注册一个工具"""
        if tool.name in self._tools:
            print(f"⚠️ 工具 '{tool.name}' 已存在，将被覆盖")
        self._tools[tool.name] = tool

    def get(self, name: str) -> ToolLike | None:
        """按名称获取工具"""
        return self._tools.get(name)

    def list_tools(self) -> list[dict]:
        """列出所有工具（给 LLM 看的格式）"""
        return [
            {"name": t.name, "description": t.describe()}
            for t in self._tools.values()
        ]

    def unregister(self, name: str) -> bool:
        """移除一个工具"""
        if name in self._tools:
            del self._tools[name]
            return True
        return False

    def __len__(self) -> int:
        return len(self._tools)

    def __contains__(self, name: str) -> bool:
        return name in self._tools


# ========== 使用示例 ==========
registry = ToolRegistry()
print(f"已注册工具数: {len(registry)}")   # 0
print("db_query" in registry)               # False
为什么用注册表而不是直接 import？
① 工具可以运行时动态加载（插件模式）
② Agent 只需知道注册表，不需要知道每个工具的具体类
③ 方便做权限控制、日志审计、工具版本管理

3.1 BaseTool 抽象基类 — 所有工具的骨架

from abc import ABC, abstractmethod
from typing import Any
from dataclasses import dataclass, field
from functools import wraps
import time, json

@dataclass
class ToolResult:
    """工具执行的统一返回格式"""
    success: bool
    data: Any = None
    error: str | None = None
    tool_name: str = ""
    duration_ms: float = 0
    metadata: dict = field(default_factory=dict)

    def to_dict(self) -> dict:
        return {
            "success": self.success,
            "data": self.data,
            "error": self.error,
            "tool": self.tool_name,
            "duration_ms": round(self.duration_ms, 2),
        }


# ========== 装饰器：自动计时 + 错误捕获 ==========
def tool_execution(func):
    """装饰器：给工具 execute 方法加上计时和异常捕获"""
    @wraps(func)
    def wrapper(self, **kwargs) -> ToolResult:
        start = time.perf_counter()
        try:
            # 先校验参数
            errors = self.validate_args(**kwargs)
            if errors:
                return ToolResult(
                    success=False,
                    error=f"参数校验失败: {', '.join(errors)}",
                    tool_name=self.name,
                )
            # 执行工具
            data = func(self, **kwargs)
            elapsed = (time.perf_counter() - start) * 1000
            return ToolResult(
                success=True, data=data,
                tool_name=self.name, duration_ms=elapsed,
            )
        except Exception as e:
            elapsed = (time.perf_counter() - start) * 1000
            return ToolResult(
                success=False, error=str(e),
                tool_name=self.name, duration_ms=elapsed,
            )
    return wrapper


class BaseTool(ABC):
    """所有工具的抽象基类"""
    name: str
    description: str
    parameters: list[ToolParam]

    @abstractmethod
    def execute(self, **kwargs) -> ToolResult: ...

    @abstractmethod
    def describe(self) -> str: ...

    def validate_args(self, **kwargs) -> list[str]:
        """校验必填参数是否存在"""
        errors = []
        for param in self.parameters:
            if param.required and param.name not in kwargs:
                errors.append(f"缺少必填参数: {param.name}")
            elif param.enum and kwargs.get(param.name) not in param.enum:
                errors.append(
                    f"参数 {param.name} 值 '{kwargs[param.name]}' "
                    不在允许范围 {param.enum} 中"
                )
        return errors

    def schema_for_llm(self) -> dict:
        """生成给 LLM Function Calling 用的 JSON Schema"""
        return {
            "name": self.name,
            "description": self.description,
            "parameters": {
                "type": "object",
                "properties": {
                    p.name: {"type": p.type, "description": p.description}
                    for p in self.parameters
                },
                "required": [p.name for p in self.parameters if p.required],
            }
        }

🔀 4. 工具编排器 — 串行 / 并行 / 条件路由 15 min

有了工具，下一步就是编排：决定多个工具按什么顺序执行。类比数仓的 DAG 调度。

4.1 编排策略对比

策略适用场景类比Python 实现
串行 Sequential 步骤有前后依赖 ETL 串行任务 for 循环逐个执行
并行 Parallel 步骤无依赖，可同时执行 多表并行抽取 asyncio.gather / ThreadPool
条件路由 Conditional 根据上一步结果决定下一步 ETL 条件分支 if-else / match-case

4.2 ToolOrchestrator 完整实现

import asyncio
from dataclasses import dataclass, field
from typing import Callable, Any
from enum import Enum, auto

class StepStatus(Enum):
    """步骤执行状态"""
    PENDING = auto()
    RUNNING = auto()
    SUCCESS = auto()
    FAILED  = auto()
    SKIPPED = auto()  # 条件路由中跳过的步骤

@dataclass
class PipelineStep:
    """编排管线的单个步骤"""
    name: str
    tool_name: str             # 要调用的工具名
    args: dict = field(default_factory=dict)
    condition: Callable[[dict], bool] | None = None  # 执行条件
    status: StepStatus = StepStatus.PENDING
    result: ToolResult | None = None
    parallel_group: int | None = None  # 并行分组编号

@dataclass
class PipelineResult:
    """管线执行结果"""
    success: bool
    steps: list[PipelineStep]
    total_duration_ms: float
    outputs: dict = field(default_factory=dict)


class ToolOrchestrator:
    """
    工具编排器 — 支持三种编排策略
    核心职责：接收一组 PipelineStep，按策略依次/并行/条件执行
    """

    def __init__(self, registry: ToolRegistry):
        self.registry = registry

    # ========== 策略1：串行执行 ==========
    def run_sequential(self, steps: list[PipelineStep]) -> PipelineResult:
        """
        串行执行：按顺序逐个执行步骤
        上一步的 result 会注入到下一步的 args 中
        """
        start = time.perf_counter()
        context = {}  # 存储中间结果，供后续步骤引用

        for step in steps:
            # 条件检查
            if step.condition and not step.condition(context):
                step.status = StepStatus.SKIPPED
                continue

            tool = self.registry.get(step.tool_name)
            if not tool:
                step.status = StepStatus.FAILED
                step.result = ToolResult(
                    success=False,
                    error=f"工具 '{step.tool_name}' 未注册")
                continue

            # 合并 args（用 context 中的变量替换占位符）
            resolved_args = self._resolve_args(step.args, context)

            step.status = StepStatus.RUNNING
            step.result = tool.execute(**resolved_args)

            if step.result.success:
                step.status = StepStatus.SUCCESS
                context[step.name] = step.result.data
            else:
                step.status = StepStatus.FAILED

        elapsed = (time.perf_counter() - start) * 1000
        all_ok = all(s.status in (StepStatus.SUCCESS, StepStatus.SKIPPED)
                   for s in steps)
        return PipelineResult(
            success=all_ok, steps=steps,
            total_duration_ms=elapsed, outputs=context)

    # ========== 策略2：并行执行 ==========
    async def run_parallel(self, steps: list[PipelineStep]) -> PipelineResult:
        """
        并行执行：所有步骤同时启动（无依赖时使用）
        """
        start = time.perf_counter()
        context = {}

        async def _run_one(step: PipelineStep):
            if step.condition and not step.condition(context):
                step.status = StepStatus.SKIPPED
                return
            tool = self.registry.get(step.tool_name)
            if not tool:
                step.status = StepStatus.FAILED
                return
            step.status = StepStatus.RUNNING
            # 模拟异步调用（实际项目用 aiohttp）
            await asyncio.sleep(0.01)  # 模拟网络延迟
            step.result = tool.execute(**step.args)
            step.status = StepStatus.SUCCESS if step.result.success else StepStatus.FAILED
            context[step.name] = step.result.data

        await asyncio.gather(*[_run_one(s) for s in steps])

        elapsed = (time.perf_counter() - start) * 1000
        all_ok = all(s.status in (StepStatus.SUCCESS, StepStatus.SKIPPED)
                   for s in steps)
        return PipelineResult(
            success=all_ok, steps=steps,
            total_duration_ms=elapsed, outputs=context)

    # ========== 策略3：条件路由 ==========
    def run_conditional(
        self,
        steps: list[PipelineStep],
        router: Callable[[dict], str | None],
    ) -> PipelineResult:
        """
        条件路由：根据 router 函数决定执行哪条分支
        router(context) → 下一步的 step.name 或 None（结束）
        """
        start = time.perf_counter()
        context = {}
        executed = set()

        while True:
            # 由 router 决定下一步
            next_step_name = router(context)
            if next_step_name is None:
                break

            # 找到对应步骤
            step = None
            for s in steps:
                if s.name == next_step_name:
                    step = s
                    break
            if step is None or step.name in executed:
                break

            tool = self.registry.get(step.tool_name)
            if not tool:
                step.status = StepStatus.FAILED
                break

            step.status = StepStatus.RUNNING
            step.result = tool.execute(**self._resolve_args(step.args, context))
            step.status = StepStatus.SUCCESS if step.result.success else StepStatus.FAILED
            context[step.name] = step.result.data
            executed.add(step.name)

        elapsed = (time.perf_counter() - start) * 1000
        return PipelineResult(
            success=all(s.status != StepStatus.FAILED for s in executed),
            steps=steps, total_duration_ms=elapsed, outputs=context)

    # ========== 辅助方法 ==========
    def _resolve_args(self, args: dict, context: dict) -> dict:
        """将 args 中的 ${step_name.key} 替换为 context 中的实际值"""
        resolved = {}
        for k, v in args.items():
            if isinstance(v, str) and v.startswith("${") and v.endswith("}"):
                ref = v[2:-1]
                parts = ref.split(".")
                resolved[k] = context
                for p in parts:
                    if isinstance(resolved, dict):
                        resolved = resolved.get(p, "")
            else:
                resolved[k] = v
        return resolved

5.1 AgentStep — 记录每一步轨迹

@dataclass
class AgentStep:
    """Agent 单步执行记录（类似审计日志）"""
    step_num: int
    thought: str        # Agent 的思考过程
    action: str | None   # 调用的工具名

5.2 SimpleAgent — 完整 ReAct 循环

import json, re

class SimpleAgent:
    """
    最小可运行的 Agent — ReAct 循环
    职责：接收用户任务 → 规划步骤 → 调用工具 → 汇总结果
    """

    def __init__(self, registry: ToolRegistry, max_steps: int = 5):
        self.registry = registry
        self.max_steps = max_steps
        self.history: list[AgentStep] = []

    def build_system_prompt(self) -> str:
        """构建系统提示词（告诉 LLM 有哪些工具可用）"""
        tools_desc = "\n".join(
            f"- {t['name']}: {t['description']}"
            for t in self.registry.list_tools()
        )
        return f"""你是 AMC 智能运维助手，负责执行数据分析和运维任务。

可用工具：
{tools_desc}

请按以下格式回复（JSON）：
{{"thought": "你的思考", "action": "工具名", "action_input": {{参数}}}}

如果所有步骤都完成了，回复：
{{"thought": "总结", "final_answer": "最终回答"}}

注意：
1. 每次只调用一个工具
2. 仔细分析工具返回结果再决定下一步
3. 最多执行 {self.max_steps} 步"""

    def run(self, user_query: str) -> dict:
        """执行完整的 ReAct 循环"""
        messages = [
            {"role": "system", "content": self.build_system_prompt()},
            {"role": "user", "content": user_query},
        ]

        for i in range(self.max_steps):
            # Step 1: 调用 LLM 获取下一步决策
            response = self._call_llm(messages)
            if not response:
                break

            # Step 2: 解析 LLM 响应
            parsed = self._parse_response(response)

            # 判断是否已经给出最终回答
            if "final_answer" in parsed:
                self.history.append(AgentStep(
                    step_num=i+1, thought=parsed["thought"],
                    action=None, observation=parsed["final_answer"]))
                return {"answer": parsed["final_answer"],
                        "steps": self.history}

            # Step 3: 执行工具
            tool_name = parsed.get("action")
            tool_args = parsed.get("action_input", {})
            tool = self.registry.get(tool_name)

            if not tool:
                observation = f"错误：工具 '{tool_name}' 不存在"
            else:
                result = tool.execute(**tool_args)
                observation = json.dumps(result.to_dict(),
                                       ensure_ascii=False)

            self.history.append(AgentStep(
                step_num=i+1,
                thought=parsed.get("thought", ""),
                action=tool_name,
                action_input=tool_args,
                observation=observation,
            ))

            # Step 4: 将观察结果追加到消息历史
            messages.append({"role": "assistant", "content": response})
            messages.append({"role": "user", "content":
                f"工具返回: {observation}\n请继续。"})

        return {"answer": "已达到最大步骤限制", "steps": self.history}

    def _call_llm(self, messages: list) -> str:
        """调用 LLM（模拟实现，实际项目替换为真实 API）"""
        # 模拟 LLM 响应（实际替换为 requests/zhipuai SDK）
        last_user = messages[-1]["content"]
        if "不良资产余额" in last_user and not self.history:
            return json.dumps({
                "thought": "用户想查不良资产余额，需要调用db_query工具",
                "action": "db_query",
                "action_input": {"metric": "不良资产余额", "period": "本月"}
            }, ensure_ascii=False)
        return json.dumps({
            "thought": "已获取数据，可以给出最终回答",
            "final_answer": "本月不良资产余额为12.8亿元。"
        }, ensure_ascii=False)

    def _parse_response(self, text: str) -> dict:
        """从 LLM 响应中提取 JSON"""
        # 尝试直接解析
        try:
            return json.loads(text)
        except json.JSONDecodeError:
            pass
        # 尝试从文本中提取 JSON 块
        match = re.search(r'\{[^{}]*\}', text, re.DOTALL)
        if match:
            try:
                return json.loads(match.group())
            except json.JSONDecodeError:
                pass
        return {"thought": text, "final_answer": text}

🚨 6. 常见陷阱 + 最佳实践 5 min

Agent 开发 Top 5 陷阱

#陷阱症状解法
1 死循环 Agent 一直调同一个工具，不停止 设置 max_steps（推荐 5-8）+ 重复检测
2 工具描述模糊 LLM 选错工具或传错参数 描述要写清用途+适用场景+参数含义，附示例
3 JSON 解析失败 LLM 返回格式不规范 多层容错 + JSON Mode + 重试纠正
4 工具副作用不可控 Agent 反复调用导致重复写入/删除 幂等设计 + Human-in-the-Loop 审批
5 上下文爆炸 多轮工具调用后 token 超限 上下文压缩 / 滑动窗口 / 摘要替换旧步骤

最佳实践速查表

# ✅ 1. 工具接口统一
#    所有工具返回 ToolResult，Agent 只认这一种格式

# ✅ 2. 参数校验前置
#    validate_args() 在 execute() 之前执行，快速失败

# ✅ 3. 执行轨迹记录

#    工具失败时 Agent 能换方案，而不是直接崩掉
