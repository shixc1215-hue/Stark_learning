2.1 timeit — 精确测量代码执行时间

评估 RAG 系统时需要测量各阶段耗时。time.time() 精度不够，timeit 自动多次运行取最佳值，排除系统抖动干扰。

import timeit
import functools

# 方式1：timeit.timeit 测量代码片段（字符串模式）
# 适合简单表达式，number 控制运行次数
elapsed = timeit.timeit("[i**2 for i in range(100)]", number=10000)
print(f"列表推导式 10000 次: {elapsed:.4f}s, 平均 {elapsed/10000*1e6:.1f}μs")

# 方式2：timeit.repeat 多轮取最小值（更稳定）
# 返回列表，每轮 number 次运行的总耗时
results = timeit.repeat("[i**2 for i in range(100)]", number=10000, repeat=5)
print(f"5轮最小值: {min(results):.4f}s")  # 取最小值排除干扰

# 方式3：测量函数（用 functools.partial 绑定参数）
def search_test(vector_store, query, k=5):
    return vector_store.query(query, k)

# partial 把函数变成无参形式，timeit 才能调用
test_fn = functools.partial(search_test, mock_store, "不良资产处置方式")
elapsed = timeit.timeit(test_fn, number=100)
print(f"检索 100 次平均: {elapsed/100*1000:.2f}ms")

# 方式4：命令行模式（快速对比）
# python -m timeit -s "import json" "json.dumps({'a':1})"
选型建议：日常开发用 time.perf_counter() 够了（Day 51 已用）；评估报告需要精确基准测试时用 timeit。repeat 取 min 而非 mean——因为最小值最接近"纯计算时间"，其他都是被干扰的。

# 进阶：带 TTL 的缓存（lru_cache 本身不支持 TTL）
# 用类包装实现过期机制
class TTLCache:
    """带过期时间的缓存，适合 Embedding 等可能更新的数据。"""
    def __init__(self, ttl_seconds=3600, maxsize=512):
        self.ttl = ttl_seconds
        self.maxsize = maxsize
        self._store = {}  # key -> (value, timestamp)

    def get(self, key):
        if key in self._store:
            value, ts = self._store[key]
            import time
            if time.time() - ts < self.ttl:
                return value  # 缓存有效
            del self._store[key]  # 过期删除
        return None  # 未命中

    def set(self, key, value):
        if len(self._store) >= self.maxsize:
            # 简单 FIFO 淘汰（生产用 OrderedDict 做 LRU）
            oldest = next(iter(self._store))
            del self._store[oldest]
        import time
        self._store[key] = (value, time.time())

# RAG 场景应用：Embedding 缓存
embedding_cache = TTLCache(ttl_seconds=7200, maxsize=1024)

def get_embedding(text: str) -> list:
    # 用文本 MD5 做 key，避免长字符串占内存
    key = hashlib.md5(text.encode()).hexdigest()
    cached = embedding_cache.get(key)
    if cached is not None:
        return cached
    # 实际调用 API
    result = call_embedding_api(text)
    embedding_cache.set(key, result)
    return result

2.3 contextlib.suppress — 优雅地忽略指定异常

评估脚本中很多操作是"尽力而为"——LLM 评分失败不应中断整个评估流程。suppress 比裸 try/except 更清晰。

from contextlib import suppress
import json

# 不优雅的方式：裸 try/except 静默吞掉异常
try:
    result = json.loads(broken_json_str)
except json.JSONDecodeError:
    result = None  # 意图不明显，可能是漏写了别的异常

# 优雅方式：明确声明"我就是要忽略这个异常"
with suppress(json.JSONDecodeError):
    result = json.loads(broken_json_str)
# 如果抛 JSONDecodeError，自动跳过；result 可能未定义
# 更安全的写法：
result = None
with suppress(json.JSONDecodeError):
    result = json.loads(broken_json_str)

# 多异常一起忽略
with suppress(json.JSONDecodeError, KeyError, TypeError):
    data = json.loads(response_text)["data"]["score"]

# RAG 评估中的实际应用：LLM-as-Judge 可能超时或返回非 JSON
def safe_llm_score(question, answer, context) -> float | None:
    """调用 LLM 打分，失败返回 None 而非崩溃。"""
    score = None
    with suppress(TimeoutError, ConnectionError, json.JSONDecodeError):
        resp = call_judge_llm(question, answer, context)
        score = float(json.loads(resp)["score"])
    return score  # 失败时为 None，调用方自行处理
三件套速查： timeit → 精确基准测试（取 min）； lru_cache → 纯函数缓存（返回值可哈希）； suppress → 明确忽略异常（比裸 except 安全）。

🔍 四、检索质量评估（Retrieval Metrics） 15 min

检索是 RAG 的基础——检索不到正确的文档，生成再好也是"一本正经地胡说"。以下四个指标从不同角度衡量检索质量。

指标含义公式关注点
Hit Rate@KTop-K 中是否至少命中一个命中样本数 / 总样本数能不能找到
Recall@K所有相关文档中被召回的比例命中文档数 / 总相关文档数找全了吗
Precision@KTop-K 中相关文档的占比命中文档数 / K找得准吗
MRR@K第一个命中结果的排名倒数1/rank（rank 为首个命中排名）找得快吗
from dataclasses import dataclass
from typing import Protocol

# ---- 检索器协议（D50 已定义，这里简化） ----
class RetrieverProtocol(Protocol):
    def retrieve(self, query: str, top_k: int) -> list[dict]:
        """返回 [{'id': str, 'score': float, 'content': str}, ...]"""
        ...

@dataclass
class RetrievalMetrics:
    """单条样本的检索评估结果。"""
    hit_rate: float       # 0 or 1
    recall: float          # [0, 1]
    precision: float       # [0, 1]
    mrr: float             # [0, 1]
    first_hit_rank: int  # 首个命中的排名（未命中为 -1）

def eval_single_retrieval(
    retrieved_ids: list[str],
    relevant_ids: list[str],
    k: int = 5
) -> RetrievalMetrics:
    """评估单条检索结果。
    Args:
        retrieved_ids: 检索器返回的文档ID列表（按相似度排序）
        relevant_ids: 标注的正确文档ID列表（ground truth）
        k: 只看 Top-K 结果
    """
    top_k_ids = retrieved_ids[:k]
    relevant_set = set(relevant_ids)

    # 命中的文档ID（交集）
    hits = [i for i, doc_id in enumerate(top_k_ids) if doc_id in relevant_set]

    # Hit Rate: 是否至少命中一个
    hit_rate = 1.0 if hits else 0.0

    # Recall: 命中数 / 总相关数
    recall = len(hits) / len(relevant_set) if relevant_set else 0.0

    # Precision: 命中数 / K
    precision = len(hits) / k

    # MRR: 第一个命中的排名倒数
    if hits:
        first_rank = hits[0] + 1  # 排名从 1 开始
        mrr = 1.0 / first_rank
        first_hit_rank = first_rank
    else:
        mrr = 0.0
        first_hit_rank = -1

    return RetrievalMetrics(
        hit_rate=hit_rate, recall=recall,
        precision=precision, mrr=mrr,
        first_hit_rank=first_hit_rank
    )

def eval_retrieval_batch(
    retriever: RetrieverProtocol,
    eval_set: list[EvalSample],
    k: int = 5
) -> dict:
    """批量评估检索质量，返回汇总指标。"""
    all_metrics = []
    for sample in eval_set:
        # 调用检索器获取 Top-K 结果
        results = retriever.retrieve(sample.question, top_k=k)
        retrieved_ids = [r["id"] for r in results]

        m = eval_single_retrieval(retrieved_ids, sample.relevant_doc_ids, k)
        all_metrics.append((sample, m))

    # 汇总统计
    n = len(all_metrics)
    summary = {
        "samples": n,
        "hit_rate@k": sum(m.hit_rate for _, m in all_metrics) / n,
        "recall@k": sum(m.recall for _, m in all_metrics) / n,
        "precision@k": sum(m.precision for _, m in all_metrics) / n,
        "mrr@k": sum(m.mrr for _, m in all_metrics) / n,
    }

    # 按难度分组统计
    for diff in ["easy", "medium", "hard"]:
        group = [(s, m) for s, m in all_metrics if s.difficulty == diff]
        if group:
            gn = len(group)
            summary[f"recall@k_{diff}"] = sum(m.recall for _, m in group) / gn

    return summary, all_metrics

🤖 五、生成质量评估（Generation Metrics） 15 min

检索找对了文档，但 LLM 可能：①编造文档中没有的信息（幻觉）、②答非所问、③忽略检索到的关键内容。三个指标分别对应这三个问题。

指标评估什么评分方式低分说明
Faithfulness（忠实度）答案是否只基于检索内容LLM-as-Judge (0-1)存在幻觉/编造
Answer Relevancy（相关性）答案是否回答了问题LLM-as-Judge (0-1)答非所问
Context Utilization（利用率）检索内容是否被充分利用LLM-as-Judge (0-1)遗漏关键信息

def extract_json_safe(text: str) -> dict | None:
    """容错 JSON 提取（Day 41 技术复用）。"""
    # 尝试直接解析
    with suppress(json.JSONDecodeError):
        return json.loads(text)
    # 尝试提取花括号内容
    import re
    match = re.search(r'\{[^}]+\}', text)
    if match:
        with suppress(json.JSONDecodeError):
            return json.loads(match.group())
    return None

    """用 LLM-as-Judge 评估单条生成结果。"""
    system_prompt = """你是一个 RAG 系统评估专家。请对以下问答对进行评分，
    返回 JSON 格式，包含三个字段（0.0-1.0）：
    - faithfulness: 答案是否完全基于参考内容，没有编造信息
    - answer_relevancy: 答案是否直接回答了问题
    - context_utilization: 答案是否充分利用了参考内容中的关键信息
    只返回 JSON，不要其他文字。"""

    user_prompt = f"""## 问题
{question}

## 参考内容（检索到的文档）
{context}

## 系统答案
{answer}

## 参考答案（人工标注，可选）
{reference or "（无）"}

请评分："""

    raw = call_judge_llm(system_prompt, user_prompt)
    parsed = extract_json_safe(raw)

    if parsed:
        return GenerationMetrics(
            faithfulness=parsed.get("faithfulness"),
            answer_relevancy=parsed.get("answer_relevancy"),
            context_utilization=parsed.get("context_utilization"),
            judge_raw=raw
        )
    return GenerationMetrics(None, None, None, raw)

# ---- 批量生成评估 ----
def eval_generation_batch(
    rag_pipeline,  # D51 的 RAGQueryPipeline
    eval_set: list[EvalSample],
) -> dict:
    """端到端评估：对每个问题走完整 RAG 管线，再评估生成质量。"""
    all_metrics = []
    for sample in eval_set:
        # 调用 RAG 管线获取答案和检索上下文
        result = rag_pipeline.query(sample.question)
        answer = result["answer"]
        context = "\n".join(result["contexts"])

        # LLM-as-Judge 评估
        m = eval_single_generation(
            question=sample.question,
            answer=answer,
            context=context,
            reference=sample.reference_answer
        )
        all_metrics.append((sample, m))

    # 汇总（过滤 None 值——LLM Judge 可能失败）
    faith_scores = [m.faithfulness for _, m in all_metrics if m.faithfulness is not None]
    rel_scores = [m.answer_relevancy for _, m in all_metrics if m.answer_relevancy is not None]
    util_scores = [m.context_utilization for _, m in all_metrics if m.context_utilization is not None]

    def avg(lst):
        return sum(lst) / len(lst) if lst else 0.0

    return {
        "samples": len(all_metrics),
        "judge_success_rate": len(faith_scores) / len(all_metrics),
        "avg_faithfulness": avg(faith_scores),
        "avg_answer_relevancy": avg(rel_scores),
        "avg_context_utilization": avg(util_scores),
        "details": all_metrics,
    }

LLM-as-Judge 的局限：
  • 位置偏差：LLM 倾向于给排在前面的答案更高分
  • 自评偏差：用同一个 LLM 做 Judge 和生成，评分可能偏高
  • 建议：Judge 模型选比生成模型更强的（如用 GPT-4 评估 GLM 生成）

指标异常可能根因优化策略影响范围
Recall@K 低 分块过大/过小；Embedding 模型差；查询太短 调 chunk_size；换 Embedding 模型；Query 扩展（HyDE） 离线索引 + 在线检索

MRR 低 相关文档排在后面 Reranker 重排；调整相似度阈值 在线检索

6.1 分块策略调优实验

chunk_size 是影响 Recall 和 Precision 的最核心参数。下面设计一个实验：用不同 chunk_size 构建索引，对比评估指标。

import timeit
from dataclasses import dataclass
from typing import Callable

@dataclass
class ChunkConfig:
    """分块参数配置。"""
    chunk_size: int = 500        # 每块字符数
    overlap: int = 50          # 重叠字符数
    separators: list[str] = None  # 分隔符优先级

    def __post_init__(self):
        if self.separators is None:
            self.separators = ["\n\n", "\n", "。", "；", " "]

@dataclass
class EvalResult:
    """单次实验结果。"""
    config_name: str
    recall: float
    precision: float
    mrr: float
    faithfulness: float
    index_time_ms: float   # 离线索引耗时
    query_time_ms: float   # 在线查询耗时

def run_chunk_experiment(
    documents: list[str],
    eval_set: list[EvalSample],
    configs: dict[str, ChunkConfig],
    embed_fn: Callable,
    llm_fn: Callable,
) -> list[EvalResult]:
    """对不同分块配置运行完整评估实验。
    每组配置：重建索引 → 检索评估 → 生成评估 → 记录耗时
    """
    results = []

    for name, config in configs.items():
        print(f"\n--- 实验: {name} (chunk={config.chunk_size}, overlap={config.overlap}) ---")

        # 1. 离线索引（计时）
        def build_index():
            chunks = []
            for doc in documents:
                # 递归分块（简化版，实际用 D50 的 RecursiveCharacterChunker）
                for i in range(0, len(doc), config.chunk_size - config.overlap):
                    chunk = doc[i:i + config.chunk_size]
                    chunks.append(chunk)
                    if i + config.chunk_size >= len(doc):
                        break
            # Embedding + 存入向量库（mock）
            return chunks

        index_time = timeit.timeit(build_index, number=1) * 1000  # ms

        # 2. 检索评估（mock）
        # 实际调用 eval_retrieval_batch
        recall = 0.70 + (config.chunk_size - 500) * 0.0003  # 模拟数据
        precision = 0.32 - (config.chunk_size - 500) * 0.0002
        mrr = 0.70

        # 3. 生成评估（mock）
        faithfulness = 0.84 + (config.overlap - 50) * 0.001

        # 4. 在线查询耗时（mock）
        query_time = 150 + config.chunk_size * 0.1

        result = EvalResult(
            config_name=name,
            recall=max(0, recall),
            precision=max(0, precision),
            mrr=mrr,
            faithfulness=min(1.0, faithfulness),
            index_time_ms=index_time,
            query_time_ms=query_time,
        )
        results.append(result)
        print(f"  Recall={result.recall:.3f}  Precision={result.precision:.3f}  "
              f"Faith={result.faithfulness:.3f}  Query={result.query_time_ms:.0f}ms")

    return results

# 对比实验：无 Reranker vs 有 Reranker
def compare_with_without_reranker(eval_set, retriever, reranker):
    """对比有无 Reranker 的检索效果。"""
    results_no_rerank = []
    results_with_rerank = []

    for sample in eval_set:
        # 无 Reranker：直接用向量检索结果
        raw_results = retriever.retrieve(sample.question, top_k=20)
        raw_ids = [r["id"] for r in raw_results[:5]]
        m1 = eval_single_retrieval(raw_ids, sample.relevant_doc_ids, k=5)
        results_no_rerank.append(m1)

        # 有 Reranker：召回 Top-20 → Reranker 精排 → 取 Top-5
        reranked = reranker.rerank(
            query=sample.question,
            documents=raw_results,
            top_k=5
        )
        reranked_ids = [r["id"] for r in reranked]
        m2 = eval_single_retrieval(reranked_ids, sample.relevant_doc_ids, k=5)
        results_with_rerank.append(m2)

    def avg(lst, attr):
        vals = [getattr(m, attr) for m in lst]
        return sum(vals) / len(vals) if vals else 0

    return {
        "无Reranker": {
            "precision@5": avg(results_no_rerank, "precision"),
            "mrr@5": avg(results_no_rerank, "mrr"),
        },
        "有Reranker": {
            "precision@5": avg(results_with_rerank, "precision"),
            "mrr@5": avg(results_with_rerank, "mrr"),
        }
    }

class ABTestRunner:
    """RAG 系统 A/B 测试运行器。"""

    def __init__(self, config: ABTestConfig):
        self.config = config
        self.results = {}  # variant_name -> [metrics, ...]

    def run_variant(
        self,
        variant_name: str,
        pipeline_factory: Callable[[], Any],  # 返回一个 RAG 管线实例
        eval_set: list[EvalSample],
    ) -> dict:
        """运行单个变体的完整评估。"""
        print(f"\n{'='*50}")
        print(f"运行变体: {variant_name}")
        print(f"{'='*50}")

        # 构建管线（不同配置用不同参数）

        # 检索评估

            "timestamp": time.strftime("%Y-%m-%d %H:%M:%S"),

                             if not k.startswith("recall@k_")},
            "generation": {k: v for k, v in gen_summary.items()
                              if k != "details"},
        }
        self.results[variant_name] = result
        return result

    def compare(self) -> str:
        """生成对比报告。"""
        if len(self.results) < 2:
            return "需要至少 2 个变体才能对比"

        names = list(self.results.keys())
        a, b = self.results[names[0]], self.results[names[1]]

        report = f"\n{'='*60}\n"
        report += f"A/B 测试对比报告: {self.config.name}\n"
        report += f"{'='*60}\n"
        report += f"{'指标':<25} {names[0]:>15} {names[1]:>15} {'变化':>10}\n"
        report += f"{'-'*60}\n"

        # 检索指标对比
        for key in ["recall@k", "precision@k", "mrr@k"]:
            va = a["retrieval"].get(key, 0)
            vb = b["retrieval"].get(key, 0)
            delta = vb - va
            arrow = "↑" if delta > 0 else "↓" if delta < 0 else "→"
            report += f"{key:<25} {va:>15.4f} {vb:>15.4f} {arrow}{abs(delta):>8.4f}\n"

        # 生成指标对比
        for key in ["avg_faithfulness", "avg_answer_relevancy", "avg_context_utilization"]:
            va = a["generation"].get(key, 0)
            vb = b["generation"].get(key, 0)
            delta = vb - va
            arrow = "↑" if delta > 0 else "↓" if delta < 0 else "→"
            report += f"{key:<25} {va:>15.4f} {vb:>15.4f} {arrow}{abs(delta):>8.4f}\n"

        report += f"{'='*60}\n"
        return report

    def save_report(self, path: str):
        """保存完整结果为 JSON。"""
        with open(path, "w", encoding="utf-8") as f:
            json.dump(self.results, f, ensure_ascii=False, indent=2)

陷阱 1：用训练数据做评估
把构建索引的文档直接拿来出题评估——当然检索得分高。正确做法：评估集的问题和文档要有一定"距离"，模拟真实用户提问。
陷阱 2：只看平均分，忽略分布
平均 Recall 80% 可能掩盖了 hard 问题只有 30%。要按难度/分类分组统计，关注最差的分组。
陷阱 3：LLM Judge 与生成模型相同
用 GLM-4 生成、再用 GLM-4 做 Judge → 自评偏差严重。Judge 模型应更强或不同（如用 GPT-4 评 GLM 生成）。
陷阱 4：一次改多个变量
同时调 chunk_size + 换 Embedding 模型 + 加 Reranker → 结果变好不知道谁的功劳。每次只改一个变量。
陷阱 5：评估集太小不具统计意义
5-10 条样本的指标波动极大（一条样本就能改变 10%+）。至少 50 条，理想 100+ 条才有参考价值。

测试集 50-100 条，覆盖多难度多分类，标注 ground truth 不要用索引文档原文出题

练习 1：实现 NDCG@K 指标

Day 35 提到过 NDCG（Normalized Discounted Cumulative Gain），它比 MRR 更精细——考虑了所有命中位置的排名衰减。请实现：

def ndcg_at_k(retrieved_ids: list[str], relevant_ids: list[str], k: int = 5) -> float:
    """计算 NDCG@K。
    DCG = sum(rel_i / log2(i+1)) for i in 1..K

练习 3：LeetCode #912 — 排序数组

用归并排序实现，理解"分治"思想——和 RAG 评估中"按难度分组统计再汇总"是同一种思维模式。

class Solution:
    def sortArray(self, nums: list[int]) -> list[int]:
        # 归并排序：分治 → 排左 → 排右 → 合并
        if len(nums) <= 1:
            return nums

        mid = len(nums) // 2
        left = self.sortArray(nums[:mid])
        right = self.sortArray(nums[mid:])

        # 合并两个有序数组
        merged = []
        i = j = 0
        while i < len(left) and j < len(right):
            if left[i] <= right[j]:
                merged.append(left[i])
                i += 1
            else:
                merged.append(right[j])
                j += 1
        merged.extend(left[i:])
        merged.extend(right[j:])
        return merged
