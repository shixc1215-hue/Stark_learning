2.1 sorted() 高级用法 — 检索排序的基石

检索的核心操作就是"按相似度从高到低排序"。Python 的 sorted() 配合 key 参数是利器。

2.2 三种检索策略对比

策略原理适用场景类比 SQL
关键词检索 精确匹配关键词 专有名词、编号、ID WHERE text LIKE '%keyword%'
向量检索 Embedding 语义相似度 语义模糊、概念解释 ORDER BY similarity DESC
混合检索 关键词 + 向量加权融合 生产环境推荐 WHERE ... ORDER BY (kw_score + sim)

2.3 关键词检索 — 字符串匹配方案

def keyword_search(query: str, documents: list[dict], top_k: int = 3) -> list[dict]:
    """最简单的关键词检索 — 类比 SQL LIKE '%keyword%'"""
    query_lower = query.lower()
    scored = []
    for doc in documents:
        text = doc["text"].lower()
        hits = sum(1 for word in query_lower.split() if word in text)
        if hits > 0:
            scored.append({**doc, "keyword_score": hits})
    return sorted(scored, key=lambda d: d["keyword_score"], reverse=True)[:top_k]

2.4 向量检索 — 复用 Day30 的余弦相似度

import math

def cosine_similarity(a: list[float], b: list[float]) -> float:
    """余弦相似度 — 方向越接近值越接近1"""
    dot = sum(x * y for x, y in zip(a, b))
    norm_a = math.sqrt(sum(x * x for x in a))
    norm_b = math.sqrt(sum(x * x for x in b))
    return dot / (norm_a * norm_b + 1e-8)  # +1e-8 防除零

def vector_search(query_vec: list[float], doc_vectors: list[dict], top_k: int = 3) -> list[dict]:
    """向量检索 — 按语义相似度排序"""
    scored = []
    for item in doc_vectors:
        sim = cosine_similarity(query_vec, item["vector"])
        scored.append({**item, "similarity": sim})
    return sorted(scored, key=lambda d: d["similarity"], reverse=True)[:top_k]

# heapq.nsmallest / nlargest 更高效（不用全排序）
import heapq

def vector_search_fast(query_vec, doc_vectors, top_k=3):
    """用堆排序找 Top-K — O(N log K) 而非 O(N log N)"""
    scored = []
    for item in doc_vectors:
        sim = cosine_similarity(query_vec, item["vector"])
        scored.append({**item, "similarity": sim})
    # nlargest 找相似度最高的 K 个
    return heapq.nlargest(top_k, scored, key=lambda d: d["similarity"])

2.5 混合检索 — RRF（Reciprocal Rank Fusion）

RRF 核心思想：不直接混合分数（量纲不同），而是混合排名。
公式：RRF_score = 1/(k + rank_keyword) + 1/(k + rank_vector)，通常 k=60。 
def reciprocal_rank_fusion(
    keyword_results: list[dict],
    vector_results: list[dict],
    k: int = 60
) -> list[dict]:
    """RRF 混合检索 — 融合关键词和向量两种排名"""
    scores = {}      # doc_id -> 累计 RRF 分数
    merged_data = {} # doc_id -> 文档内容

    # 关键词排名贡献
    for rank, doc in enumerate(keyword_results, start=1):
        doc_id = doc["id"]
        scores[doc_id] = scores.get(doc_id, 0) + 1 / (k + rank)
        merged_data[doc_id] = doc

    # 向量排名贡献
    for rank, doc in enumerate(vector_results, start=1):
        doc_id = doc["id"]
        scores[doc_id] = scores.get(doc_id, 0) + 1 / (k + rank)
        merged_data[doc_id] = doc

    # 按 RRF 分数降序
    fused = [
        {**merged_data[did], "rrf_score": score}
        for did, score in scores.items()
    ]
    return sorted(fused, key=lambda d: d["rrf_score"], reverse=True)

# 测试
kw_results = [{"id": "A", "text": "债权转让规范"}, {"id": "B", "text": "风险分类指引"}]
vec_results = [{"id": "B", "text": "风险分类指引"}, {"id": "C", "text": "资产保全制度"}]
fused = reciprocal_rank_fusion(kw_results, vec_results)
for item in fused:
    print(f"  {item['id']}: RRF={item['rrf_score']:.4f} | {item['text']}")
# A: RRF=0.0164 | 债权转让规范（仅关键词命中）
# C: RRF=0.0164 | 资产保全制度（仅向量命中）
# B: RRF=0.0328 | 风险分类指引（两种都命中，分数翻倍！）
为什么用排名而不是分数？关键词分数（命中数）和向量分数（余弦相似度）量纲不同，直接加权没意义。RRF 把两者统一转成排名再融合，简单有效，是工业界最常用的混合检索方案。
🔬 Query 变换：让搜索更聪明 ⏱ 20 min

用户输入的查询往往不是最优的搜索语句。比如"什么叫五级分类"太口语化，"不良资产处置方式有哪些"范围太大。Query 变换就是在搜索前先"改写"查询，提高检索质量。

3.1 常见的 Query 变换技术

技术做法解决什么问题复杂度
Query 改写 让 LLM 把口语化查询改写成更清晰的搜索语句 查询太口语化/模糊 低
Query 扩展 添加同义词、相关术语 查询缺少关键术语 中
HyDE 让 LLM 先生成假设性回答，用回答去检索 问题和文档表达差异大 高
Multi-Query 生成多个变体查询，分别检索后合并 查询角度单一 中

def rewrite_query(user_query: str) -> str:

        temperature=0.0,  # 低温度保证稳定

3.4 HyDE：假设性文档嵌入

HyDE 思路：用户问"五级分类怎么认定？"，直接拿这个问题去检索效果一般。但如果先让 LLM 生成一段"看起来像答案"的文字，再拿这段文字去检索，匹配度往往更高 — 因为"答案"和"文档"的语义空间更接近。 
def hyde_search(user_query: str, collection, top_k: int = 3) -> list[dict]:
    """HyDE: 先让 LLM 生成假设性回答，再用回答去检索"""
    # Step 1: 让 LLM 生成假设性文档
    response = client.chat.completions.create(
        model="glm-4-flash",
        messages=[
            {
                "role": "system",
                "content": "请根据用户问题，生成一段看起来像官方文档的假设性回答。"
                           "不需要完全准确，目的是用于语义检索。"
            },
            {"role": "user", "content": user_query}
        ],
        temperature=0.3,
    )
    hypothetical_doc = response.choices[0].message.content

    # Step 2: 用假设性文档去向量库检索
    results = collection.query(
        query_texts=[hypothetical_doc],
        n_results=top_k,
    )
    return results

print("--- HyDE ---")

5.1 三种核心后处理

# ---------- 1. Context Window 截断 ----------
def truncate_context(documents: list[str], max_chars: int = 4000) -> list[str]:
    """把检索结果截断到 LLM 的上下文窗口内
    类比 SQL: LIMIT 只取前N行，这里按字符数截断"""
    result = []
    total = 0
    for doc in documents:
        if total + len(doc) > max_chars:
            break
        result.append(doc)
        total += len(doc)
    return result

# ---------- 2. 相似度阈值过滤 ----------
def filter_by_threshold(
    documents: list[str],
    distances: list[float],
    max_distance: float = 1.0  # 根据实际情况调整
) -> list[str]:
    """过滤掉相似度太低的结果
    类比 SQL: WHERE score > threshold"""
    return [doc for doc, dist in zip(documents, distances) if dist <= max_distance]

# ---------- 3. 文档去重 ----------
def deduplicate_documents(documents: list[str]) -> list[str]:
    """去除内容重复或高度相似的文档片段"""
    seen = set()
    result = []
    for doc in documents:
        fingerprint = doc.strip()[:50]  # 用前50字符作为简易指纹
        if fingerprint not in seen:
            seen.add(fingerprint)
            result.append(doc)
    return result

5.2 完整的检索后处理流水线

from dataclasses import dataclass

@dataclass
class RetrievedChunk:
    """检索结果的数据结构"""
    text: str
    source: str       # 来源文件名
    distance: float    # 距离（越小越相似）
    metadata: dict     # 元数据

def post_process(
    chunks: list[RetrievedChunk],
    max_chars: int = 4000,
    max_distance: float = 1.0,
) -> list[RetrievedChunk]:
    """完整的后处理流水线：阈值过滤 -> 去重 -> 截断"""
    # Step 1: 相似度阈值过滤
    filtered = [c for c in chunks if c.distance <= max_distance]
    print(f"  阈值过滤: {len(chunks)} -> {len(filtered)} 条")

    # Step 2: 去重
    deduped = []
    seen = set()
    for chunk in filtered:
        fp = chunk.text.strip()[:50]
        if fp not in seen:
            seen.add(fp)
            deduped.append(chunk)
    print(f"  去重: {len(filtered)} -> {len(deduped)} 条")

    # Step 3: 按字符数截断（已经按相似度排好序，优先保留高相似度的）
    truncated = []
    total = 0
    for chunk in deduped:
        if total + len(chunk.text) > max_chars:
            break
        truncated.append(chunk)
        total += len(chunk.text)
    print(f"  截断: {len(deduped)} -> {len(truncated)} 条, 共 {total} 字符")

    return truncated
🎯 LLM 实战：完整 RAG 检索器 ⏱ 15 min

把今天学的内容组装成一个完整的 RAGRetriever 类：

class RAGRetriever:
    """完整的 RAG 检索器 — 组合查询改写 + 向量检索 + 后处理"""

    def __init__(self, collection, llm_client, top_k: int = 10, max_chars: int = 4000):
        self.collection = collection
        self.llm = llm_client
        self.top_k = top_k
        self.max_chars = max_chars

    def retrieve(self, query: str, rewrite: bool = True) -> RetrievalResult:
        """完整的检索流程"""
        # Step 1: Query 改写（可选）
        rewritten = rewrite_query(query) if rewrite else query

        # Step 2: 向量库检索
        results = self.collection.query(
            query_texts=[rewritten],
            n_results=self.top_k,
        )

        # Step 3: 组装 Chunk 列表
        chunks = []
        for doc, dist, meta in zip(
            results["documents"][0],
            results["distances"][0],
            results["metadatas"][0],
        ):
            chunks.append(RetrievedChunk(
                text=doc,
                source=meta.get("source", "unknown"),
                distance=dist,
                metadata=meta,
            ))

        # Step 4: 后处理
        final_chunks = post_process(chunks, max_chars=self.max_chars)

        return RetrievalResult(
            chunks=final_chunks,
            query=query,
            rewritten_query=rewritten,
            total_candidates=len(chunks),
            final_count=len(final_chunks),
        )

# 使用示例
retriever = RAGRetriever(collection, client)
result = retriever.retrieve("不良资产有哪些处置方式？")
print(f"候选: {result.total_candidates} -> 最终: {result.final_count}")
print(f"改写后: {result.rewritten_query}")
print(f"Context:\n{result.format_context()[:200]}...")
检索管线全链路类比 SQL：
Query 改写 = 解析需求补条件 → 向量库检索 = SELECT * FROM docs ORDER BY similarity → 后处理 = WHERE/HAVING/LIMIT/OFFSET

# 技巧1: 命中率日志 — 监控检索质量
import logging
logger = logging.getLogger("rag.retrieval")

def retrieve_with_logging(retriever, query: str):
    result = retriever.retrieve(query)
    logger.info(
        "检索日志 | query=%s | rewritten=%s | 候选=%d | 最终=%d | "
        "avg_dist=%.4f | sources=%s",
        query, result.rewritten_query,
        result.total_candidates, result.final_count,
        sum(c.distance for c in result.chunks) / max(len(result.chunks), 1),
        [c.source for c in result.chunks],
    )
    return result

# 技巧2: 空结果处理 — 兜底策略
def retrieve_with_fallback(retriever, query: str, min_results: int = 2):
    """如果检索结果太少，扩大范围重试"""
    result = retriever.retrieve(query)
    if result.final_count < min_results:
        logger.warning(f"检索结果不足({result.final_count}<{min_results})，扩大范围重试")
        # 扩大 top_k 重试
        retriever.top_k *= 2
        result = retriever.retrieve(query, rewrite=False)  # 不再改写，避免二次失真
        retriever.top_k //= 2  # 恢复
    return result

检索 Top-5 常见陷阱：
  1. Query 太短 — "五级分类"只有4个字，Embedding 信息不足。用 Query 改写补充。
  2. 不设阈值 — Top-K 结果可能全不相关。一定要设 max_distance 过滤。
  3. 忽略元数据 — 查 2024 年的政策但 2019 年的文档排在前面。加 where 过滤。
  4. Context 超长 — 10 个片段全塞进去超过 Token 限制。做截断。
  5. 不做去重 — 同一段内容被多个分块命中，浪费 Context 空间。

练习 3：LeetCode #347 — 前 K 个高频元素

给定一个数组，返回出现频率前 K 高的元素。提示：用 heapq.nlargest(k, counter.items(), key=itemgetter(1))，跟检索 Top-K 是同一套路。

# 模板
from collections import Counter
from heapq import nlargest
from operator import itemgetter

def top_k_frequent(nums: list[int], k: int) -> list[int]:
    counts = Counter(nums)
    return [item[0] for item in nlargest(k, counts.items(), key=itemgetter(1))]

# 测试
print(top_k_frequent([1,1,1,2,2,3], 2))  # [1, 2]
