2.1 结构化 JSON 日志

Day 27 学过 logging，今天把它升级成 JSON 格式 —— 日志直接写入文件，后续可以用 Python 或 SQL 分析：

import json import logging from datetime import datetime from pathlib import Path class JSONFormatter(logging.Formatter): """把每条日志转成 JSON，方便后续分析""" def format(self, record: logging.LogRecord) -> str: # 把 record 的额外字段取出来 log_data = { "time": datetime.fromtimestamp(record.created).isoformat(), "level": record.levelname, "module": record.module, "message": record.getMessage(), } # record.__dict__ 里会有 extra 传入的自定义字段 for key in ("agent_id", "tool_name", "step", "duration_ms"): if key in record.__dict__: log_data[key] = record.__dict__[key] return json.dumps(log_data, ensure_ascii=False) # ---- 配置 logger ---- logger = logging.getLogger("amc_agent") logger.setLevel(logging.DEBUG) # 同时输出到控制台（人类可读）和文件（机器可解析） console_handler = logging.StreamHandler() console_handler.setFormatter(logging.Formatter( "[%(levelname)s] %(message)s" )) logger.addHandler(console_handler) # JSON 文件 handler —— 每条一行 JSONL 格式 log_dir = Path("./logs") log_dir.mkdir(exist_ok=True) file_handler = logging.FileHandler(log_dir / "agent.jsonl") file_handler.setFormatter(JSONFormatter()) logger.addHandler(file_handler) # ---- 使用 ---- logger.info( "Agent 启动", extra={"agent_id": "agent-001", "step": "init"} ) logger.info( "工具调用完成", extra={ "agent_id": "agent-001", "tool_name": "db_query", "duration_ms": 342 } )
为什么用 JSONL（每行一条 JSON）？ ① 大文件也能逐行读取，不用全部加载到内存 ② grep/awk 等命令行工具直接能用 ③ 和数仓的 JSON 解析完美兼容 —— 你甚至可以用 Spark 读 JSONL 做日志分析。

2.2 审计追踪器

审计日志和普通日志不同：它记录的是谁、在什么时候、做了什么决策、结果是什么。对于 Agent 系统，每个决策点都要留痕：

from dataclasses import dataclass, field, asdict from typing import Optional from enum import Enum, auto class DecisionType(Enum): ROUTE = auto() # 路由决策：选择了哪个 Agent TOOL = auto() # 工具决策：调用了哪个工具 APPROVE = auto() # 审批决策：人工批准/拒绝 RETRY = auto() # 重试决策：自动重试/放弃 FALLBACK = auto() # 降级决策：主方案失败切备选 @dataclass class AuditRecord: """单条审计记录 —— Agent 的每一个关键决策""" run_id: str # 一次完整执行的唯一标识 timestamp: str # ISO 8601 时间 decision_type: DecisionType # 决策类型 agent_id: Optional[str] = None # 哪个 Agent 做的决策 detail: str = "" # 决策描述 result: str = "success" # 决策结果 duration_ms: Optional[int] = None # 耗时（毫秒） metadata: dict = field(default_factory=dict) # 附加信息 class AuditLogger: """审计日志管理器 —— 追踪 Agent 每一步决策""" def __init__(self, log_path: str = "./logs/audit.jsonl"): self.log_path = Path(log_path) self.log_path.parent.mkdir(parents=True, exist_ok=True) self._records: list[AuditRecord] = [] def log( self, decision_type: DecisionType, detail: str, run_id: str, agent_id: Optional[str] = None, result: str = "success", duration_ms: Optional[int] = None, **metadata ): """记录一条审计日志（同时写内存 + 追加文件）""" record = AuditRecord( run_id=run_id, timestamp=datetime.now().isoformat(), decision_type=decision_type, agent_id=agent_id, detail=detail, result=result, duration_ms=duration_ms, metadata=metadata, ) self._records.append(record) # 追加写入 JSONL 文件（数仓风格的行式存储） with open(self.log_path, "a", encoding="utf-8") as f: f.write(json.dumps(asdict(record), ensure_ascii=False) + "\n") def get_summary(self, run_id: str) -> dict: """按 run_id 汇总一次执行的所有决策""" steps = [r for r in self._records if r.run_id == run_id] return { "run_id": run_id, "total_steps": len(steps), "decisions": [ { "type": r.decision_type.name, "detail": r.detail, "result": r.result, "duration_ms": r.duration_ms, } for r in steps ], "success_rate": sum( 1 for r in steps if r.result == "success" ) / max(len(steps), 1), } # ---- 使用示例 ---- audit = AuditLogger() run_id = f"run-{datetime.now().strftime('%Y%m%d-%H%M%S')}" # 记录路由决策 audit.log( decision_type=DecisionType.ROUTE, detail="用户问题匹配到数据查询 Agent", run_id=run_id, agent_id="data-agent", ) # 记录工具调用 audit.log( decision_type=DecisionType.TOOL, detail="执行 db_query 查询不良资产余额", run_id=run_id, agent_id="data-agent", duration_ms=256, ) # 记录审批决策 audit.log( decision_type=DecisionType.APPROVE, detail="风险预警报告需人工确认", run_id=run_id, result="approved", metadata={"approver": "张主管", "comment": "数据准确，同意发送"}, ) # 查看汇总 print(json.dumps(audit.get_summary(run_id), indent=2, ensure_ascii=False))
和数仓审计的类比：你在数仓里做的每次跑批都有调度日志（Airflow/DolphinScheduler），Agent 系统的审计日志本质相同 —— 记录每个"步骤"的输入、输出、状态。区别只是 Agent 的"步骤"由 LLM 自主决定，而非你写死的 DAG。

3.2 性能指标收集器

用 time.perf_counter 做高精度计时，收集每个阶段的耗时分布：

from time import perf_counter from collections import defaultdict class MetricsCollector: """Agent 性能指标收集器 —— 每个阶段耗时一目了然""" def __init__(self): self._timings: dict[str, list[float]] = defaultdict(list) self._counts: dict[str, int] = defaultdict(int) self._errors: dict[str, int] = defaultdict(int) self._start_time: Optional[float] = None def start_run(self): """标记一次完整执行的开始""" self._start_time = perf_counter() @contextmanager def timer(self, stage: str): """上下文管理器：自动计时并记录""" start = perf_counter() try: yield finally: duration = (perf_counter() - start) * 1000 # 转毫秒 self._timings[stage].append(duration) self._counts[stage] += 1 def record_error(self, stage: str): self._errors[stage] += 1 def report(self) -> dict: """生成性能报告""" total_ms = (perf_counter() - self._start_time) * 1000 if self._start_time else 0 result = {"total_ms": round(total_ms, 1), "stages": {}} for stage, times in self._timings.items(): result["stages"][stage] = { "count": len(times), "avg_ms": round(sum(times)/len(times), 1), "max_ms": round(max(times), 1), "errors": self._errors.get(stage, 0), } return result

3.3 AMC 智能分析平台 —— 完整主入口

把所有组件拼到一起。这段代码展示了一次完整执行的全链路：

# main.py —— AMC 智能分析平台完整主入口 import asyncio from datetime import datetime from collections import deque # ============================================================ # 1. 初始化基础设施 # ============================================================ config = AMCConfig() logger = logging.getLogger("amc_agent") metrics = MetricsCollector() audit = AuditLogger() # ============================================================ # 2. 注册工具（来自 D53） # ============================================================ # 假设已有以下工具类：DBQueryTool, KnowledgeSearchTool, # ReportGeneratorTool, RiskAlertTool # 每个工具都实现 execute(query: str) -> str 接口 registry = ToolRegistry() registry.register("db_query", DBQueryTool()) registry.register("knowledge_search", KnowledgeSearchTool()) registry.register("generate_report", ReportGeneratorTool()) registry.register("risk_alert", RiskAlertTool()) # ============================================================ # 3. 配置 Agent 核心（来自 D54） # ============================================================ # 简化的 mock LLM 调用 —— 生产环境替换为真实 API async def mock_llm(messages: list[dict]) -> str: """模拟 LLM 响应（演示用，生产替换为智谱 API）""" await asyncio.sleep(0.1) # 模拟网络延迟 # 简单模拟：根据最后一条用户消息返回工具调用 user_msg = messages[-1]["content"].lower() if "余额" in user_msg: return json.dumps({ "thought": "需要查询数据库获取余额信息", "tool": "db_query", "input": "SELECT 余额 FROM 不良资产表 WHERE 日期='2026-07'" }) elif "报告" in user_msg: return json.dumps({ "thought": "用户需要生成分析报告", "tool": "generate_report", "input": " AMC 2026年7月不良资产分析报告" }) else: return json.dumps({ "thought": "直接回答用户问题", "answer": "根据 AMC 相关规定..." }) # ============================================================ # 4. 构建 Agent 核心循环 # ============================================================ async def run_agent(user_query: str, run_id: str) -> dict: """ Agent 完整执行流程： 用户提问 → 记忆加载 → LLM 推理 → 工具调用 → 结果生成 """ metrics.start_run() memory = deque(maxlen=config.memory_window) steps: list[dict] = [] # 系统提示词 system_prompt = """你是一个 AMC（资产管理公司）智能分析助手。 你可以使用以下工具回答用户关于不良资产管理的问题： - db_query: 查询数据库获取业务数据 - knowledge_search: 搜索知识库获取政策和制度信息 - generate_report: 生成结构化分析报告 - risk_alert: 发送风险预警通知 请先用中文思考你的分析思路，然后选择合适的工具。 回答格式必须是 JSON： {"thought": "你的思考", "tool": "工具名", "input": "工具输入"} 或 {"thought": "你的思考", "answer": "直接回答"} """ messages = [{"role": "system", "content": system_prompt}] messages.append({"role": "user", "content": user_query}) for step_idx in range(config.max_steps): # ---- 调用 LLM ---- with metrics.timer("llm_inference"): response_text = await mock_llm(messages) try: response = json.loads(response_text) except json.JSONDecodeError: logger.warning(f"LLM 返回非法 JSON: {response_text[:100]}") metrics.record_error("llm_inference") break # ---- 判断是直接回答还是工具调用 ---- if "answer" in response: # 直接回答 → 结束循环 steps.append({ "step": step_idx, "thought": response["thought"], "answer": response["answer"], }) logger.info( f"Agent 直接回答（第{step_idx}步）", extra={"run_id": run_id, "step": step_idx} ) break # ---- 执行工具调用 ---- tool_name = response["tool"] tool_input = response["input"] # 记录工具决策到审计日志 audit.log( decision_type=DecisionType.TOOL, detail=f"调用 {tool_name}: {tool_input[:50]}", run_id=run_id, ) # 执行工具 with metrics.timer("tool_execution"): tool = registry.get(tool_name) tool_result = await tool.execute(tool_input) # 记录步骤 steps.append({ "step": step_idx, "thought": response["thought"], "tool": tool_name, "input": tool_input, "result": tool_result[:200], # 截断防止上下文爆炸 }) # 把工具结果反馈给 LLM（继续循环） messages.append({"role": "assistant", "content": response_text}) messages.append({ "role": "user", "content": f"工具 {tool_name} 返回结果：{tool_result}\n请根据结果继续分析。" }) # 保存到记忆 memory.append({"role": "user", "content": user_query}) memory.append({"role": "assistant", "content": response_text[:200}) return { "run_id": run_id, "query": user_query, "steps": steps, "final_answer": steps[-1].get("answer", "（工具执行完毕）"), "metrics": metrics.report(), "audit_summary": audit.get_summary(run_id), } # ============================================================ # 5. 运行 # ============================================================ async def main(): run_id = f"run-{datetime.now().strftime('%Y%m%d-%H%M%S')}" result = await run_agent( user_query="查询2026年7月不良资产余额，并生成分析报告", run_id=run_id, ) # 打印完整执行报告 print("\n" + "="*50) print(f" AMC 智能分析平台 · 执行报告") print(f" Run ID: {run_id}") print("="*50) for step in result["steps"]: print(f"\n📋 Step {step['step']}: {step.get('thought', '')}") if "tool" in step: print(f" 🔧 工具: {step['tool']}({step['input'][:50]}...)") print(f" 📤 结果: {step['result'][:100]}...") else: print(f" 💬 回答: {step.get('answer', '')[:100]}") print(f"\n⏱️ 性能指标: {json.dumps(result['metrics'], indent=2)}") print(f"\n📝 审计摘要: {json.dumps(result['audit_summary'], indent=2)}") asyncio.run(main())

from collections import Counter import heapq def topKFrequent(nums: list[int], k: int) -> list[int]: # Counter 统计频率 + nlargest 取 Top-K（O(n log k)） return [item for item, _ in heapq.nlargest(k, Counter(nums).items(), key=lambda x: x[1])] # 测试 print(topKFrequent([1,1,1,2,2,3], 2)) # [1, 2]
