template_str = """

env = Environment(loader=FileSystemLoader("templates"))
template = env.get_template("report.html.j2")

# Jinja2 过滤器（管道符 | 调用）
# {{ value|round(2) }}  →  四舍五入2位
# {{ value|default("N/A") }}  →  默认值
# {{ items|join(", ") }}  →  列表拼接
# {{ score * 100|round(1) }}  →  注意运算优先级！

# 自定义过滤器（注册到 Environment）
def severity_color(severity):
    "根据严重级别返回CSS颜色类"
    colors = {"CRITICAL": "red", "HIGH": "orange", "MEDIUM": "yellow", "LOW": "green"}
    return colors.get(severity, "blue")

env.filters["severity_color"] = severity_color
# 模板中: <span class="{{ item.severity|severity_color }}">
数仓人速记：Jinja2 的模板语法和 SQL 的 CASE WHEN 思路一样 —— 根据条件渲染不同内容。{{ '✅' if passed else '❌' }} 等价于 SQL 的 CASE WHEN passed THEN '✅' ELSE '❌' END。

def chart_to_base64(fig) -> str:
    "将 matplotlib 图表转为 base64 字符串（内嵌HTML）"
    buf = BytesIO()
    fig.savefig(buf, format="png", dpi=150, bbox_inches="tight")

# 质量趋势图：最近N天得分折线
def plot_quality_trend(scores: dict) -> str:
    """scores = {"2026-08-08": 92.5, "2026-08-09": 88.3, ...}"""
    dates = list(scores.keys())
    values = list(scores.values())

    fig, ax = plt.subplots(figsize=(8, 3))
    ax.plot(dates, values, "o-", color="#3b82f6", linewidth=2, markersize=6)
    ax.axhline(y=80, color="#dc2626", linestyle="--", alpha=0.5, label="告警线(80分)")
    ax.set_ylim(0, 100)
    ax.set_title("数据质量趋势", fontsize=14)
    ax.set_ylabel("质量得分")
    for i, v in enumerate(values):
        ax.annotate(f"{v}", (dates[i], v), textcoords="offset points", xytext=(0,8), ha="center", fontsize=10)
    ax.legend()
    return chart_to_base64(fig)

2.3 Python 核心补充：string.Template 安全替换

当你不想用 Jinja2（轻量场景），可以用标准库的 string.Template：

from string import Template

# 安全！不会被注入恶意代码（不像 f-string 那样执行表达式）
t = Template("质量得分：$score / 100，异常数：$count")
result = t.substitute(score=92.5, count=3)
# → "质量得分：92.5 / 100，异常数：3"

# safe_substitute：缺失变量不报错，保留占位符
t.safe_substitute(score=92.5)
# → "质量得分：92.5 / 100，异常数：$count"
安全提示：拼接 HTML/SQL 时，永远不要直接用 f-string 插入用户数据 —— 存在注入风险。string.Template 和 Jinja2 都是更安全的替代。

from dataclasses import dataclass, field
from datetime import datetime
from pathlib import Path
from jinja2 import Template
from io import BytesIO
import base64
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt

@dataclass
class RuleResultItem:
    """单条规则检测结果"""
    name: str
    passed: bool
    violations: list = field(default_factory=list)
    severity: str = "MEDIUM"
    details: str = ""

@dataclass
class LLMAnomalyItem:
    """LLM检测到的异常项"""
    field: str
    description: str
    root_cause: str
    fix_suggestion: str
    priority: str = "HIGH"

class QualityReportBuilder:
    """质检报告生成器：整合规则结果 + LLM异常 + 趋势图表 → HTML"""

    def __init__(self, output_dir: str = "reports"):
        self.output_dir = Path(output_dir)
        self.output_dir.mkdir(parents=True, exist_ok=True)

    def build(
        self,
        rule_results: list[RuleResultItem],
        llm_anomalies: list[LLMAnomalyItem] | None = None,
        overall_score: float = 100.0,
        duration: float = 0.0,
        trend_scores: dict | None = None,
    ) -> str:
        "生成完整HTML质检报告，返回文件路径"
        llm_anomalies = llm_anomalies or []

        # 1. 统计指标
        total_checks = len(rule_results)
        anomaly_count = sum(
            len(r.violations) for r in rule_results if not r.passed
        )

        # 2. 生成趋势图表（如果有历史数据）
        chart_image = ""
        if trend_scores and len(trend_scores) >= 2:
            chart_image = self._render_trend_chart(trend_scores)

        # 3. Jinja2 渲染
        template = Template(REPORT_HTML)  # 上面的模板
        html = template.render(
            report_date=datetime.now().strftime("%Y-%m-%d %H:%M"),
            overall_score=overall_score,
            total_checks=total_checks,
            anomaly_count=anomaly_count,
            duration=round(duration, 1),
            rule_results=rule_results,
            llm_anomalies=llm_anomalies,
            chart_image=chart_image,
        )

        # 4. 写入文件
        filename = f"quality_report_{datetime.now().strftime('%Y%m%d_%H%M%S')}.html"
        filepath = self.output_dir / filename
        filepath.write_text(html, encoding="utf-8")

4.1 告警分级策略

质检结果不能一锅端告警 —— 要像数仓的 ETL 监控一样，按严重程度分级处理。

级别阈值条件通知渠道响应时间类比数仓
CRITICAL得分 < 60 或主键重复电话 + 企业微信 + 邮件30分钟内ETL 全量失败
HIGH得分 60-80 或空值率 > 20%企业微信 + 邮件2小时内增量同步延迟
MEDIUM得分 80-90 或个别规则失败邮件汇总当天处理单表数据倾斜
LOW得分 ≥ 90日志记录下个迭代常规监控

4.2 AlertManager 告警管理器

# alert_manager.py
from dataclasses import dataclass, field
from enum import Enum
from datetime import datetime
import smtplib
from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart

class AlertLevel(Enum):
    CRITICAL = "CRITICAL"
    HIGH = "HIGH"
    MEDIUM = "MEDIUM"
    LOW = "LOW"

@dataclass
class AlertRule:
    """告警触发规则"""
    level: AlertLevel
    condition: callable  # (score, anomalies) → bool
    channels: list[str]     # ["email", "webhook", "sms"]
    cooldown_min: int = 60  # 同级别冷却时间（防刷屏）

@dataclass
class AlertRecord:
    """告警记录（防重复发送）"""
    level: AlertLevel
    triggered_at: datetime
    score: float
    anomaly_count: int

class AlertManager:
    """告警管理器：分级触发 + 冷却防刷 + 多渠道通知"""

    def __init__(self):
        self._rules: list[AlertRule] = []
        self._history: list[AlertRecord] = []
        self._setup_default_rules()

    def _setup_default_rules(self):
        """预设四级告警规则"""
        self._rules = [
            AlertRule(
                level=AlertLevel.CRITICAL,
                condition=lambda s, a: s < 60 or a > 10,
                channels=["webhook", "email"],
                cooldown_min=30,
            ),
            AlertRule(
                level=AlertLevel.HIGH,
                condition=lambda s, a: 60 <= s < 80 or a > 5,
                channels=["webhook", "email"],
                cooldown_min=60,
            ),
            AlertRule(
                level=AlertLevel.MEDIUM,
                condition=lambda s, a: 80 <= s < 90 or a > 0,
                channels=["email"],
                cooldown_min=120,
            ),
            AlertRule(
                level=AlertLevel.LOW,
                condition=lambda s, a: s >= 90,
                channels=["log"],
                cooldown_min=0,  # LOW级别不限频
            ),
        ]

    def check(self, score: float, anomaly_count: int) -> list[AlertRule]:
        "检查是否需要告警，返回触发的规则列表（已过滤冷却）"
        now = datetime.now()
        triggered = []
        for rule in self._rules:
            if rule.condition(score, anomaly_count):
                # 冷却检查：同一级别在 cooldown_min 内是否已发过
                last = None
                for rec in self._history:
                    if rec.level == rule.level:
                        last = rec
                if last is None or \
                   (now - last.triggered_at).total_seconds() > rule.cooldown_min * 60:
                    triggered.append(rule)
                    self._history.append(
                        AlertRecord(rule.level, now, score, anomaly_count)
                    )
        return triggered

    def send_webhook(self, title: str, content: str, webhook_url: str):
        """发送企业微信/飞书 Webhook 通知（兼容格式）"""
        import requests, json
        payload = {
            "msgtype": "markdown",
            "markdown": {
                "title": title,
                "text": f"## {title}\n{content}"
            }
        }
        resp = requests.post(webhook_url, json=payload, timeout=10)
        resp.raise_for_status()

    def send_email(self, subject: str, body: str,
                   to: str, smtp_config: dict):
        """发送邮件通知（Day58 学过，回顾）"""
        msg = MIMEMultipart("alternative")
        msg["Subject"] = subject
        msg["From"] = smtp_config["sender"]
        msg[attach(MIMEText(body, "html", "utf-8"))
        with smtplib.SMTP_SSL(smtp_config["host"], smtp_config["port"]) as server:
            server.login(smtp_config["user"], smtp_config["password"])
            server.sendmail(smtp_config["sender"], to, msg.as_string())
数仓人秒懂：告警分级 ≈ ETL 监控的 告警等级。CRITICAL = 全量ETL挂了必须马上修；HIGH = 增量延迟；MEDIUM = 个别表异常；LOW = 日常巡检。冷却机制防止同一个问题反复告警刷屏（数仓里叫"告警抑制"）。

# fix_suggester.py — LLM 自动修复建议生成器
from dataclasses import dataclass
import re, json

@dataclass
class FixSuggestion:
    """修复建议（Pydantic 结构化输出）"""
    fix_type: str         # "sql" / "manual" / "ignore"
    fix_sql: str | None    # 修复SQL（fix_type=sql时有值）
    reason: str           # 为什么这么修
    risk: str            # 风险等级：LOW/MEDIUM/HIGH
    estimated_rows: int   # 预计影响行数

class FixSuggester:
    """基于 LLM 的修复建议生成器"""

    # 禁止的 SQL 操作（安全红线）
    FORBIDDEN = ["DROP", "TRUNCATE", "ALTER", "DELETE FROM", "GRANT"]

    def __init__(self, call_llm: callable):
        self._call_llm = call_llm  # LLM 调用函数

    def generate(self, anomaly: LLMAnomalyItem, table_schema: str) -> FixSuggestion:
        "为单条异常生成修复建议"
        prompt = f"""你是数据质量修复专家。根据以下信息生成修复方案。

【异常信息】
- 字段：{anomaly.field}
- 问题描述：{anomaly.description}
- 根因分析：{anomaly.root_cause}

【表结构】
{table_schema}

【要求】
1. 优先生成 UPDATE 修复 SQL（绝不能包含 DROP/TRUNCATE/DELETE FROM/ALTER）
2. 如果无法自动修复，返回 fix_type="manual"
3. 如果是误报，返回 fix_type="ignore"

请严格按JSON格式返回：
{{"fix_type": "sql", "fix_sql": "UPDATE ...", "reason": "...", "risk": "LOW", "estimated_rows": 5}}"""

        raw = self._call_llm(prompt)
        return self._parse_fix(raw)

    def _parse_fix(self, raw: str) -> FixSuggestion:
        "三层容错解析 LLM 返回的修复JSON"
        # 尝试1：直接 json.loads
        try:
            data = json.loads(raw)
            return self._to_fix(data)
        except (json.JSONDecodeError, KeyError):
            pass

        # 尝试2：正则提取 JSON 块
        match = re.search(r'\{[^{}]*\}', raw, re.DOTALL)
        if match:
            try:
                data = json.loads(match.group())
                return self._to_fix(data)
            except (json.JSONDecodeError, KeyError):
                pass

        # 尝试3：返回 fallback
        return FixSuggestion(
            fix_type="manual", fix_sql=None,
            reason="LLM返回格式异常，需要人工分析",
            risk="HIGH", estimated_rows=0,
        )

    def _to_fix(self, data: dict) -> FixSuggestion:
        "安全校验 + 构建 FixSuggestion"
        fix_sql = data.get("fix_sql")

        # 安全检查：禁止危险操作
        if fix_sql:
            sql_upper = fix_sql.upper()
            for forbidden in self.FORBIDDEN:
                if forbidden in sql_upper:
                    return FixSuggestion(
                        fix_type="manual", fix_sql=None,
                        reason=f"安全拦截：SQL包含禁止操作 {forbidden}",
                        risk="HIGH", estimated_rows=0,
                    )
            # 强制加上 WHERE 条件检查
            if "UPDATE" in sql_upper and "WHERE" not in sql_upper:
                return FixSuggestion(
                    fix_type="manual", fix_sql=None,
                    reason="安全拦截：UPDATE语句缺少WHERE条件",
                    risk="HIGH", estimated_rows=0,
                )

        return FixSuggestion(
            fix_type=data.get("fix_type", "manual"),
            fix_sql=fix_sql,
            reason=data.get("reason", "无"),
            risk=data.get("risk", "MEDIUM"),
            estimated_rows=data.get("estimated_rows", 0),
        )

    def batch_generate(self, anomalies: list, table_schema: str) -> list[FixSuggestion]:
        "批量生成修复建议（过滤 ignore 类型）"
        fixes = []
        for a in anomalies:
            fix = self.generate(a, table_schema)
            if fix.fix_type != "ignore":
                fixes.append(fix)
        return fixes
安全红线：自动修复 SQL 必须强制人工确认后才能执行！FixSuggester 只负责建议，不执行 SQL。安全检查三层：禁止 DDL/DROP → UPDATE 必须有 WHERE → 输出结构化 JSON。

# pipeline.py — P4 完整质检管线
import time, json, logging
from pathlib import Path
from datetime import datetime

logger = logging.getLogger("quality.pipeline")

class QualityPipeline:
    """
    P4 完整质检管线（5阶段）
    Day59: 规则扫描
    Day60: LLM 语义检测 + 归因
    Day61: 报告渲染 + 告警 + 修复建议
    """

    def __init__(self, rule_engine, llm_detector, attributor,
                 report_builder, alert_mgr, fix_suggester):
        self.rule_engine = rule_engine      # Day59
        self.llm_detector = llm_detector   # Day60
        self.attributor = attributor        # Day60
        self.report_builder = report_builder  # Day61
        self.alert_mgr = alert_mgr          # Day61
        self.fix_suggester = fix_suggester  # Day61

    def run(self, data: list[dict], table_name: str,
            table_schema: str = "",
            trend_scores: dict | None = None) -> dict:
        "执行完整质检管线，返回结果摘要"
        start = time.perf_counter()
        logger.info(f"[Pipeline] 开始质检: {table_name}, {len(data)}行")

        # ===== 阶段1：规则扫描 (Day59) =====
        logger.info("[Stage 1] 规则引擎扫描...")
        rule_report = self.rule_engine.check(data)

        # ===== 阶段2：智能采样 (Day60) =====
        logger.info("[Stage 2] 智能采样...")
        suspicious_rows = [r for r in data if self._is_suspicious(r, rule_report)]
        normal_rows = [r for r in data if r not in suspicious_rows]
        sample = suspicious_rows[:len(normal_rows)] + normal_rows[:len(suspicious_rows)]

        # ===== 阶段3：LLM 语义检测 (Day60) =====
        llm_anomalies = []
        if sample:
            logger.info(f"[Stage 3] LLM检测: {len(sample)}行采样")
            llm_results = self.llm_detector.detect(sample, table_schema)
            llm_anomalies = [
                LLMAnomalyItem(
                    field=a["field"],
                    description=a["description"],
                    root_cause=a["root_cause"],
                    fix_suggestion="待生成",
                )
                for a in llm_results
            ]

        # ===== 阶段4：修复建议生成 (Day61) =====
        fixes = []
        if llm_anomalies and table_schema:
            logger.info(f"[Stage 4] 生成修复建议: {len(llm_anomalies)}条异常")
            fixes = self.fix_suggester.batch_generate(llm_anomalies, table_schema)
            # 把修复建议回填到异常项
            for a, f in zip(llm_anomalies, fixes):
                a.fix_suggestion = f.reason

        # ===== 阶段5：报告 + 告警 (Day61) =====
        duration = time.perf_counter() - start
        logger.info(f"[Stage 5] 生成报告 + 告警, 耗时{duration:.1f}s")

        report_path = self.report_builder.build(
            rule_results=rule_report.rule_results,
            llm_anomalies=llm_anomalies,
            overall_score=rule_report.overall_score,
            duration=duration,
            trend_scores=trend_scores,
        )

        # 告警检查
        alerts = self.alert_mgr.check(
            report_report.overall_score,
            len(llm_anomalies) + len(rule_report.failed_rules),
        )
        for alert in alerts:
            logger.warning(f"[Alert] {alert.level.value} 触发告警")
            # self.alert_mgr.send_webhook(...) 实际发送

        return {
            "table": table_name,
            "score": report_report.overall_score,
            "rule_issues": len(report_report.failed_rules),
            "llm_issues": len(llm_anomalies),
            "fixes": len(fixes),
            "report": report_path,
            "alerts": len(alerts),
            "duration": round(duration, 2),
        }

    def _is_suspicious(self, row, report) -> bool:
        "判断某行是否可疑（被规则命中）"
        for result in report.rule_results:
            if not result.passed and row in result.violations:
                return True
        return False

⚠️ 常见陷阱 Top 5
  1. matplotlib 不 close(fig) → 内存泄漏，每次生成图表吃几十MB。养成习惯：fig.savefig(...) → plt.close(fig)
  2. base64 图片过大 → HTML报告几MB。解决：降低 dpi（120足够）+ bbox_inches="tight" 裁白边
  3. Jinja2 模板 XSS → 用户数据直接渲染到 HTML。解决：用 {{ var|e }} 自动转义，或 autoescape=True
  4. 告警风暴 → 同一问题反复告警。解决：冷却机制（cooldown_min）+ 告警聚合（多条合并为一条）
  5. 自动修复 SQL 执行 → 永远不要自动执行！FixSuggester 只生成建议，执行必须人工确认 + 审批流程
