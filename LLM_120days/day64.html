2.1 deque 高级用法 — 滑动窗口 + maxlen

deque(maxlen=N) 自动丢弃超出部分，非常适合做"保留最近 N 轮对话"的滑动窗口。Day 63 用过基础版，今天看进阶场景。

from collections import deque

# maxlen 自动淘汰最老的元素 — O(1) 复杂度
recent = deque(maxlen=5)
for i in range(8):
    recent.append(f"turn_{i}")
print(list(recent))  # ['turn_3', 'turn_4', 'turn_5', 'turn_6', 'turn_7']

# 实际场景：保留最近N轮对话（每轮=用户消息+助手回复）
class ConversationWindow:
    def __init__(self, max_turns: int = 10, max_chars: int = 4000):
        self._turns = deque(maxlen=max_turns)  # 最多保留N轮
        self._max_chars = max_chars            # 字符数硬上限
        self._total_chars = 0

    def add_turn(self, role: str, content: str):
        """添加一轮对话，超字符上限则裁剪"""
        entry = {"role": role, "content": content}
        self._total_chars += len(content)

        # 超字符上限 → 从队头淘汰旧轮次
        while self._total_chars > self._max_chars and self._turns:
            removed = self._turns.popleft()
            self._total_chars -= len(removed["content"])

        self._turns.append(entry)

    def get_history(self) -> list[dict]:
        """获取当前窗口内的对话历史"""
        return list(self._turns)

    def summary_stats(self) -> dict:
        return {
            "turn_count": len(self._turns),
            "total_chars": self._total_chars,
            "max_capacity": self._turns.maxlen
        }

# 测试

win.add_turn("user", "查询资产A001")
win.add_turn("assistant", "资产A001余额为100万")
win.add_turn("user", "还有哪些风险")

# 实际场景：将老对话压缩为摘要行
def summarize_turns(turns: list[dict], max_summary_len: int = 200) -> str:
    """将多轮对话压缩为一句摘要"""
    key_points = []
    for t in turns:
        role = "用户" if t["role"] == "user" else "助手"
        brief = textwrap.shorten(t["content"], width=30, placeholder="...")
        key_points.append(f"{role}:{brief}")
    full = "; ".join(key_points)
    return textwrap.shorten(full, width=max_summary_len, placeholder="...")

def append_jsonl(filepath: str, record: dict):
    """追加一条 JSON 记录到文件末尾"""
    with open(filepath, "a", encoding="utf-8") as f:
        f.write(json.dumps(record, ensure_ascii=False) + "\n")

# 手写 JSONL 逐行读取
def read_jsonl(filepath: str) -> list[dict]:
    records = []
    with open(filepath, "r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if line:
                records.append(json.loads(line))
    return records

3.1 ContextCompressor — 上下文压缩器

长对话的 Token 会指数增长。策略：最近 N 轮保留原文，更早的轮次压缩为摘要。这和数仓里的"明细 + 汇总"分层思路一致。

from dataclasses import dataclass, field
from collections import deque
import textwrap, json, time

@dataclass
class CompressedContext:
    """压缩后的上下文，给 LLM 用"""
    summary: str               # 历史摘要
    recent_turns: list[dict]  # 最近N轮原文
    total_chars: int           # 压缩后总字符数
    compression_ratio: float   # 压缩比（原始/压缩后）

@dataclass
class ContextCompressor:
    """双策略上下文压缩器"""
    recent_window: int = 6       # 保留最近N轮原文
    summary_max_chars: int = 300 # 摘要最大字符数
    _all_turns: list = field(default_factory=list)

    def add_turn(self, role: str, content: str):
        """添加一轮对话"""
        self._all_turns.append({"role": role, "content": content})

    def compress(self) -> CompressedContext:
        """
        压缩策略：
        1. 最近 recent_window 轮 → 保留原文（明细层）
        2. 更早的轮次 → 压缩为摘要（汇总层）
        类似数仓：DWD明细 + DWS汇总
        """
        if len(self._all_turns) <= self.recent_window:
            # 对话还不够长，不需要压缩
            return CompressedContext(
                summary="",
                recent_turns=[*self._all_turns],
                total_chars=sum(len(t["content"]) for t in self._all_turns),
                compression_ratio=1.0
            )

        # 分层：历史摘要 + 最近原文
        older_turns = self._all_turns[:-self.recent_window]
        recent_turns = self._all_turns[-self.recent_window:]

        # 老对话压缩为摘要（实际中可调用 LLM 生成摘要）
        summary = self._build_summary(older_turns)

        recent_chars = sum(len(t["content"]) for t in recent_turns)
        original_chars = sum(len(t["content"]) for t in self._all_turns)
        total_chars = len(summary) + recent_chars

        return CompressedContext(
            summary=summary,
            recent_turns=recent_turns,
            total_chars=total_chars,
            compression_ratio=round(original_chars / max(total_chars, 1), 2)
        )

    def _build_summary(self, turns: list[dict]) -> str:
        """将老对话压缩为摘要行"""
        # 简化版：提取每轮要点拼接（实际可调 LLM 生成摘要）
        points = []
        for t in turns:
            brief = textwrap.shorten(
                t["content"], width=25, placeholder="..."
            )
            role = "用户" if t["role"] == "user" else "助手"
            points.append(f"{role}:{brief}")
        raw = "; ".join(points)
        return textwrap.shorten(
            raw, width=self.summary_max_chars,
            placeholder="...(更多历史已省略)"
        )

    def build_llm_messages(self) -> list[dict]:
        """构建给 LLM 的 messages 列表"""
        ctx = self.compress()
        messages = []
        # 如果有历史摘要，作为系统消息的一部分
        if ctx.summary:
            messages.append({
                "role": "system",
                "content": f"以下是之前的对话摘要:\n{ctx.summary}"
            })
        # 最近N轮原文
        messages.extend(ctx.recent_turns)
        return messages

# 测试

3.2 DialogueEvaluator — 对话质量评估器

怎么知道多轮对话"质量好不好"？定义 5 个可自动计算的指标：

指标含义计算方式健康阈值
completion_rate槽位填写完成率已完成意图数 / 总意图数≥ 0.7
avg_turns每个意图平均轮数总轮数 / 已完成意图数≤ 5
fallback_rate兜底回复占比fallback 次数 / 总轮次≤ 0.15
response_time_p9595 分位响应时间排序取 95% 位置≤ 3000ms
slot_hit_rate实体提取命中率提取到的槽位数 / 需要的槽位数
from dataclasses import dataclass, field
from statistics import mean

@dataclass
class DialogueMetrics:
    """一次会话的质量指标"""
    session_id: str
    total_turns: int = 0
    completed_intents: int = 0
    total_intents: int = 0
    fallback_count: int = 0
    response_times: list = field(default_factory=list)
    slots_filled: int = 0
    slots_needed: int = 0

    @property
    def completion_rate(self) -> float:
        return self.completed_intents / max(self.total_intents, 1)

    @property
    def avg_turns_per_intent(self) -> float:
        return round(self.total_turns / max(self.completed_intents, 1), 1)

    @property
    def fallback_rate(self) -> float:
        return self.fallback_count / max(self.total_turns, 1)

    @property
    def response_time_p95(self) -> float:
        if not self.response_times:
            return 0
        sorted_times = sorted(self.response_times)
        idx = int(len(sorted_times) * 0.95)
        return sorted_times[min(idx, len(sorted_times) - 1)]

    @property
    def slot_hit_rate(self) -> float:
        return self.slots_filled / max(self.slots_needed, 1)

    def health_report(self) -> dict:
        """生成健康报告，标注异常项"""
        report = {
            "session_id": self.session_id,
            "completion_rate": self.completion_rate,
            "avg_turns": self.avg_turns_per_intent,
            "fallback_rate": self.fallback_rate,
            "p95_ms": self.response_time_p95,
            "slot_hit": self.slot_hit_rate,
            "issues": []
        }
        # 健康阈值检查
        if self.completion_rate < 0.7:
            report["issues"].append(
                f"完成率过低({self.completion_rate:.0%})，检查槽位设计")
        if self.fallback_rate > 0.15:
            report["issues"].append(
                f"兜底率过高({self.fallback_rate:.0%})，扩充规则或训练数据")
        if self.response_time_p95 > 3000:
            report["issues"].append(
                f"P95响应过慢({self.response_time_p95:.0f}ms)，检查LLM或网络")
        return report

# 测试

    fallback_count=1,
    response_times=[200, 350, 800, 1500, 3200],
    slots_filled=8, slots_needed=10
)

🌐 FastAPI WebSocket 实时对话 15 min

4.1 为什么用 WebSocket 而不是 HTTP？

特性HTTP 请求-响应WebSocket 全双工
连接方式每次请求新建 TCP 连接一次握手，持久连接
通信方向客户端发起，服务端回复双方随时发送
适用场景查数据、提交表单实时对话、通知推送
延迟每次都有握手开销消息即时送达
数仓类比批量跑批查询Kafka 实时流消费
核心概念：HTTP 就像"打电话问一个问题，挂掉，再打下一个"；WebSocket 就像"建立通话后一直聊"，双方随时说话，延迟更低、体验更好。智能客服天然适合 WebSocket。

4.2 完整 WebSocket 聊天服务

# chat_ws.py — FastAPI WebSocket 聊天服务
from fastapi import FastAPI, WebSocket, WebSocketDisconnect
from fastapi.middleware.cors import CORSMiddleware
from contextlib import asynccontextmanager
import json, time, uuid
from collections import defaultdict

app = FastAPI(title="AMC 智能客服")

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_methods=["*"],
    allow_headers=["*"],
)

# ========== 连接管理 ==========
class ConnectionManager:
    """管理所有活跃的 WebSocket 连接"""
    def __init__(self):
        # session_id → WebSocket 实例
        self._connections: dict[str, WebSocket] = {}
        # session_id → 对话历史
        self._histories: dict[str, list] = defaultdict(list)

    async def connect(self, ws: WebSocket, session_id: str):
        await ws.accept()
        self._connections[session_id] = ws
        print(f"[连接] {session_id} 已建立")

    def disconnect(self, session_id: str):
        self._connections.pop(session_id, None)
        print(f"[断开] {session_id} 已离线")

    async def send_json(self, session_id: str, data: dict):
        """向指定会话发送 JSON 消息"""
        ws = self._connections.get(session_id)
        if ws:
            await ws.send_json(data)

manager = ConnectionManager()

# ========== 简化版意图处理（实际替换为 Day 62 的 IntentClassifier）==========
def simple_process(message: str, history: list) -> dict:
    """简化版对话处理（演示用，实际接入 ConversationManager）"""
    start = time.time()

    # 简单规则匹配
    msg_lower = message.lower()
    if "你好" in msg_lower or "hello" in msg_lower:
        reply = "您好！我是AMC智能客服，可以帮您查询资产信息、了解处置进度、分析风险报告。请问需要什么帮助？"
        intent = "greeting"
    elif "资产" in msg_lower or "查询" in msg_lower:
        reply = "请提供资产编号，我来为您查询详细信息。例如：A001、B003。"
        intent = "query_asset"
    elif "风险" in msg_lower or "报告" in msg_lower:
        reply = "正在为您生成风险分析报告，请稍候...（实际中这里调用 LLM + RAG）"
        intent = "risk_report"
    else:
        reply = "抱歉，我暂时无法理解您的问题。您可以尝试询问：资产查询、风险报告、处置进度等。"
        intent = "fallback"

    elapsed_ms = int((time.time() - start) * 1000)

    return {
        "reply": reply,
        "intent": intent,
        "response_time_ms": elapsed_ms,
        "turn": len(history) + 1
    }

# ========== WebSocket 端点 ==========
@app.websocket("/ws/chat/{session_id}")
async def chat_endpoint(ws: WebSocket, session_id: str):
    await manager.connect(ws, session_id)
    history = manager._histories[session_id]

    try:
        # 发送欢迎消息
        await manager.send_json(session_id, {
            "type": "system",
            "content": "连接成功，AMC智能客服已就绪",
            "session_id": session_id
        })

        # 持续监听客户端消息
        while True:
            # 接收客户端发来的 JSON 消息
            data = await ws.receive_json()
            user_msg = data.get("message", "")

            # 记录用户消息
            history.append({"role": "user", "content": user_msg})

            # 处理对话（实际中接入 ConversationManager）
            result = simple_process(user_msg, history)

            # 记录助手回复
            history.append({"role": "assistant", "content": result["reply"]})

            # 发送回复
            await manager.send_json(session_id, {
                "type": "reply",
                "content": result["reply"],
                "intent": result["intent"],
                "turn": result["turn"],
                "response_time_ms": result["response_time_ms"]
            })

    except WebSocketDisconnect:
        manager.disconnect(session_id)
        # 保存对话日志
        print(f"[会话结束] {session_id}，共 {len(history)} 轮")

# ========== HTTP 健康检查 ==========
@app.get("/health")
async def health():
    return {
        "status": "ok",
        "active_sessions": len(manager._connections)
    }

# ========== 启动命令 ==========
# uvicorn chat_ws:app --host 0.0.0.0 --port 8000 --reload
关键区别：WebSocket 端点用 @app.websocket 装饰器，接收/发送用 await ws.receive_json() / await ws.send_json()。连接一旦建立就在 while True 循环中持续通信，直到客户端断开触发 WebSocketDisconnect。

import asyncio
async def heartbeat_watchdog(ws, timeout=30):

        await asyncio.sleep(timeout)
        try:
            await asyncio.wait_for(ws.send_json({"type": "ping"}), timeout=5)

  3. LeetCode #146 — LRU 缓存：用 OrderedDict 实现一个 LRU 缓存（容量 K，超容量淘汰最久未使用的）。这在对话缓存、Embedding 缓存中非常常用。 
    from collections import OrderedDict

class LRUCache:
    def __init__(self, capacity: int):
        self._cache = OrderedDict()
        self._cap = capacity

    def get(self, key: str) -> str | None:
        if key not in self._cache:
            return None
        self._cache.move_to_end(key)  # 标记为最近使用
        return self._cache[key]

    def put(self, key: str, value: str):
        if key in self._cache:
            self._cache.move_to_end(key)
        self._cache[key] = value
        if len(self._cache) > self._cap:
            self._cache.popitem(last=False)  # 淘汰最旧的
