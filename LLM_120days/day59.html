🐍 Python 知识：ABC 抽象基类 + 规则引擎设计模式 ~15 分钟

规则引擎的核心设计思想：定义统一的规则接口，每种规则只需实现 check() 方法。这正好是面向对象编程中抽象基类（ABC）的经典场景。

🧩1. ABC 抽象基类 — 规则引擎的骨架

from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from typing import Any
from enum import Enum

class Severity(Enum):
    """严重度分级 — 类似数仓中的告警级别"""
    WARNING = "warning"     # 轻微异常，不影响流程
    ERROR = "error"         # 数据质量问题，需要修复
    CRITICAL = "critical"   # 严重错误，必须阻断

@dataclass
class RuleResult:
    """单条规则的检测结果"""
    rule_id: str            # 规则ID，如 "null_check_001"
    rule_name: str          # 规则名称
    severity: Severity      # 严重度
    passed: bool            # 是否通过
    column: str              # 检测的列名
    detail: str = ""         # 详情描述
    failed_count: int = 0    # 失败记录数
    total_count: int = 0    # 总记录数
    sample_values: list = field(default_factory=list)  # 异常样本

    @property
    def fail_rate(self) -> float:
        """失败率"""
        return self.failed_count / self.total_count if self.total_count > 0 else 0.0


class BaseRule(ABC):
    """质检规则抽象基类 — 所有规则必须继承它"""

    def __init__(self, rule_id: str, rule_name: str, severity: Severity = Severity.ERROR):
        self.rule_id = rule_id
        self.rule_name = rule_name
        self.severity = severity

    @abstractmethod
    def check(self, data: list[dict], column: str) -> RuleResult:
        """
        核心检测方法 — 子类必须实现
        :param data: 数据行列表 [{"col1": val1, ...}, ...]
        :param column: 要检测的列名
        :return: RuleResult 检测结果
        """
        ...

    def __repr__(self):
        return f"{self.__class__.__name__}(id={self.rule_id}, name={self.rule_name})"
设计要点：ABC 的 @abstractmethod 强制子类实现 check()。如果忘了实现，实例化时就直接报错 — 编译期就能发现问题，而不是运行时。

🔧2. functools.reduce — 链式聚合结果

当规则引擎执行完所有规则后，需要把多个 RuleResult 聚合成一份总报告。reduce 是函数式编程的利器：

from functools import reduce

# --- reduce 基础用法 ---
# reduce(函数, 列表, 初始值)
# 原理：初始值 与 列表第1个元素 → 新值，再与第2个 → 新值，依次折叠

numbers = [1, 2, 3, 4, 5]
total = reduce(lambda acc, x: acc + x, numbers, 0)
print(total)  # 15

# --- 等价于 ---
acc = 0
for x in numbers:
    acc = acc + x  # 每次把累积值和当前元素"折叠"在一起

# --- 实际场景：合并多个规则结果列表 ---
results_list = [
    [RuleResult("r1", "空值检查", Severity.ERROR, False, "col_a", failed_count=5, total_count=100)],
    [RuleResult("r2", "范围检查", Severity.WARNING, True, "col_b", total_count=100)],
    [RuleResult("r3", "格式检查", Severity.ERROR, False, "col_c", failed_count=2, total_count=100)],
]

# reduce 把多个列表"拍平"成一个
all_results = reduce(lambda acc, x: acc + x, results_list, [])
print(f"共 {len(all_results)} 条检测结果")  # 3

# --- 统计：计算总失败数 ---
total_failed = reduce(
    lambda acc, r: acc + r.failed_count,
    all_results,
    0
)
print(f"总失败记录: {total_failed}")  # 7

🔗3. itertools.chain — 扁平化多数据源

from itertools import chain

# 场景：质检多个数据表，每个表产生一组结果
table_a_results = [RuleResult("r1", ...), RuleResult("r2", ...)]
table_b_results = [RuleResult("r3", ...)]
table_c_results = [RuleResult("r4", ...), RuleResult("r5", ...), RuleResult("r6", ...)]

# chain: 比 reduce + 列表拼接更高效（惰性迭代器，不创建中间列表）
all_results = list(chain(table_a_results, table_b_results, table_c_results))
print(f"跨表共 {len(all_results)} 条结果")  # 6

# 实际场景：遍历多个 CSV 文件，合并数据行
import csv
from pathlib import Path

csv_files = Path("data").glob("amc_*.csv")
all_rows = list(chain.from_iterable(  # chain.from_iterable 把"列表的列表"铺平
    csv.DictReader(f.open(encoding="utf-8"))
    for f in csv_files
))
print(f"合并 {len(csv_files)} 个文件，共 {len(all_rows)} 行")
reduce vs chain 选择指南： reduce 适合需要"累积计算"的场景（求和/合并/分组）；chain 适合"拼接多个可迭代对象"的场景。简单拼接优先用 chain，更高效。

📊1. 空值检查（Null Check）

class NullCheckRule(BaseRule):
    """检查指定列的空值率是否超过阈值"""

    def __init__(self, rule_id: str, column: str,
                 max_null_rate: float = 0.05,
                 severity: Severity = Severity.ERROR):
        super().__init__(rule_id, f"{column} 空值检查", severity)
        self.column = column
        self.max_null_rate = max_null_rate  # 空值率上限，默认5%

    def check(self, data: list[dict], column: str = None) -> RuleResult:
        col = column or self.column
        total = len(data)
        if total == 0:
            return RuleResult(self.rule_id, self.rule_name, self.severity,
                             True, col, "无数据", 0, 0)

        null_count = sum(1 for row in data
                        if row.get(col) is None or str(row.get(col, "")).strip() == "")
        null_rate = null_count / total
        passed = null_rate <= self.max_null_rate

        # 采集最多5个空值样本
        samples = [row for row in data
                   if row.get(col) is None or str(row.get(col, "")).strip() == ""][:5]

        return RuleResult(
            rule_id=self.rule_id, rule_name=self.rule_name,
            severity=self.severity, passed=passed, column=col,
            detail=f"空值率 {null_rate:.2%}（阈值 {self.max_null_rate:.2%}），"
                   f"空值 {null_count}/{total} 条",
            failed_count=null_count, total_count=total,
            sample_values=samples
        )

📏2. 范围检查（Range Check）

class RangeCheckRule(BaseRule):
    """检查数值列是否在指定范围内"""

    def __init__(self, rule_id: str, column: str,
                 min_val: float = None, max_val: float = None,
                 severity: Severity = Severity.ERROR):
        super().__init__(rule_id, f"{column} 范围检查", severity)
        self.column = column
        self.min_val = min_val  # 最小值（None表示不检查下界）
        self.max_val = max_val  # 最大值（None表示不检查上界）

    def check(self, data: list[dict], column: str = None) -> RuleResult:
        col = column or self.column
        total = len(data)
        out_of_range = []

        for i, row in enumerate(data):
            val = row.get(col)
            if val is None:
                continue  # 空值交给 NullCheckRule 处理
            try:
                num = float(val)
                if (self.min_val is not None and num < self.min_val) or \
                   (self.max_val is not None and num > self.max_val):
                    out_of_range.append((i, val))
            except (ValueError, TypeError):
                out_of_range.append((i, val))  # 非数值也算异常

        passed = len(out_of_range) == 0
        samples = [f"行{i+1}: {v}" for i, v in out_of_range[:5]]

        return RuleResult(
            rule_id=self.rule_id, rule_name=self.rule_name,
            severity=self.severity, passed=passed, column=col,
            detail=f"范围 [{self.min_val}, {self.max_val}]，"
                   f"越界 {len(out_of_range)}/{total} 条",
            failed_count=len(out_of_range), total_count=total,
            sample_values=samples
        )

🔑3. 唯一性检查（Uniqueness Check）

class UniqueCheckRule(BaseRule):
    """检查列值是否唯一 — 主键/业务键必备规则"""

    def __init__(self, rule_id: str, column: str,
                 severity: Severity = Severity.CRITICAL):
        super().__init__(rule_id, f"{column} 唯一性检查", severity)
        self.column = column

    def check(self, data: list[dict], column: str = None) -> RuleResult:
        col = column or self.column
        total = len(data)
        values = [row.get(col) for row in data if row.get(col) is not None]

        # Counter 统计频率，找出重复值
        freq = Counter(values)
        duplicates = {v: c for v, c in freq.items() if c > 1}
        dup_count = sum(c for c in duplicates.values()) - len(duplicates)
        # 注意：dup_count 是"多余"的记录数，不是重复值个数

        passed = len(duplicates) == 0
        samples = [f"{v} 出现 {c} 次" for v, c in list(duplicates.items())[:5]]

        return RuleResult(
            rule_id=self.rule_id, rule_name=self.rule_name,
            severity=self.severity, passed=passed, column=col,
            detail=f"发现 {len(duplicates)} 个重复值，涉及 {dup_count} 条多余记录",
            failed_count=dup_count, total_count=total,
            sample_values=samples
        )

import re

class PatternCheckRule(BaseRule):
    """检查列值是否符合正则表达式 — 适用于身份证号、手机号、日期等"""

    def __init__(self, rule_id: str, column: str,
                 pattern: str, pattern_desc: str = "",
                 severity: Severity = Severity.ERROR):
        super().__init__(rule_id, f"{column} 格式检查", severity)
        self.column = column
        self.pattern = pattern        # 正则表达式
        self.pattern_desc = pattern_desc  # 人类可读的格式描述
        self._compiled = re.compile(pattern)

    def check(self, data: list[dict], column: str = None) -> RuleResult:
        col = column or self.column
        total = len(data)
        mismatch = []

        for row in data:
            val = row.get(col)
            if val is None or str(val).strip() == "":
                continue
            if not self._compiled.match(str(val).strip()):
                mismatch.append(str(val))

        passed = len(mismatch) == 0

        return RuleResult(
            rule_id=self.rule_id, rule_name=self.rule_name,
            severity=self.severity, passed=passed, column=col,
            detail=f"格式要求: {self.pattern_desc or self.pattern}，"
                   f"不匹配 {len(mismatch)}/{total} 条",
            failed_count=len(mismatch), total_count=total,
            sample_values=mismatch[:5]
        )

🔗5. 引用完整性检查（Referential Check）

class ReferentialCheckRule(BaseRule):
    """检查列值是否都在参照集合中 — 类似 SQL 的外键约束"""

    def __init__(self, rule_id: str, column: str,
                 ref_values: set | list,
                 severity: Severity = Severity.ERROR):
        super().__init__(rule_id, f"{column} 引用完整性", severity)
        self.column = column
        self.ref_values = set(ref_values)  # 转为 set，O(1) 查找

    def check(self, data: list[dict], column: str = None) -> RuleResult:
        col = column or self.column
        total = len(data)
        not_found = []

        for row in data:
            val = row.get(col)
            if val is not None and val not in self.ref_values:
                not_found.append(str(val))

        passed = len(not_found) == 0
        # 去重统计
        unique_missing = set(not_found)

        return RuleResult(
            rule_id=self.rule_id, rule_name=self.rule_name,
            severity=self.severity, passed=passed, column=col,
            detail=f"参照集大小 {len(self.ref_values)}，"
                   f"不在参照集中的值 {len(not_found)} 条（{len(unique_missing)} 个唯一值）",
            failed_count=len(not_found), total_count=total,
            sample_values=list(unique_missing)[:5]
        )

📈6. 统计突变检查（Statistical Drift Check）

class StatDriftCheckRule(BaseRule):
    """检查列的统计值是否与基线偏离过大 — 行数突变/均值漂移"""

    def __init__(self, rule_id: str, column: str,
                 baseline_mean: float = None,
                 baseline_std: float = None,
                 baseline_count: int = None,
                 mean_threshold: float = 3.0,    # 均值偏离N倍标准差
                 count_threshold: float = 0.5,    # 行数偏离基线50%即告警
                 severity: Severity = Severity.WARNING):
        super().__init__(rule_id, f"{column} 统计突变检查", severity)
        self.column = column
        self.baseline_mean = baseline_mean
        self.baseline_std = baseline_std
        self.baseline_count = baseline_count
        self.mean_threshold = mean_threshold
        self.count_threshold = count_threshold

    def check(self, data: list[dict], column: str = None) -> RuleResult:
        col = column or self.column
        total = len(data)
        issues = []

        # 行数突变检测
        if self.baseline_count and self.baseline_count > 0:
            count_change = abs(total - self.baseline_count) / self.baseline_count
            if count_change > self.count_threshold:
                issues.append(
                    f"行数 {total}，基线 {self.baseline_count}，"
                    f"偏离 {count_change:.1%}"
                )

        # 均值漂移检测
        if self.baseline_mean is not None and self.baseline_std is not None:
            values = [float(row[col]) for row in data if row.get(col) is not None]
            if values:
                current_mean = mean(values)
                if self.baseline_std > 0:
                    z_score = abs(current_mean - self.baseline_mean) / self.baseline_std
                    if z_score > self.mean_threshold:
                        issues.append(
                            f"均值 {current_mean:.2f}，基线 {self.baseline_mean:.2f}，"
                            f"Z-score={z_score:.1f}（阈值 {self.mean_threshold}）"
                        )

        passed = len(issues) == 0

        return RuleResult(
            rule_id=self.rule_id, rule_name=self.rule_name,
            severity=self.severity, passed=passed, column=col,
            detail="; ".join(issues) if issues else "统计指标正常",
            failed_count=0, total_count=total,
            sample_values=issues
        )

class BusinessRuleCheck(BaseRule):
    """通用业务规则检查 — 支持自定义判断函数"""

    def __init__(self, rule_id: str, rule_name: str,
                 check_fn,  # 判断函数: (row) -> (bool_is_ok, str_reason)
                 severity: Severity = Severity.ERROR):
        super().__init__(rule_id, rule_name, severity)
        self.check_fn = check_fn

    def check(self, data: list[dict], column: str = None) -> RuleResult:
        total = len(data)
        violations = []

        for i, row in enumerate(data):
            try:
                is_ok, reason = self.check_fn(row)
                if not is_ok:
                    violations.append(f"行{i+1}: {reason}")
            except Exception as e:
                violations.append(f"行{i+1}: 规则执行异常 - {e}")

        passed = len(violations) == 0

        return RuleResult(
            rule_id=self.rule_id, rule_name=self.rule_name,
            severity=self.severity, passed=passed, column="多列",
            detail=f"违反业务规则 {len(violations)}/{total} 条",
            failed_count=len(violations), total_count=total,
            sample_values=violations[:5]
        )

@dataclass
class QualityReport:
    """质检总报告"""
    table_name: str
    total_rows: int
    total_rules: int
    results: list[RuleResult] = field(default_factory=list)
    timestamp: str = ""

    @property
    def passed_rules(self) -> int:
        return sum(1 for r in self.results if r.passed)

    @property
    def failed_rules(self) -> int:
        return sum(1 for r in self.results if not r.passed)

    @property
    def critical_count(self) -> int:
        return sum(1 for r in self.results
                   if not r.passed and r.severity == Severity.CRITICAL)

    @property
    def overall_score(self) -> float:
        """总得分: 通过率 × 100"""
        return (self.passed_rules / self.total_rules * 100) if self.total_rules > 0 else 100.0

    @property
    def is_healthy(self) -> bool:
        """是否健康（无 CRITICAL 失败）"""
        return self.critical_count == 0

    def summary(self) -> str:
        """文本摘要"""
        status = "✅ 健康" if self.is_healthy else "❌ 需修复"
        lines = [
            f"📊 质检报告: {self.table_name} ({self.timestamp})",
            f"   数据量: {self.total_rows} 行 | 规则数: {self.total_rules}",
            f"   通过: {self.passed_rules} | 失败: {self.failed_rules} | CRITICAL: {self.critical_count}",
            f"   得分: {self.overall_score:.1f}/100 | 状态: {status}",
        ]
        for r in self.results:
            icon = "✅" if r.passed else {"WARNING": "⚠️", "ERROR": "❌", "CRITICAL": "🚨"}[r.severity.value]
            lines.append(f"   {icon} [{r.severity.value}] {r.rule_name}: {r.detail}")
        return "\n".join(lines)


class RuleEngine:
    """质检规则引擎 — 注册规则 → 执行 → 生成报告"""

    def __init__(self):
        self._rules: dict[str, BaseRule] = {}  # rule_id → rule

    def register(self, rule: BaseRule) -> "RuleEngine":
        """注册规则（支持链式调用）"""
        self._rules[rule.rule_id] = rule
        return self  # 返回 self，支持 engine.register(r1).register(r2)

    def unregister(self, rule_id: str) -> None:
        """移除规则"""
        self._rules.pop(rule_id, None)

    @property
    def rule_count(self) -> int:
        return len(self._rules)

    def run(self, data: list[dict], table_name: str = "unknown",
            column_override: dict = None) -> QualityReport:
        """
        执行所有规则并生成报告
        :param data: 数据行列表
        :param table_name: 表名（用于报告）
        :param column_override: 列名映射 {"原规则column": "实际column"}
        :return: QualityReport
        """
        from datetime import datetime
        column_override = column_override or {}
        results = []

        for rule_id, rule in self._rules.items():
            # 如果有列名映射，使用映射后的列名
            col = column_override.get(rule.column, rule.column) if hasattr(rule, 'column') else None
            try:
                result = rule.check(data, col)
                results.append(result)
            except Exception as e:
                # 规则执行异常，记录为失败
                results.append(RuleResult(
                    rule_id=rule_id, rule_name=rule.rule_name,
                    severity=Severity.CRITICAL, passed=False,
                    column=str(col), detail=f"规则执行异常: {e}"
                ))

        return QualityReport(
            table_name=table_name,
            total_rows=len(data),
            total_rules=len(results),
            results=results,
            timestamp=datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        )

    def run_multi_table(self, tables: dict[str, list[dict]]) -> list[QualityReport]:
        """批量质检多张表"""
        reports = []
        for table_name, data in tables.items():
            report = self.run(data, table_name)
            reports.append(report)
        return reports

练习 3：LeetCode #692 — 前K个高频单词（用 Counter + heapq）

给定单词列表和整数 k，返回前 k 个出现频率最高的单词。频率相同时按字典序升序。

# 提示：Counter.most_common(k) 返回 (word, count) 列表
# 但频率相同时需要自定义排序 key：(-count, word)
from collections import Counter

def top_k_frequent(words: list[str], k: int) -> list[str]:
    counts = Counter(words)
    # 排序：频率降序 → 字典序升序
    sorted_words = sorted(counts.keys(),
                          key=lambda w: (-counts[w], w))
    return sorted_words[:k]

# 测试

    ["i", "love", "leetcode", "i", "love", "coding"], 2
))  # ["i", "love"]
