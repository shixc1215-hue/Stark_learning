2.1 APScheduler 定时任务框架

Python 最主流的定时任务库，支持三种触发器：

# pip install apscheduler
from apscheduler.schedulers.blocking import BlockingScheduler
from apscheduler.schedulers.background import BackgroundScheduler
from apscheduler.triggers.cron import CronTrigger
from apscheduler.triggers.interval import IntervalTrigger
from apscheduler.triggers.date import DateTrigger
from datetime import datetime

# ---- 创建调度器（后台模式，不阻塞主线程）----
scheduler = BackgroundScheduler()

# ---- 三种触发器 ----

# 1) interval：固定间隔执行（类似 every 30 minutes）
scheduler.add_job(
    generate_daily_report,
    IntervalTrigger(hours=1),          # 每小时
    id="hourly_report"
)

# 2) cron：类 Cron 表达式（类似 Airflow 的 schedule）
# 格式：分 时 日 月 星期
scheduler.add_job(
    generate_daily_report,
    CronTrigger(hour=8, minute=0),     # 每天早8点
    id="daily_report_8am"
)

# 3) date：一次性定时执行
scheduler.add_job(
    send_summary,
    DateTrigger(run_date=datetime(2026, 8, 8, 18, 0)),  # 指定时间
    id="one_time_summary"
)

# ---- 启动调度器 ----
scheduler.start()
print("调度器已启动，任务数:", len(scheduler.get_jobs()))
💡 BlockingScheduler vs BackgroundScheduler：
BlockingScheduler 会阻塞当前线程（适合独立脚本）；BackgroundScheduler 在后台运行（适合嵌入 FastAPI 等服务）。生产环境推荐后者。

2.3 邮件发送 — smtplib

import smtplib
from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart
from email.mime.base import MIMEBase
from email import encoders
from pathlib import Path

def send_email(
    to: str,
    subject: str,
    html_body: str,
    attachments: list[Path] | None = None,
    smtp_host: str = "smtp.qq.com",
    smtp_port: int = 465,
    username: str = "your@qq.com",
    password: str = "xxx"  # QQ邮箱用授权码，不是密码
) -> bool:
    """发送 HTML 邮件，支持附件"""
    try:
        msg = MIMEMultipart()
        msg["From"] = f"AMC报表系统 <{username}>"
        msg["To"] = to
        msg["Subject"] = subject

        # HTML 正文
        msg.attach(MIMEText(html_body, "html", "utf-8"))

        # 附件（报表 PDF/HTML）
        for file_path in (attachments or []):
            with open(file_path, "rb") as f:
                part = MIMEBase("application", "octet-stream")
                part.set_payload(f.read())
                encoders.encode_base64(part)
                part.add_header(
                    "Content-Disposition",
                    f"attachment; filename={file_path.name}"
                )
                msg.attach(part)

        # 发送
        with smtplib.SMTP_SSL(smtp_host, smtp_port) as server:
            server.login(username, password)
            server.sendmail(username, to, msg.as_string())

        print(f"邮件发送成功 → {to}")
        return True

    except Exception as e:
        print(f"邮件发送失败: {e}")
        return False
⚠️ QQ/网易邮箱注意：必须先开启 SMTP 服务并获取授权码，不能用登录密码。设置路径：QQ邮箱 → 设置 → 账户 → POP3/SMTP 服务 → 开启 → 生成授权码。

2.4 Webhook 通知 — 飞书/钉钉/企业微信

import requests
import json

def send_webhook(
    url: str,
    title: str,
    content: str,
    msg_type: str = "interactive"  # "text" | "interactive"
) -> bool:
    """
    发送 Webhook 通知
    兼容飞书/钉钉/企业微信的通用 Webhook 格式
    """
    try:
        if msg_type == "text":
            payload = {
                "msg_type": "text",
                "content": {"text": f"{title}\n{content}"}
            }
        else:
            # 飞书 interactive 卡片格式
            payload = {
                "msg_type": "interactive",
                "card": {
                    "header": {
                        "title": {"tag": "plain_text", "content": title},
                        "template": "blue"
                    },
                    "elements": [
                        {"tag": "markdown", "content": content},
                        {"tag": "hr"},
                        {"tag": "note", "elements": [
                            {"tag": "plain_text", "content": "AMC 报表系统自动生成"}
                        ]}
                    ]
                }
            }

        resp = requests.post(url, json=payload, timeout=10)
        result = resp.json()

        if result.get("code", 0) == 0 or resp.status_code == 200:
            print(f"Webhook 通知成功 → {title}")
            return True
        else:
            print(f"Webhook 失败: {result}")
            return False

    except Exception as e:
        print(f"Webhook 异常: {e}")
        return False

2.5 报表文件存储与归档

import os
from pathlib import Path
from datetime import datetime
import shutil

class ReportArchiver:
    """报表文件归档管理器"""

    def __init__(self, base_dir: str = "reports"):
        self.base_dir = Path(base_dir)
        # 按日期归档: reports/2026-08/07/
        self.today_dir = self.base_dir / datetime.now().strftime("%Y-%m/%d")
        self.today_dir.mkdir(parents=True, exist_ok=True)

    def save(self, filename: str, content: str) -> Path:
        """保存报表文件，自动带时间戳"""
        timestamp = datetime.now().strftime("%H%M%S")
        safe_name = f"{timestamp}_{filename}"
        path = self.today_dir / safe_name
        path.write_text(content, encoding="utf-8")
        print(f"报表已保存: {path}")
        return path

    def save_binary(self, filename: str, data: bytes) -> Path:
        """保存二进制文件（如 PDF 图片）"""
        timestamp = datetime.now().strftime("%H%M%S")
        safe_name = f"{timestamp}_{filename}"
        path = self.today_dir / safe_name
        path.write_bytes(data)
        return path

    def cleanup_old(self, keep_days: int = 30):
        """清理超过 N 天的旧报表"""
        from datetime import timedelta
        cutoff = datetime.now() - timedelta(days=keep_days)

        for dir_path in self.base_dir.rglob("*"):
            if dir_path.is_dir():
                try:
                    # 解析日期目录名
                    date_str = dir_path.name
                    dir_date = datetime.strptime(date_str, "%Y-%m-%d")
                    if dir_date < cutoff:
                        shutil.rmtree(dir_path)
                        print(f"已清理旧目录: {dir_path}")
                except (ValueError, OSError):
                    pass  # 跳过无法解析的目录

    def get_latest(self, limit: int = 5) -> list[Path]:
        """获取最近的 N 份报表"""
        all_reports = sorted(
            self.base_dir.rglob("*.html"),
            key=lambda p: p.stat().st_mtime,
            reverse=True
        )
        return all_reports[:limit]

    def disk_usage(self) -> dict:
        """统计报表占用空间"""
        total_size = sum(
            f.stat().st_size for f in self.base_dir.rglob("*") if f.is_file()
        )
        return {
            "total_files": sum(1 for _ in self.base_dir.rglob("*") if _.is_file()),
            "total_size_mb": round(total_size / 1024 / 1024, 2),
            "base_dir": str(self.base_dir)
        }

3.1 通知内容智能生成

不同渠道需要不同格式：邮件要正式详细，Webhook 要简洁有力。

from dataclasses import dataclass, field
from typing import Literal

@dataclass
class NotificationContent:
    """通知内容（LLM 生成 + 模板组装）"""
    subject: str           # 邮件标题
    email_body: str         # 邮件 HTML 正文
    webhook_title: str      # 飞书卡片标题
    webhook_content: str    # 飞书卡片内容（Markdown）
    priority: Literal["high", "medium", "low"] = "medium"
    recipients: list[str] = field(default_factory=list)


class NotificationFormatter:
    """基于 LLM 的通知内容格式化器"""

    EMAIL_PROMPT = """你是AMC（资产管理公司）报表通知助手。根据报表摘要，生成通知内容。

## 报表摘要
{summary}

## 要求
1. 生成邮件标题（简洁，20字以内）
2. 生成邮件HTML正文（格式美观，关键数据用加粗，包含3-5个要点）
3. 生成Webhook卡片内容（Markdown格式，控制在200字以内，适合飞书展示）
4. 判断优先级：涉及重大风险→high，常规数据→medium，正常无异常→low

请严格按以下JSON格式输出：
{{
  "subject": "邮件标题",
  "email_body": "<邮件HTML正文>",
  "webhook_title": "卡片标题",
  "webhook_content": "Markdown内容",
  "priority": "high/medium/low"
}}"""

    def __init__(self, api_key: str):
        self.api_key = api_key

    def format(self, report_summary: str) -> NotificationContent:
        """调用 LLM 生成多格式通知内容"""
        from zhipuai import ZhipuAI
        client = ZhipuAI(api_key=self.api_key)

        resp = client.chat.completions.create(
            model="glm-4-flash",
            messages=[
                {"role": "system", "content": "你是通知格式化助手，只输出JSON。"},
                {"role": "user", "content": self.EMAIL_PROMPT.format(
                    summary=report_summary
                )}
            ],
            temperature=0.3,  # 低随机性，确保格式稳定
        )

        import json
        try:
            data = json.loads(resp.choices[0].message.content)
            return NotificationContent(
                subject=data["subject"],
                email_body=data["email_body"],
                webhook_title=data["webhook_title"],
                webhook_content=data["webhook_content"],
                priority=data.get("priority", "medium")
            )
        except (json.JSONDecodeError, KeyError) as e:
            # Fallback：LLM 输出异常时用模板兜底
            return self._fallback_format(report_summary)

    def _fallback_format(self, summary: str) -> NotificationContent:
        """降级：用固定模板生成通知"""
        return NotificationContent(
            subject="[AMC] 每日数据报表已生成",
            email_body=f"<h2>AMC 日报</h2><p>{summary}</p>",
            webhook_title="📊 AMC 日报",
            webhook_content=summary[:200],
            priority="medium"
        )

3.2 多渠道路由策略

from enum import Enum

class ChannelType(Enum):
    EMAIL = "email"
    WEBHOOK = "webhook"
    CONSOLE = "console"

class NotificationRouter:
    """通知渠道路由器 — 根据优先级选择渠道"""

    # 路由规则：优先级 → 渠道列表
    ROUTING_RULES = {
        "high":   [ChannelType.EMAIL, ChannelType.WEBHOOK, ChannelType.CONSOLE],
        "medium": [ChannelType.WEBHOOK, ChannelType.CONSOLE],
        "low":    [ChannelType.CONSOLE],
    }

    def __init__(self):
        self.channels = {
            ChannelType.EMAIL: {
                "enabled": True,
                "recipients": ["team@amc.com"],
                "webhook_url": None
            },
            ChannelType.WEBHOOK: {
                "enabled": True,
                "webhook_url": "https://open.feishu.cn/open-apis/bot/v2/hook/xxx"
            },
            ChannelType.CONSOLE: {
                "enabled": True
            }
        }

    def route(self, notification: NotificationContent) -> list[ChannelType]:
        """根据优先级决定发送渠道"""
        rules = self.ROUTING_RULES.get(notification.priority, ["console"])
        active = [ch for ch in rules if self.channels[ch]["enabled"]]
        print(f"📋 优先级={notification.priority}, "
              f"路由渠道={[ch.value for ch in active]}")
        return active

    def send(self, notification: NotificationContent) -> dict:
        """执行多渠道分发"""
        channels = self.route(notification)
        results = {}

        for ch in channels:
            if ch == ChannelType.EMAIL:
                results["email"] = send_email(
                    to="team@amc.com",
                    subject=notification.subject,
                    html_body=notification.email_body
                )
            elif ch == ChannelType.WEBHOOK:
                url = self.channels[ch]["webhook_url"]
                results["webhook"] = send_webhook(
                    url=url,
                    title=notification.webhook_title,
                    content=notification.webhook_content
                )
            elif ch == ChannelType.CONSOLE:
                print(f"🔔 [{notification.priority.upper()}] "
                      f"{notification.webhook_title}")
                print(f"   {notification.webhook_content[:100]}...")
                results["console"] = True

        return results

# scheduling/scheduler.py
from apscheduler.schedulers.background import BackgroundScheduler
from apscheduler.triggers.cron import CronTrigger
from apscheduler.events import EVENT_JOB_EXECUTED, EVENT_JOB_ERROR
from datetime import datetime
import logging

logger = logging.getLogger("scheduler")

class ReportScheduler:
    """AMC 报表调度管理器"""

    # ---- 预定义任务配置 ----
    JOBS = {
        "daily_report": {
            "cron": "0 8 * * 1-5",      # 工作日早8点
            "description": "AMC每日经营日报",
            "priority": "medium"
        },
        "weekly_risk": {
            "cron": "0 8 * * 0",          # 每周日早8点
            "description": "周度风险分析报告",
            "priority": "medium"
        },
        "monthly_summary": {
            "cron": "0 9 1 * *",          # 每月1号早9点
            "description": "月度资产汇总报告",
            "priority": "high"
        },
        "risk_alert": {
            "interval_minutes": 30,       # 每30分钟
            "description": "实时风险预警检查",
            "priority": "high"
        }
    }

    def __init__(self):
        self.scheduler = BackgroundScheduler(
            timezone="Asia/Shanghai",      # 明确时区！
            job_defaults={
                "coalesce": True,          # 合并错过的执行
                "max_instances": 1,         # 同一任务不并发
                "misfire_grace_time": 3600  # 允许1小时内补执行
            }
        )
        self._setup_event_listeners()

    def _setup_event_listeners(self):
        """监听任务执行事件，记录日志"""
        self.scheduler.add_listener(
            self._on_job_success,
            EVENT_JOB_EXECUTED
        )
        self.scheduler.add_listener(
            self._on_job_error,
            EVENT_JOB_ERROR
        )

    def _on_job_success(self, event):
        logger.info(f"✅ 任务完成: {event.job_id}")

    def _on_job_error(self, event):
        logger.error(f"❌ 任务失败: {event.job_id}, "
                     f"异常: {event.exception}")

    def register_jobs(self, job_func):
        """注册所有预定义任务"""
        for job_id, config in self.JOBS.items():
            if "cron" in config:
                # 解析 cron 表达式
                parts = config["cron"].split()
                trigger = CronTrigger(
                    minute=parts[0],
                    hour=parts[1],
                    day=parts[2],
                    month=parts[3],
                    day_of_week=parts[4]
                )
            else:
                trigger = IntervalTrigger(
                    minutes=config.get("interval_minutes", 60)
                )

            self.scheduler.add_job(
                job_func,
                trigger=trigger,
                id=job_id,
                name=config["description"],
                kwargs={"job_id": job_id, "priority": config["priority"]}
            )
            logger.info(f"📋 已注册: {job_id} ({config['description']})")

    def start(self):
        """启动调度器"""
        self.scheduler.start()
        logger.info("🚀 调度器已启动")

    def stop(self):
        """优雅停止"""
        self.scheduler.shutdown(wait=True)
        logger.info("🛑 调度器已停止")

    def status(self) -> dict:
        """获取所有任务状态"""
        jobs = self.scheduler.get_jobs()
        return {
            "running": self.scheduler.running,
            "job_count": len(jobs),
            "jobs": [
                {
                    "id": j.id,
                    "name": j.name,
                    "next_run": str(j.next_run_time) if j.next_run_time else None,
                    "trigger": str(j.trigger)
                }
                for j in jobs
            ]
        }

4.3 任务执行函数 — 完整链路

# scheduling/jobs.py
import logging
from datetime import datetime
from rendering.builder import ReportBuilder  # D57 的报表构建器
from nl2sql.engine import NL2SQLEngine        # D56 的 NL2SQL 引擎
from scheduling.notifier import NotificationRouter, NotificationFormatter
from scheduling.archiver import ReportArchiver

logger = logging.getLogger("jobs")

async def execute_report_job(
    job_id: str,
    priority: str,
    natural_query: str = "生成今日AMC经营日报"
):
    """
    完整的报表生成 + 分发 任务
    一次调度的完整执行流程：
    1. NL2SQL 查询数据
    2. LLM 生成报表
    3. 渲染 HTML + 图表
    4. 归档文件
    5. 格式化通知
    6. 多渠道分发
    """
    start_time = datetime.now()
    job_logger = lambda msg: logger.info(f"[{job_id}] {msg}")
    job_logger(f"========== 开始执行 ==========")

    try:
        # Step 1: NL2SQL 获取数据
        job_logger("Step 1/5: NL2SQL 查询数据...")
        engine = NL2SQLEngine()
        sql_result = engine.query(natural_query)

        # Step 2: LLM 生成报表分析
        job_logger("Step 2/5: LLM 生成报表分析...")
        builder = ReportBuilder()
        html_report = await builder.build_report(
            query=natural_query,
            data=sql_result,
            title=f"AMC {job_id}"
        )

        # Step 3: 归档报表
        job_logger("Step 3/5: 归档报表文件...")
        archiver = ReportArchiver()
        report_path = archiver.save(f"{job_id}.html", html_report)

        # Step 4: LLM 格式化通知内容
        job_logger("Step 4/5: 生成通知内容...")
        formatter = NotificationFormatter(api_key="your-key")
        notification = formatter.format(
            report_summary=sql_result["summary"]
        )

        # Step 5: 多渠道分发
        job_logger("Step 5/5: 分发通知...")
        router = NotificationRouter()
        send_results = router.send(notification)

        # 汇总结果
        elapsed = (datetime.now() - start_time).total_seconds()
        job_logger(f"========== 执行完成 ({elapsed:.1f}s) ==========")

        return {
            "job_id": job_id,
            "status": "success",
            "report_path": str(report_path),
            "notification": send_results,
            "elapsed_seconds": elapsed
        }

    except Exception as e:
        logger.error(f"[{job_id}] 任务失败: {e}", exc_info=True)
        # 失败时也要通知（高优先级）
        if priority == "high":
            router = NotificationRouter()
            router.send(NotificationContent(
                subject=f"[告警] {job_id} 执行失败",
                email_body=f"<p>任务 {job_id} 失败: {e}</p>",
                webhook_title="⚠️ 报表生成失败",
                webhook_content=f"任务 {job_id} 异常: {str(e)[:100]}",
                priority="high"
            ))
        return {
            "job_id": job_id,
            "status": "failed",
            "error": str(e)
        }

    # 1. 创建调度器
    scheduler = ReportScheduler()

    # 2. 注册任务（async 函数需要包装）
    def sync_job_wrapper(job_id: str, priority: str):
        """将 async 任务包装为同步调用"""
        asyncio.run(execute_report_job(
            job_id=job_id, priority=priority
        ))

    scheduler.register_jobs(sync_job_wrapper)

    # 3. 优雅退出信号处理
    def shutdown(signum, frame):
        print("\n🛑 收到停止信号，正在关闭调度器...")
        scheduler.stop()
        sys.exit(0)

    signal.signal(signal.SIGINT, shutdown)   # Ctrl+C
    signal.signal(signal.SIGTERM, shutdown)   # kill

    # 4. 启动
    scheduler.start()
    print("\n" + "="*50)
    print("  AMC 报表调度系统已启动")
    print("  按 Ctrl+C 停止")
    print("="*50 + "\n")

    # 5. 查看任务状态
    import json
    print(json.dumps(scheduler.status(), indent=2, ensure_ascii=False))

    # 6. 阻塞主线程，保持进程运行
    try:
        while True:
            import time
            time.sleep(3600)  # 每小时检查一次
    except KeyboardInterrupt:
        scheduler.stop()

if __name__ == "__main__":
    main()
💡 生产部署建议：独立调度器适合内网环境。如果是 Web 服务，推荐把 BackgroundScheduler 嵌入 FastAPI 的 lifespan 中，随服务自动启停，避免多进程管理。

🚨 常见陷阱 Top 5 5 min
❌ 陷阱 1：时区问题
APScheduler 默认时区取决于系统。生产环境必须显式指定 timezone="Asia/Shanghai"，否则 Cron 触发时间会差8小时。
❌ 陷阱 2：多进程重复执行
gunicorn/uvicorn 启动多个 worker 时，每个 worker 都会创建 BackgroundScheduler，导致任务重复执行。解决方案：只用 1 个 worker，或用 uwsgi.mule 隔离。
❌ 陷阱 3：邮件授权码 ≠ 登录密码
QQ邮箱/网易邮箱的 SMTP 密码是授权码，不是登录密码。且授权码不要硬编码在代码中，用环境变量或 .env 管理。
⚠️ 陷阱 4：任务超时无处理
LLM API 调用可能超时或无响应。每个任务应设置 timeout，并在 EVENT_JOB_ERROR 中处理，避免一个任务卡死影响后续调度。
⚠️ 陷阱 5：报表堆积不清理
每日生成报表会逐渐占满磁盘。ReportArchiver.cleanup_old() 应在每次调度后自动执行，或单独设一个清理任务。
