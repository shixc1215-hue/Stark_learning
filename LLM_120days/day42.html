🐍 Python 知识补充：typing.Protocol 与结构化子类型 15 分钟

今天是综合实战日，我们学一个能极大提升代码架构质量的特性——typing.Protocol。它比抽象基类（ABC）更灵活，是实现"面向接口编程"的利器。

Protocol vs ABC：什么时候用什么？

# ============ ABC（抽象基类）—— 需要显式继承 ============
from abc import ABC, abstractmethod

class Retriever(ABC):
    @abstractmethod
    def retrieve(self, query: str, top_k: int = 5) -> list:
        """检索相关文档"""

# 子类必须显式继承
class VectorRetriever(Retriever):  # ← 必须写 (Retriever)
    def retrieve(self, query: str, top_k: int = 5) -> list:
        return []  # 实际实现

# ============ Protocol —— 隐式满足（结构化子类型）============
from typing import Protocol

class Retriever(Protocol):
    def retrieve(self, query: str, top_k: int = 5) -> list: ...

# 任何有 retrieve(query, top_k) 方法的类自动满足协议！
class ChromaRetriever:  # ← 不需要继承！
    def retrieve(self, query: str, top_k: int = 5) -> list:
        return []  # 自动被视为 Retriever

# 类型检查器知道 ChromaRetriever 满足 Retriever 协议
def search(retriever: Retriever, q: str) -> list:
    return retriever.retrieve(q)
核心区别：ABC 是"名义子类型"（必须显式声明继承），Protocol 是"结构化子类型"（只要有相同的方法签名就满足）。在我们的 AMC 系统中，用 Protocol 定义 Embedder、Retriever、Reranker、Generator 四大协议，任何实现都自动兼容。

runtime_checkable：让 Protocol 可以 isinstance 检查

from typing import Protocol, runtime_checkable

@runtime_checkable
class Tool(Protocol):
    def execute(self, **kwargs) -> str: ...

# 现在可以用 isinstance 做运行时检查
class SQLQueryTool:
    def execute(self, **kwargs) -> str:
        return "SELECT * FROM amc_report"

tool = SQLQueryTool()
assert isinstance(tool, Tool)  # ✓ 通过

实战建议：在 AMC 系统中，我们为每个核心组件定义 Protocol（Embedder / Retriever / Reranker / Generator / Tool），这样后续换模型（比如从智谱换成 DeepSeek）只需实现新类，不需要改任何上层代码。这就是 SOLID 中的"依赖倒置原则"。

chunker.py — 递归文本分块

# indexer/chunker.py
import re
from models import Document, Chunk

class RecursiveChunker:
    """
    递归分块器 —— 按层级分隔符切分文本
    类比数仓: 按 partition → sub-partition 逐级细分
    """

    # 分隔符优先级：先按大标题分，再按小标题，最后按段落
    SEPARATORS = [
        r"\n(?=#{1,3}\s)",   # Markdown 标题 (#/##/###)
        r"\n(?=\d+\.)",       # 编号列表 (1. 2. 3.)
        r"\n{2,}",             # 段落（连续换行）
        r"\n",                 # 单行换行
        r"[。！？；]",         # 中文句末标点
    ]

    def __init__(self, chunk_size: int = 512, overlap: int = 50):
        self.chunk_size = chunk_size
        self.overlap = overlap

    def split_text(self, text: str,
                   separators: list[str] | None = None) -> list[str]:
        """递归分块：用当前分隔符切，太大就换更细的分隔符"""
        if separators is None:
            separators = self.SEPARATORS

        # 基础情况：文本足够小，直接返回
        if len(text) <= self.chunk_size:
            return [text] if text.strip() else []

        # 尝试当前最粗的分隔符
        sep = separators[0]
        parts = re.split(sep, text)
        parts = [p.strip() for p in parts if p.strip()]

        # 如果所有块都 <= chunk_size，成功
        if all(len(p) <= self.chunk_size for p in parts):
            return parts

        # 有大块？递归用更细的分隔符
        result = []
        for part in parts:
            if len(part) <= self.chunk_size:
                result.append(part)
            elif len(separators) > 1:
                result.extend(
                    self.split_text(part, separators[1:])
                )
            else:
                # 最细分隔符还是太大，硬切
                result.extend(self._hard_split(part))
        return result

    def _hard_split(self, text: str) -> list[str]:
        """硬切 + overlap 保证不丢上下文"""
        chunks = []
        start = 0
        while start < len(text):
            end = start + self.chunk_size
            chunks.append(text[start:end])
            start = end - self.overlap  # 重叠部分
        return chunks

    def chunk_document(self, doc: Document) -> list[Chunk]:
        """完整文档分块，保留来源信息"""
        raw_chunks = self.split_text(doc.content)
        chunks = []
        for i, content in enumerate(raw_chunks):
            chunks.append(Chunk(
                chunk_id=f"{doc.doc_id}_{i:04d}",
                content=content,
                doc_id=doc.doc_id,
                chunk_index=i,
                metadata={**doc.metadata, "chunk_size": len(content)}
            ))
        return chunks

embedder.py — 智谱 Embedding 服务

# indexer/embedder.py
import hashlib
from functools import lru_cache

class ZhipuEmbedder:
    """
    智谱 Embedding 服务
    注意：它隐式满足 Embedder Protocol（有 embed + embed_batch）
    """

    def __init__(self, api_key: str, model: str = "embedding-3"):
        self.api_key = api_key
        self.model = model
        self._cache: dict[str, list[float]] = {}

    def _call_api(self, texts: list[str]) -> list[list[float]]:
        """调用智谱 Embedding API（实际用 requests）"""
        import requests
        url = "https://open.bigmodel.cn/api/paas/v4/embeddings"
        headers = {"Authorization": f"Bearer {self.api_key}"}
        body = {"model": self.model, "input": texts}
        resp = requests.post(url, headers=headers, json=body)
        resp.raise_for_status()
        # 按 index 排序后返回
        data = sorted(resp.json()["data"], key=lambda x: x["index"])
        return [item["embedding"] for item in data]

    def embed(self, text: str) -> list[float]:
        """单条文本 Embedding（带缓存）"""
        key = hashlib.md5(text.encode()).hexdigest()[:16]
        if key in self._cache:
            return self._cache[key]
        result = self.embed_batch([text])
        self._cache[key] = result[0]
        return result[0]

    def embed_batch(self, texts: list[str]) -> list[list[float]]:
        """批量 Embedding（自动分批，每批最多 20 条）"""
        all_embeddings = []
        batch_size = 20
        for i in range(0, len(texts), batch_size):
            batch = texts[i:i + batch_size]
            # 先查缓存
            uncached = []
            cache_indices = []
            for j, text in enumerate(batch):
                key = hashlib.md5(text.encode()).hexdigest()[:16]
                if key in self._cache:
                    cache_indices.append((j, self._cache[key]))
                else:
                    uncached.append(text)

            # 未缓存的调 API
            if uncached:
                new_embeddings = self._call_api(uncached)
                uncached_idx = 0
                for j, text in enumerate(batch):
                    key = hashlib.md5(text.encode()).hexdigest()[:16]
                    if key not in self._cache:
                        self._cache[key] = new_embeddings[uncached_idx]
                        uncached_idx += 1

            all_embeddings.extend([
                self._cache[hashlib.md5(t.encode()).hexdigest()[:16]]
                for t in batch
            ])
        return all_embeddings

# indexer/vectorstore.py
import chromadb

class AMCVectorStore:
    """AMC 知识库 —— 基于 ChromaDB 的向量存储"""

    def __init__(self, persist_dir: str, collection_name: str):
        self.client = chromadb.PersistentClient(path=persist_dir)
        self.collection = self.client.get_or_create_collection(
            name=collection_name,
            metadata={"hnsw:space": "cosine"}  # 余弦相似度
        )

    def add_chunks(self, chunks: list, embeddings: list):
        """批量写入分块 + 向量"""
        self.collection.upsert(
            ids=[c.chunk_id for c in chunks],
            embeddings=embeddings,
            documents=[c.content for c in chunks],
            metadatas=[{
                "doc_id": c.doc_id,
                "chunk_index": c.chunk_index,
                **c.metadata
            } for c in chunks]
        )

    def search(self, query_embedding: list, n_results: int = 10,
             where: dict | None = None) -> list:
        """语义搜索（支持元数据过滤）"""
        kwargs = {
            "query_embeddings": [query_embedding],
            "n_results": n_results,
        }
        if where:
            kwargs["where"] = where

        results = self.collection.query(**kwargs)
        return results

# agent/tools.py
from models import SearchResult
from typing import Callable

class ToolRegistry:
    """工具注册表 —— Agent 可调用的所有工具"""

    def __init__(self):
        self._tools: dict[str, dict] = {}

    def register(self, name: str, description: str,
                func: Callable, parameters: dict):
        self._tools[name] = {
            "description": description,
            "function": func,
            "parameters": parameters,
        }

    def get_tools_schema(self) -> list[dict]:
        """生成 Function Calling 的 tools 参数"""
        return [
            {
                "type": "function",
                "function": {
                    "name": name,
                    "description": info["description"],
                    "parameters": info["parameters"],
                }
            }
            for name, info in self._tools.items()
        ]

    def execute(self, name: str, **kwargs) -> str:
        if name not in self._tools:
            return f"错误: 未知工具 '{name}'"
        return self._tools[name]["function"](**kwargs)


# ===== 注册 AMC 专用工具 =====
def build_amc_tools(reranker, generator, retriever, embedder) -> ToolRegistry:
    """工厂函数：创建并注册所有工具"""
    registry = ToolRegistry()

    # 工具1: 知识库检索
    def knowledge_search(query: str, top_k: int = 5) -> str:
        """从 AMC 知识库检索相关内容"""
        emb = embedder.embed(query)
        raw_results = retriever.search(emb, n_results=top_k)
        # 转换为 SearchResult 再重排
        results = [SearchResult(
            chunk_id=r["id"],
            content=r["document"],
            score=r["distance"],
            source=r["metadata"].get("doc_id", "unknown")
        ) for r in raw_results]
        # 重排序
        reranked = reranker.rerank(query, results, top_k=top_k)
        # 格式化输出
        output_parts = []
        for i, r in enumerate(reranked):
            output_parts.append(
                f"[{i+1}] (分数:{r.score:.3f}) {r.content[:200]}"
            )
        return "\n".join(output_parts)

    registry.register(
        name="knowledge_search",
        description="从 AMC 知识库检索相关政策、规范、业务规则。"
                    "输入: 检索查询字符串。输出: 相关知识片段列表。",

            "type": "object",
            "properties": {
                "query": {"type": "string", "description": "检索问题"},
                "top_k": {"type": "integer", "description": "返回条数"},

            "required": ["query"]
        }
    )

    # 工具2: SQL 查询（模拟 AMC 数仓）

        return f"SQL 执行成功，返回空结果。SQL: {sql}"

    registry.register(
        name="sql_query",
        description="查询 AMC 数仓中的不良资产数据。"

def call_llm(messages: list[dict],
            tools: list[dict] | None = None) -> dict:
    """调用 LLM API（智谱 GLM）"""
    import requests
    from config import config
    url = "https://open.bigmodel.cn/api/paas/v4/chat/completions"
    headers = {"Authorization": f"Bearer {config.zhipu_api_key}"}
    body = {
        "model": config.zhipu_model,
        "messages": messages,
        "temperature": config.agent_temperature,
    }
    if tools:
        body["tools"] = tools
    resp = requests.post(url, headers=headers, json=body, timeout=60)
    resp.raise_for_status()
    return resp.json()["choices"][0]["message"]


class ReActAgent:
    """
    ReAct Agent —— 思考→行动→观察 循环
    整合 Day 36-40 所有 Agent 技巧:
    - ReAct 循环 (Day 36)
    - Function Calling (Day 37)
    - 记忆管理 (Day 38)
    - 错误恢复 (Day 39)
    - RAG+Agent 联合 (Day 40)
    - Prompt 约束 (Day 41)
    """

    SYSTEM_PROMPT = """你是AMC（资产管理公司）智能分析助手。你的职责：
1. 回答关于不良资产管理、监管报送、风险分析的问题
2. 查询数仓数据并给出分析
3. 生成风险分析报告

可用工具:
- knowledge_search: 检索 AMC 知识库（政策、规范、业务规则）
- sql_query: 查询 AMC 数仓数据
- generate_report: 生成风险分析报告

工作流程（严格遵循）:
Thought: 分析问题，决定下一步
Action: 调用一个工具
Observation: 查看工具结果
... 重复直到有足够信息 ...
Final Answer: 给出最终回答

注意:
- 优先使用 knowledge_search 查找知识
- 需要数据时使用 sql_query
- 需要报告时先收集信息再用 generate_report
- 回答要专业、准确、基于事实
- 不确定的数据要明确说明"""

    def __init__(self, tool_registry, max_steps: int = 8):
        self.tools = tool_registry
        self.max_steps = max_steps
        self.trace: list[AgentStep] = []
        self.conversation: list[dict] = []

    def run(self, question: str) -> QueryResult:
        """运行 Agent —— ReAct + FC 混合模式"""
        self.trace = []
        self.conversation = [
            {"role": "system", "content": self.SYSTEM_PROMPT},
            {"role": "user", "content": question},
        ]
        tools_schema = self.tools.get_tools_schema()

        for step in range(self.max_steps):
            # 第一步和后续都用 FC 模式
            msg = call_llm(self.conversation, tools=tools_schema)

            # 如果没有工具调用 → 生成最终回答
            if not msg.get("tool_calls"):
                agent_step = AgentStep(
                    step_num=step,
                    status=AgentStatus.FINISHED,
                    thought="Agent 给出最终回答",
                )
                self.trace.append(agent_step)
                return QueryResult(
                    answer=msg["content"],
                    sources=[],
                    agent_trace=self.trace
                )

            # 处理工具调用
            self.conversation.append(msg)
            for tc in msg["tool_calls"]:
                tool_name = tc["function"]["name"]
                try:
                    args = json.loads(tc["function"]["arguments"])
                except json.JSONDecodeError:
                    args = {}

                # 执行工具
                try:
                    result = self.tools.execute(tool_name, **args)
                    status = AgentStatus.TOOL_CALL
                except Exception as e:
                    result = f"工具执行失败: {e}"
                    status = AgentStatus.TOOL_CALL

                # 记录步骤
                agent_step = AgentStep(
                    step_num=step, status=status,
                    thought=msg["content"][:200],
                    tool_name=tool_name,
                    tool_input=args,
                    tool_output=result[:500],
                )
                self.trace.append(agent_step)

                # 回传工具结果给 LLM
                self.conversation.append({
                    "role": "tool",
                    "tool_call_id": tc["id"],
                    "content": result,
                })

        # 超过最大步数
        return QueryResult(
            answer="抱歉，分析过程过于复杂，请简化问题后重试。",
            sources=[], agent_trace=self.trace
        )

    def show_trace(self) -> str:
        """格式化输出 Agent 执行轨迹"""
        lines = ["=== Agent 执行轨迹 ==="]
        for step in self.trace:
            icon = {AgentStatus.THINKING: "💭",
                   AgentStatus.TOOL_CALL: "🔧",
                   AgentStatus.GENERATING: "✍️",
                   AgentStatus.FINISHED: "✅"}.get(step.status, "?")
            lines.append(f"\n{icon} Step {step.step_num}: [{step.status.value}] ")
            if step.tool_name:
                lines.append(f"   工具: {step.tool_name}")
                lines.append(f"   输入: {step.tool_input}")
                lines.append(f"   输出: {step.tool_output[:100]}...")
        return "\n".join(lines)

# ========== Phase 1: 离线索引 ==========
def build_index(documents: list[Document]):
    """
    离线索引管线：文档 → 分块 → Embedding → 向量库
    类比数仓: ODS → DWD → DWS → ADS
    """
    print("📦 [1/4] 文本分块...")
    chunker = RecursiveChunker(
        chunk_size=config.chunk_size,
        overlap=config.chunk_overlap
    )
    all_chunks = []
    for doc in documents:
        chunks = chunker.chunk_document(doc)
        all_chunks.extend(chunks)
        print(f"   {doc.doc_id}: {len(chunks)} 个分块")
    print(f"   总计: {len(all_chunks)} 个分块")

    print("\n🧮 [2/4] 生成 Embedding...")
    embedder = ZhipuEmbedder(config.zhipu_api_key, config.embedding_model)
    texts = [c.content for c in all_chunks]
    embeddings = embedder.embed_batch(texts)
    print(f"   完成: {len(embeddings)} 个向量")

    print("\n💾 [3/4] 写入向量库...")
    store = AMCVectorStore(config.chroma_dir, config.collection_name)
    store.add_chunks(all_chunks, embeddings)
    print(f"   完成: {store.collection.count()} 条记录")

    print("\n✅ [4/4] 索引构建完成！")
    return chunker, embedder, store

def start_agent(embedder, store):
    """启动 Agent 问答系统"""

    from query.reranker import SimpleReranker
    from query.retriever import ChromaRetriever
    from query.generator import PromptGenerator

    reranker = SimpleReranker()
    generator = PromptGenerator()

    # 构建 Agent（注入所有工具）

        reranker=reranker, generator=generator,
        retriever=retriever, embedder=embedder
    )
    agent = ReActAgent(tool_registry, max_steps=config.agent_max_steps)

    return agent

⚠️ 综合项目常见陷阱 Top 5 5 分钟
陷阱 1：全量加载 vs 懒加载
离线索引时不要一次性把所有文档 Embedding。用生成器逐文档处理，内存占用从 O(N) 降到 O(1)。分块时同理——split_text 返回 list，但 chunk_document 可以改成生成器。
陷阱 2：Agent 死循环
Agent 可能在两个工具之间反复调用。解决方案：max_steps 硬限制 + 重复检测（如果连续 3 次调用同一工具，自动终止）+ Token 预算控制。
陷阱 3：工具描述太简短
FC 模式下，LLM 完全依赖工具描述来决定调用哪个工具。描述要包含：功能说明、适用场景、输入输出格式、注意事项。至少 3-5 句话。
陷阱 4：向量库和模型不匹配
如果 Embedding 模型更换（比如从 embedding-3 换成 bge-large），所有已存储的向量需要重新生成。解决方案：在向量库 metadata 中记录模型版本，启动时检查。
陷阱 5：缺少评估闭环
很多 RAG 系统上线后就不再评估。实际项目要定期跑评估（Day 35 学的），监控 P@K、MRR、Faithfulness 等指标的趋势。

def calculate_ratio(numerator: float, denominator: float,
                    ratio_name: str) -> str:
    """计算业务比率"""
    if denominator == 0:
        return f"{ratio_name}: 无法计算（分母为0）"
    ratio = numerator / denominator * 100
    return f"{ratio_name}: {ratio:.2f}% (分子={numerator}, 分母={denominator})"

练习 3：LeetCode #239 — 滑动窗口最大值

用 collections.deque（Day 38 学过的）实现 O(n) 解法。这道题是单调队列的经典应用，也是理解 Agent 中滑动窗口记忆机制的底层基础。

from collections import deque

def maxSlidingWindow(nums: list[int], k: int) -> list[int]:
    """单调递减队列：队首始终是当前窗口最大值"""
    result = []
    dq = deque()  # 存索引，值单调递减

    for i, num in enumerate(nums):
        # 移除超出窗口的元素
        while dq and dq[0] <= i - k:
            dq.popleft()

        # 移除比当前元素小的（它们不可能是最大值）
        while dq and nums[dq[-1]] < num:
            dq.pop()

        dq.append(i)

        # 窗口形成后，记录最大值
        if i >= k - 1:
            result.append(nums[dq[0]])

    return result

# 测试
