🧠 模型量化：从 32GB 到 4GB 的魔法 ~15 min

前面 Day 43-44 我们学会了微调模型，但训练出来的模型动辄几十GB，怎么在实际环境中跑起来？这就是今天的主角——模型量化。

量化（Quantization）：将模型权重从高精度（FP32 / FP16）压缩到低精度（INT8 / INT4），类似把一张高清图压缩成标清——体积缩小数倍，肉眼几乎看不出区别。
为什么需要量化？

以一个 7B 参数的模型为例（7B = 70亿个参数）：

精度每个参数7B 模型大小需要显存质量损失
FP32（float32）4 字节~28 GB专业 GPU无（基准）
FP16（半精度）2 字节~14 GB高端 GPU极小
INT81 字节~7 GB中端 GPU轻微
INT40.5 字节~3.5 GB消费级 GPU / CPU可控
数仓类比：量化就像数据仓库中的列式压缩——把 DECIMAL(20,6) 压成 INT，查询精度够用但存储降 80%。
量化核心算法

1. 对称量化（Symmetric Quantization）

# 对称量化：将浮点数映射到 [-127, 127] 的整数范围
import numpy as np

def symmetric_quantize(weights, bits=8):
    # 计算量化范围：最大绝对值
    scale = np.max(np.abs(weights))
    # 量化：乘以缩放因子，四舍五入到整数
    q_max = 2 ** (bits - 1) - 1  # 8bit → 127
    quantized = np.round(weights / scale * q_max).astype(np.int8)
    return quantized, scale

def symmetric_dequantize(quantized, scale):
    # 反量化：还原回浮点数
    return quantized.astype(np.float32) * scale / 127.0

# 演示
weights = np.array([0.3, -0.8, 1.5, -0.1])
q, s = symmetric_quantize(weights)
print(f"原始:    {weights}")
print(f"量化后: {q}")
print(f"反量化: {symmetric_dequantize(q, s)}")
# 输出：原始和反量化有微小差异，这就是量化误差

2. 非对称量化（Asymmetric Quantization）

# 非对称量化：使用 zero_point 偏移，更精确地映射非对称分布
def asymmetric_quantize(weights, bits=8):
    # 零点 + 缩放，类似线性映射 y = kx + b
    w_min, w_max = weights.min(), weights.max()
    q_min, q_max = 0, 2**bits - 1
    scale = (w_max - w_min) / (q_max - q_min)
    zero_point = round(-w_min / scale)
    quantized = np.round(weights / scale + zero_point).astype(np.uint8)
    return quantized, scale, zero_point

# 对称 vs 非对称：非对称在权重分布不均匀时更精确
# 实际生产中 GPTQ、AWQ 等算法在非对称量化基础上做了大量优化
主流量化方法对比
方法精度特点适用场景
GPTQINT4/INT8逐层量化 + Hessian 矩阵校准离线量化，质量高

📦 GGUF 格式：Ollama 的模型集装箱 ~10 min

GGUF（GPT-Generated Unified Format）是 llama.cpp 团队设计的模型文件格式，Ollama 使用的就是 GGUF。理解它，就像理解数据仓库中的 Parquet 文件一样重要。

GGUF 文件结构
┌──────────────────────────────────────────────┐ │ GGUF 文件结构 │ ├──────────────────────────────────────────────┤ │ Magic Number (GGUF) 4 bytes │ │ Version 4 bytes │ │ ───────────────────────────────────── │ │ Metadata (JSON-like) │ │ ├── model.name: "qwen2.5-7b" │ │ ├── model.params: 7B │ │ ├── general.quantization_version: 2 │ │ └── tokenizer.model: ... │ │ ───────────────────────────────────── │ │ Tensor Data (二进制块) │ │ ├── tensor_1 (INT4 量化权重) │ │ ├── tensor_2 (INT4 量化权重) │ │ └── ... │ │ ───────────────────────────────────── │ │ Vocabulary (词表) │ └──────────────────────────────────────────────┘
量化级别速查

下载模型时经常看到 Q4_K_M、Q5_K_S 这样的后缀，含义如下：

后缀每位参数7B 模型质量推荐场景
Q4_04.5 bit~3.9 GB较差磁盘极紧
Q4_K_M4.8 bit~4.4 GB不错⭐ 默认首选
Q5_K_M5.7 bit~5.1 GB很好质量优先
Q8_08.5 bit~7.7 GB几乎无损学术/基准测试
F1616 bit~14 GB无损微调基座
选择建议：日常使用选 Q4_K_M（速度与质量的甜点）；对质量要求高选 Q5_K_M；做评测或微调选 F16。
Python 读取 GGUF 元数据
import struct

def read_gguf_metadata(filepath):
    """读取 GGUF 文件的元数据部分（前几个 key-value）"""
    with open(filepath, 'rb') as f:
        # 1. 读取 Magic Number
        magic = f.read(4)
        if magic != b'GGUF':
            raise ValueError("不是有效的 GGUF 文件")

        # 2. 读取版本号
        version = struct.unpack('<I', f.read(4))[0]

        # 3. 读取 metadata count
        n_kv = struct.unpack('<Q', f.read(8))[0]

        # 4. 简单打印前 5 个 key
        metadata = {}
        for i in range(min(5, n_kv)):
            # GGUF key 是 uint32 长度 + UTF-8 字符串
            key_len = struct.unpack('<Q', f.read(8))[0]
            key = f.read(key_len).decode('utf-8')
            # value 类型（uint32）
            vtype = struct.unpack('<I', f.read(4))[0]
            print(f"  Key: {key}, ValueType: {vtype}")

    return version, n_kv

def chat_with_ollama(prompt, model="qwen2.5:7b", system=None):
    """通过 Ollama REST API 进行对话"""
    messages = []
    if system:
        messages.append({"role": "system", "content": system})
    messages.append({"role": "user", "content": prompt})

            "messages": messages,

                "temperature": 0.3,
                "num_predict": 512,
            }
        },
        timeout=120
    )
    resp.raise_for_status()
    return resp.json()["message"]["content"]

🔧 LLM 实战：Ollama + FastAPI 构建本地 LLM 服务 ~15 min

把 Day 25-26 学的 FastAPI 和今天的 Ollama 结合，构建一个完全离线的 LLM 服务！

# llm_server.py — 本地 LLM API 服务（零外部依赖）
# 无需 API Key，无网络延迟，数据不出机器

from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel, Field
from typing import Optional
import httpx
import asyncio
from datetime import datetime
from contextlib import asynccontextmanager

# ===== 请求/响应模型 =====
class ChatRequest(BaseModel):
    message: str = Field(..., description="用户消息", min_length=1)
    model: str = Field(default="qwen2.5:7b", description="模型名称")
    system: Optional[str] = Field(default=None, description="系统提示词")
    temperature: float = Field(default=0.3, ge=0, le=2)

class ChatResponse(BaseModel):
    reply: str
    model: str
    tokens: int
    elapsed_ms: float

class ModelInfo(BaseModel):
    name: str
    size_gb: float
    quantization: str

# ===== 服务类 =====
class OllamaService:
    """封装 Ollama API 调用（依赖注入友好的单例）"""

    def __init__(self, base_url="http://localhost:11434"):
        self.base_url = base_url
        self._client: Optional[httpx.AsyncClient] = None

    async def startup(self):
        """应用启动时初始化 HTTP 客户端"""
        self._client = httpx.AsyncClient(
            base_url=self.base_url,
            timeout=httpx.Timeout(300.0, connect=10.0)
        )

    async def shutdown(self):
        """应用关闭时释放连接"""
        if self._client:
            await self._client.aclose()

    async def chat(self, message: str, model="qwen2.5:7b",
                    system=None, temperature=0.3) -> dict:
        """调用 Ollama Chat API"""
        messages = []
        if system:
            messages.append({"role": "system", "content": system})
        messages.append({"role": "user", "content": message})

        start = datetime.now()
        resp = await self._client.post("/api/chat", json={
            "model": model,
            "messages": messages,
            "stream": False,
            "options": {"temperature": temperature}
        })
        elapsed = (datetime.now() - start).total_seconds() * 1000

        resp.raise_for_status()
        data = resp.json()
        return {
            "reply": data["message"]["content"],
            "tokens": data.get("eval_count", 0),
            "elapsed_ms": elapsed,
        }

    async def list_models(self) -> list[dict]:
        """列出已安装的模型"""
        resp = await self._client.get("/api/tags")
        resp.raise_for_status()
        models = []
        for m in resp.json().get("models", []):
            size_gb = m["size"] / (1024**3)
            models.append({
                "name": m["name"],
                "size_gb": round(size_gb, 2),
                "quantization": m.get("details", {}).get(
                    "quantization_level", "unknown"
                )
            })
        return models

# ===== FastAPI 应用 =====
llm_service = OllamaService()

@asynccontextmanager
async def lifespan(app: FastAPI):
    await llm_service.startup()
    print("🚀 LLM Server 启动完成")
    yield
    await llm_service.shutdown()
    print("👋 LLM Server 关闭")

app = FastAPI(title="本地 LLM 服务", lifespan=lifespan)
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_methods=["*"],
    allow_headers=["*"],
)

@app.get("/health")
async def health():
    return {"status": "ok", "engine": "ollama"}

@app.get("/models", response_model=list[ModelInfo])
async def list_models():
    models = await llm_service.list_models()
    return [ModelInfo(**m) for m in models]

@app.post("/chat", response_model=ChatResponse)
async def chat(req: ChatRequest):
    try:
        result = await llm_service.chat(
            message=req.message,
            model=req.model,
            system=req.system,
            temperature=req.temperature,
        )
        return ChatResponse(
            reply=result["reply"],
            model=req.model,
            tokens=result["tokens"],
            elapsed_ms=result["elapsed_ms"],
        )
    except httpx.ConnectError:
        raise HTTPException(
            status_code=503,
            detail="Ollama 服务未启动，请先运行 ollama serve"
        )

Python 模型性能基准测试
import time
import asyncio
from dataclasses import dataclass, field

@dataclass
class BenchResult:
    """模型基准测试结果"""
    model: str
    prompt_tokens: int = 0
    completion_tokens: int = 0
    time_seconds: float = 0
    tokens_per_second: float = 0.0

    def __post_init__(self):
        # 自动计算 tokens/s
        if self.time_seconds > 0:
            self.tokens_per_second = round(
                self.completion_tokens / self.time_seconds, 2
            )

async def benchmark_model(
    model: str,
    prompt: str,
    rounds: int = 3
) -> BenchResult:
    """对单个模型进行多轮基准测试"""
    import httpx

    async with httpx.AsyncClient(
        base_url="http://localhost:11434", timeout=300
    ) as client:
        total_time = 0
        total_tokens = 0

        for i in range(rounds):
            start = time.perf_counter()
            resp = await client.post("/api/chat", json={
                "model": model,
                "messages": [{"role": "user", "content": prompt}],
                "stream": False,
            })
            elapsed = time.perf_counter() - start
            data = resp.json()

            total_time += elapsed
            total_tokens += data.get("eval_count", 0)

        return BenchResult(
            model=model,
            prompt_tokens=data.get("prompt_eval_count", 0),
            completion_tokens=total_tokens,
            time_seconds=round(total_time, 2),
        )
