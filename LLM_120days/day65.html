什么是数据血缘？

数据血缘（Data Lineage）追踪数据从源头到终点的完整流动路径——一张 ADS 报表字段，来自哪个 DWS 的哪列、经了怎样的计算逻辑、依赖哪些 DIM 维表。
在数仓分层（ODS→DWD→DWS→ADS）中，血缘是数据治理的基石：影响分析、故障排查、合规审计都离不开它。

1. AST 抽象语法树（Python 内置 ast 模块）

AST 是代码的结构化表示。数仓中大量 SQL 逻辑，解析血缘本质上就是"解析代码结构"——Python 的 ast 模块是最佳入门。

# ast 模块：把 Python 代码变成一棵可遍历的树
import ast

# 1. 解析代码为 AST
source = """
total = amount * rate
result = round(total, 2)
if total > 1000:
    print("high")
"""
tree = ast.parse(source)

# 2. 遍历所有节点 — 类似数仓的 ETL 扫描
class VarVisitor(ast.NodeVisitor):
    """收集所有被赋值的变量名"""

    def __init__(self):
        self.assigned_vars = []   # 赋值目标
        self.used_vars = []       # 引用的变量

    def visit_Assign(self, node):
        # 提取赋值左边的变量名
        for target in node.targets:
            if isinstance(target, ast.Name):
                self.assigned_vars.append(target.id)
        # 继续遍历子节点（右边的表达式）
        self.generic_visit(node)

    def visit_Name(self, node):
        # 记录所有被引用的变量（非赋值目标）
        self.used_vars.append(node.id)
        self.generic_visit(node)

visitor = VarVisitor()
visitor.visit(tree)
print(f"赋值变量: {visitor.assigned_vars}")  # ['total', 'result']
print(f"引用变量: {visitor.used_vars}")      # ['amount', 'rate', 'total', 'round', ...]
数仓类比：AST 遍历 = 数据探查 profiling。visit_Assign 就像"找到目标字段"，visit_Name 就像"找到所有被引用的源字段"。血缘解析的思路完全一样！

2. 正则多模式匹配（sqlparse 的底层原理）

SQL 解析常用"正则 + 组合"策略：把复杂的 SQL 拆成 INSERT、SELECT、FROM、JOIN 等子句，分别用正则提取。

import re

# SQL 血缘解析常用的正则模式组合
SQL_PATTERNS = {
    # 1. 匹配 CREATE TABLE xxx AS / INSERT INTO xxx
    'target_table': re.compile(
        r'(?:CREATE\s+TABLE|INSERT\s+INTO)\s+`?(\w+)`?',
        re.IGNORECASE
    ),

    # 2. 匹配 FROM / JOIN 后的表名
    'source_table': re.compile(
        r'(?:FROM|JOIN)\s+`?(\w+)`?',
        re.IGNORECASE
    ),

    # 3. 匹配 SELECT 后的字段名（含别名 alias）
    'select_columns': re.compile(
        r'SELECT\s+(.*?)\s+FROM',
        re.IGNORECASE | re.DOTALL
    ),

    # 4. 匹配字段别名: col AS alias / col alias
    'column_alias': re.compile(
        r'`?(\w+)`?\s+(?:AS\s+)?`?(\w+)`?',
        re.IGNORECASE
    ),
}

# 实战：从 SQL 提取表级血缘
sql = """
INSERT INTO dws_bid_daily_stat
SELECT
    b.bid_id,
    b.company_name,
    SUM(p.price) AS total_amount,
    COUNT(*) AS bid_count
FROM ods_bid_info b
LEFT JOIN ods_bid_price p ON b.bid_id = p.bid_id
WHERE b.status = 'active'
GROUP BY b.bid_id, b.company_name
"""

# 提取目标表
target = SQL_PATTERNS['target_table'].search(sql)
print(f"目标表: {target.group(1) if target else 'N/A'}")
# 输出: 目标表: dws_bid_daily_stat

# 提取所有源表
sources = SQL_PATTERNS['source_table'].findall(sql)
print(f"源表: {sources}")
# 输出: 源表: ['ods_bid_info', 'ods_bid_price']

# 查询某张表的完整上游
def get_upstream(table, graph, visited=None):
    """递归获取完整上游链路（类似数仓的递归依赖查询）"""
    if visited is None:
        visited = set()
    if table in visited:
        return []  # 防止循环依赖
    visited.add(table)
    result = list(graph.get(table, []))
    for parent in result:
        result.extend(get_upstream(parent, graph, visited))
    return result

print(f"ads_top_bidders 完整上游: {get_upstream('ads_top_bidders', lineage)}")
# ['dws_bid_daily_stat', 'ods_bid_info', 'ods_bid_price', 'dim_company']
关键理解：递归上游查询 = 数仓中的"影响分析"——改了 ODS 表，能自动追溯到哪些 DWS/ADS 受影响。

from dataclasses import dataclass, field
from enum import Enum
from typing import Optional

class TableLayer(Enum):
    """数仓分层枚举"""
    ODS = "ODS"     # 操作数据层（原始数据）
    DIM = "DIM"     # 维度表层
    DWD = "DWD"     # 明细数据层
    DWS = "DWS"     # 汇总数据层
    ADS = "ADS"     # 应用数据层
    TMP = "TMP"     # 临时表

@dataclass
class ColumnMeta:
    """列级元数据——每张表的每个字段"""
    name: str                       # 字段名
    data_type: str = "VARCHAR"   # 数据类型
    comment: str = ""               # 字段注释
    is_pk: bool = False              # 是否主键
    is_nullable: bool = True         # 是否可空
    source_expr: str = ""          # 来源表达式（如 SUM(price)）

@dataclass
class TableMeta:
    """表级元数据——对应你数仓中的一张表"""
    table_name: str
    layer: TableLayer
    database: str = "ybt_dw"
    comment: str = ""
    columns: list[ColumnMeta] = field(default_factory=list)
    create_sql: str = ""         # 建表 SQL（用于解析血缘）
    row_count: int = 0

    def get_column(self, name: str) -> Optional[ColumnMeta]:
        """按字段名查找列元数据"""
        for col in self.columns:
            if col.name.lower() == name.lower():
                return col
        return None

@dataclass
class LineageEdge:
    """血缘关系边——从 source 到 target 的依赖"""
    source_table: str     # 上游表
    source_column: str    # 上游列
    target_table: str     # 下游表
    target_column: str    # 下游列
    transform_type: str = "direct"  # direct/agg/filter/derive
    transform_expr: str = ""      # 转换表达式
    sql_origin: str = ""           # 来源 SQL

表级血缘解析

import re
from dataclasses import dataclass, field

@dataclass
class TableLineageResult:
    """单条 SQL 解析出的表级血缘"""
    target_table: str                          # 下游目标表
    source_tables: list[str] = field(default_factory=list)  # 上游源表
    join_type: dict = field(default_factory=dict)  # JOIN 类型: {表名: 'LEFT/INNER/...'}
    where_conditions: list[str] = field(default_factory=list)  # WHERE 条件

class SQLTableParser:
    """
    SQL 表级血缘解析器
    核心思路：正则模式组合提取 → 表名标准化 → 去重
    """

    # 预编译正则（性能优化：编译一次，反复使用）
    RE_TARGET = re.compile(
        r'(?:CREATE\s+TABLE\s+(?:IF\s+NOT\s+EXISTS\s+)?)?`?(\w+)`?'
        r'\s+(?:AS\s+)?SELECT', re.IGNORECASE
    )
    RE_INSERT = re.compile(
        r'INSERT\s+(?:INTO|OVERWRITE)\s+`?(\w+)`?', re.IGNORECASE
    )
    RE_FROM = re.compile(r'FROM\s+`?(\w+)`?', re.IGNORECASE)
    RE_JOIN = re.compile(
        r'(LEFT|RIGHT|INNER|FULL|CROSS)?\s*JOIN\s+`?(\w+)`?', re.IGNORECASE
    )
    RE_SUBQUERY = re.compile(r'\(\s*SELECT', re.IGNORECASE)
    # 需要排除的关键字（不是表名）
    EXCLUDE_KEYWORDS = {
        'SELECT', 'WHERE', 'AND', 'OR', 'ON', 'AS',
        'GROUP', 'ORDER', 'BY', 'HAVING', 'LIMIT',
        'UNION', 'EXCEPT', 'INTERSECT', 'SET', 'INTO',
    }

    def parse(self, sql: str) -> TableLineageResult:
        """解析一条 SQL 的表级血缘"""
        sql_clean = self._remove_comments(sql)
        sql_clean = self._remove_subqueries(sql_clean)

        # 1. 找目标表
        target = self._extract_target(sql_clean)

        # 2. 找所有源表（FROM + JOIN）
        sources, join_types = self._extract_sources(sql_clean)

        # 3. 找 WHERE 条件
        where = self._extract_where(sql_clean)

        return TableLineageResult(
            target_table=target,
            source_tables=sources,
            join_type=join_types,
            where_conditions=where,
        )

    def _remove_comments(self, sql: str) -> str:
        """去除 SQL 注释（单行 -- 和多行 /* */）"""
        sql = re.sub(r'--.*$', '', sql, flags=re.MULTILINE)
        sql = re.sub(r'/\*.*?\*/', '', sql, flags=re.DOTALL)
        return sql.strip()

    def _remove_subqueries(self, sql: str) -> str:
        """简化处理：将子查询标记为 [SUBQUERY]，避免误解析内部 FROM"""
        return re.sub(
            r'\(\s*SELECT\b[^)]*\)', '[SUBQUERY]', sql,
            flags=re.IGNORECASE | re.DOTALL
        )

    def _extract_target(self, sql: str) -> str:
        """提取目标表名"""
        m = self.RE_TARGET.search(sql) or self.RE_INSERT.search(sql)
        return m.group(1) if m else "UNKNOWN"

    def _extract_sources(self, sql: str) -> tuple[list[str], dict]:
        """提取源表和 JOIN 类型"""
        sources = []
        join_types = {}

        # FROM 子句
        for m in self.RE_FROM.finditer(sql):
            tbl = m.group(1)
            if tbl.upper() not in self.EXCLUDE_KEYWORDS:
                sources.append(tbl)

        # JOIN 子句（含类型）
        for m in self.RE_JOIN.finditer(sql):
            jtype = m.group(1) or "INNER"  # 省略默认 INNER
            tbl = m.group(2)
            if tbl.upper() not in self.EXCLUDE_KEYWORDS:
                sources.append(tbl)
                join_types[tbl] = jtype.upper()

        return list(dict.fromkeys(sources)), join_types  # 去重保序

    def _extract_where(self, sql: str) -> list[str]:
        """提取 WHERE 条件（简化版）"""
        m = re.search(r'WHERE\s+(.*?)(?:\s+GROUP\s+|\s+ORDER\s+|\s+LIMIT\s+|$)',
                     sql, re.IGNORECASE | re.DOTALL)
        if not m:
            return []
        # 按 AND 拆分多个条件
        conds = re.split(r'\s+AND\s+', m.group(1), flags=re.IGNORECASE)
        return [c.strip() for c in conds if c.strip()]

列级血缘解析

import re
from dataclasses import dataclass, field

@dataclass
class ColumnLineage:
    """单列的血缘信息"""
    target_col: str                  # 下游字段名
    source_cols: list[str]           # 依赖的上游字段
    transform_type: str             # direct / agg / derive / filter
    expression: str                 # 完整表达式

class SQLColumnParser:
    """
    列级血缘解析器
    核心思路：
    1. 提取 SELECT 子句
    2. 识别列别名（AS）
    3. 分析表达式中的源字段引用
    """

    # 聚合函数模式
    RE_AGG = re.compile(
        r'\b(SUM|AVG|COUNT|MAX|MIN|GROUP_CONCAT)\s*\([^)]*\)', re.IGNORECASE
    )
    # 字段引用模式: table.column 或单独 column
    RE_COL_REF = re.compile(r'`?(\w+)\.`?(\w+)`?')
    # CASE WHEN 模式（派生字段）
    RE_CASE = re.compile(r'\bCASE\b', re.IGNORECASE)

    def parse_select_columns(self, sql: str) -> list[ColumnLineage]:
        """解析 SELECT 子句的列级血缘"""
        # 1. 提取 SELECT ... FROM 之间的内容
        m = re.search(
            r'SELECT\s+(.*?)\s+FROM', sql,
            re.IGNORECASE | re.DOTALL
        )
        if not m:
            return []

        select_clause = m.group(1).strip()
        results = []

        # 2. 按逗号拆分各列（注意排除函数内的逗号）
        columns = self._split_columns(select_clause)

        for col_expr in columns:
            col_expr = col_expr.strip()
            if not col_expr:
                continue

            # 3. 提取别名
            alias = self._extract_alias(col_expr)
            expr = self._remove_alias(col_expr)

            # 4. 识别转换类型
            transform = self._classify_transform(expr)

            # 5. 提取源字段
            source_cols = self._extract_source_cols(expr)

            results.append(ColumnLineage(
                target_col=alias,
                source_cols=source_cols,
                transform_type=transform,
                expression=expr,
            ))

        return results

    def _split_columns(self, select_clause: str) -> list[str]:
        """
        智能拆分 SELECT 列（尊重函数括号内的逗号）
        思路：用"括号深度"判断逗号是否属于函数内部
        """
        columns = []
        depth = 0
        current = []

        for char in select_clause:
            if char == '(':
                depth += 1
            elif char == ')':
                depth = max(0, depth - 1)
            elif char == ',' and depth == 0:
                columns.append(''.join(current).strip())
                current = []
                continue
            current.append(char)

        if current:
            columns.append(''.join(current).strip())

        return columns

    def _extract_alias(self, col_expr: str) -> str:
        """提取列别名"""
        m = re.search(
            r'(?:AS\s+)?`?(\w+)`?\s*$', col_expr, re.IGNORECASE
        )
        return m.group(1) if m else col_expr.split('.')[-1]

    def _remove_alias(self, col_expr: str) -> str:
        """移除别名，保留表达式部分"""
        return re.sub(
            r'\s+(?:AS\s+)?`?\w+`?\s*$', '',
            col_expr, flags=re.IGNORECASE
        ).strip()

    def _classify_transform(self, expr: str) -> str:
        """分类转换类型"""
        if self.RE_AGG.search(expr):
            return "agg"      # 聚合: SUM(), COUNT()
        if self.RE_CASE.search(expr):
            return "derive"   # 派生: CASE WHEN
        if '.' in expr and expr.count('.') == 1:
            return "direct"   # 直接映射: b.bid_id
        return "derive"       # 其他: 默认派生

    def _extract_source_cols(self, expr: str) -> list[str]:
        """提取表达式中的源字段引用"""
        # 移除函数名和关键字，只保留字段引用
        clean = re.sub(r'\b(?:SUM|AVG|COUNT|MAX|MIN|GROUP_CONCAT|DISTINCT)\b',
                      '', expr, flags=re.IGNORECASE)
        clean = re.sub(r'[()*+\-/<>=!&|]', ' ', clean)

        cols = []
        for part in clean.split():
            # 匹配 table.column 格式
            m = re.match(r'^`?(\w+)`?\.`?(\w+)`?$', part)
            if m:
                cols.append(f"{m.group(1)}.{m.group(2)}")
        return list(dict.fromkeys(cols))

场景：为什么需要 LLM？

维度正则解析LLM 辅助
表级血缘✅ 准确率高（FROM/JOIN 规范）没必要，浪费 Token
列级血缘（简单映射）✅ b.bid_id → bid_id没必要
列级血缘（复杂表达式）⚠️ IFNULL(a,0)*rate+bonus✅ 能理解语义
业务含义标注❌ 无法理解 total_amount 是"含税"✅ 根据注释+上下文推断
血缘断裂检测⚠️ 只能看结构✅ 判断逻辑是否合理

LLM 血缘标注器

from dataclasses import dataclass
from typing import Optional

@dataclass
class LLMAnnotation:
    """LLM 对血缘关系的语义标注"""
    table_name: str
    column_name: str
    business_meaning: str     # 业务含义（中文）
    data_quality_risk: str    # 数据质量风险提示
    suggestion: str           # 优化建议

class LineageAnnotator:
    """
    用 LLM 对血缘关系进行语义标注
    适用场景：复杂表达式 / 业务含义推断 / 风险提示
    """

    ANNOTATE_PROMPT = """你是一个数据仓库血缘分析专家。
请分析以下 SQL 字段的血缘关系，给出语义标注。

## 目标表信息
- 表名: {target_table}
- 表注释: {target_comment}
- 层级: {target_layer}

## 字段信息
- 字段名: {column_name}
- 表达式: {expression}
- 上游依赖: {source_columns}

## 源表信息
{source_table_info}

## 请以 JSON 格式返回：
{{
  "business_meaning": "该字段在业务上的含义",
  "data_quality_risk": "该字段可能存在的数据质量风险",
  "suggestion": "对该字段或表达式的优化建议"
}}

只返回 JSON，不要其他内容。"""

    def annotate(self, column_lineage, target_meta, source_metas) -> LLMAnnotation:
        """
        对单列血缘进行 LLM 语义标注
        """
        prompt = self.ANNOTATE_PROMPT.format(
            target_table=target_meta.table_name,
            target_comment=target_meta.comment,
            target_layer=target_meta.layer.value,
            column_name=column_lineage.target_col,
            expression=column_lineage.expression,
            source_columns=", ".join(column_lineage.source_cols),
            source_table_info=self._build_source_info(
                column_lineage.source_cols, source_metas
            ),
        )

        # 调用 LLM（这里用 mock 演示，实际用 zhipuai SDK）
        response = self._call_llm(prompt)
        return self._parse_response(response, column_lineage)

    def _build_source_info(self, source_cols, source_metas):
        """构建源表字段信息描述"""
        info_lines = []
        for col in source_cols:
            parts = col.split('.')
            if len(parts) == 2:
                tbl, col_name = parts
                for meta in source_metas:
                    if meta.table_name == tbl:
                        col_meta = meta.get_column(col_name)
                        desc = col_meta.comment if col_meta else "未知"
                        info_lines.append(
                            f"- {tbl}.{col_name}: {desc}"
                        )
                        break
        return "\n".join(info_lines) or "无源表信息"

    def _call_llm(self, prompt: str) -> dict:
        """调用 LLM（mock 版，实际替换为 zhipuai）"""
        # 实际代码：
        # from zhipuai import ZhipuAI
        # client = ZhipuAI(api_key="xxx")
        # resp = client.chat.completions.create(
        #     model="glm-4-flash",
        #     messages=[{"role":"user","content":prompt}],
        #     temperature=0.1,
        # )
        # return json.loads(resp.choices[0].message.content)
        return {
            "business_meaning": "各投标的总金额汇总，反映项目资金规模",
            # 模拟返回，实际由 LLM 生成
            "data_quality_risk": "若 ods_bid_price.price 为 NULL，SUM 会自动跳过，
可能导致 total_amount 比预期偏低",
            "suggestion": "建议添加 COALESCE(p.price, 0) 处理空值，
并增加 COUNT(p.price) 校验非空行数",
        }

    def _parse_response(self, response, column_lineage) -> LLMAnnotation:
        """解析 LLM 返回（含容错）"""
        return LLMAnnotation(
            table_name="dws_bid_daily_stat",
            column_name=column_lineage.target_col,
            business_meaning=response.get("business_meaning", ""),
            data_quality_risk=response.get("data_quality_risk", ""),
            suggestion=response.get("suggestion", ""),
        )
