🐍 Python 知识：Agent 必备数据结构 ⏱ 15分钟

Agent 进阶需要三个 Python 标准库工具，它们分别对应 Agent 的消息队列、配置合并和状态管理。

3.1 collections.deque — 高效双端队列

Agent 的对话历史就是一个先进先出（FIFO）的消息队列。deque 比 list 快得多，且自带长度限制——非常适合做"只保留最近 N 轮对话"的滑动窗口。

deque 基础：Agent 对话窗口
from collections import deque

# 创建一个最多保留 10 条消息的对话窗口
# maxlen 超出时自动丢弃左边（最早）的元素
chat_history = deque(maxlen=10)

# 添加消息（右边 = 最新）
chat_history.append({"role": "user", "content": "查询不良资产余额"})
chat_history.append({"role": "assistant", "content": "余额为500万"})
print(len(chat_history))  # 2

# 继续添加，超过 maxlen=10 时自动淘汰最旧的
for i in range(15):
    chat_history.append({"role": "user", "content": f"消息{i}"})
print(len(chat_history))  # 10 ← 自动裁剪到 maxlen

# deque 转回 list（传给 LLM API 需要 list 格式）
messages_list = list(chat_history)
print(len(messages_list))  # 10

# 左端操作（不常用，但了解一下）
chat_history.appendleft({"role": "system", "content": "你是AMC助手"})
# 注意：appendleft 也会触发 maxlen 淘汰，但淘汰的是右端
💡 为什么不直接用 list？
list 的 insert(0, x) 需要移动所有元素，O(n) 复杂度。deque 两端操作都是 O(1)。当 Agent 跑了几十轮对话后，deque 的性能优势明显。另外 maxlen 自动裁剪，list 需要手动 if len(msgs) > N: msgs = msgs[-N:]。
3.2 ChainMap — 多层配置合并

Agent 的配置往往来自多个来源：默认配置 → 环境变量 → 用户传入 → 命令行参数。ChainMap 把多个 dict 链在一起，按优先级查找——类似 CSS 的层叠。

ChainMap：Agent 配置优先级链
from collections import ChainMap

# 模拟四层配置（优先级从低到高）
default_config = {
    "model": "glm-4-flash",
    "temperature": 0.7,
    "max_tokens": 2048,
    "timeout": 30,
}
env_config = {
    "model": "glm-4-plus",   # 环境变量覆盖了 model
    "api_key": "env_xxx",
}
user_config = {
    "temperature": 0.3,  # 用户要求更精确
}

# ChainMap: user > env > default（第一个参数优先级最高）
config = ChainMap(user_config, env_config, default_config)

print(config["model"])        # "glm-4-plus" ← 来自 env_config
print(config["temperature"]) # 0.3 ← 来自 user_config
print(config["max_tokens"])  # 2048 ← 来自 default_config
print(config["timeout"])      # 30 ← 来自 default_config

# 查找某个 key 存在于哪一层
print(config.maps)  # [user_config, env_config, default_config]

# 新增配置（默认写入第一层 = user_config）
config["debug"] = True
print(user_config)  # {'temperature': 0.3, 'debug': True}

# 注意：ChainMap 是视图，修改会影响原始 dict！
# 如需隔离，用 ChainMap(user.copy(), env.copy(), default.copy())
数仓类比：ChainMap 就像数仓中 COALESCE(col1, col2, col3) 的配置版——按优先级逐层找，第一个非空值胜出。也可以理解为"环境变量 > 配置文件 > 默认值"的经典优先级模式。
3.3 enum — 枚举状态管理

Agent 有明确的运行阶段：思考→行动→观察→回答。用字符串 `"thinking"` / `"acting"` 容易拼错，用枚举类型安全又清晰。

enum：Agent 状态机
from enum import Enum, auto

class AgentStatus(Enum):
    """Agent 运行状态枚举"""
    IDLE = auto()       # 空闲，等待任务
    THINKING = auto()   # 正在思考下一步
    TOOL_CALL = auto()  # 正在调用工具
    OBSERVING = auto()  # 观察工具结果
    FINISHED = auto()   # 任务完成

# 枚举值是单例，比较用 is 或 ==
status = AgentStatus.IDLE
if status == AgentStatus.IDLE:
    print("Agent 空闲中")

# 枚举自带 .name 和 .value
print(AgentStatus.THINKING.name)   # "THINKING"
print(AgentStatus.THINKING.value)  # 2

# 遍历所有状态（画状态图时有用）
for s in AgentStatus:
    print(f"{s.name} = {s.value}")

# 从字符串解析枚举（解析 LLM 返回的状态）
status_str = "TOOL_CALL"
status = AgentStatus[status_str]  # AgentStatus.TOOL_CALL

# 安全解析（字符串可能不匹配）
try:
    status = AgentStatus["INVALID"]
except KeyError:
    print("未知状态，使用默认 IDLE")
    status = AgentStatus.IDLE
💡 enum vs 字符串常量：如果你写了 if status == "THINKNG"（拼错一个字母），Python 不会报错但逻辑就错了。用枚举 if status == AgentStatus.THINKING，IDE 会自动补全，拼写错误会直接 KeyError。
Python 工具Agent 中的用途核心优势 deque(maxlen=N)对话历史滑动窗口O(1) 两端操作 + 自动裁剪 ChainMap多层配置优先级合并无拷贝视图，按优先级查找 EnumAgent 状态 / 消息类型枚举类型安全 + IDE 补全 + 自文档化
🔗 ToolChain：工具链管道编排 ⏱ 15分钟
核心思想：Day 37 的 Agent 是 LLM 自己选择调哪个工具。但很多场景工具的执行顺序是固定的——就像 ETL 管道：先抽取 → 再转换 → 再加载。ToolChain 把工具串成固定管道，上一个的输出自动流入下一个的输入。
4.1 管道模式 vs 自由选择
Day 37 FC 模式（LLM选工具）ToolChain 管道模式 适用场景开放式问答，工具不确定固定流程，步骤明确 例子"查一下不良资产数据""分析报表→生成摘要→发送邮件" 谁决定顺序LLM 推理开发者预设 可靠性可能选错工具100% 按预设执行 灵活性高中（可加条件分支）
4.2 实现 ToolChain 管道
ToolChain：管道式工具编排器
from dataclasses import dataclass, field
from typing import Any, Callable
from enum import Enum, auto
import json
import time

class PipeStatus(Enum):
    SUCCESS = auto()
    SKIP = auto()     # 跳过此步骤（条件分支）
    ERROR = auto()

@dataclass
class PipeResult:
    """管道中单个步骤的执行结果"""
    status: PipeStatus
    data: Any = None
    error: str | None = None
    step_name: str = ""
    duration: float = 0.0

class ToolChain:
    """
    工具链管道编排器
    - 每个 Tool 是一个函数: (input_data) -> PipeResult
    - 上一步的 PipeResult.data 自动流入下一步
    - 支持 SKIP（跳过）和 ERROR（中断管道）
    """

    def __init__(self, name: str = "default"):
        self.name = name
        self._tools: list[tuple[str, Callable]] = []  # [(name, func), ...]
        self._log: list[dict] = []  # 执行日志

    def add(self, name: str, tool: Callable) -> "ToolChain":
        """添加工具步骤（支持链式调用）"""
        self._tools.append((name, tool))
        return self  # 链式调用: chain.add("a", f1).add("b", f2)

    def run(self, input_data: Any = None) -> PipeResult:
        """按顺序执行管道中的所有工具"""
        current = input_data

        for step_name, tool in self._tools:
            start = time.time()
            try:
                result = tool(current)
                if not isinstance(result, PipeResult):
                    # 如果工具直接返回数据，自动包装
                    result = PipeResult(
                        status=PipeStatus.SUCCESS,
                        data=result,
                        step_name=step_name,
                    )
                result.step_name = step_name
                result.duration = time.time() - start
            except Exception as e:
                result = PipeResult(
                    status=PipeStatus.ERROR,
                    error=str(e),
                    step_name=step_name,
                    duration=time.time() - start,
                )

            self._log.append({
                "step": step_name,
                "status": result.status.name,
                "duration": round(result.duration, 3),
                "error": result.error,
            })

            # 根据状态决定是否继续
            if result.status == PipeStatus.ERROR:
                return result  # 中断管道
            if result.status == PipeStatus.SUCCESS:
                current = result.data  # 流入下一步
            # SKIP: current 不变，继续下一步

        return PipeResult(status=PipeStatus.SUCCESS, data=current)

    def show_log(self):
        """打印执行日志"""
        for entry in self._log:
            icon = "✅" if entry["status"] == "SUCCESS" else \
                   "⏭️" if entry["status"] == "SKIP" else "❌"
            print(f"  {icon} {entry['step'] [{entry['duration']]s")
            if entry["error"]:
                print(f"     ⚠ {entry['error']")

用 ToolChain 编排 AMC 数据处理流程
# ===== 模拟 AMC 报表数据 =====
mock_report = {
    "report_id": "AMC-2026-Q2-001",
    "raw_text": """
    二季度不良资产处置报告
    处置总额：3.5亿元
    核销：1.2亿，转让：1.5亿，重组：0.8亿
    同比增长：15.3%
    不良率：1.85%，较上季度下降0.12个百分点
    """,
}

# ===== 定义管道中的工具函数 =====

def extract_data(raw_data) -> PipeResult:
    """步骤1：从原始报表中提取结构化数据"""
    text = raw_data.get("raw_text", "")
    # 模拟提取（实际场景会用 LLM 或正则）
    structured = {
        "report_id": raw_data["report_id"],
        "total_amount": 3.5,
        "breakdown": {"核销": 1.2, "转让": 1.5, "重组": 0.8},
        "yoy_growth": 15.3,
        "npl_ratio": 1.85,
        "npl_change": -0.12,
    }
    return PipeResult(status=PipeStatus.SUCCESS, data=structured)

def validate_data(data) -> PipeResult:
    """步骤2：校验数据质量（条件分支：不合格则跳过后续）"""
    # 检查不良率是否异常
    if data["npl_ratio"] > 5.0:
        return PipeResult(
            status=PipeStatus.ERROR,
            error=f"不良率异常: {data['npl_ratio']}% 超过阈值5%",
            data=data,
        )
    # 检查金额是否完整
    breakdown_sum = sum(data["breakdown"].values())
    if abs(breakdown_sum - data["total_amount"]) > 0.1:
        return PipeResult(
            status=PipeStatus.ERROR,
            error="分项金额之和不等于总额",
        )
    return PipeResult(status=PipeStatus.SUCCESS, data=data)

def generate_summary(data) -> PipeResult:
    """步骤3：用 LLM 生成分析摘要（这里模拟）"""
    summary = (
        f"AMC 二季度不良资产处置{data['total_amount']}亿元，"
        f"同比增长{data['yoy_growth']}%。"
        f"不良率{data['npl_ratio']}%，"
        f"较上季度{'下降' if data['npl_change'] < 0 else '上升'}"
        f"{abs(data['npl_change'])}个百分点。处置方式："
        f"{'、'.join(f'{k}{v}亿' for k, v in data['breakdown'].items())}。"
    )
    data["summary"] = summary
    return PipeResult(status=PipeStatus.SUCCESS, data=data)

def format_output(data) -> PipeResult:
    """步骤4：格式化输出"""
    output = {
        "report_id": data["report_id"],
        "summary": data["summary"],
        "metrics": {
            "处置总额(亿)": data["total_amount"],
            "不良率(%)": data["npl_ratio"],
            "同比增长(%)": data["yoy_growth"],
        }
    }
    return PipeResult(status=PipeStatus.SUCCESS, data=output)

# ===== 组装并执行管道 =====
chain = (
    ToolChain("AMC报表处理")
    .add("提取数据", extract_data)
    .add("数据校验", validate_data)
    .add("生成摘要", generate_summary)
    .add("格式化输出", format_output)
)

result = chain.run(mock_report)

if result.status == PipeStatus.SUCCESS:
    print(json.dumps(result.data, ensure_ascii=False, indent=2))
    chain.show_log()
else:
    print(f"管道中断: {result.error}")
    chain.show_log()
⚡ ToolChain vs Function Calling 互补而非替代：当流程确定时用 ToolChain（更可控），当需要 LLM 灵活判断时用 FC。很多高级框架（LangGraph）本质就是两者的结合。
🧠 AgentMemory：对话记忆机制 ⏱ 15分钟
为什么需要记忆？没有记忆的 Agent 每次都从零开始。对话到第三轮时它忘了第一轮说了什么。记忆分两层：短期记忆（当前对话上下文）和长期记忆（跨对话的持久知识）。
短期记忆（Working Memory）长期记忆（Long-term Memory） 类比大脑的工作记忆（7±2 项）笔记/知识库 存储deque 滑动窗口向量数据库 / JSON文件 范围当前会话内跨会话持久化 读取全量（最近N轮）语义检索相关片段 实现deque(maxlen=N)Embedding + 向量相似度
5.1 实现 AgentMemory 完整类
AgentMemory：双层记忆系统
from collections import deque
from dataclasses import dataclass, field
from typing import Any
import json

@dataclass
class MemoryEntry:
    """一条记忆条目"""
    content: str
    role: str  # "user" / "assistant" / "system" / "observation"
    metadata: dict = field(default_factory=dict)

class AgentMemory:
    """
    Agent 双层记忆系统

    短期记忆: deque 滑动窗口，保留最近 N 轮对话
    长期记忆: list 存储，支持关键词检索（生产中替换为向量检索）
    """

    def __init__(self, short_term_limit=10, long_term_limit=100):
        # 短期记忆：deque 自动滑动窗口
        self.short_term = deque(maxlen=short_term_limit)
        # 长期记忆：全量存储（带容量限制）
        self.long_term: list[MemoryEntry] = []
        self._long_term_limit = long_term_limit

    def add(self, content: str, role: str = "user", save_long=False, **meta):
        """添加一条记忆"""
        entry = MemoryEntry(content=content, role=role, metadata=meta)

        # 始终写入短期记忆
        self.short_term.append(entry)

        # 按需写入长期记忆
        if save_long:
            self.long_term.append(entry)
            # 超出容量时淘汰最旧的
            if len(self.long_term) > self._long_term_limit:
                self.long_term = self.long_term[-self._long_term_limit:]

    def get_context(self, include_long=True, max_long=3) -> list[dict]:
        """
        获取传给 LLM 的完整上下文

        Returns:
            [{"role": "user", "content": "..."}, ...]
        """
        messages = []

        # 1. 从长期记忆中检索相关片段（简单版：取最近几条）
        if include_long and self.long_term:
            recent_long = self.long_term[-max_long:]
            messages.append({
                "role": "system",
                "content": "[历史记忆摘要] " + "; ".join(
                    f"{e.role}:{e.content}" for e in recent_long
                ),
            })

        # 2. 短期记忆全量（已经是裁剪后的）
        messages.extend([
            {"role": e.role, "content": e.content}
            for e in self.short_term
        ])

        return messages

    def search_long(self, keyword: str, top_k=3) -> list[MemoryEntry]:
        """在长期记忆中按关键词检索（生产中替换为向量检索）"""
        keyword = keyword.lower()
        scored = []
        for entry in self.long_term:
            score = entry.content.lower().count(keyword)
            if score > 0:
                scored.append((score, entry))
        scored.sort(key=lambda x: x[0], reverse=True)
        return [entry for _, entry in scored[:top_k]]

    def clear_short(self):
        """清空短期记忆（新会话开始时）"""
        self.short_term.clear()

    def stats(self) -> dict:
        """记忆统计"""
        return {
            "short_term": len(self.short_term),
            "short_term_max": self.short_term.maxlen,
            "long_term": len(self.long_term),
        }
使用 AgentMemory
# 创建记忆系统
memory = AgentMemory(short_term_limit=6, long_term_limit=50)

# ===== 第一轮对话 =====
memory.add("查询AMC不良资产余额", "user")
memory.add("当前不良资产余额为500万元", "assistant", save_long=True)

# ===== 第二轮对话 =====
memory.add("去年同期是多少？", "user")
memory.add("去年同期为380万元，同比增长31.6%", "assistant", save_long=True)

# 获取传给 LLM 的上下文
context = memory.get_context()
for msg in context:
    role_icon = {"system": "🔧", "user": "👤", "assistant": "🤖"}
    print(f"{role_icon.get(msg['role'], '?')}  {msg['role']: {msg['content'][:50]}...")

# ===== 新会话：短期清空，长期保留 =====
memory.clear_short()
memory.add("帮我查一下上次的余额数据", "user")

# 长期记忆会自动注入，Agent 知道"上次"指的是什么
context = memory.get_context(include_long=True)

# 搜索长期记忆
results = memory.search_long("余额")
for r in results:
    print(f"  找到: {r.content} ({r.role})")

print("记忆统计:", memory.stats())

MultiAgentRouter：多Agent路由调度
from dataclasses import dataclass, field
from typing import Any, Callable
import re

@dataclass
class AgentRoute:
    """一个 Agent 的路由注册信息"""
    name: str
    description: str        # 这个 Agent 擅长什么
    keywords: list[str]     # 触发关键词
    handler: Callable       # 处理函数: (query) -> str

class MultiAgentRouter:
    """
    多 Agent 路由器

    简单版：基于关键词匹配
    进阶版：用 LLM 做意图分类（本节实现简单版，注释标明进阶方式）
    """

    def __init__(self, default_agent: str = "general"):
        self._routes: dict[str, AgentRoute] = {}
        self.default_agent = default_agent

    def register(self, name: str, description: str,
                keywords: list[str], handler: Callable):
        """注册一个 Agent"""
        self._routes[name] = AgentRoute(
            name=name,
            description=description,
            keywords=[k.lower() for k in keywords],
            handler=handler,
        )

    def _route(self, query: str) -> str:
        """根据用户输入选择路由（简单版：关键词匹配）"""
        query_lower = query.lower()

        # 统计每个 Agent 的关键词命中数
        scores = {}
        for name, route in self._routes.items():
            score = sum(1 for kw in route.keywords if kw in query_lower)
            scores[name] = score

        # 选得分最高的
        best = max(scores, key=scores.get)
        if scores[best] == 0:
            return self.default_agent
        return best

    def dispatch(self, query: str) -> dict:
        """分发并执行"""
        agent_name = self._route(query)
        route = self._routes[agent_name]

        # 执行 Agent 的处理函数
        result = route.handler(query)

        return {
            "query": query,
            "agent": agent_name,
            "description": route.description,
            "result": result,
        }

    def list_agents(self):
        """列出所有注册的 Agent"""
        for name, route in self._routes.items():
            print(f"  🤖 {name}: {route.description}")
            print("     关键词: " + ", ".join(route.keywords))

import json

def llm_route(self, query: str, llm_client) -> str:
    """
    用 LLM 做意图分类（进阶版路由）

    发送给 LLM 一个结构化 prompt，让它返回 agent 名称。
    这比关键词匹配更准确，能理解语义。
    """
    agent_list = ", ".join(
        f"{name}({r.description})"
        for name, r in self._routes.items()
        if name != self.default_agent
    )

    prompt = f"""你是一个意图分类器。根据用户问题，选择最合适的Agent。

可用的Agent：
{agent_list}
{self.default_agent}(兜底处理)

只返回Agent名称，不要其他内容。

用户问题：{query}"""

    # 调用 LLM（简化版）
    response = llm_client.chat(prompt)
    agent_name = response.strip().strip('"')

    # 安全检查：确保返回的是已注册的 Agent
    if agent_name not in self._routes:
        agent_name = self.default_agent

    return agent_name
