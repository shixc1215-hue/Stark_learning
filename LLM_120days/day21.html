🔧 一、@staticmethod —— 不需要 self 的工具函数 ⏱ 20min
核心思想：静态方法是"属于类但不需要实例"的函数。它不接收 self 也不接收 cls，就是一个普通函数碰巧放在类的命名空间里——逻辑上归这个类管，但不需要任何实例数据。
1.1 语法对比：普通方法 vs 静态方法
class LoanCalculator:
    """AMC 不良贷款计算器 —— 静态方法示例"""

    def __init__(self, principal: float, rate: float, months: int):
        self.principal = principal    # 本金（万元）
        self.rate = rate              # 月利率（小数）
        self.months = months          # 期数

    # —— 普通实例方法：需要 self（实例数据）——
    def monthly_payment(self) -> float:
        """等额本息月供，依赖实例属性"""
        r = self.rate
        n = self.months
        p = self.principal
        if r == 0:
            return p / n
        return p * r * (1 + r)**n / ((1 + r)**n - 1)

    # —— @staticmethod：不需要 self ——
    @staticmethod
    def risk_level(non_perf_ratio: float) -> str:
        """
        根据不良率判断风险等级
        纯工具函数：输入不良率，输出等级字符串
        注意：没有 self 参数！不需要实例！
        """
        if non_perf_ratio >= 0.05:
            return "高风险"
        elif non_perf_ratio >= 0.03:
            return "中风险"
        elif non_perf_ratio >= 0.01:
            return "低风险"
        else:
            return "正常"

    @staticmethod
    def format_amount(amount: float) -> str:
        """金额格式化：100.5 → '100.50万元'"""
        return f"{amount:,.2f}万元"


# ✅ 静态方法不需要创建实例就能调用！
print(LoanCalculator.risk_level(0.045))      # → 中风险
print(LoanCalculator.format_amount(5000.123)) # → 5,000.12万元

# 当然，通过实例调用也可以（但不推荐，因为不需要实例）
calc = LoanCalculator(5000, 0.005, 60)
print(calc.monthly_payment())       # → 普通方法，需要实例
print(calc.risk_level(0.045))       # → 静态方法，通过实例调用（不推荐）
💡 什么时候用 @staticmethod？
  • 逻辑上属于这个类（放在这里最合理），但不需要访问实例属性或类属性
  • 纯计算函数、格式化工具、校验规则——输入输出完全自包含
  • 想想 AMC：验证身份证号的规则属于 AMCCustomer 类，但验证过程不需要某个具体客户的数据
⚠️ 常见误区：如果一个方法只用了 self 的一个属性，那它仍然是实例方法，不应该改成静态方法。静态方法的判断标准是"完全不依赖实例/类状态"。
1.2 实战：AMC 数据校验工具集
class DataValidator:
    """AMC 数据质量校验工具 —— 所有方法都是 @staticmethod"""

    @staticmethod
    def is_valid_amount(amount: float) -> bool:
        """校验金额：必须非负，精确到分"""
        return isinstance(amount, (int, float)) and amount >= 0

    @staticmethod
    def is_valid_date(date_str: str) -> bool:
        """校验日期格式 YYYY-MM-DD"""
        import re
        return bool(re.match(r'^\d{4}-\d{2}-\d{2}$', date_str))

    @staticmethod
    def is_valid_id_card(id_card: str) -> bool:
        """简单校验身份证号长度"""
        return len(id_card) == 18 and id_card.isdigit()[:-1]

    @staticmethod
    def validate_row(row: dict, rules: list) -> list:
        """
        批量校验一行数据
        rules: [("金额", "amount"), ("日期", "date"), ...]
        返回错误列表，空列表表示通过
        """
        errors = []
        validators = {
            "amount": DataValidator.is_valid_amount,
            "date": DataValidator.is_valid_date,
            "id_card": DataValidator.is_valid_id_card,
        }
        for field, rule_type in rules:
            value = row.get(field)
            validator = validators.get(rule_type)
            if validator and not validator(value):
                errors.append(f"{field} 校验失败")
        return errors


# 使用示例 —— 不需要实例化
row = {"债务人说": "张三", "不良余额": -100, "入账日期": "2026-06-19"}
errors = DataValidator.validate_row(row, [("不良余额", "amount"), ("入账日期", "date")])
print(errors)  # → ['不良余额 校验失败']

2.2 工厂方法 —— AMC 报告的多种创建方式
from datetime import date

class AMCReport:
    """AMC 不良资产报告 —— 用 @classmethod 实现工厂模式"""

    def __init__(self, report_id: str, debtor_name: str,
                 principal: float, non_perf_balance: float,
                 report_date: str):
        self.report_id = report_id
        self.debtor_name = debtor_name
        self.principal = principal          # 贷款本金
        self.non_perf_balance = non_perf_balance  # 不良余额
        self.report_date = report_date
        self._validated = False

    # —— 普通实例方法 ——
    def non_perf_ratio(self) -> float:
        """计算不良率"""
        return self.non_perf_balance / self.principal if self.principal else 0

    def __repr__(self):
        return (f"AMCReport('{self.report_id}', {self.debtor_name}, "
                f"不良率={self.non_perf_ratio():.2%})")

    # ======== @classmethod 工厂方法 ========

    @classmethod
    def from_dict(cls, data: dict) -> "AMCReport":
        """
        从字典创建报告（最常用的工厂方法）
        cls 就是 AMCReport 类本身
        cls(...) 等同于 AMCReport(...)
        """
        return cls(
            report_id=data["报告编号"],
            debtor_name=data["债务人说"],
            principal=data["贷款本金"],
            non_perf_balance=data["不良余额"],
            report_date=data["报告日期"],
        )

    @classmethod
    def from_csv_row(cls, row: str, delimiter: str = ",") -> "AMCReport":
        """
        从CSV行创建报告
        格式：报告编号,债务人说,贷款本金,不良余额,报告日期
        """
        parts = row.strip().split(delimiter)
        return cls(
            report_id=parts[0],
            debtor_name=parts[1],
            principal=float(parts[2]),
            non_perf_balance=float(parts[3]),
            report_date=parts[4],
        )

    @classmethod
    def create_monthly_summary(cls, reports: list) -> dict:
        """
        类级别的聚合方法：汇总多份报告
        注意：这里返回 dict 而非实例，展示 @classmethod 的另一种用法
        """
        total_principal = sum(r.principal for r in reports)
        total_non_perf = sum(r.non_perf_balance for r in reports)
        return {
            "报告数量": len(reports),
            "总本金": total_principal,
            "总不良余额": total_non_perf,
            "综合不良率": total_non_perf / total_principal if total_principal else 0,
        }


# ======== 使用示例 ========

# 方式1：直接构造
r1 = AMCReport("R001", "XX公司", 10000, 500, "2026-06-19")

# 方式2：从字典构造（工厂方法）
data = {"报告编号": "R002", "债务人说": "YY公司", "贷款本金": 20000, "不良余额": 1500, "报告日期": "2026-06-19"}
r2 = AMCReport.from_dict(data)   # ← 注意：类名.方法名 调用

# 方式3：从CSV行构造
r3 = AMCReport.from_csv_row("R003,ZZ公司,50000,3000,2026-06-19")

print(r1)  # AMCReport('R001', XX公司, 不良率=5.00%)
print(r2)  # AMCReport('R002', YY公司, 不良率=7.50%)

# 方式4：类级别聚合
summary = AMCReport.create_monthly_summary([r1, r2, r3])
print(f"综合不良率: {summary['综合不良率']:.2%}")  # → 综合不良率: 6.83%
💡 工厂方法的优势：
  • 给类提供多种构造方式，用户不需要记住每种方式的参数顺序
  • 返回值类型是 cls，所以继承时自动适配子类（子类调用工厂方法会返回子类实例！）
  • 比在 __init__ 里写一堆 if-else 判断参数类型优雅得多
2.3 继承中的 cls 魔法
class DetailedReport(AMCReport):
    """详细报告 —— 继承 AMCReport"""

    def __init__(self, *args, risk_level: str = "未知", **kwargs):
        super().__init__(*args, **kwargs)
        self.risk_level = risk_level

    def __repr__(self):
        return f"DetailedReport('{self.report_id}', 风险={self.risk_level})"

    @classmethod
    def from_dict(cls, data: dict) -> "DetailedReport":
        """重写工厂方法 —— 自动添加风险等级"""
        instance = super().from_dict(data)
        # 基于不良率自动判断风险
        ratio = instance.non_perf_ratio()
        if ratio >= 0.05:
            instance.risk_level = "高风险"
        elif ratio >= 0.03:
            instance.risk_level = "中风险"
        else:
            instance.risk_level = "低风险"
        return instance


# 关键！子类调用工厂方法，cls 指向子类
data = {"报告编号": "R004", "债务人说": "AAA公司", "贷款本金": 10000,
        "不良余额": 800, "报告日期": "2026-06-19"}

r = DetailedReport.from_dict(data)
print(r)  # → DetailedReport('R004', 风险=高风险)
print(type(r))  # → <class '__main__.DetailedReport'>
# cls 自动是 DetailedReport，不是 AMCReport！
✨ 三、魔术方法 —— Python 的"协议编程" ⏱ 30min
核心思想：魔术方法（也叫 dunder 方法）是 Python 内置的协议接口。定义了它们，你的类就能像内置类型一样被 print()、len()、for、+ 等操作。这不是黑魔法，而是 Python 的"契约"——你遵守协议，Python 给你便利。
3.1 字符串表示：__str__ vs __repr__
class Asset:
    """AMC 资产记录"""

    def __init__(self, asset_id: str, debtor: str, balance: float):
        self.asset_id = asset_id
        self.debtor = debtor
        self.balance = balance

    def __str__(self):
        """
        用户友好的字符串 —— print() 和 str() 调用
        目标：给人看，通俗易懂
        """
        return f"{self.debtor}（{self.asset_id}）：{self.balance:,.0f}万元"

    def __repr__(self):
        """
        开发者友好的字符串 —— 交互式环境/调试显示
        目标：能还原对象，或至少包含关键信息
        最好让 eval(repr(obj)) == obj（虽然不强制）
        """
        return f"Asset('{self.asset_id}', '{self.debtor}', {self.balance})"


asset = Asset("A001", "XX公司", 5000.5)

print(str(asset))    # → XX公司（A001）：5,001万元（给用户看的）
print(repr(asset))   # → Asset('A001', 'XX公司', 5000.5)（给开发者看的）

# 直接 print 调用 __str__
print(asset)         # → XX公司（A001）：5,001万元

# 在列表、字典等容器中显示，调用 __repr__
assets = [asset, Asset("A002", "YY公司", 3000)]
print(assets)
# → [Asset('A001', 'XX公司', 5000.5), Asset('A002', 'YY公司', 3000)]
⚠️ 只有 __repr__ 没有 __str__ 时，print() 会回退到 __repr__。建议至少实现 __repr__，再按需加 __str__。
3.2 长度与布尔值：__len__ / __bool__
class AssetPortfolio:
    """AMC 资产组合"""

    def __init__(self, name: str):
        self.name = name
        self._assets: list[Asset] = []  # 内部存储

    def add(self, asset: Asset):
        self._assets.append(asset)

    def __len__(self):
        """
        len(portfolio) 调用此方法
        让你的对象像列表一样支持 len()
        """
        return len(self._assets)

    def __bool__(self):
        """
        if portfolio: 调用此方法
        自定义对象的"真假"判断
        如果没有定义，默认用 __len__（非0为True）
        """
        return len(self._assets) > 0

    def __repr__(self):
        return f"AssetPortfolio('{self.name}', {len(self)}笔资产)"

print(bool(portfolio))      # → False

print(len(portfolio))       # → 1
print(bool(portfolio))      # → True

if portfolio:

3.3 容器协议：__getitem__ / __contains__ / __iter__
class AssetPortfolio:
    """AMC 资产组合 —— 完整容器协议"""

    def __init__(self, name: str):
        self.name = name
        self._assets: list[Asset] = []

    def add(self, asset: Asset):
        self._assets.append(asset)

    def __len__(self):
        return len(self._assets)

    def __getitem__(self, key):
        """
        索引访问：portfolio[0]、portfolio[1:3]
        key 可以是 int（索引）或 slice（切片）
        """
        if isinstance(key, int):
            return self._assets[key]
        elif isinstance(key, slice):
            # 返回新的 AssetPortfolio，保持类型一致
            result = AssetPortfolio(f"{self.name}（切片）")
            for asset in self._assets[key]:
                result.add(asset)
            return result
        raise TypeError("只支持整数索引和切片")

    def __contains__(self, asset_id: str) -> bool:
        """
        'in' 操作符：asset_id in portfolio
        返回 True/False
        """
        return any(a.asset_id == asset_id for a in self._assets)

    def __iter__(self):
        """
        迭代协议：for asset in portfolio
        返回迭代器
        """
        return iter(self._assets)

    def __repr__(self):
        return f"AssetPortfolio('{self.name}', {len(self)}笔资产)"

3.4 运算符重载：__add__ / __eq__ / __lt__
class Asset:
    """AMC 资产 —— 支持运算符"""

    def __init__(self, asset_id: str, debtor: str, balance: float):
        self.asset_id = asset_id
        self.debtor = debtor
        self.balance = balance

    def __repr__(self):
        return f"Asset('{self.asset_id}', balance={self.balance})"

    def __add__(self, other):
        """
        a1 + a2 → 合并两笔资产
        Python 在执行 + 时，自动调用左边对象的 __add__
        """
        return Asset(
            asset_id=f"{self.asset_id}+{other.asset_id}",
            debtor=f"{self.debtor},{other.debtor}",
            balance=self.balance + other.balance,
        )

    def __eq__(self, other):
        """a1 == a2 → 比较两笔资产是否相等（按ID和余额）"""
        if not isinstance(other, Asset):
            return NotImplemented  # 让 Python 处理类型不匹配
        return self.asset_id == other.asset_id and self.balance == other.balance

    def __lt__(self, other):
        """a1 < a2 → 按余额比较"""
        return self.balance < other.balance

    def __hash__(self):
        """定义了 __eq__ 后，必须定义 __hash__ 才能用集合/字典"""
        return hash((self.asset_id, self.balance))


# ======== 运算符实战 ========
a1 = Asset("A001", "甲公司", 5000)
a2 = Asset("A002", "乙公司", 3000)
a3 = Asset("A001", "丙公司", 5000)  # 同ID同余额

# 加法合并
merged = a1 + a2
print(merged)  # → Asset('A001+A002', balance=8000)

# 相等比较
print(a1 == a3)   # → True（ID和余额相同）
print(a1 == a2)   # → False

# 大小比较 + 排序
assets = [a2, a1]
print(sorted(assets))  # → [Asset('A002', balance=3000), Asset('A001', balance=5000)]

# 放入集合（需要 __hash__）
asset_set = {a1, a2, a3}
print(len(asset_set))  # → 2（a1 和 a3 哈希相同，被视为同一资产）
💡 运算符重载的注意事项：
  • 定义了 __eq__ 但不定义 __hash__，对象会变成 unhashable（不能放集合/当字典键）
  • __lt__ 定义后，sorted()、min()、max() 全部自动生效
  • 类型不匹配时返回 NotImplemented（不是 False！），让 Python 尝试右操作数的反向方法
3.5 上下文管理器：__enter__ / __exit__
class ReportSession:
    """AMC 报告生成会话 —— 上下文管理器"""

    def __init__(self, session_name: str):
        self.session_name = session_name
        self.reports = []
        self._active = False

    def __enter__(self):
        """
        with 语句进入时调用
        应该返回 self（或者你想暴露的资源）
        """
        print(f"[{self.session_name}] 会话开始...")
        self._active = True
        self.start_time = __import__('time').time()
        return self  # as 后面的变量就是这个返回值

    def __exit__(self, exc_type, exc_val, exc_tb):
        """
        with 语句退出时调用（无论是否异常）
        exc_type: 异常类型（None = 正常退出）
        返回 True: 异常被处理，不会传播
        返回 False: 异常继续传播
        """
        elapsed = __import__('time').time() - self.start_time
        self._active = False
        if exc_type is None:
            print(f"[{self.session_name}] 会话结束，生成 {len(self.reports)} 份报告，耗时 {elapsed:.1f}s")
        else:
            print(f"[{self.session_name}] 会话异常终止：{exc_val}")
        return False  # 不吞异常

    def add_report(self, report):
        if not self._active:
            raise RuntimeError("会话未激活，请使用 with 语句")
        self.reports.append(report)


# ======== 使用 with 语句 ========
with ReportSession("2026Q2_C表生成") as session:
    session.add_report("C001")
    session.add_report("C002")
    session.add_report("C003")
# 输出：
# [2026Q2_C表生成] 会话开始...
# [2026Q2_C表生成] 会话结束，生成 3 份报告，耗时 0.0s
🤖 四、LLM 实战 —— 用魔术方法构建 Prompt 模板类 ⏱ 25min
实战场景：构建一个 PromptTemplate 类，利用魔术方法让它像字符串一样好用：+ 拼接模板、len() 计算长度、for 遍历变量、str() 渲染输出。 
class PromptTemplate:
    """
    Prompt 模板类 —— 用魔术方法让它像原生字符串一样好用
    支持 {变量} 占位符，类似 Python 的 format() 或 Jinja2
    """

    def __init__(self, template: str, variables: list[str] = None):
        self.template = template
        self.variables = variables or self._extract_variables(template)
        self._metadata = {}  # 元数据（版本、标签等）

    @staticmethod
    def _extract_variables(template: str) -> list[str]:
        """从模板中提取 {变量名} 列表"""
        import re
        return list(set(re.findall(r'\{(\w+)\}', template)))

    def __repr__(self):
        return f"PromptTemplate(vars={self.variables})"

    def __str__(self):
        """直接转字符串 → 原始模板"""
        return self.template

    def __len__(self):
        """len(template) → 模板字符数"""
        return len(self.template)

    def __add__(self, other):
        """
        t1 + t2 → 合并两个模板
        自动合并变量列表
        """
        if isinstance(other, str):
            other = PromptTemplate(other)
        combined = f"{self.template}\n{other.template}"
        merged_vars = list(set(self.variables + other.variables))
        return PromptTemplate(combined, merged_vars)

    def __contains__(self, var_name: str) -> bool:
        """'system' in template → 检查是否包含某个变量"""
        return var_name in self.variables

    def __iter__(self):
        """for var in template → 遍历所有变量名"""
        return iter(self.variables)

    def __call__(self, **kwargs) -> str:
        """
        template(role="系统", content="...") → 渲染模板
        让模板对象可以像函数一样调用！
        """
        try:
            return self.template.format(**kwargs)
        except KeyError as e:
            missing = e.args[0]
            raise ValueError(f"缺少必要变量：{{{missing}}}，需要提供：{self.variables}") from None

    # ======== @classmethod 工厂方法 ========

    @classmethod
    def from_system_user(cls, system: str, user: str) -> "PromptTemplate":
        """快捷创建 system + user 双模板"""
        combined = f"System: {system}\n\nUser: {user}"
        return cls(combined)

    @classmethod
    def from_file(cls, filepath: str) -> "PromptTemplate":
        """从文件加载模板"""
        with open(filepath, 'r', encoding='utf-8') as f:
            return cls(f.read().strip())

class LLMMessageList:
    """
    对话消息列表 —— 支持完整的容器协议
    自动计算 token 数量，方便管理上下文窗口
    """

    def __init__(self):
        self._messages = []  # [{"role": "system"/"user"/"assistant", "content": "..."}]

    def __len__(self):
        """len(messages) → 消息条数"""
        return len(self._messages)

    def __repr__(self):
        return f"LLMMessageList({len(self)}条消息)"

    def __getitem__(self, key):
        """messages[0]、messages[-1]、messages[1:3]"""
        return self._messages[key]

    def __iter__(self):
        """for msg in messages"""
        return iter(self._messages)

    def __contains__(self, role: str) -> bool:
        """'assistant' in messages → 是否有AI回复"""
        return any(m["role"] == role for m in self._messages)

    def __bool__(self):
        """if messages: → 是否有消息"""
        return len(self._messages) > 0

    def add_system(self, content: str):
        self._messages.append({"role": "system", "content": content})

    def add_user(self, content: str):
        self._messages.append({"role": "user", "content": content})

    def add_assistant(self, content: str):
        self._messages.append({"role": "assistant", "content": content})

    def to_openai(self) -> list[dict]:
        """转换为 OpenAI API 格式"""
        return self._messages.copy()

    @property
    def total_chars(self) -> int:
        """所有消息的总字符数"""
        return sum(len(m["content"]) for m in self._messages)

    def trim_to_chars(self, max_chars: int):
        """
        裁剪历史消息，保留 system + 最近的消息
        模拟 token 限制（简化版用字符数代替）
        """
        system_msgs = [m for m in self._messages if m["role"] == "system"]
        other_msgs = [m for m in self._messages if m["role"] != "system"]

        # 从最早的开始删，直到总字符数满足要求
        total = sum(len(m["content"]) for m in system_msgs)
        keep = []
        for msg in reversed(other_msgs):
            total += len(msg["content"])
            if total > max_chars:
                break
            keep.append(msg)

        self._messages = system_msgs + list(reversed(keep))

练习3（LeetCode #155）：最小栈

设计一个栈，支持 push、pop、top、getMin 四个操作，全部 O(1) 时间复杂度。提示：__len__ 返回栈大小，__bool__ 判断是否为空。

class MinStack:
    def __init__(self):
        self._stack = []       # 数据栈
        self._min_stack = []   # 最小值栈

    def push(self, val: int):
        self._stack.append(val)
        # 最小值栈同步记录当前最小值
        min_val = min(val, self._min_stack[-1]) if self._min_stack else val
        self._min_stack.append(min_val)

    def pop(self) -> int:
        self._min_stack.pop()
        return self._stack.pop()

    def top(self) -> int:
        return self._stack[-1]

    def getMin(self) -> int:
        return self._min_stack[-1]

    def __len__(self):
        return len(self._stack)

    def __bool__(self):
        return len(self._stack) > 0
